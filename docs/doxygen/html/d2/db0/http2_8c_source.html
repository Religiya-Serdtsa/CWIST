309int h2_decode_integer(const unsigned char *buf,
310 size_t len,
311 size_t *pos,
312 uint8_t prefix_bits,
313 uint32_t *value) {
314 if (*pos >= len || prefix_bits == 0 || prefix_bits > 8) return -1;
315 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
316 uint32_t n = buf[*pos] & mask;
317 (*pos)++;
318 if (n < mask) {
319 *value = n;
320 return 0;
321 }
322
323 uint32_t m = 0;
324 while (*pos < len) {
325 uint8_t b = buf[(*pos)++];
326 if (m > 28) return -1;
327 n += (uint32_t)(b & 0x7f) << m;
328 if ((b & 0x80) == 0) {
329 *value = n;
330 return 0;
331 }
332 m += 7;
333 }
334 return -1;
335}

647
648 h2_huffman_node *node = h2_huffman_root;
649 for (size_t i = 0; i < src_len; ++i) {
650 unsigned char byte = src[i];
651 for (int b = 7; b >= 0; --b) {
652 int bit = (byte >> b) & 1;
653 node = node->child[bit];
654 if (!node) { free(out); return NULL; }
655 if (node->is_terminal) {
656 if (node->symbol == 256) {
657 free(out);
658 return NULL;
659 }
660 if (out_pos + 1 >= cap) {
661 cap *= 2;
662 char *tmp = (char *)realloc(out, cap);
663 if (!tmp) { free(out); return NULL; }
664 out = tmp;
665 }
666 out[out_pos++] = (char)node->symbol;
667 node = h2_huffman_root;
668 }
669 }
670 }
671
672 if (node != h2_huffman_root) {
673 h2_huffman_node *check = node;
674 while (check && !check->is_terminal) {
675 check = check->child[1];
676 }
677 if (!check || check->symbol != 256) {
678 free(out);
679 return NULL;
680 }
681 }
682
683 out[out_pos] = '\0';
684 if (out_len) *out_len = out_pos;
685 return out;
686}

688char *h2_decode_string(const unsigned char *buf, size_t len, size_t *pos) {
689 if (*pos >= len) return NULL;
690 bool huffman = (buf[*pos] & 0x80) != 0;
691 uint32_t str_len = 0;
692 if (h2_decode_integer(buf, len, pos, 7, &str_len) != 0) return NULL;
693 if (*pos + str_len > len) return NULL;
694
695 if (huffman) {
696 size_t decoded_len = 0;
697 char *out = h2_huffman_decode(buf + *pos, str_len, &decoded_len);
698 *pos += str_len;
699 return out;
700 }
701
702 char *out = (char *)malloc((size_t)str_len + 1);
703 if (!out) return NULL;
704 memcpy(out, buf + *pos, str_len);
705 out[str_len] = '\0';
706 *pos += str_len;
707 return out;
708}

748static void h2_decode_header_block(cwist_http_request *req, const unsigned char *payload, size_t len) {
749 size_t pos = 0;
750
751 while (pos < len) {
752 uint8_t b = payload[pos];
753 uint32_t name_index = 0;
754 char *name = NULL;
755 char *value = NULL;
756
757 if (b & 0x80) {
758 uint32_t index = 0;
759 if (h2_decode_integer(payload, len, &pos, 7, &index) != 0) break;
760 const cwist_http2_static_header *entry = h2_static_header(index);
761 if (entry && entry->name) h2_apply_header(req, entry->name, entry->value);
762 continue;
763 }
764
765 if ((b & 0xE0) == 0x20) {
766 uint32_t new_size = 0;
767 if (h2_decode_integer(payload, len, &pos, 5, &new_size) != 0) break;
768 continue;
769 }
770
771 if ((b & 0x40) == 0x40) {
772 if (h2_decode_integer(payload, len, &pos, 6, &name_index) != 0) break;
773 } else if ((b & 0xf0) == 0x00) {
774 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0) break;
775 } else if ((b & 0xf0) == 0x10) {
776 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0) break;
777 } else {
778 break;
779 }
780
781 if (name_index > 0) {
782 const cwist_http2_static_header *entry = h2_static_header(name_index);
783 if (!entry || !entry->name) {
784 char *discard = h2_decode_string(payload, len, &pos);
785 free(discard);
786 continue;
787 }
788 name = strdup(entry->name);
789 } else {
790 name = h2_decode_string(payload, len, &pos);
791 }
792 value = h2_decode_string(payload, len, &pos);
793 if (!name || !value) {
794 free(name);
795 free(value);
796 break;
797 }
798 h2_apply_header(req, name, value);
799 free(name);
800 free(value);
801 }
802}

806static size_t h2_encode_integer(unsigned char *dst, size_t dst_cap, uint32_t value, uint8_t prefix_bits) {
807 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
808 unsigned char first = dst[0] & ~mask;
809 if (value < mask) {
810 dst[0] = first | (uint8_t)value;
811 return 1;
812 }
813 dst[0] = first | mask;
814 value -= mask;
815 size_t i = 1;
816 while (value >= 128) {
817 if (i >= dst_cap) return 0;
818 dst[i++] = (unsigned char)((value & 0x7F) | 0x80);
819 value >>= 7;
820 }
821 if (i >= dst_cap) return 0;
822 dst[i++] = (unsigned char)value;
823 return i;
824}

826static size_t h2_encode_string(unsigned char *dst, size_t dst_cap, const char *str) {
827 size_t len = strlen(str);
828 dst[0] = 0x00;
829 size_t n = h2_encode_integer(dst, dst_cap, (uint32_t)len, 7);
830 if (n == 0 || n + len > dst_cap) return 0;
831 memcpy(dst + n, str, len);
832 return n + len;
833}

844static size_t h2_encode_response_headers(cwist_http_response *res,
845 unsigned char *dst, size_t dst_cap) {
846 size_t pos = 0;
847
848 switch (res->status_code) {
849 case 200: if (pos >= dst_cap) return 0; dst[pos++] = 0x88; break;
850 case 204: if (pos >= dst_cap) return 0; dst[pos++] = 0x89; break;
851 case 206: if (pos >= dst_cap) return 0; dst[pos++] = 0x8a; break;
852 case 304: if (pos >= dst_cap) return 0; dst[pos++] = 0x8b; break;
853 case 400: if (pos >= dst_cap) return 0; dst[pos++] = 0x8c; break;
854 case 404: if (pos >= dst_cap) return 0; dst[pos++] = 0x8d; break;
855 case 500: if (pos >= dst_cap) return 0; dst[pos++] = 0x8e; break;
856 default: {
857 char status_str[16];
858 snprintf(status_str, sizeof(status_str), "%d", res->status_code);
859 if (pos + 1 > dst_cap) return 0;
860 dst[pos] = 0x00;
861 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
862 if (n == 0) return 0;
863 pos += n;
864 n = h2_encode_string(dst + pos, dst_cap - pos, ":status");
865 if (n == 0) return 0;
866 pos += n;
867 n = h2_encode_string(dst + pos, dst_cap - pos, status_str);
868 if (n == 0) return 0;
869 pos += n;
870 break;
871 }
872 }
873
874 size_t body_len = 0;
875 if (res->use_file_stream) body_len = res->file_stream_len;
876 else if (res->is_ptr_body) body_len = res->ptr_body_len;
877 else if (res->body) body_len = res->body->size;
878
879 if (!headers_have_content_length(res->headers)) {
880 char cl_str[32];
881 snprintf(cl_str, sizeof(cl_str), "%zu", body_len);
882 int name_idx = h2_static_table_find_name("content-length");
883 if (name_idx > 0) {
884 if (pos + 1 > dst_cap) return 0;
885 dst[pos] = 0x00;
886 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, (uint32_t)name_idx, 4);
887 if (n == 0) return 0;
888 pos += n;
889 } else {
890 if (pos + 1 > dst_cap) return 0;
891 dst[pos] = 0x00;
892 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
893 if (n == 0) return 0;
894 pos += n;
895 n = h2_encode_string(dst + pos, dst_cap - pos, "content-length");
896 if (n == 0) return 0;
897 pos += n;
898 }
899 size_t n = h2_encode_string(dst + pos, dst_cap - pos, cl_str);
900 if (n == 0) return 0;
901 pos += n;
902 }
903
904 cwist_http_header_node *curr = res->headers;
905 while (curr) {
906 if (!curr->key || !curr->key->data || !curr->value || !curr->value->data) {
907 curr = curr->next;
908 continue;
909 }
910 if (strcasecmp(curr->key->data, "connection") == 0 ||
911 strcasecmp(curr->key->data, "keep-alive") == 0 ||
912 strcasecmp(curr->key->data, "transfer-encoding") == 0 ||
913 strcasecmp(curr->key->data, "upgrade") == 0) {
914 curr = curr->next;
915 continue;
916 }
917
918 char lower_name[256];
919 size_t name_len = strlen(curr->key->data);
920 if (name_len >= sizeof(lower_name)) name_len = sizeof(lower_name) - 1;
921 for (size_t i = 0; i < name_len; ++i) {
922 lower_name[i] = (char)tolower((unsigned char)curr->key->data[i]);
923 }
924 lower_name[name_len] = '\0';
925
926 int name_idx = h2_static_table_find_name(lower_name);
927 if (name_idx > 0) {
928 if (pos + 1 > dst_cap) return 0;
929 dst[pos] = 0x00;
930 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, (uint32_t)name_idx, 4);
931 if (n == 0) return 0;
932 pos += n;
933 } else {
934 if (pos + 1 > dst_cap) return 0;
935 dst[pos] = 0x00;
936 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
937 if (n == 0) return 0;
938 pos += n;
939 n = h2_encode_string(dst + pos, dst_cap - pos, lower_name);
940 if (n == 0) return 0;
941 pos += n;
942 }
943 size_t n = h2_encode_string(dst + pos, dst_cap - pos, curr->value->data);
944 if (n == 0) return 0;
945 pos += n;
946
947 curr = curr->next;
948 }
949
950 return pos;
951}

955static int h2_send_response_raw(cwist_https_connection *conn, uint32_t stream_id,
956 cwist_http_response *res, uint32_t max_frame_size) {
957 unsigned char header_block[8192];
958 size_t block_len = h2_encode_response_headers(res, header_block, sizeof(header_block));
959 if (block_len == 0) return -1;
960
961 uint8_t header_flags = CWIST_HTTP2_FLAG_END_HEADERS;
962
963 size_t body_len = 0;
964 const unsigned char *body_data = NULL;
965 if (res->use_file_stream) {
966 body_len = res->file_stream_len;
967 } else if (res->is_ptr_body) {
968 body_len = res->ptr_body_len;
969 body_data = (const unsigned char *)res->ptr_body;
970 } else if (res->body) {
971 body_len = res->body->size;
972 body_data = (const unsigned char *)res->body->data;
973 }
974
975 if (body_len == 0) header_flags |= CWIST_HTTP2_FLAG_END_STREAM;
976
977 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_HEADERS, header_flags, stream_id,
978 header_block, (uint32_t)block_len) != 0) {
979 return -1;
980 }
981
982 if (res->use_file_stream && res->file_stream_fd >= 0) {
983 off_t offset = res->file_stream_offset;
984 size_t remaining = res->file_stream_len;
985 while (remaining > 0) {
986 uint32_t chunk = (uint32_t)(remaining > max_frame_size ? max_frame_size : remaining);
987 unsigned char *chunk_buf = (unsigned char *)malloc(chunk);
988 if (!chunk_buf) return -1;
989 ssize_t r = pread(res->file_stream_fd, chunk_buf, chunk, offset);
990 if (r <= 0) { free(chunk_buf); return -1; }
991 uint8_t flags = (remaining == (size_t)r) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
992 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id, chunk_buf, (uint32_t)r) != 0) {
993 free(chunk_buf); return -1;
994 }
995 free(chunk_buf);
996 offset += r;
997 remaining -= (size_t)r;
998 }
999 } else {
1000 size_t sent = 0;
1001 while (sent < body_len) {
1002 size_t remaining = body_len - sent;
1003 uint32_t chunk = (uint32_t)(remaining > max_frame_size ? max_frame_size : remaining);
1004 uint8_t flags = (sent + chunk == body_len) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
1005 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id,
1006 body_data + sent, chunk) != 0) {
1007 return -1;
1008 }
1009 sent += chunk;
1010 }
1011 }
1012 return 0;
1013}

1105 }
1106

1109

1123 }

1125 h2_decode_header_block(s->req, payload + block_offset, block_len);
1126 }
1127 return 0;
1128 }
1129
1130 /* Need continuation */
1131 hc->expecting_continuation = true;

1139 return -1;

1169 if (!s) {
1170 s = h2_stream_create(hc, hc->cont_stream_id);

1183 s->req->private_data = hc->conn;
1184 }

1186 h2_decode_header_block(s->req, hc->cont_buf, hc->cont_len);
1187 }
1188 hc->expecting_continuation = false;
1189 hc->cont_len = 0;
1190 hc->cont_stream_id = 0;
1191 }
1192 return 0;

1195/* --- SETTINGS Parser --- */
1196
1197static int h2_handle_settings(h2_conn *hc, const unsigned char *payload, size_t len, bool ack) {
1198 if (ack) {
1199 return 0;
1200 }
1201
1202 if (len % 6 != 0) {
1203 return -1; /* FRAME_SIZE_ERROR */
1204 }
1205
1206 for (size_t i = 0; i + 6 <= len; i += 6) {
1207 uint16_t id = ((uint16_t)payload[i] << 8) | payload[i + 1];
1208 uint32_t value = ((uint32_t)payload[i + 2] << 24) |
1209 ((uint32_t)payload[i + 3] << 16) |
1210 ((uint32_t)payload[i + 4] << 8) |
1211 (uint32_t)payload[i + 5];
1212 switch (id) {
1213 case 0x1: /* SETTINGS_HEADER_TABLE_SIZE */
1214 break;
1215 case 0x2: /* SETTINGS_ENABLE_PUSH */
1216 if (value > 1) return -1; /* PROTOCOL_ERROR */
1217 break;
1218 case 0x3: /* SETTINGS_MAX_CONCURRENT_STREAMS */
1219 hc->peer_max_concurrent_streams = value;
1220 break;
1221 case 0x4: /* SETTINGS_INITIAL_WINDOW_SIZE */
1222 if (value > 2147483647U) return -1; /* FLOW_CONTROL_ERROR */
1223 {
1224 int32_t delta = (int32_t)value - (int32_t)hc->peer_initial_window_size;
1225 hc->peer_initial_window_size = value;
1226 h2_stream *s = hc->streams;
1227 while (s) {
1228 s->send_window += delta;
1229 s = s->next;
1230 }
1231 }
1232 break;
1233 case 0x5: /* SETTINGS_MAX_FRAME_SIZE */
1234 if (value < 16384 || value > 16777215) return -1; /* PROTOCOL_ERROR */
1235 hc->peer_max_frame_size = value;
1236 break;
1237 case 0x6: /* SETTINGS_MAX_HEADER_LIST_SIZE */
1238 break;
1239 }

1246static int h2_handle_window_update(h2_conn *hc, uint32_t stream_id,
1247 const unsigned char *payload, size_t len) {

1250 int32_t increment = (int32_t)(
1251 (((uint32_t)payload[0] & 0x7f) << 24) |
1252 ((uint32_t)payload[1] << 16) |

1265 * or may be ignored depending on state. We ignore for leniency. */
1266 return 0;
1267 }
1268 s->send_window += increment;
1269 if (s->send_window > 2147483647 || s->send_window < 0) return -1;
1270 }

1312/* --- Connection Dispatcher --- */
1313
1314cwist_error_t cwist_http2_serve_connection(
1315 cwist_https_connection *conn,
1316 void *user_ctx,
1317 cwist_http2_request_handler_func handler
1318) {
1319 cwist_error_t result;
1320
1321 if (!conn || !handler) {
1322 result = make_error(CWIST_ERR_INT16);
1323 result.error.err_i16 = -1;
1324 return result;
1325 }
1326
1327 if (cwist_http2_verify_preface(conn) != 0) {
1328 result = make_error(CWIST_ERR_INT16);
1329 result.error.err_i16 = -1;
1330 return result;
1331 }
1332
1333 h2_conn hc;
1334 h2_conn_init(&hc, conn);
1335
1336 unsigned char settings[6] = {0x00, 0x01, 0x00, 0x00, 0x00, 0x00};
1337 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_SETTINGS, 0, 0, settings, sizeof(settings)) != 0) {
1338 h2_conn_destroy(&hc);
1339 result = make_error(CWIST_ERR_INT16);
1340 result.error.err_i16 = -1;
1341 return result;
1342 }
1343
1344 bool connected = true;
1345 while (connected && atomic_load(&g_cwist_running)) {
1346 unsigned char hdr[9];
1347 int offset = 0;
1348 while (offset < 9) {
1349 int n = h2_read(conn, hdr + offset, 9 - offset);
1350 if (n <= 0) { connected = false; break; }
1351 offset += n;
1352 }
1353 if (!connected) break;
1354
1355 uint32_t len = ((uint32_t)hdr[0] << 16) | ((uint32_t)hdr[1] << 8) | hdr[2];
1356 uint8_t type = hdr[3];
1357 uint8_t flags = hdr[4];
1358 uint32_t stream_id = (((uint32_t)hdr[5] & 0x7f) << 24) |
1359 ((uint32_t)hdr[6] << 16) |
1360 ((uint32_t)hdr[7] << 8) |
1361 hdr[8];
1362
1363 /* Frame size validation */
1364 if (len > CWIST_HTTP2_MAX_FRAME_SIZE) {
1365 if (type == CWIST_HTTP2_FRAME_SETTINGS || type == CWIST_HTTP2_FRAME_HEADERS ||
1366 type == CWIST_HTTP2_FRAME_CONTINUATION || type == CWIST_HTTP2_FRAME_PUSH_PROMISE) {
1367 if (type == CWIST_HTTP2_FRAME_SETTINGS || len > hc.peer_max_frame_size) {
1368 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FRAME_SIZE_ERROR);
1369 connected = false;
1370 break;
1371 }
1372 }
1373 }
1374
1375 unsigned char *payload = NULL;
1376 if (len > 0) {
1377 payload = (unsigned char *)malloc(len);
1378 if (!payload) { connected = false; break; }
1379 int off = 0;
1380 while (off < (int)len) {
1381 int r = h2_read(conn, payload + off, len - off);
1382 if (r <= 0) { connected = false; break; }
1383 off += r;
1384 }
1385 }
1386 if (!connected) { free(payload); break; }
1387
1388 switch (type) {
1389 case CWIST_HTTP2_FRAME_SETTINGS: {
1390 if (h2_handle_settings(&hc, payload, len, flags & CWIST_HTTP2_FLAG_ACK) != 0) {
1391 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1392 connected = false;
1393 } else if ((flags & CWIST_HTTP2_FLAG_ACK) == 0) {
1394 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_SETTINGS, CWIST_HTTP2_FLAG_ACK, 0, NULL, 0) != 0) {
1395 connected = false;
1396 }
1397 }
1398 break;
1399 }
1400
1401 case CWIST_HTTP2_FRAME_HEADERS: {
1402 if (stream_id == 0 || (stream_id % 2) == 0) {
1403 h2_send_goaway(&hc, 0, H2_ERR_PROTOCOL_ERROR);
1404 connected = false;
1405 break;
1406 }
1407 /* Strip PADDED and PRIORITY from payload for decoding */
1408 size_t block_offset = 0;
1409 size_t block_len = len;
1410 if ((flags & CWIST_HTTP2_FLAG_PADDED) != 0 && block_len > 0) {
1411 uint8_t pad_len = payload[block_offset++];
1412 block_len--;
1413 if (pad_len <= block_len) block_len -= pad_len;
1414 }
1415 if ((flags & CWIST_HTTP2_FLAG_PRIORITY) != 0 && block_len >= 5) {
1416 block_offset += 5;
1417 block_len -= 5;
1418 }
1419 if (h2_begin_headers(&hc, stream_id,
1420 payload + block_offset, block_len,
1421 flags & CWIST_HTTP2_FLAG_END_HEADERS,
1422 flags & CWIST_HTTP2_FLAG_END_STREAM) != 0) {
1423 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1424 connected = false;
1425 break;
1426 }
1427 hc.last_processed_stream_id = stream_id;
1428 if (flags & CWIST_HTTP2_FLAG_END_STREAM) {
1429 h2_stream *s = h2_stream_find(&hc, stream_id);
1430 if (s && s->req) {
1431 cwist_http_response *res = cwist_http_response_create();
1432 if (res) {
1433 handler(user_ctx, s->req, res);
1434 h2_inject_alt_svc(conn, res);
1435 if (h2_send_response_hc(&hc, stream_id, res) != 0) connected = false;
1436 cwist_http_response_destroy(res);
1437 }
1438 h2_stream_remove(&hc, stream_id);
1439 }
1440 }
1441 break;
1442 }
1443
1444 case CWIST_HTTP2_FRAME_CONTINUATION: {
1445 if (stream_id == 0 || !hc.expecting_continuation || hc.cont_stream_id != stream_id) {
1446 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1447 connected = false;
1448 break;
1449 }
1450 if (h2_handle_continuation(&hc, stream_id, payload, len,
1451 flags & CWIST_HTTP2_FLAG_END_HEADERS) != 0) {
1452 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1453 connected = false;
1454 break;
1455 }
1456 if (flags & CWIST_HTTP2_FLAG_END_STREAM) {
1457 /* END_STREAM on CONTINUATION is a protocol violation */
1458 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1459 connected = false;
1460 break;
1461 }
1462 if ((flags & CWIST_HTTP2_FLAG_END_HEADERS) && hc.cont_end_stream) {
1463 h2_stream *s = h2_stream_find(&hc, stream_id);
1464 if (s && s->req) {
1465 cwist_http_response *res = cwist_http_response_create();
1466 if (res) {
1467 handler(user_ctx, s->req, res);
1468 h2_inject_alt_svc(conn, res);
1469 if (h2_send_response_hc(&hc, stream_id, res) != 0) connected = false;
1470 cwist_http_response_destroy(res);
1471 }
1472 h2_stream_remove(&hc, stream_id);
1473 }
1474 }
1475 break;
1476 }
1477
1478 case CWIST_HTTP2_FRAME_DATA: {
1479 if (stream_id == 0) {
1480 h2_send_goaway(&hc, 0, H2_ERR_PROTOCOL_ERROR);
1481 connected = false;
1482 break;
1483 }
1484 h2_stream *s = h2_stream_find(&hc, stream_id);
1485 if (!s) {
1486 /* DATA on closed/unknown stream: must check if it violates flow control */
1487 /* For simplicity, we treat it as a stream error by ignoring the data
1488 * but still accounting connection-level window. */
1489 hc.conn_recv_window -= (int32_t)len;
1490 if (hc.conn_recv_window < 0) {
1491 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FLOW_CONTROL_ERROR);
1492 connected = false;
1493 }
1494 break;
1495 }
1496 if (len > 0) {
1497 hc.conn_recv_window -= (int32_t)len;
1498 s->recv_window -= (int32_t)len;
1499 if (hc.conn_recv_window < 0 || s->recv_window < 0) {
1500 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FLOW_CONTROL_ERROR);
1501 connected = false;
1502 break;
1503 }
1504 }
1505 if (payload && len > 0) {
1506 cwist_sstring_append_len(s->req->body, (const char *)payload, len);
1507 }
1508 if (flags & CWIST_HTTP2_FLAG_END_STREAM) {
1509 cwist_http_response *res = cwist_http_response_create();
1510 if (res) {
1511 handler(user_ctx, s->req, res);
1512 h2_inject_alt_svc(conn, res);
1513 if (h2_send_response_hc(&hc, stream_id, res) != 0) connected = false;
1514 cwist_http_response_destroy(res);
1515 }
1516 h2_stream_remove(&hc, stream_id);
1517 }
1518 if (connected) {
1519 h2_auto_window_update(&hc, s);
1520 }
1521 break;
1522 }
1523
1524 case CWIST_HTTP2_FRAME_RST_STREAM: {
1525 if (stream_id == 0 || len != 4) {
1526 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1527 connected = false;
1528 break;
1529 }
1530 h2_stream_remove(&hc, stream_id);
1531 break;
1532 }
1533
1534 case CWIST_HTTP2_FRAME_PING: {
1535 if (len != 8) {
1536 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FRAME_SIZE_ERROR);
1537 connected = false;
1538 break;
1539 }
1540 if ((flags & CWIST_HTTP2_FLAG_ACK) == 0) {
1541 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_PING, CWIST_HTTP2_FLAG_ACK, 0, payload, 8) != 0) {
1542 connected = false;
1543 }
1544 }
1545 break;
1546 }
1547
1548 case CWIST_HTTP2_FRAME_GOAWAY: {
1549 connected = false;
1550 break;
1551 }
1552
1553 case CWIST_HTTP2_FRAME_WINDOW_UPDATE: {
1554 if (len != 4) {
1555 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FRAME_SIZE_ERROR);
1556 connected = false;
1557 break;
1558 }
1559 if (h2_handle_window_update(&hc, stream_id, payload, len) != 0) {
1560 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FLOW_CONTROL_ERROR);
1561 connected = false;
1562 }
1563 break;
1564 }
1565
1566 default:
1567 /* Unknown frames MUST be ignored (RFC 7540 Section 5.5) */
1568 break;
1569 }
1570
1571 free(payload);
1572 }
1573
1574 h2_conn_destroy(&hc);
1575
1576 result = make_error(CWIST_ERR_INT16);
1577 result.error.err_i16 = 0;
1578 return result;
1579}
