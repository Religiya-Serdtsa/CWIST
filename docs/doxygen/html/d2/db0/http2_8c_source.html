280/* Poll the socket for the direction OpenSSL is waiting on, or for plain
281 * socket writability. Returns 0 when ready, -1 on timeout/error. */
282static int h2_wait_socket(cwist_https_connection *conn, int ssl_error, int timeout_ms) {
283 struct pollfd pfd = { .fd = conn->fd, .events = 0 };
284 if (conn->ssl) {
285 if (ssl_error == SSL_ERROR_WANT_READ) {
286 pfd.events = POLLIN;
287 } else if (ssl_error == SSL_ERROR_WANT_WRITE) {
288 pfd.events = POLLOUT;
289 } else {
290 return -1;
291 }
292 } else {
293 pfd.events = POLLOUT;
294 }
295 int ret = poll(&pfd, 1, timeout_ms);
296 if (ret <= 0) return -1;
297 if (pfd.revents & (POLLERR | POLLHUP | POLLNVAL)) return -1;
298 return 0;
299}

301/* Poll until frame bytes can be read or the connection idle deadline
302 * expires. Skips the poll when decrypted bytes are already buffered.
303 * Returns 0 when readable, -1 on idle timeout or socket error. */
304static int h2_wait_readable(h2_conn *hc) {
305 while (hc->conn->ssl ? (SSL_pending(hc->conn->ssl) <= 0) : true) {
306 uint64_t idle_ms = (uint64_t)h2_idle_timeout_ms();
307 uint64_t now = h2_now_ms();
308 if (now - hc->last_activity >= idle_ms) return -1;
309 uint64_t idle_left = idle_ms - (now - hc->last_activity);
310 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
311 if (idle_left < (uint64_t)wait_ms) wait_ms = (int)idle_left;
312 struct pollfd pfd = { .fd = hc->conn->fd, .events = POLLIN };
313 int pret = poll(&pfd, 1, wait_ms);
314 if (pret < 0) return -1;
315 if (pret > 0) return 0;
316 }
317 return 0;
318}

320/* Read exactly len bytes, tolerating SSL_ERROR_WANT_READ/WANT_WRITE (and
321 * plain EAGAIN) on the non-blocking sockets the pool hands us. Bounded by
322 * the connection idle deadline so a stalled peer cannot pin a worker.
323 * Returns 0 on success, -1 on EOF, timeout, or socket error. */
324static int h2_read_full(h2_conn *hc, void *buf, size_t len) {
325 unsigned char *p = (unsigned char *)buf;
326 size_t got = 0;
327 while (got < len) {
328 int n = h2_read(hc->conn, p + got, (int)(len - got));
329 if (n > 0) {
330 got += (size_t)n;
331 hc->last_activity = h2_now_ms();
332 continue;
333 }
334 if (n < 0) {
335 int retryable = 0;
336 int ssl_err = 0;
337 if (hc->conn->ssl) {
338 ssl_err = SSL_get_error(hc->conn->ssl, n);
339 retryable = (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE);
340 } else {
341 retryable = (errno == EAGAIN || errno == EWOULDBLOCK || errno == EINTR);
342 }
343 if (retryable) {
344 uint64_t idle_ms = (uint64_t)h2_idle_timeout_ms();
345 uint64_t now = h2_now_ms();
346 if (now - hc->last_activity >= idle_ms) return -1;
347 uint64_t left = idle_ms - (now - hc->last_activity);
348 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
349 if (left < (uint64_t)wait_ms) wait_ms = (int)left;
350 if (hc->conn->ssl) {
351 if (h2_wait_socket(hc->conn, ssl_err, wait_ms) != 0) return -1;
352 } else {
353 struct pollfd pfd = { .fd = hc->conn->fd, .events = POLLIN };
354 if (poll(&pfd, 1, wait_ms) <= 0) return -1;
355 }
356 continue;
357 }
358 }
359 return -1; /* EOF (0) or a real read error */
360 }
361 return 0;
362}

364static int h2_write_all(cwist_https_connection *conn, const void *buf, size_t len) {
365 const unsigned char *p = (const unsigned char *)buf;
366 int retries = 0;
367 /* 100ms polls; keep the total wait under the connection idle deadline. */
368 int max_retries = h2_idle_timeout_ms() / 100;
369 if (max_retries > 1000) max_retries = 1000;
370 if (max_retries < 1) max_retries = 1;
371 while (len > 0) {
372 int n = h2_write(conn, p, (int)len);
373 if (n <= 0) {
374 if (n < 0) {
375 if (conn->ssl) {
376 int err = SSL_get_error(conn->ssl, n);
377 if (err == SSL_ERROR_WANT_WRITE || err == SSL_ERROR_WANT_READ) {
378 if (++retries > max_retries) return -1; /* ~overall timeout guard */
379 if (h2_wait_socket(conn, err, 100) != 0) return -1;
380 continue;
381 }
382 } else {
383 if (errno == EAGAIN || errno == EWOULDBLOCK || errno == EINTR) {
384 if (++retries > max_retries) return -1;
385 if (h2_wait_socket(conn, 0, 100) != 0) return -1;
386 continue;
387 }
388 }
389 }
390 return -1;
391 }
392 retries = 0;
393 p += n;
394 len -= (size_t)n;
395 }
396 return 0;
397}

470int h2_decode_integer(const unsigned char *buf,
471 size_t len,
472 size_t *pos,
473 uint8_t prefix_bits,
474 uint32_t *value) {
475 if (*pos >= len || prefix_bits == 0 || prefix_bits > 8) return -1;
476 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
477 uint32_t n = buf[*pos] & mask;
478 (*pos)++;
479 if (n < mask) {
480 *value = n;
481 return 0;
482 }
483
484 uint32_t m = 0;
485 while (*pos < len) {
486 uint8_t b = buf[(*pos)++];
487 if (m > 28) return -1;
488 n += (uint32_t)(b & 0x7f) << m;
489 if ((b & 0x80) == 0) {
490 *value = n;
491 return 0;
492 }
493 m += 7;
494 }
495 return -1;
496}

834char *h2_huffman_decode(const unsigned char *src, size_t src_len, size_t *out_len) {
835 pthread_once(&h2_huffman_once, h2_huffman_init);
836 if (!h2_huffman_root) return NULL;
837 size_t cap = src_len * 2 + 1;
838 if (cap < 16) cap = 16;
839 char *out = (char *)cwist_alloc(cap);
840 if (!out) return NULL;
841 size_t out_pos = 0;
842
843 uint32_t bit_buf = 0;
844 int bit_cnt = 0;
845 size_t src_idx = 0;
846 h2_huffman_node *node = h2_huffman_root;
847
848 while (src_idx < src_len || bit_cnt > 0) {
849 /* Keep the bit buffer filled with up to 24 bits. */
850 while (bit_cnt <= 24 && src_idx < src_len) {
851 bit_buf = (bit_buf << 8) | src[src_idx++];
852 bit_cnt += 8;
853 }
854
855 /* Speed-up: If we are at the root, try decoding 8 bits using the LUT. */
856 if (node == h2_huffman_root && bit_cnt >= 8) {
857 uint8_t index = (uint8_t)(bit_buf >> (bit_cnt - 8));
858 h2_huffman_lut_entry entry = h2_huffman_root_lut[index];
859 if (entry.symbol != -1) {
860 if (entry.symbol == 256) {
861 cwist_free(out);
862 return NULL;
863 }
864 if (out_pos + 1 >= cap) {
865 cap *= 2;
866 char *tmp = (char *)cwist_realloc(out, cap);
867 if (!tmp) { cwist_free(out); return NULL; }
868 out = tmp;
869 }
870 out[out_pos++] = (char)entry.symbol;
871 bit_cnt -= entry.bits;
872 continue;
873 }
874 }
875
876 /* Fallback: bit-by-bit traversal. */
877 if (bit_cnt > 0) {
878 int bit = (bit_buf >> (bit_cnt - 1)) & 1;
879 bit_cnt--;
880 node = node->child[bit];
881 if (!node) { cwist_free(out); return NULL; }
882 if (node->is_terminal) {
883 if (node->symbol == 256) {
884 cwist_free(out);
885 return NULL;
886 }
887 if (out_pos + 1 >= cap) {
888 cap *= 2;
889 char *tmp = (char *)cwist_realloc(out, cap);
890 if (!tmp) { cwist_free(out); return NULL; }
891 out = tmp;
892 }
893 out[out_pos++] = (char)node->symbol;
894 node = h2_huffman_root;
895 }
896 } else {
897 break;
898 }
899 }
900
901 if (node != h2_huffman_root) {
902 h2_huffman_node *check = node;
903 while (check && !check->is_terminal) {
904 check = check->child[1];
905 }
906 if (!check || check->symbol != 256) {
907 cwist_free(out);
908 return NULL;
909 }
910 }
911
912 out[out_pos] = '\0';
913 if (out_len) *out_len = out_pos;
914 return out;
915}

983 char *name = NULL;
984 char *value = NULL;

990 if (entry && entry->name) h2_apply_header(req, entry->name, entry->value);
991 continue;
992 }
993

995 uint32_t new_size = 0;
996 if (h2_decode_integer(payload, len, &pos, 5, &new_size) != 0) break;

1001 if (h2_decode_integer(payload, len, &pos, 6, &name_index) != 0) break;
1002 } else if ((b & 0xf0) == 0x00) {
1003 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0) break;
1004 } else if ((b & 0xf0) == 0x10) {
1005 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0) break;
1006 } else {

1008 }

1010 if (name_index > 0) {

1015 continue;
1016 }
1017 name = strdup(entry->name);
1018 } else {
1019 name = h2_decode_string(payload, len, &pos);

1021 value = h2_decode_string(payload, len, &pos);
1022 if (!name || !value) {
1023 cwist_free(name);
1024 cwist_free(value);
1025 break;
1026 }
1027 h2_apply_header(req, name, value);
1028 free(name);
1029 free(value);
1030 }

1035static size_t h2_encode_integer(unsigned char *dst, size_t dst_cap, uint32_t value, uint8_t prefix_bits) {
1036 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
1037 unsigned char first = dst[0] & ~mask;
1038 if (value < mask) {
1039 dst[0] = first | (uint8_t)value;
1040 return 1;
1041 }
1042 dst[0] = first | mask;
1043 value -= mask;
1044 size_t i = 1;
1045 while (value >= 128) {
1046 if (i >= dst_cap) return 0;
1047 dst[i++] = (unsigned char)((value & 0x7F) | 0x80);
1048 value >>= 7;
1049 }
1050 if (i >= dst_cap) return 0;
1051 dst[i++] = (unsigned char)value;
1052 return i;
1053}

1060 memcpy(dst + n, str, len);
1061 return n + len;

1208/* Send an encoded header block as HEADERS, splitting into CONTINUATION
1209 * frames when it exceeds the peer's advertised max frame size. base_flags
1210 * (e.g. END_STREAM) applies to the HEADERS frame only. */
1211static int h2_send_header_block(cwist_https_connection *conn, uint32_t stream_id,
1212 const unsigned char *block, size_t block_len,
1213 uint32_t max_frame, uint8_t base_flags) {
1214 size_t sent = 0;
1215 uint8_t type = CWIST_HTTP2_FRAME_HEADERS;
1216 uint8_t flags = base_flags;
1217 while (true) {
1218 size_t chunk = block_len - sent;
1219 if (chunk > max_frame) chunk = max_frame;
1220 uint8_t frame_flags = flags;
1221 if (sent + chunk == block_len) frame_flags |= CWIST_HTTP2_FLAG_END_HEADERS;
1222 if (h2_write_frame(conn, type, frame_flags, stream_id,
1223 block + sent, (uint32_t)chunk) != 0) {
1224 return -1;
1225 }
1226 sent += chunk;
1227 if (sent == block_len) return 0;
1228 type = CWIST_HTTP2_FRAME_CONTINUATION;
1229 flags = 0; /* END_STREAM is illegal on CONTINUATION */
1230 }
1231}
1232
1233/* Encode response headers into a heap buffer that doubles until the block
1234 * fits (up to 1 MiB). The old fixed 8 KiB stack buffer made any response
1235 * with a large header set (big cookies, JWTs) abort the whole connection,
1236 * which browsers surface as ERR_INVALID_RESPONSE. */
1237static unsigned char *h2_encode_response_block(cwist_http_response *res, size_t *out_len) {
1238 size_t cap = 16384;
1239 unsigned char *buf = NULL;
1240 while (cap <= (1u << 20)) {
1241 unsigned char *nb = (unsigned char *)cwist_realloc(buf, cap);
1242 if (!nb) {
1243 cwist_free(buf);
1244 return NULL;
1245 }
1246 buf = nb;
1247 size_t n = h2_encode_response_headers(res, buf, cap);
1248 if (n > 0) {
1249 *out_len = n;
1250 return buf;
1251 }
1252 cap *= 2;
1253 }
1254 cwist_free(buf);
1255 return NULL;
1256}

1264static int h2_send_seq_data_frames(cwist_https_connection *conn,
1265 uint32_t stream_id,
1266 const uint8_t *body_data,
1267 size_t body_len,
1268 uint32_t max_frame_size,
1269 uint8_t *send_xor) {
1270 uint16_t chunk_payload = CWIST_HTTP2_SEQ_CHUNK_PAYLOAD(max_frame_size);
1271 if (chunk_payload == 0) return -1;
1272
1273 cwist_seq_message_t msg;
1274 if (!cwist_seq_split(body_data, body_len, chunk_payload, &msg)) return -1;
1275
1276 for (size_t i = 0; i < msg.count; i++) {
1277 uint8_t flags = (i + 1 == msg.count) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
1278 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id,
1279 msg.chunks[i], (uint32_t)msg.chunk_lens[i]) != 0) {
1280 cwist_seq_message_free(&msg);
1281 return -1;
1282 }
1283 if (send_xor) *send_xor ^= h2_xor_bytes(msg.chunks[i], msg.chunk_lens[i]);
1284 }
1285
1286 cwist_seq_message_free(&msg);
1287 return 0;
1288}

1290static int h2_send_response_raw(cwist_https_connection *conn, uint32_t stream_id,
1291 cwist_http_response *res, uint32_t max_frame_size) {
1292 size_t block_len = 0;
1293 unsigned char *block = h2_encode_response_block(res, &block_len);
1294 if (!block) return -1;
1295
1296 size_t body_len = 0;
1297 const unsigned char *body_data = NULL;
1298 if (res->use_file_stream) {
1299 body_len = res->file_stream_len;
1300 } else if (res->is_ptr_body) {
1301 body_len = res->ptr_body_len;
1302 body_data = (const unsigned char *)res->ptr_body;
1303 } else if (res->body) {
1304 body_len = res->body->size;
1305 body_data = (const unsigned char *)res->body->data;
1306 }
1307
1308 uint8_t header_flags = 0;
1309 if (body_len == 0 || h2_status_forbids_body(res->status_code)) {
1310 body_len = 0;
1311 header_flags |= CWIST_HTTP2_FLAG_END_STREAM;
1312 }
1313
1314 int rc = h2_send_header_block(conn, stream_id, block, block_len, max_frame_size, header_flags);

1317
1318 if (body_len == 0) return 0;
1319
1320 if (res->use_file_stream && res->file_stream_fd >= 0) {
1321 /* Load file contents and sequence them. Server push rarely streams
1322 * huge files, so a single read is acceptable here. */

1328 cwist_free(file_buf);
1329 return rc;
1330 }
1331
1332 return h2_send_seq_data_frames(conn, stream_id, body_data, body_len, max_frame_size, NULL);
1333}

1336 /* h2_read_full tolerates WANT_READ on the non-blocking TLS sockets;
1337 * 0/EOF and real errors stay fatal. */
1338 if (h2_read_full(hc, buf, (size_t)len) != 0) return -1;
1339 return len;

1342/* Returns 0 normally, -1 when the peer vanished mid-frame (EOF/error). */
1343static int h2_process_incoming_frames_nonblocking(h2_conn *hc, h2_stream *s) {
1344 struct pollfd pfd;
1345 pfd.fd = hc->conn->fd;
1346 pfd.events = POLLIN;
1347
1348 while (poll(&pfd, 1, 0) > 0 && (pfd.revents & POLLIN)) {
1349 unsigned char hdr[9];
1350 int n = h2_read_all(hc, hdr, 9);
1351 if (n != 9) return -1;
1352
1353 uint32_t len = ((uint32_t)hdr[0] << 16) | ((uint32_t)hdr[1] << 8) | hdr[2];
1354 uint8_t type = hdr[3];
1355 uint8_t flags = hdr[4];
1356 uint32_t stream_id = (((uint32_t)hdr[5] & 0x7f) << 24) |
1357 ((uint32_t)hdr[6] << 16) |
1358 ((uint32_t)hdr[7] << 8) |
1359 hdr[8];
1360
1361 unsigned char *payload = NULL;
1362 if (len > 0) {
1363 payload = (unsigned char *)cwist_alloc(len);
1364 if (!payload) return -1;
1365 int r = h2_read_all(hc, payload, len);
1366 if (r != (int)len) {
1367 cwist_free(payload);
1368 return -1;
1369 }
1370 }
1371
1372 hc->last_activity = h2_now_ms();
1373
1374 if (type == 0x08) { // WINDOW_UPDATE
1375 if (len == 4) {
1376 int32_t increment = (((int32_t)payload[0] & 0x7f) << 24) |
1377 ((int32_t)payload[1] << 16) |
1378 ((int32_t)payload[2] << 8) |
1379 (int32_t)payload[3];
1380 if (increment > 0) {
1381 if (stream_id == 0) {
1382 hc->conn_send_window += increment;
1383 } else if (s && stream_id == s->stream_id) {
1384 s->send_window += increment;
1385 } else {
1386 h2_stream *other = h2_stream_find(hc, stream_id);
1387 if (other) other->send_window += increment;
1388 }
1389 }
1390 }
1391 } else if (type == 0x06) { // PING
1392 if ((flags & 0x01) == 0 && len == 8) { // ACK flag is 0x01
1393 h2_write_frame(hc->conn, 0x06, 0x01, 0, payload, 8);
1394 }
1395 } else if (type == 0x03) { // RST_STREAM
1396 if (s && stream_id == s->stream_id) {
1397 s->send_window = -999; // abort code
1398 }
1399 }
1400 cwist_free(payload);
1401 }
1402 return 0;
1403}

1408static uint16_t h2_seq_chunk_payload(h2_conn *hc, h2_stream *s, uint32_t max_frame_size) {
1409 uint16_t max_payload = CWIST_HTTP2_SEQ_CHUNK_PAYLOAD(max_frame_size);
1410 int32_t eff_window = hc->conn_send_window;
1411 if (s && s->send_window < eff_window) eff_window = s->send_window;
1412 if (eff_window <= (int32_t)CWIST_SEQ_HEADER_SIZE) return 0;
1413 uint32_t allowed_payload = (uint32_t)(eff_window - (int32_t)CWIST_SEQ_HEADER_SIZE);
1414 return (allowed_payload < (uint32_t)max_payload) ? (uint16_t)allowed_payload : max_payload;
1415}

1420static int h2_send_seq_file_body(h2_conn *hc, h2_stream *s, uint32_t stream_id,
1421 int fd, off_t offset, size_t body_len,
1422 uint32_t max_frame_size) {
1423 uint16_t chunk_payload = h2_seq_chunk_payload(hc, s, max_frame_size);
1424 if (chunk_payload == 0 || body_len == 0) return -1;
1425
1426 uint32_t total_chunks = (uint32_t)((body_len + chunk_payload - 1) / chunk_payload);
1427 size_t remaining = body_len;
1428 off_t cur_offset = offset;
1429
1430 for (uint32_t i = 0; i < total_chunks && remaining > 0; i++) {
1431 while (hc->conn_send_window <= 0 || (s && s->send_window <= 0)) {
1432 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) return -1;
1433 if (s && s->send_window == -999) return -1;
1434 if (hc->conn_send_window <= 0 || (s && s->send_window <= 0)) {
1435 /* Give up if no frame arrived within the idle deadline. */
1436 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;
1437 struct timespec ts = {0, 2000000};
1438 nanosleep(&ts, NULL);
1439 }
1440 }
1441
1442 size_t plen = remaining > chunk_payload ? chunk_payload : remaining;
1443 size_t chunk_len = CWIST_SEQ_HEADER_SIZE + plen;
1444 if ((int32_t)chunk_len > hc->conn_send_window ||
1445 (s && (int32_t)chunk_len > s->send_window)) {
1446 /* Window too small for a full sequenced chunk; wait for update. */
1447 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) return -1;
1448 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;
1449 struct timespec ts = {0, 2000000};
1450 nanosleep(&ts, NULL);
1451 i--;
1452 continue;
1453 }
1454
1455 unsigned char *chunk = (unsigned char *)cwist_alloc(chunk_len);
1456 if (!chunk) return -1;
1457 ssize_t r = pread(fd, chunk + CWIST_SEQ_HEADER_SIZE, plen, cur_offset);
1458 if (r <= 0 || (size_t)r != plen) { cwist_free(chunk); return -1; }
1459 cwist_seq_chunk_build_header(chunk, i + 1, (uint16_t)total_chunks,
1460 (uint16_t)plen, chunk_payload);
1461 uint8_t flags = (i + 1 == total_chunks) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
1462 if (h2_write_frame(hc->conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id,
1463 chunk, (uint32_t)chunk_len) != 0) {
1464 cwist_free(chunk); return -1;
1465 }
1466 if (s) s->send_xor ^= h2_xor_bytes(chunk, chunk_len);
1467 cwist_free(chunk);
1468 hc->conn_send_window -= (int32_t)chunk_len;
1469 if (s) s->send_window -= (int32_t)chunk_len;
1470 cur_offset += (off_t)plen;
1471 remaining -= plen;
1472 }
1473 return 0;
1474}

1491 if (s && s->send_window == -999) return -1;

1495 struct timespec ts = {0, 2000000};

1497 }

1500 size_t chunk_len = msg.chunk_lens[i];
1501 if ((int32_t)chunk_len > hc->conn_send_window ||
1502 (s && (int32_t)chunk_len > s->send_window)) {
1503 /* Window too small for a full sequenced chunk; wait for update. */

1509 continue;

1515 cwist_seq_message_free(&msg);
1516 return -1;

1519 hc->conn_send_window -= (int32_t)chunk_len;
1520 if (s) s->send_window -= (int32_t)chunk_len;
1521 }
1522
1523 cwist_seq_message_free(&msg);
1524 return 0;
1525}

1527static int h2_send_response_hc(h2_conn *hc, uint32_t stream_id, cwist_http_response *res) {
1528 h2_stream *s = h2_stream_find(hc, stream_id);
1529
1530 size_t block_len = 0;
1531 unsigned char *block = h2_encode_response_block(res, &block_len);
1532 if (!block) return -1;
1533
1534 size_t body_len = 0;
1535 const unsigned char *body_data = NULL;
1536 if (res->use_file_stream) {
1537 body_len = res->file_stream_len;
1538 } else if (res->is_ptr_body) {
1539 body_len = res->ptr_body_len;
1540 body_data = (const unsigned char *)res->ptr_body;
1541 } else if (res->body) {
1542 body_len = res->body->size;
1543 body_data = (const unsigned char *)res->body->data;
1544 }
1545
1546 /* HEAD responses and 1xx/204/304 statuses carry no content. The
1547 * content-length header (already encoded from the real body length for
1548 * HEAD; suppressed by the encoder for bodyless statuses) stays, but no
1549 * DATA frames may follow. */
1550 bool no_content = h2_status_forbids_body(res->status_code) ||
1551 (s && s->req && s->req->method == CWIST_HTTP_HEAD);
1552
1553 uint8_t header_flags = 0;
1554 if (body_len == 0 || no_content) {
1555 body_len = 0;
1556 header_flags |= CWIST_HTTP2_FLAG_END_STREAM;
1557 }
1558
1559 uint32_t max_frame = hc->peer_max_frame_size;
1560 if (max_frame == 0) max_frame = CWIST_HTTP2_MAX_FRAME_SIZE;
1561
1562 int rc = h2_send_header_block(hc->conn, stream_id, block, block_len, max_frame, header_flags);
1563 cwist_free(block);
1564 if (rc != 0) return -1;
1565
1566 if (body_len == 0) return 0;
1567
1568 if (hc->sequenced_data) {
1569 if (res->use_file_stream && res->file_stream_fd >= 0) {
1570 return h2_send_seq_file_body(hc, s, stream_id, res->file_stream_fd,
1571 res->file_stream_offset, body_len, max_frame);
1572 }
1573 return h2_send_seq_memory_body(hc, s, stream_id, body_data, body_len, max_frame);
1574 }
1575
1576 /* Fallback: standard HTTP/2 DATA frames without sequence headers. */
1577 if (res->use_file_stream && res->file_stream_fd >= 0) {
1578 off_t offset = res->file_stream_offset;
1579 size_t remaining = res->file_stream_len;
1580 while (remaining > 0) {
1581 while (hc->conn_send_window <= 0 || (s && s->send_window <= 0)) {
1582 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) return -1;
1583 if (s && s->send_window == -999) return -1;
1584 if (hc->conn_send_window <= 0 || (s && s->send_window <= 0)) {
1585 /* Give up if no frame arrived within the idle deadline. */
1586 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;
1587 struct timespec ts = {0, 2000000};
1588 nanosleep(&ts, NULL);
1589 }
1590 }
1591
1592 uint32_t chunk = (uint32_t)(remaining > max_frame ? max_frame : remaining);
1593 uint32_t allowed = chunk;
1594 if ((int32_t)allowed > hc->conn_send_window) allowed = (uint32_t)hc->conn_send_window;
1595 if (s && (int32_t)allowed > s->send_window) allowed = (uint32_t)s->send_window;
1596
1597 unsigned char *chunk_buf = (unsigned char *)cwist_alloc(allowed);
1598 if (!chunk_buf) return -1;
1599 ssize_t r = pread(res->file_stream_fd, chunk_buf, allowed, offset);
1600 if (r <= 0) { free(chunk_buf); return -1; }
1601 uint8_t flags = (remaining == (size_t)r) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
1602 if (h2_write_frame(hc->conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id, chunk_buf, (uint32_t)r) != 0) {
1603 free(chunk_buf); return -1;
1604 }
1605 if (s) s->send_xor ^= h2_xor_bytes(chunk_buf, (size_t)r);
1606 free(chunk_buf);
1607 hc->conn_send_window -= (int32_t)r;
1608 if (s) s->send_window -= (int32_t)r;
1609 offset += r;
1610 remaining -= (size_t)r;
1611 }
1612 } else {
1613 size_t sent = 0;
1614 while (sent < body_len) {
1615 while (hc->conn_send_window <= 0 || (s && s->send_window <= 0)) {
1616 if (h2_process_incoming_frames_nonblocking(hc, s) != 0) return -1;
1617 if (s && s->send_window == -999) return -1;
1618 if (hc->conn_send_window <= 0 || (s && s->send_window <= 0)) {
1619 /* Give up if no frame arrived within the idle deadline. */
1620 if (h2_now_ms() - hc->last_activity >= (uint64_t)h2_idle_timeout_ms()) return -1;
1621 struct timespec ts = {0, 2000000};
1622 nanosleep(&ts, NULL);
1623 }
1624 }
1625
1626 size_t remaining = body_len - sent;
1627 uint32_t chunk = (uint32_t)(remaining > max_frame ? max_frame : remaining);
1628 uint32_t allowed = chunk;
1629 if ((int32_t)allowed > hc->conn_send_window) allowed = (uint32_t)hc->conn_send_window;
1630 if (s && (int32_t)allowed > s->send_window) allowed = (uint32_t)s->send_window;
1631
1632 uint8_t flags = (sent + allowed == body_len) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
1633 if (h2_write_frame(hc->conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id,
1634 body_data + sent, allowed) != 0) {
1635 return -1;
1636 }
1637 if (s) s->send_xor ^= h2_xor_bytes(body_data + sent, allowed);
1638 hc->conn_send_window -= (int32_t)allowed;
1639 if (s) s->send_window -= (int32_t)allowed;
1640 sent += allowed;
1641 }
1642 }
1643 return 0;
1644}

1648static int h2_begin_headers(h2_conn *hc, uint32_t stream_id,
1649 const unsigned char *payload, size_t len,
1650 bool end_headers, bool end_stream) {
1651 if (hc->expecting_continuation) {
1652 return -1; /* PROTOCOL_ERROR: HEADERS while expecting CONTINUATION */
1653 }
1654
1655 size_t block_offset = 0;
1656 size_t block_len = len;
1657
1658 if (end_headers) {
1659 h2_stream *s = h2_stream_find(hc, stream_id);
1660 if (!s) {
1661 s = h2_stream_create(hc, stream_id);
1662 if (!s) return -1;
1663 s->req = cwist_http_request_create();
1664 if (!s->req) {
1665 h2_stream_remove(hc, stream_id);
1666 return -1;
1667 }
1668 cwist_sstring_assign(s->req->version, "HTTP/2");
1669 s->req->stream_id = stream_id;
1670 s->req->private_data = hc->conn;
1671 }
1672 if (payload && block_len > 0) {
1673 h2_decode_header_block(s->req, payload + block_offset, block_len);
1674 }
1675 return 0;
1676 }
1677
1678 /* Need continuation */
1679 hc->expecting_continuation = true;
1680 hc->cont_stream_id = stream_id;
1681 hc->cont_end_stream = end_stream;
1682 hc->cont_len = block_len;
1683 hc->cont_cap = block_len < 1024 ? 1024 : block_len * 2;
1684 hc->cont_buf = (unsigned char *)cwist_alloc(hc->cont_cap);
1685 if (!hc->cont_buf) {
1686 hc->expecting_continuation = false;
1687 return -1;
1688 }
1689 if (payload && block_len > 0) {
1690 memcpy(hc->cont_buf, payload + block_offset, block_len);
1691 }
1692 return 0;
1693}

1695static int h2_handle_continuation(h2_conn *hc, uint32_t stream_id,
1696 const unsigned char *payload, size_t len,
1697 bool end_headers) {
1698 if (!hc->expecting_continuation || hc->cont_stream_id != stream_id) {
1699 return -1; /* PROTOCOL_ERROR */
1700 }
1701
1702 if (len > 0) {
1703 if (hc->cont_len + len > hc->cont_cap) {
1704 size_t new_cap = hc->cont_cap * 2;
1705 while (new_cap < hc->cont_len + len) new_cap *= 2;
1706 unsigned char *tmp = (unsigned char *)cwist_realloc(hc->cont_buf, new_cap);
1707 if (!tmp) return -1;
1708 hc->cont_buf = tmp;
1709 hc->cont_cap = new_cap;
1710 }
1711 memcpy(hc->cont_buf + hc->cont_len, payload, len);
1712 hc->cont_len += len;
1713 }
1714
1715 if (end_headers) {
1716 h2_stream *s = h2_stream_find(hc, hc->cont_stream_id);
1717 if (!s) {
1718 s = h2_stream_create(hc, hc->cont_stream_id);
1719 if (!s) {
1720 hc->expecting_continuation = false;
1721 return -1;
1722 }
1723 s->req = cwist_http_request_create();
1724 if (!s->req) {
1725 h2_stream_remove(hc, hc->cont_stream_id);
1726 hc->expecting_continuation = false;
1727 return -1;
1728 }
1729 cwist_sstring_assign(s->req->version, "HTTP/2");
1730 s->req->stream_id = hc->cont_stream_id;
1731 s->req->private_data = hc->conn;
1732 }
1733 if (hc->cont_buf && hc->cont_len > 0) {
1734 h2_decode_header_block(s->req, hc->cont_buf, hc->cont_len);
1735 }
1736 hc->expecting_continuation = false;
1737 hc->cont_len = 0;
1738 hc->cont_stream_id = 0;
1739 }
1740 return 0;
1741}

1743/* --- SETTINGS Parser --- */
1744
1745static int h2_handle_settings(h2_conn *hc, const unsigned char *payload, size_t len, bool ack) {
1746 if (ack) {
1747 return 0;
1748 }
1749
1750 if (len % 6 != 0) {
1751 return -1; /* FRAME_SIZE_ERROR */
1752 }
1753
1754 for (size_t i = 0; i + 6 <= len; i += 6) {
1755 uint16_t id = ((uint16_t)payload[i] << 8) | payload[i + 1];
1756 uint32_t value = ((uint32_t)payload[i + 2] << 24) |
1757 ((uint32_t)payload[i + 3] << 16) |
1758 ((uint32_t)payload[i + 4] << 8) |
1759 (uint32_t)payload[i + 5];
1760 switch (id) {
1761 case 0x1: /* SETTINGS_HEADER_TABLE_SIZE */
1762 break;
1763 case 0x2: /* SETTINGS_ENABLE_PUSH */
1764 if (value > 1) return -1; /* PROTOCOL_ERROR */
1765 break;
1766 case 0x3: /* SETTINGS_MAX_CONCURRENT_STREAMS */
1767 hc->peer_max_concurrent_streams = value;
1768 break;
1769 case 0x4: /* SETTINGS_INITIAL_WINDOW_SIZE */
1770 if (value > 2147483647U) return -1; /* FLOW_CONTROL_ERROR */
1771 {
1772 int32_t delta = (int32_t)value - (int32_t)hc->peer_initial_window_size;
1773 hc->peer_initial_window_size = value;
1774 h2_stream *s = hc->streams;
1775 while (s) {
1776 s->send_window += delta;
1777 s = s->next;
1778 }
1779 }
1780 break;
1781 case 0x5: /* SETTINGS_MAX_FRAME_SIZE */
1782 if (value < 16384 || value > 16777215) return -1; /* PROTOCOL_ERROR */
1783 hc->peer_max_frame_size = value;
1784 break;
1785 case 0x6: /* SETTINGS_MAX_HEADER_LIST_SIZE */
1786 break;
1787 }
1788 }
1789 return 0;
1790}

1792/* --- WINDOW_UPDATE Parser --- */
1793
1794static int h2_handle_window_update(h2_conn *hc, uint32_t stream_id,
1795 const unsigned char *payload, size_t len) {
1796 if (len != 4) return -1;
1797
1798 int32_t increment = (int32_t)(
1799 (((uint32_t)payload[0] & 0x7f) << 24) |
1800 ((uint32_t)payload[1] << 16) |
1801 ((uint32_t)payload[2] << 8) |
1802 (uint32_t)payload[3]
1803 );
1804 if (increment <= 0) return -1;
1805
1806 if (stream_id == 0) {
1807 hc->conn_send_window += increment;
1808 if (hc->conn_send_window > 2147483647 || hc->conn_send_window < 0) return -1;
1809 } else {
1810 h2_stream *s = h2_stream_find(hc, stream_id);
1811 if (!s) {
1812 /* RFC 7540: WINDOW_UPDATE on closed stream is a STREAM_CLOSED error
1813 * or may be ignored depending on state. We ignore for leniency. */
1814 return 0;
1815 }
1816 s->send_window += increment;
1817 if (s->send_window > 2147483647 || s->send_window < 0) return -1;
1818 }
1819 return 0;
1820}

1822/* --- Preface Verification --- */
1823
1824static int cwist_http2_verify_preface(h2_conn *hc) {
1825 char buffer[CWIST_HTTP2_CONNECTION_PREFACE_LEN];
1826 /* The preface can arrive in a later TLS record than the handshake, so a
1827 * non-blocking SSL_read may legitimately report WANT_READ here; wait it
1828 * out instead of dropping the connection. */
1829 if (h2_read_full(hc, buffer, CWIST_HTTP2_CONNECTION_PREFACE_LEN) != 0) {
1830 return -1;
1831 }
1832
1833 if (memcmp(buffer, CWIST_HTTP2_CONNECTION_PREFACE, CWIST_HTTP2_CONNECTION_PREFACE_LEN) != 0) {
1834 return -1;
1835 }
1836
1837 return 0;
1838}

1842static void h2_inject_alt_svc(cwist_https_connection *conn, cwist_http_response *res) {
1843 if (conn->http3_enabled) {

1845 socklen_t ss_len = sizeof(ss);
1846 int port = 443;

1850 } else if (ss.ss_family == AF_INET6) {

1856 cwist_http_header_add(&res->headers, "Alt-Svc", alt_svc);
1857 }

1860/* --- Connection Dispatcher --- */
1861
1862cwist_error_t cwist_http2_serve_connection(
1863 cwist_https_connection *conn,
1864 void *user_ctx,
1865 cwist_http2_request_handler_func handler
1866) {
1867 cwist_error_t result;
1868
1869 if (!conn || !handler) {
1870 result = make_error(CWIST_ERR_INT16);
1871 result.error.err_i16 = -1;
1872 return result;
1873 }
1874
1875 h2_conn hc;
1876 h2_conn_init(&hc, conn);
1877
1878 if (cwist_http2_verify_preface(&hc) != 0) {
1879 h2_conn_destroy(&hc);
1880 result = make_error(CWIST_ERR_INT16);
1881 result.error.err_i16 = -1;
1882 return result;
1883 }
1884
1885 unsigned char settings[12] = {
1886 0x00, 0x01, 0x00, 0x00, 0x10, 0x00, // SETTINGS_HEADER_TABLE_SIZE = 4096
1887 0x00, 0x04, 0x7f, 0xff, 0xff, 0xff // SETTINGS_INITIAL_WINDOW_SIZE = 2147483647 (2GB)
1888 };
1889 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_SETTINGS, 0, 0, settings, sizeof(settings)) != 0) {
1890 h2_conn_destroy(&hc);
1891 result = make_error(CWIST_ERR_INT16);
1892 result.error.err_i16 = -1;
1893 return result;
1894 }
1895
1896 /* Upgrade connection flow control window to 2GB */
1897 h2_send_window_update(&hc, 0, 2147483647 - 65535);
1898
1899 bool connected = true;
1900 bool sent_goaway = false;
1901 while (connected && atomic_load(&g_cwist_running)) {
1902 unsigned char hdr[9];
1903 /* Bound the wait for the next frame with the idle deadline so an
1904 * idle connection cannot monopolize a pool worker forever. */
1905 if (h2_wait_readable(&hc) != 0) {
1906 if (!sent_goaway &&
1907 h2_now_ms() - hc.last_activity >= (uint64_t)h2_idle_timeout_ms()) {
1908 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_NO_ERROR);
1909 sent_goaway = true;
1910 }
1911 connected = false;
1912 break;
1913 }
1914 /* Frame bytes already in flight may still straddle TLS records or
1915 * TCP segments; h2_read_full waits out WANT_READ instead of
1916 * treating it as a dropped connection. */
1917 if (h2_read_full(&hc, hdr, 9) != 0) { connected = false; break; }
1918
1919 uint32_t len = ((uint32_t)hdr[0] << 16) | ((uint32_t)hdr[1] << 8) | hdr[2];
1920 uint8_t type = hdr[3];
1921 uint8_t flags = hdr[4];
1922 uint32_t stream_id = (((uint32_t)hdr[5] & 0x7f) << 24) |
1923 ((uint32_t)hdr[6] << 16) |
1924 ((uint32_t)hdr[7] << 8) |
1925 hdr[8];
1926
1927 /* Frame size validation */
1928 if (len > CWIST_HTTP2_MAX_FRAME_SIZE) {
1929 if (type == CWIST_HTTP2_FRAME_SETTINGS || type == CWIST_HTTP2_FRAME_HEADERS ||
1930 type == CWIST_HTTP2_FRAME_CONTINUATION || type == CWIST_HTTP2_FRAME_PUSH_PROMISE) {
1931 if (type == CWIST_HTTP2_FRAME_SETTINGS || len > hc.peer_max_frame_size) {
1932 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FRAME_SIZE_ERROR);
1933 connected = false;
1934 break;
1935 }
1936 }
1937 }
1938
1939 unsigned char *payload = NULL;
1940 if (len > 0) {
1941 payload = (unsigned char *)cwist_alloc(len);
1942 if (!payload) { connected = false; break; }
1943 if (h2_read_full(&hc, payload, len) != 0) connected = false;
1944 }
1945 if (!connected) { cwist_free(payload); break; }
1946
1947 hc.last_activity = h2_now_ms();
1948
1949 switch (type) {
1950 case CWIST_HTTP2_FRAME_SETTINGS: {
1951 if (h2_handle_settings(&hc, payload, len, flags & CWIST_HTTP2_FLAG_ACK) != 0) {
1952 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1953 connected = false;
1954 } else if ((flags & CWIST_HTTP2_FLAG_ACK) == 0) {
1955 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_SETTINGS, CWIST_HTTP2_FLAG_ACK, 0, NULL, 0) != 0) {
1956 connected = false;
1957 }
1958 }
1959 break;
1960 }
1961
1962 case CWIST_HTTP2_FRAME_HEADERS: {
1963 if (stream_id == 0 || (stream_id % 2) == 0) {
1964 h2_send_goaway(&hc, 0, H2_ERR_PROTOCOL_ERROR);
1965 connected = false;
1966 break;
1967 }
1968 /* Strip PADDED and PRIORITY from payload for decoding */
1969 size_t block_offset = 0;
1970 size_t block_len = len;
1971 if ((flags & CWIST_HTTP2_FLAG_PADDED) != 0 && block_len > 0) {
1972 uint8_t pad_len = payload[block_offset++];
1973 block_len--;
1974 if (pad_len <= block_len) block_len -= pad_len;
1975 }
1976 if ((flags & CWIST_HTTP2_FLAG_PRIORITY) != 0 && block_len >= 5) {
1977 block_offset += 5;
1978 block_len -= 5;
1979 }
1980 if (h2_begin_headers(&hc, stream_id,
1981 payload + block_offset, block_len,
1982 flags & CWIST_HTTP2_FLAG_END_HEADERS,
1983 flags & CWIST_HTTP2_FLAG_END_STREAM) != 0) {
1984 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
1985 connected = false;
1986 break;
1987 }
1988 hc.last_processed_stream_id = stream_id;
1989 if (flags & CWIST_HTTP2_FLAG_END_STREAM) {
1990 h2_stream *s = h2_stream_find(&hc, stream_id);
1991 if (s && s->req) {
1992 cwist_http_response *res = cwist_http_response_create();
1993 if (res) {
1994 handler(user_ctx, s->req, res);
1995 h2_inject_alt_svc(conn, res);
1996 if (h2_send_response_hc(&hc, stream_id, res) != 0) connected = false;
1997 cwist_http_response_destroy(res);
1998 }
1999 h2_stream_remove(&hc, stream_id);
2000 }
2001 }
2002 break;
2003 }
2004
2005 case CWIST_HTTP2_FRAME_CONTINUATION: {
2006 if (stream_id == 0 || !hc.expecting_continuation || hc.cont_stream_id != stream_id) {
2007 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
2008 connected = false;
2009 break;
2010 }
2011 if (h2_handle_continuation(&hc, stream_id, payload, len,
2012 flags & CWIST_HTTP2_FLAG_END_HEADERS) != 0) {
2013 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
2014 connected = false;
2015 break;
2016 }
2017 if (flags & CWIST_HTTP2_FLAG_END_STREAM) {
2018 /* END_STREAM on CONTINUATION is a protocol violation */
2019 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
2020 connected = false;
2021 break;
2022 }
2023 if ((flags & CWIST_HTTP2_FLAG_END_HEADERS) && hc.cont_end_stream) {
2024 h2_stream *s = h2_stream_find(&hc, stream_id);
2025 if (s && s->req) {
2026 cwist_http_response *res = cwist_http_response_create();
2027 if (res) {
2028 handler(user_ctx, s->req, res);
2029 h2_inject_alt_svc(conn, res);
2030 if (h2_send_response_hc(&hc, stream_id, res) != 0) connected = false;
2031 cwist_http_response_destroy(res);
2032 }
2033 h2_stream_remove(&hc, stream_id);
2034 }
2035 }
2036 break;
2037 }
2038
2039 case CWIST_HTTP2_FRAME_DATA: {
2040 if (stream_id == 0) {
2041 h2_send_goaway(&hc, 0, H2_ERR_PROTOCOL_ERROR);
2042 connected = false;
2043 break;
2044 }
2045 h2_stream *s = h2_stream_find(&hc, stream_id);
2046 if (!s) {
2047 /* DATA on closed/unknown stream: must check if it violates flow control */
2048 /* For simplicity, we treat it as a stream error by ignoring the data
2049 * but still accounting connection-level window. */
2050 hc.conn_recv_window -= (int32_t)len;
2051 if (hc.conn_recv_window < 0) {
2052 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FLOW_CONTROL_ERROR);
2053 connected = false;
2054 }
2055 break;
2056 }
2057 if (len > 0) {
2058 hc.conn_recv_window -= (int32_t)len;
2059 s->recv_window -= (int32_t)len;
2060 if (hc.conn_recv_window < 0 || s->recv_window < 0) {
2061 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FLOW_CONTROL_ERROR);
2062 connected = false;
2063 break;
2064 }
2065 }
2066 if (payload && len > 0) {
2067 if (hc.sequenced_data && len >= CWIST_SEQ_HEADER_SIZE) {
2068 cwist_seq_chunk_t chunk;
2069 if (cwist_seq_chunk_parse(payload, len, &chunk)) {
2070 if (!s->body_assembler) {
2071 s->body_assembler = cwist_seq_assembler_create_limited(
2072 CWIST_HTTP_MAX_BODY_SIZE);
2073 if (!s->body_assembler) {
2074 connected = false;
2075 break;
2076 }
2077 }
2078 if (!cwist_seq_assembler_feed(s->body_assembler, &chunk)) {
2079 /* A conflicting duplicate is corruption, not a
2080 * retransmission. Refuse the stream without
2081 * exposing a mixed body to the application. */
2082 uint8_t rst[4] = {0, 0, 0, H2_ERR_PROTOCOL_ERROR};
2083 h2_write_frame(hc.conn, CWIST_HTTP2_FRAME_RST_STREAM,
2084 0, stream_id, rst, sizeof(rst));
2085 h2_stream_remove(&hc, stream_id);
2086 break;
2087 }
2088 }
2089 /* Malformed sequence chunks are ignored (discarded). */
2090 } else if (!hc.sequenced_data) {
2091 if (s->req->body->size + len > CWIST_HTTP_MAX_BODY_SIZE) {
2092 uint8_t rst[4] = {0, 0, 0, H2_ERR_REFUSED_STREAM};
2093 h2_write_frame(hc.conn, CWIST_HTTP2_FRAME_RST_STREAM, 0, stream_id, rst, 4);
2094 h2_stream_remove(&hc, stream_id);
2095 break;
2096 }
2097 s->recv_xor ^= h2_xor_bytes(payload, len);
2098 cwist_sstring_append_len(s->req->body, (const char *)payload, len);
2099 }
2100 }
2101 if (flags & CWIST_HTTP2_FLAG_END_STREAM) {
2102 /* No assembler means no sequenced chunks ever arrived;
2103 * an empty DATA END_STREAM is a complete empty body. */
2104 bool sequence_complete = !hc.sequenced_data || !s->body_assembler;
2105 if (hc.sequenced_data && s->body_assembler) {
2106 const uint8_t *assembled = NULL;
2107 size_t assembled_len = 0;
2108 if (cwist_seq_assembler_get_data(s->body_assembler, &assembled, &assembled_len)) {
2109 if (s->req->body->size + assembled_len <= CWIST_HTTP_MAX_BODY_SIZE) {
2110 s->recv_xor ^= h2_xor_bytes(assembled, assembled_len);
2111 cwist_sstring_append_len(s->req->body, (const char *)assembled, assembled_len);
2112 sequence_complete = true;
2113 }
2114 }
2115 }
2116 if (!sequence_complete) {
2117 /* TASFA-style ARQ boundary: the assembler retains the
2118 * exact missing sequence numbers as recovery targets;
2119 * HTTP/2 represents "retry this unprocessed request"
2120 * with REFUSED_STREAM. Never call the handler with a
2121 * partial sequenced body. */
2122 uint8_t rst[4] = {0, 0, 0, H2_ERR_REFUSED_STREAM};
2123 h2_write_frame(hc.conn, CWIST_HTTP2_FRAME_RST_STREAM,
2124 0, stream_id, rst, sizeof(rst));
2125 h2_auto_window_update(&hc, s);
2126 h2_stream_remove(&hc, stream_id);
2127 break;
2128 }
2129 cwist_http_response *res = cwist_http_response_create();
2130 if (res) {
2131 handler(user_ctx, s->req, res);
2132 h2_inject_alt_svc(conn, res);
2133 if (h2_send_response_hc(&hc, stream_id, res) != 0) connected = false;
2134 cwist_http_response_destroy(res);
2135 }
2136 /* Keep the stream state alive until its final window
2137 * credit is returned; h2_auto_window_update reads it. */
2138 if (connected) h2_auto_window_update(&hc, s);
2139 h2_stream_remove(&hc, stream_id);
2140 s = NULL;
2141 }
2142 if (connected && s) {
2143 h2_auto_window_update(&hc, s);
2144 }
2145 break;
2146 }
2147
2148 case CWIST_HTTP2_FRAME_RST_STREAM: {
2149 if (stream_id == 0 || len != 4) {
2150 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_PROTOCOL_ERROR);
2151 connected = false;
2152 break;
2153 }
2154 h2_stream_remove(&hc, stream_id);
2155 break;
2156 }
2157
2158 case CWIST_HTTP2_FRAME_PING: {
2159 if (len != 8) {
2160 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FRAME_SIZE_ERROR);
2161 connected = false;
2162 break;
2163 }
2164 if ((flags & CWIST_HTTP2_FLAG_ACK) == 0) {
2165 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_PING, CWIST_HTTP2_FLAG_ACK, 0, payload, 8) != 0) {
2166 connected = false;
2167 }
2168 }
2169 break;
2170 }
2171
2172 case CWIST_HTTP2_FRAME_GOAWAY: {
2173 connected = false;
2174 break;
2175 }
2176
2177 case CWIST_HTTP2_FRAME_WINDOW_UPDATE: {
2178 if (len != 4) {
2179 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FRAME_SIZE_ERROR);
2180 connected = false;
2181 break;
2182 }
2183 if (h2_handle_window_update(&hc, stream_id, payload, len) != 0) {
2184 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_FLOW_CONTROL_ERROR);
2185 connected = false;
2186 }
2187 break;
2188 }
2189
2190 default:
2191 /* Unknown frames MUST be ignored (RFC 7540 Section 5.5) */
2192 break;
2193 }
2194
2195 cwist_free(payload);
2196 }
2197
2198 if (connected && !atomic_load(&g_cwist_running) && !sent_goaway) {
2199 h2_send_goaway(&hc, hc.last_processed_stream_id, H2_ERR_NO_ERROR);
2200 sent_goaway = true;
2201 }
2202
2203 h2_conn_destroy(&hc);
2204
2205 result = make_error(CWIST_ERR_INT16);
2206 result.error.err_i16 = 0;
2207 return result;
2208}
