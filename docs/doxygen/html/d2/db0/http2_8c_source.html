195int h2_decode_integer(const unsigned char *buf,
196 size_t len,
197 size_t *pos,
198 uint8_t prefix_bits,
199 uint32_t *value) {
200 if (*pos >= len || prefix_bits == 0 || prefix_bits > 8) return -1;
201 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
202 uint32_t n = buf[*pos] & mask;
203 (*pos)++;
204 if (n < mask) {
205 *value = n;
206 return 0;
207 }
208
209 uint32_t m = 0;
210 while (*pos < len) {
211 uint8_t b = buf[(*pos)++];
212 if (m > 28) return -1;
213 n += (uint32_t)(b & 0x7f) << m;
214 if ((b & 0x80) == 0) {
215 *value = n;
216 return 0;
217 }
218 m += 7;
219 }
220 return -1;
221}

526char *h2_huffman_decode(const unsigned char *src, size_t src_len, size_t *out_len) {
527 h2_huffman_init();
528 size_t cap = src_len * 2 + 1;
529 if (cap < 16) cap = 16;
530 char *out = (char *)malloc(cap);
531 if (!out) return NULL;
532 size_t out_pos = 0;
533
534 h2_huffman_node *node = h2_huffman_root;
535 for (size_t i = 0; i < src_len; ++i) {
536 unsigned char byte = src[i];
537 for (int b = 7; b >= 0; --b) {
538 int bit = (byte >> b) & 1;
539 node = node->child[bit];
540 if (!node) { free(out); return NULL; }
541 if (node->is_terminal) {
542 if (node->symbol == 256) {
543 /* EOS reached before end of input - error per RFC 7541 */
544 free(out);
545 return NULL;
546 }
547 if (out_pos + 1 >= cap) {
548 cap *= 2;
549 char *tmp = (char *)realloc(out, cap);
550 if (!tmp) { free(out); return NULL; }
551 out = tmp;
552 }
553 out[out_pos++] = (char)node->symbol;
554 node = h2_huffman_root;
555 }
556 }
557 }
558
559 /* If we ended mid-tree, the remaining bits must be a prefix of EOS (all 1s).
560 * Check that the current node is on the path to EOS. */
561 if (node != h2_huffman_root) {
562 /* Walk the remaining bits as 1s to see if we reach EOS */
563 h2_huffman_node *check = node;
564 while (check && !check->is_terminal) {
565 check = check->child[1];
566 }
567 if (!check || check->symbol != 256) {
568 free(out);
569 return NULL;
570 }
571 }
572
573 out[out_pos] = '\0';
574 if (out_len) *out_len = out_pos;
575 return out;
576}

578char *h2_decode_string(const unsigned char *buf, size_t len, size_t *pos) {
579 if (*pos >= len) return NULL;
580 bool huffman = (buf[*pos] & 0x80) != 0;
581 uint32_t str_len = 0;
582 if (h2_decode_integer(buf, len, pos, 7, &str_len) != 0) return NULL;
583 if (*pos + str_len > len) return NULL;
584
585 if (huffman) {
586 size_t decoded_len = 0;
587 char *out = h2_huffman_decode(buf + *pos, str_len, &decoded_len);
588 *pos += str_len;
589 return out;
590 }
591
592 char *out = (char *)malloc((size_t)str_len + 1);
593 if (!out) return NULL;
594 memcpy(out, buf + *pos, str_len);
595 out[str_len] = '\0';
596 *pos += str_len;
597 return out;
598}

650static void h2_decode_header_block(cwist_http_request *req, const unsigned char *payload, size_t len) {
651 size_t pos = 0;
652
653 while (pos < len) {
654 uint8_t b = payload[pos];
655 uint32_t name_index = 0;
656 char *name = NULL;
657 char *value = NULL;
658
659 if (b & 0x80) {
660 uint32_t index = 0;
661 if (h2_decode_integer(payload, len, &pos, 7, &index) != 0) break;
662 const cwist_http2_static_header *entry = h2_static_header(index);
663 if (entry && entry->name) h2_apply_header(req, entry->name, entry->value);
664 continue;
665 }
666
667 if ((b & 0xE0) == 0x20) {
668 uint32_t new_size = 0;
669 if (h2_decode_integer(payload, len, &pos, 5, &new_size) != 0) break;
670 continue;
671 }
672
673 if ((b & 0x40) == 0x40) {
674 if (h2_decode_integer(payload, len, &pos, 6, &name_index) != 0) break;
675 } else if ((b & 0xf0) == 0x00) {
676 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0) break;
677 } else if ((b & 0xf0) == 0x10) {
678 if (h2_decode_integer(payload, len, &pos, 4, &name_index) != 0) break;
679 } else {
680 break;
681 }
682
683 if (name_index > 0) {
684 const cwist_http2_static_header *entry = h2_static_header(name_index);
685 if (!entry || !entry->name) {
686 /* Dynamic table reference or invalid index - skip value and continue */
687 char *discard = h2_decode_string(payload, len, &pos);
688 free(discard);
689 continue;
690 }
691 name = strdup(entry->name);
692 } else {
693 name = h2_decode_string(payload, len, &pos);
694 }
695 value = h2_decode_string(payload, len, &pos);
696 if (!name || !value) {
697 free(name);
698 free(value);
699 break;
700 }
701 h2_apply_header(req, name, value);
702 free(name);
703 free(value);
704 }

712static size_t h2_encode_integer(unsigned char *dst, size_t dst_cap, uint32_t value, uint8_t prefix_bits) {
713 uint8_t mask = (uint8_t)((1u << prefix_bits) - 1u);
714 unsigned char first = dst[0] & ~mask;
715 if (value < mask) {
716 dst[0] = first | (uint8_t)value;
717 return 1;
718 }
719 dst[0] = first | mask;
720 value -= mask;
721 size_t i = 1;
722 while (value >= 128) {
723 if (i >= dst_cap) return 0;
724 dst[i++] = (unsigned char)((value & 0x7F) | 0x80);
725 value >>= 7;
726 }
727 if (i >= dst_cap) return 0;
728 dst[i++] = (unsigned char)value;
729 return i;
730}

735static size_t h2_encode_string(unsigned char *dst, size_t dst_cap, const char *str) {
736 size_t len = strlen(str);
737 dst[0] = 0x00; /* literal, no huffman */
738 size_t n = h2_encode_integer(dst, dst_cap, (uint32_t)len, 7);
739 if (n == 0 || n + len > dst_cap) return 0;
740 memcpy(dst + n, str, len);
741 return n + len;
742}

759static size_t h2_encode_response_headers(cwist_http_response *res,
760 unsigned char *dst, size_t dst_cap) {
761 size_t pos = 0;
762
763 /* :status */
764 switch (res->status_code) {
765 case 200: if (pos >= dst_cap) return 0; dst[pos++] = 0x88; break;
766 case 204: if (pos >= dst_cap) return 0; dst[pos++] = 0x89; break;
767 case 206: if (pos >= dst_cap) return 0; dst[pos++] = 0x8a; break;
768 case 304: if (pos >= dst_cap) return 0; dst[pos++] = 0x8b; break;
769 case 400: if (pos >= dst_cap) return 0; dst[pos++] = 0x8c; break;
770 case 404: if (pos >= dst_cap) return 0; dst[pos++] = 0x8d; break;
771 case 500: if (pos >= dst_cap) return 0; dst[pos++] = 0x8e; break;
772 default: {
773 char status_str[16];
774 int status_len = snprintf(status_str, sizeof(status_str), "%d", res->status_code);
775 if (pos + 1 > dst_cap) return 0;
776 dst[pos] = 0x00; /* literal without indexing, literal name */
777 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
778 if (n == 0) return 0;
779 pos += n;
780 n = h2_encode_string(dst + pos, dst_cap - pos, ":status");
781 if (n == 0) return 0;
782 pos += n;
783 n = h2_encode_string(dst + pos, dst_cap - pos, status_str);
784 if (n == 0) return 0;
785 pos += n;
786 break;
787 }
788 }
789
790 /* Auto content-length */
791 size_t body_len = 0;
792 if (res->use_file_stream) body_len = res->file_stream_len;
793 else if (res->is_ptr_body) body_len = res->ptr_body_len;
794 else if (res->body) body_len = res->body->size;
795
796 if (!headers_have_content_length(res->headers)) {
797 char cl_str[32];
798 int cl_len = snprintf(cl_str, sizeof(cl_str), "%zu", body_len);
799 int name_idx = h2_static_table_find_name("content-length");
800 if (name_idx > 0) {
801 if (pos + 1 > dst_cap) return 0;
802 dst[pos] = 0x00; /* literal without indexing, indexed name */
803 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, (uint32_t)name_idx, 4);
804 if (n == 0) return 0;
805 pos += n;
806 } else {
807 if (pos + 1 > dst_cap) return 0;
808 dst[pos] = 0x00;
809 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
810 if (n == 0) return 0;
811 pos += n;
812 n = h2_encode_string(dst + pos, dst_cap - pos, "content-length");
813 if (n == 0) return 0;
814 pos += n;
815 }
816 size_t n = h2_encode_string(dst + pos, dst_cap - pos, cl_str);
817 if (n == 0) return 0;
818 pos += n;
819 }
820
821 /* User headers */
822 cwist_http_header_node *curr = res->headers;
823 while (curr) {
824 if (!curr->key || !curr->key->data || !curr->value || !curr->value->data) {
825 curr = curr->next;
826 continue;
827 }
828 /* Skip connection-specific headers (HTTP/2 forbids these in responses) */
829 if (strcasecmp(curr->key->data, "connection") == 0 ||
830 strcasecmp(curr->key->data, "keep-alive") == 0 ||
831 strcasecmp(curr->key->data, "transfer-encoding") == 0 ||
832 strcasecmp(curr->key->data, "upgrade") == 0) {
833 curr = curr->next;
834 continue;
835 }
836
837 char lower_name[256];
838 size_t name_len = strlen(curr->key->data);
839 if (name_len >= sizeof(lower_name)) name_len = sizeof(lower_name) - 1;
840 for (size_t i = 0; i < name_len; ++i) {
841 lower_name[i] = (char)tolower((unsigned char)curr->key->data[i]);
842 }
843 lower_name[name_len] = '\0';
844
845 int name_idx = h2_static_table_find_name(lower_name);
846 if (name_idx > 0) {
847 if (pos + 1 > dst_cap) return 0;
848 dst[pos] = 0x00; /* literal without indexing, indexed name */
849 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, (uint32_t)name_idx, 4);
850 if (n == 0) return 0;
851 pos += n;
852 } else {
853 if (pos + 1 > dst_cap) return 0;
854 dst[pos] = 0x00;
855 size_t n = h2_encode_integer(dst + pos, dst_cap - pos, 0, 4);
856 if (n == 0) return 0;
857 pos += n;
858 n = h2_encode_string(dst + pos, dst_cap - pos, lower_name);
859 if (n == 0) return 0;
860 pos += n;
861 }
862 size_t n = h2_encode_string(dst + pos, dst_cap - pos, curr->value->data);
863 if (n == 0) return 0;
864 pos += n;
865
866 curr = curr->next;
867 }
868
869 return pos;
870}

875static int h2_send_response(cwist_https_connection *conn, uint32_t stream_id, cwist_http_response *res) {
876 unsigned char header_block[8192];
877 size_t block_len = h2_encode_response_headers(res, header_block, sizeof(header_block));
878 if (block_len == 0) return -1;
879
880 uint8_t header_flags = CWIST_HTTP2_FLAG_END_HEADERS;
881
882 size_t body_len = 0;
883 const unsigned char *body_data = NULL;
884 if (res->use_file_stream) {
885 body_len = res->file_stream_len;
886 } else if (res->is_ptr_body) {
887 body_len = res->ptr_body_len;
888 body_data = (const unsigned char *)res->ptr_body;
889 } else if (res->body) {
890 body_len = res->body->size;
891 body_data = (const unsigned char *)res->body->data;
892 }
893
894 if (body_len == 0) header_flags |= CWIST_HTTP2_FLAG_END_STREAM;
895
896 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_HEADERS, header_flags, stream_id,
897 header_block, (uint32_t)block_len) != 0) {
898 return -1;
899 }
900
901 if (res->use_file_stream && res->file_stream_fd >= 0) {
902 off_t offset = res->file_stream_offset;
903 size_t remaining = res->file_stream_len;
904 while (remaining > 0) {
905 uint32_t chunk = (uint32_t)(remaining > CWIST_HTTP2_MAX_FRAME_SIZE ?
906 CWIST_HTTP2_MAX_FRAME_SIZE : remaining);
907 unsigned char *chunk_buf = (unsigned char *)malloc(chunk);
908 if (!chunk_buf) return -1;
909 ssize_t r = pread(res->file_stream_fd, chunk_buf, chunk, offset);
910 if (r <= 0) { free(chunk_buf); return -1; }
911 uint8_t flags = (remaining == (size_t)r) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
912 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id, chunk_buf, (uint32_t)r) != 0) {
913 free(chunk_buf); return -1;
914 }
915 free(chunk_buf);
916 offset += r;
917 remaining -= (size_t)r;
918 }
919 } else {
920 size_t sent = 0;
921 while (sent < body_len) {
922 size_t remaining = body_len - sent;
923 uint32_t chunk = (uint32_t)(remaining > CWIST_HTTP2_MAX_FRAME_SIZE ?
924 CWIST_HTTP2_MAX_FRAME_SIZE : remaining);
925 uint8_t flags = (sent + chunk == body_len) ? CWIST_HTTP2_FLAG_END_STREAM : 0;
926 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_DATA, flags, stream_id,
927 body_data + sent, chunk) != 0) {
928 return -1;
929 }
930 sent += chunk;
931 }
932 }
933 return 0;
934}

970cwist_error_t cwist_http2_serve_connection(
971 cwist_https_connection *conn,
972 void *user_ctx,
973 cwist_http2_request_handler_func handler
974) {
975 cwist_error_t result;
976
977 /* 1. Validation */
978 if (!conn || !handler) {
979 result = make_error(CWIST_ERR_INT16);
980 result.error.err_i16 = -1;
981 return result;
982 }
983
984 /* 2. Verify HTTP/2 Connection Preface */
985 if (cwist_http2_verify_preface(conn) != 0) {
986 result = make_error(CWIST_ERR_INT16);
987 result.error.err_i16 = -1;
988 return result;
989 }
990
991 unsigned char settings[6] = {0x00, 0x01, 0x00, 0x00, 0x00, 0x00}; /* SETTINGS_HEADER_TABLE_SIZE = 0 */
992 if (h2_write_frame(conn, CWIST_HTTP2_FRAME_SETTINGS, 0, 0, settings, sizeof(settings)) != 0) {
993 result = make_error(CWIST_ERR_INT16);
994 result.error.err_i16 = -1;
995 return result;
996 }
997
998 /* 3. Main Event Loop */
999 bool connected = true;
1000 cwist_http_request *pending_req = NULL;
1001 uint32_t pending_stream_id = 0;
1002 while (connected && atomic_load(&g_cwist_running)) {
1003 unsigned char hdr[9];
1004 int n = 0;
1005 int offset = 0;
1006 // Read 9-byte frame header
1007 while (offset < 9) {
1008 n = h2_read(conn, hdr + offset, 9 - offset);
1009 if (n <= 0) { connected = false; break; }
1010 offset += n;
1011 }
1012 if (!connected) break;
1013
1014 uint32_t len = ((uint32_t)hdr[0] << 16) | ((uint32_t)hdr[1] << 8) | hdr[2];
1015 uint8_t type = hdr[3];
1016 uint32_t stream_id = (((uint32_t)hdr[5] & 0x7f) << 24) | ((uint32_t)hdr[6] << 16) | ((uint32_t)hdr[7] << 8) | hdr[8];
1017
1018 if (len > CWIST_HTTP2_MAX_FRAME_SIZE) {
1019 connected = false;
1020 break;
1021 }
1022
1023 unsigned char *payload = NULL;
1024 if (len > 0) {
1025 payload = malloc(len);
1026 int off = 0;
1027 while (off < (int)len) {
1028 int r = h2_read(conn, payload + off, len - off);
1029 if (r <= 0) { connected = false; break; }
1030 off += r;
1031 }
1032 }
1033 if (!connected) { free(payload); break; }
1034
1035 if (type == CWIST_HTTP2_FRAME_SETTINGS) {
1036 if ((hdr[4] & CWIST_HTTP2_FLAG_ACK) == 0) {
1037 h2_write_frame(conn, CWIST_HTTP2_FRAME_SETTINGS, CWIST_HTTP2_FLAG_ACK, 0, NULL, 0);
1038 }
1039 } else if (type == CWIST_HTTP2_FRAME_HEADERS && stream_id != 0) {
1040 cwist_http_request *req = cwist_http_request_create();
1041 if (!req) { free(payload); break; }
1042 cwist_sstring_assign(req->version, "HTTP/2");
1043 req->stream_id = stream_id;
1044 req->private_data = conn;
1045
1046 size_t block_offset = 0;
1047 size_t block_len = len;
1048 if ((hdr[4] & CWIST_HTTP2_FLAG_PADDED) != 0 && block_len > 0) {
1049 uint8_t pad_len = payload[block_offset++];
1050 block_len--;
1051 if (pad_len <= block_len) block_len -= pad_len;
1052 }
1053 if ((hdr[4] & CWIST_HTTP2_FLAG_PRIORITY) != 0 && block_len >= 5) {
1054 block_offset += 5;
1055 block_len -= 5;
1056 }
1057 if (payload && block_offset <= len) {
1058 h2_decode_header_block(req, payload + block_offset, block_len);
1059 }
1060
1061 if (hdr[4] & CWIST_HTTP2_FLAG_END_STREAM) {
1062 cwist_http_response *res = cwist_http_response_create();
1063 if (res) {
1064 handler(user_ctx, req, res);
1065 if (conn->http3_enabled) {
1066 struct sockaddr_storage ss;
1067 socklen_t ss_len = sizeof(ss);
1068 int port = 443;
1069 if (getsockname(conn->fd, (struct sockaddr *)&ss, &ss_len) == 0) {
1070 if (ss.ss_family == AF_INET) {
1071 port = ntohs(((struct sockaddr_in *)&ss)->sin_port);
1072 } else if (ss.ss_family == AF_INET6) {
1073 port = ntohs(((struct sockaddr_in6 *)&ss)->sin6_port);
1074 }
1075 }
1076 char alt_svc[64];
1077 snprintf(alt_svc, sizeof(alt_svc), "h3=\":%d\"; ma=86400", port);
1078 cwist_http_header_add(&res->headers, "Alt-Svc", alt_svc);
1079 }
1080 if (h2_send_response(conn, stream_id, res) != 0) connected = false;
1081 cwist_http_response_destroy(res);
1082 }
1083 cwist_http_request_destroy(req);
1084 } else {
1085 pending_req = req;
1086 pending_stream_id = stream_id;
1087 }
1088 } else if (type == CWIST_HTTP2_FRAME_DATA && stream_id != 0) {
1089 if (pending_req && pending_stream_id == stream_id) {
1090 if (payload && len > 0) {
1091 cwist_sstring_append_len(pending_req->body, (const char *)payload, len);
1092 }
1093 if (hdr[4] & CWIST_HTTP2_FLAG_END_STREAM) {
1094 cwist_http_response *res = cwist_http_response_create();
1095 if (res) {
1096 handler(user_ctx, pending_req, res);
1097 if (conn->http3_enabled) {
1098 struct sockaddr_storage ss;
1099 socklen_t ss_len = sizeof(ss);
1100 int port = 443;
1101 if (getsockname(conn->fd, (struct sockaddr *)&ss, &ss_len) == 0) {
1102 if (ss.ss_family == AF_INET) {
1103 port = ntohs(((struct sockaddr_in *)&ss)->sin_port);
1104 } else if (ss.ss_family == AF_INET6) {
1105 port = ntohs(((struct sockaddr_in6 *)&ss)->sin6_port);
1106 }
1107 }
1108 char alt_svc[64];
1109 snprintf(alt_svc, sizeof(alt_svc), "h3=\":%d\"; ma=86400", port);
1110 cwist_http_header_add(&res->headers, "Alt-Svc", alt_svc);
1111 }
1112 if (h2_send_response(conn, stream_id, res) != 0) connected = false;
1113 cwist_http_response_destroy(res);
1114 }
1115 cwist_http_request_destroy(pending_req);
1116 pending_req = NULL;
1117 pending_stream_id = 0;
1118 }
1119 }
1120 }
1121
1122 free(payload);
1123 }
1124
1125 result = make_error(CWIST_ERR_INT16);
1126 result.error.err_i16 = 0;
1127 return result;
1128}

1150int cwist_http2_push_resource(cwist_http_request *req,
1151 const char *path,
1152 const char *content_type,
1153 const unsigned char *data,
1154 size_t data_len) {
1155 if (!req || !req->private_data || !path) return -1;
1156
1157 cwist_https_connection *conn = (cwist_https_connection *)req->private_data;
1158 uint32_t original_stream_id = req->stream_id;
1159
1160 /* Find :authority from the original request headers */
1161 const char *authority = NULL;
1162 cwist_http_header_node *h = req->headers;
1163 while (h) {
1164 if (h->key && h->key->data && strcasecmp(h->key->data, "host") == 0) {
1165 authority = h->value ? h->value->data : NULL;
1166 break;
1167 }
1168 h = h->next;
1169 }
1170 if (!authority) return -1;
1171
1172 /* Allocate a new server-initiated stream ID (even number) */
1173 static _Atomic uint32_t next_server_stream = 2;
1174 uint32_t promised_stream_id = atomic_fetch_add(&next_server_stream, 2);
1175
1176 /* Encode PUSH_PROMISE pseudo-headers (:method, :scheme, :authority, :path) */
1177 unsigned char header_block[4096];
1178 size_t pos = 0;
1179
1180 /* :method = GET → static table index 2 (0x82) */
1181 if (pos >= sizeof(header_block)) return -1;
1182 header_block[pos++] = 0x82;
1183
1184 /* :scheme = https → static table index 7 (0x87) */
1185 if (pos >= sizeof(header_block)) return -1;
1186 header_block[pos++] = 0x87;
1187
1188 /* :authority = host (literal without indexing) */
1189 if (pos + 1 > sizeof(header_block)) return -1;
1190 header_block[pos] = 0x00;
1191 size_t n = h2_encode_integer(header_block + pos, sizeof(header_block) - pos, 0, 4);
1192 if (n == 0) return -1;
1193 pos += n;
1194 n = h2_encode_string(header_block + pos, sizeof(header_block) - pos, ":authority");
1195 if (n == 0) return -1;
1196 pos += n;
1197 n = h2_encode_string(header_block + pos, sizeof(header_block) - pos, authority);
1198 if (n == 0) return -1;
1199 pos += n;
1200
1201 /* :path = path (literal without indexing) */
1202 if (pos + 1 > sizeof(header_block)) return -1;
1203 header_block[pos] = 0x00;
1204 n = h2_encode_integer(header_block + pos, sizeof(header_block) - pos, 0, 4);
1205 if (n == 0) return -1;
1206 pos += n;
1207 n = h2_encode_string(header_block + pos, sizeof(header_block) - pos, ":path");
1208 if (n == 0) return -1;
1209 pos += n;
1210 n = h2_encode_string(header_block + pos, sizeof(header_block) - pos, path);
1211 if (n == 0) return -1;
1212 pos += n;
1213
1214 /* Build PUSH_PROMISE frame payload: promised_stream_id (4 bytes) + header_block */
1215 unsigned char pp_payload[4096 + 4];
1216 pp_payload[0] = (unsigned char)((promised_stream_id >> 24) & 0x7f);
1217 pp_payload[1] = (unsigned char)((promised_stream_id >> 16) & 0xff);
1218 pp_payload[2] = (unsigned char)((promised_stream_id >> 8) & 0xff);
1219 pp_payload[3] = (unsigned char)(promised_stream_id & 0xff);
1220 memcpy(pp_payload + 4, header_block, pos);
1221
1222 /* Send PUSH_PROMISE frame */
1223 if (h2_write_frame(conn, 0x05, CWIST_HTTP2_FLAG_END_HEADERS,
1224 original_stream_id, pp_payload, (uint32_t)(pos + 4)) != 0) {
1225 return -1;
1226 }
1227
1228 /* Build and send the pushed response on the promised stream */
1229 cwist_http_response *res = cwist_http_response_create();
1230 if (!res) return -1;
1231 res->status_code = CWIST_HTTP_OK;
1232 if (content_type) {
1233 cwist_http_header_add(&res->headers, "content-type", content_type);
1234 }
1235 if (data && data_len > 0) {
1236 cwist_sstring_assign_len(res->body, (const char *)data, data_len);
1237 }
1238
1239 int ret = h2_send_response(conn, promised_stream_id, res);
1240 cwist_http_response_destroy(res);
1241 return ret;
1242}
