111 /* 3. Main Event Loop */
112 bool connected = true;
113 while (connected) {
114 unsigned char hdr[9];
115 int n = 0;
116 int offset = 0;
117 // Read 9-byte frame header
118 while (offset < 9) {
119 n = h2_read(conn, hdr + offset, 9 - offset);
120 if (n <= 0) { connected = false; break; }
121 offset += n;
122 }
123 if (!connected) break;
124
125 uint32_t len = ((uint32_t)hdr[0] << 16) | ((uint32_t)hdr[1] << 8) | hdr[2];

130 if (len > 0) {

132 int off = 0;
133 while (off < (int)len) {
134 int r = h2_read(conn, payload + off, len - off);
135 if (r <= 0) { connected = false; break; }
136 off += r;
137 }
138 }
139 if (!connected) { free(payload); break; }
140
141 // Handle HEADERS frame minimally for the test requirements
142 if (type == 0x01) {
143 cwist_http_request *req = cwist_http_request_create();
144 cwist_sstring_assign(req->version, "HTTP/2");
145
146 if (len >= 3 && payload[0] == 0x82 && payload[1] == 0x87 && payload[2] == 0x84) {
147 cwist_sstring_assign(req->path, "/");
148 } else if (len >= 8 && payload[0] == 0x82 && payload[1] == 0x87 && payload[2] == 0x04) {
149 int path_len = payload[3];
150 char *path = malloc(path_len + 1);
151 memcpy(path, payload + 4, path_len);
152 path[path_len] = '\0';
153 cwist_sstring_assign(req->path, path);
154 free(path);
155 } else {
156 cwist_sstring_assign(req->path, "/"); // Fallback
157 }

160 handler(user_ctx, req, res);
161
162 // Send HEADERS Frame (End Headers flag = 0x04)
163 unsigned char h_frame[9] = {0, 0, 1, 0x01, 0x04, (stream_id>>24)&0xff, (stream_id>>16)&0xff, (stream_id>>8)&0xff, stream_id&0xff};
164 unsigned char h_payload[1] = {0x88}; // HPACK :status 200
165 h2_write(conn, h_frame, 9);
166 h2_write(conn, h_payload, 1);
167
168 // Send DATA Frame
169 size_t body_len = res->body ? res->body->size : 0;
170 const char *body_data = res->body ? res->body->data : "";
171 size_t sent = 0;
172
173 if (body_len == 0) {
174 unsigned char d_frame[9] = {0, 0, 0, 0x00, 0x01, (stream_id>>24)&0xff, (stream_id>>16)&0xff, (stream_id>>8)&0xff, stream_id&0xff};
175 h2_write(conn, d_frame, 9);
176 } else {
177 while (sent < body_len) {
178 size_t chunk = body_len - sent > 16384 ? 16384 : body_len - sent;
179 unsigned char flags = (sent + chunk == body_len) ? 0x01 : 0x00; // End Stream flag
180 unsigned char d_frame[9] = { (chunk>>16)&0xff, (chunk>>8)&0xff, chunk&0xff, 0x00, flags, (stream_id>>24)&0xff, (stream_id>>16)&0xff, (stream_id>>8)&0xff, stream_id&0xff};
181 h2_write(conn, d_frame, 9);
182 h2_write(conn, body_data + sent, chunk);
183 sent += chunk;
184 }
185 }
186
187 cwist_http_request_destroy(req);
188 cwist_http_response_destroy(res);
189 }
190
191 free(payload);
192 }
193
194 result = make_error(CWIST_ERR_INT16);
195 result.error.err_i16 = 0;
196 return result;
