42static int cwist_ssl_wait(int fd, int ssl_error, int timeout_ms) {
43 struct pollfd pfd = { .fd = fd, .events = 0 };
44 if (ssl_error == SSL_ERROR_WANT_READ) {
45 pfd.events = POLLIN;
46 } else if (ssl_error == SSL_ERROR_WANT_WRITE) {
47 pfd.events = POLLOUT;
48 } else {
49 return -1;
50 }
51
52 int ret = poll(&pfd, 1, timeout_ms);
53 if (ret <= 0) return -1;
54 if (pfd.revents & (POLLERR | POLLHUP | POLLNVAL)) return -1;
55 return 0;
56}

94static void *https_pool_worker(void *arg) {
95 (void)arg;
96 while (1) {
97 pthread_mutex_lock(&g_https_pool.mutex);
98 while (g_https_pool.count == 0 && !g_https_pool.shutdown) {
99 pthread_cond_wait(&g_https_pool.cond_not_empty, &g_https_pool.mutex);
100 }
101 if (g_https_pool.shutdown) {
102 pthread_mutex_unlock(&g_https_pool.mutex);
103 break;
104 }
105 https_pool_task_t task = g_https_pool.queue[g_https_pool.head];
106 g_https_pool.head = (g_https_pool.head + 1) % HTTPS_TASK_QUEUE_SIZE;
107 g_https_pool.count--;
108 pthread_cond_signal(&g_https_pool.cond_not_full);
109 pthread_mutex_unlock(&g_https_pool.mutex);
110
111 struct https_thread_payload *payload = cwist_alloc(sizeof(*payload));
112 if (payload) {
113 payload->client_fd = task.client_fd;
114 payload->ctx = task.ctx;
115 payload->handler = task.handler;
116 payload->user_ctx = task.user_ctx;
117 https_thread_handler(payload);
118 } else {
119 close(task.client_fd);
120 }
121 }
122 return NULL;
123}

125int https_pool_init(void) {
126 if (g_https_pool_initialized) return 0;
127 memset(&g_https_pool, 0, sizeof(g_https_pool));
128 pthread_mutex_init(&g_https_pool.mutex, NULL);
129 pthread_cond_init(&g_https_pool.cond_not_empty, NULL);
130 pthread_cond_init(&g_https_pool.cond_not_full, NULL);
131 for (int i = 0; i < get_optimal_thread_count(); i++) {
132 if (pthread_create(&g_https_pool.threads[i], NULL, https_pool_worker, NULL) != 0) {
133 return -1;

137 return 0;

140void https_pool_submit(int client_fd, cwist_https_context *ctx, void (*handler)(cwist_https_connection *, void *), void *user_ctx) {
141 pthread_mutex_lock(&g_https_pool.mutex);
142 while (g_https_pool.count >= HTTPS_TASK_QUEUE_SIZE && !g_https_pool.shutdown) {
143 pthread_cond_wait(&g_https_pool.cond_not_full, &g_https_pool.mutex);
144 }
145 if (g_https_pool.shutdown) {
146 pthread_mutex_unlock(&g_https_pool.mutex);
147 close(client_fd);
148 return;
149 }
150 g_https_pool.queue[g_https_pool.tail].client_fd = client_fd;
151 g_https_pool.queue[g_https_pool.tail].ctx = ctx;
152 g_https_pool.queue[g_https_pool.tail].handler = handler;
153 g_https_pool.queue[g_https_pool.tail].user_ctx = user_ctx;
154 g_https_pool.tail = (g_https_pool.tail + 1) % HTTPS_TASK_QUEUE_SIZE;
155 g_https_pool.count++;
156 pthread_cond_signal(&g_https_pool.cond_not_empty);
157 pthread_mutex_unlock(&g_https_pool.mutex);
158}

160void https_pool_destroy(void) {
161 if (!g_https_pool_initialized) return;
162 pthread_mutex_lock(&g_https_pool.mutex);
163 g_https_pool.shutdown = 1;
164 pthread_cond_broadcast(&g_https_pool.cond_not_empty);
165 pthread_mutex_unlock(&g_https_pool.mutex);
166 for (int i = 0; i < get_optimal_thread_count(); i++) {
167 pthread_join(g_https_pool.threads[i], NULL);
168 }
169 pthread_mutex_destroy(&g_https_pool.mutex);
170 pthread_cond_destroy(&g_https_pool.cond_not_empty);
171 pthread_cond_destroy(&g_https_pool.cond_not_full);
172
173 g_https_pool_initialized = false;
174}

216static void cwist_https_apply_base_tls_defaults(SSL_CTX *ssl_ctx) {
217 SSL_CTX_set_options(ssl_ctx, SSL_OP_NO_COMPRESSION);
218#ifdef SSL_OP_NO_RENEGOTIATION

223 /* Session resumption. TLS 1.3 tickets are stateless and already on by
224 * default; this additionally enables the server-side session cache for
225 * TLS 1.2 session IDs and sets the session-id context without which

292/* --- Shared session ticket key -------------------------------------------
293 * cwist_app_listen() forks one worker per CPU core. BoringSSL/OpenSSL
294 * generate their internal session-ticket keys lazily on first use, i.e.
295 * AFTER the fork, so every worker would mint its own key and tickets issued
296 * by one worker could never be resumed on another (SO_REUSEPORT scatters
297 * connections across workers). Instead we generate one random key at
298 * context creation (in the parent, pre-fork) and install an explicit
299 * ticket-key callback; every forked worker inherits identical key material
300 * and tickets resume regardless of which worker accepts the connection.
301 *
302 * Trade-off: the key lives for the process lifetime (no rotation), which is
303 * the standard pre-fork server model (same as a fixed nginx ticket key).
304 * ------------------------------------------------------------------------- */
305typedef struct cwist_tls_ticket_key {
306 unsigned char name[16]; /* key_name sent in the ticket */
307 unsigned char aes_key[32]; /* AES-256-CBC encryption key */
308 unsigned char hmac_key[32]; /* HMAC-SHA-256 MAC key */
309} cwist_tls_ticket_key;
310
311static int g_ticket_key_ex_data_idx = -1;
312static pthread_once_t g_ticket_key_ex_data_once = PTHREAD_ONCE_INIT;

314static void cwist_tls_ticket_key_ex_data_init(void) {
315 g_ticket_key_ex_data_idx = SSL_CTX_get_ex_new_index(0, NULL, NULL, NULL, NULL);
316}

318static int cwist_tls_ticket_key_cb(SSL *ssl, uint8_t *key_name, uint8_t *iv,
319 EVP_CIPHER_CTX *ectx, HMAC_CTX *hctx, int encrypt) {
320 SSL_CTX *ssl_ctx = ssl ? SSL_get_SSL_CTX(ssl) : NULL;
321 const cwist_tls_ticket_key *key = NULL;
322 if (ssl_ctx && g_ticket_key_ex_data_idx >= 0) {
323 key = (const cwist_tls_ticket_key *)SSL_CTX_get_ex_data(ssl_ctx, g_ticket_key_ex_data_idx);
324 }
325 if (!key) return 0;
326
327 if (encrypt) {
328 memcpy(key_name, key->name, sizeof(key->name));
329 if (RAND_bytes(iv, 16) != 1) return 0;
330 if (EVP_EncryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
331 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1) return 0;
332 return 1;
333 }
334
335 if (memcmp(key_name, key->name, sizeof(key->name)) != 0) {
336 return 0; /* Unknown key: decline the ticket, do a full handshake. */
337 }
338 if (EVP_DecryptInit_ex(ectx, EVP_aes_256_cbc(), NULL, key->aes_key, iv) != 1) return 0;
339 if (HMAC_Init_ex(hctx, key->hmac_key, sizeof(key->hmac_key), EVP_sha256(), NULL) != 1) return 0;
340 return 1;
341}

348static void cwist_tls_setup_shared_ticket_key(SSL_CTX *ssl_ctx, void **out_key) {
349 *out_key = NULL;
350 pthread_once(&g_ticket_key_ex_data_once, cwist_tls_ticket_key_ex_data_init);
351 if (g_ticket_key_ex_data_idx < 0) return;
352
353 cwist_tls_ticket_key *key = (cwist_tls_ticket_key *)cwist_alloc(sizeof(*key));
354 if (!key) return;
355 if (RAND_bytes((uint8_t *)key, sizeof(*key)) != 1) {
356 cwist_free(key);
357 return; /* Fall back to the library-internal (per-worker) keys. */
358 }
359 SSL_CTX_set_ex_data(ssl_ctx, g_ticket_key_ex_data_idx, key);
360 SSL_CTX_set_tlsext_ticket_key_cb(ssl_ctx, cwist_tls_ticket_key_cb);
361 *out_key = key;
362}

368cwist_error_t cwist_https_init_context_with_options(cwist_https_context **ctx,
369 const char *cert_path,
370 const char *key_path,
371 const cwist_https_options *options,
372 cwist_app *app) {
373 cwist_error_t err = make_error(CWIST_ERR_INT16);
374 bool enable_http3 = options && options->enable_http3;
375 bool enable_http2 = options && (options->enable_http2 || enable_http3);
376
377 if (!ctx || !cert_path || !key_path) {
378 err.error.err_i16 = -1;
379 return err;
380 }
381
382 // Initialize OpenSSL
383 SSL_load_error_strings();
384 OpenSSL_add_ssl_algorithms();
385
386 const SSL_METHOD *method = TLS_server_method();
387 SSL_CTX *ssl_ctx = SSL_CTX_new(method);
388 if (!ssl_ctx) {
389 return make_ssl_error("Unable to create SSL context");
390 }
391
392 if (SSL_CTX_set_min_proto_version(ssl_ctx, TLS1_2_VERSION) != 1) {
393 SSL_CTX_free(ssl_ctx);
394 return make_ssl_error("Unable to enforce TLS minimum version");
395 }
396 cwist_https_apply_base_tls_defaults(ssl_ctx);
397
398 if (!cwist_tls_apply_pqc_layer(app, ssl_ctx)) {
399 SSL_CTX_free(ssl_ctx);
400 return make_ssl_error("PQC layer configuration failed");
401 }
402
403 if (enable_http2) {
404 err = cwist_https_apply_http2_tls_profile(ssl_ctx);
405 if (err.errtype != CWIST_ERR_INT16 || err.error.err_i16 != 0) {
406 SSL_CTX_free(ssl_ctx);
407 return err;
408 }
409 }
410
411 // Load Cert and Key
412 if (SSL_CTX_use_certificate_chain_file(ssl_ctx, cert_path) <= 0) {
413 SSL_CTX_free(ssl_ctx);
414 return make_ssl_error("Unable to load certificate");
415 }
416
417 if (cwist_tls_autoload_intermediates(ssl_ctx) < 0) {
418 SSL_CTX_free(ssl_ctx);
419 return make_ssl_error("Unable to complete certificate chain");
420 }
421
422 if (SSL_CTX_use_PrivateKey_file(ssl_ctx, key_path, SSL_FILETYPE_PEM) <= 0) {
423 SSL_CTX_free(ssl_ctx);
424 return make_ssl_error("Unable to load private key");
425 }
426
427 // Verify key matches cert
428 if (!SSL_CTX_check_private_key(ssl_ctx)) {
429 SSL_CTX_free(ssl_ctx);
430 return make_ssl_error("Private key does not match certificate");
431 }
432
433 *ctx = (cwist_https_context*)cwist_alloc(sizeof(cwist_https_context));
434 if (!*ctx) {
435 SSL_CTX_free(ssl_ctx);
436 err.error.err_i16 = -1;
437 return err;
438 }
439 (*ctx)->ctx = ssl_ctx;
440 (*ctx)->http2_enabled = enable_http2;
441 (*ctx)->http3_enabled = enable_http3;
442 (*ctx)->ticket_key = NULL;
443
444 /* Pre-fork shared ticket key so every forked worker can resume tickets
445 * issued by any other worker. Must run before workers are forked. */
446 cwist_tls_setup_shared_ticket_key(ssl_ctx, &(*ctx)->ticket_key);
447
448 SSL_CTX_set_alpn_select_cb(ssl_ctx,
449 cwist_https_alpn_select_cb,
450 (void *)*ctx);
451
452 err.error.err_i16 = 0; // Success
453 return err;
454}

483cwist_error_t cwist_https_accept(cwist_https_context *ctx, int client_fd, cwist_https_connection **conn) {
484 cwist_error_t err = make_error(CWIST_ERR_INT16);
485
486 if (!ctx || !ctx->ctx || client_fd < 0) {
487 err.error.err_i16 = -1;
488 return err;
489 }
490
491 SSL *ssl = SSL_new(ctx->ctx);
492 if (!ssl) {
493 return make_ssl_error("Failed to create SSL structure");
494 }
495
496 SSL_set_fd(ssl, client_fd);
497
498 /* Bound the whole handshake so a client dribbling bytes cannot pin a
499 * pool worker forever. */
500 uint64_t handshake_deadline = cwist_https_now_ms() + CWIST_HTTPS_HANDSHAKE_TIMEOUT_MS;
501 int rc;
502 while ((rc = SSL_accept(ssl)) <= 0) {
503 int ssl_err = SSL_get_error(ssl, rc);
504 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
505 uint64_t now = cwist_https_now_ms();
506 if (now >= handshake_deadline) {
507 cwist_error_t err_obj = make_ssl_error("SSL handshake timed out");
508 SSL_free(ssl);
509 return err_obj;
510 }
511 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
512 uint64_t remaining = handshake_deadline - now;
513 if (remaining < (uint64_t)wait_ms) wait_ms = (int)remaining;
514 if (cwist_ssl_wait(client_fd, ssl_err, wait_ms) != 0) {
515 cwist_error_t err_obj = make_ssl_error("SSL handshake timed out or socket error");
516 SSL_free(ssl);
517 return err_obj;
518 }
519 continue;
520 }
521 cwist_error_t err_obj = make_ssl_error("SSL handshake failed");
522 SSL_free(ssl);
523 return err_obj;
524 }
525
526 *conn = (cwist_https_connection*)cwist_alloc(sizeof(cwist_https_connection));
527 if (!*conn) {
528 SSL_free(ssl);
529 err.error.err_i16 = -1;
530 return err;
531 }
532
533 (*conn)->fd = client_fd;
534 (*conn)->ssl = ssl;
535 (*conn)->read_buf = cwist_alloc(CWIST_HTTP_READ_BUFFER_SIZE);
536 if (!(*conn)->read_buf) {
537 SSL_free(ssl);
538 cwist_free(*conn);
539 *conn = NULL;
540 err.error.err_i16 = -1;
541 return err;
542 }
543 (*conn)->buf_len = 0;
544 (*conn)->read_buf[0] = '\0';
545 (*conn)->negotiated_http2 = false;
546 (*conn)->negotiated_protocol = CWIST_HTTPS_PROTOCOL_HTTP11;
547 (*conn)->http3_enabled = ctx->http3_enabled;
548
549 const unsigned char *alpn = NULL;
550 unsigned int alpn_len = 0;
551 SSL_get0_alpn_selected(ssl, &alpn, &alpn_len);
552 if (alpn && alpn_len == 2 && memcmp(alpn, "h2", 2) == 0) {
553 (*conn)->negotiated_http2 = true;
554 (*conn)->negotiated_protocol = CWIST_HTTPS_PROTOCOL_HTTP2;
555 }
556 /* h3 is QUIC-only and never negotiated over TCP TLS */
557
558 err.error.err_i16 = 0;
559 return err;
560}

594cwist_http_request *cwist_https_receive_request(cwist_https_connection *conn) {
595 if (!conn || !conn->ssl || !conn->read_buf) return NULL;
596
597 size_t total_received = conn->buf_len;
598 char *header_end = NULL;
599
600 /* Total deadline for assembling the request headers. */
601 uint64_t headers_deadline = cwist_https_now_ms() + CWIST_HTTP_HEADERS_TIMEOUT_MS;
602
603 /* Rescanning from offset 0 on every read is O(n^2) over dribbled
604 * headers; resume 3 bytes back so a terminator split across two reads
605 * is still found. */
606 size_t scan_from = 0;
607
608 while (1) {
609 header_end = strstr(conn->read_buf + scan_from, "\r\n\r\n");
610 if (header_end) break;
611 scan_from = total_received > 3 ? total_received - 3 : 0;
612
613 if (total_received >= CWIST_HTTP_READ_BUFFER_SIZE - 1) {
614 return NULL;
615 }
616
617 uint64_t now = cwist_https_now_ms();
618 if (now >= headers_deadline) {
619 return NULL;
620 }
621 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
622 uint64_t remaining = headers_deadline - now;
623 if (remaining < (uint64_t)wait_ms) wait_ms = (int)remaining;
624
625 struct pollfd pfd = { .fd = conn->fd, .events = POLLIN };
626 int pret = poll(&pfd, 1, wait_ms);
627 if (pret <= 0) {
628 return NULL;
629 }
630
631 int bytes = SSL_read(conn->ssl, conn->read_buf + total_received, (int)(CWIST_HTTP_READ_BUFFER_SIZE - 1 - total_received));
632 if (bytes <= 0) {
633 int ssl_err = SSL_get_error(conn->ssl, bytes);
634 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
635 if (cwist_ssl_wait(conn->fd, ssl_err, CWIST_HTTP_TIMEOUT_MS) != 0) return NULL;
636 continue;
637 }
638 return NULL;
639 }
640
641 total_received += (size_t)bytes;
642 conn->read_buf[total_received] = '\0';
643 }
644
645 cwist_http_request *req = cwist_http_parse_request(conn->read_buf);
646 if (!req) return NULL;
647
648 req->client_fd = conn->fd;
649
650 size_t header_len = (header_end + 4) - conn->read_buf;
651 size_t body_received = total_received - header_len;
652
653 if (req->content_length > 0) {
654 if (req->content_length > CWIST_HTTP_MAX_BODY_SIZE) {
655 cwist_http_request_destroy(req);
656 return NULL;
657 }
658
659 char *body = cwist_alloc(req->content_length + 1);
660 if (!body) {
661 cwist_http_request_destroy(req);
662 return NULL;
663 }
664
665 size_t to_copy = body_received < req->content_length ? body_received : req->content_length;
666 memcpy(body, header_end + 4, to_copy);
667 size_t current_body_len = to_copy;
668
669 /* No total cap (slow 1 GiB uploads must keep working); abort only
670 * when no bytes arrive for a cumulative idle span. Any successful
671 * read resets the idle clock. */
672 uint64_t body_idle_start = cwist_https_now_ms();
673
674 while (current_body_len < req->content_length) {
675 uint64_t now = cwist_https_now_ms();
676 if (now - body_idle_start >= CWIST_HTTP_BODY_IDLE_TIMEOUT_MS) {
677 cwist_free(body);
678 cwist_http_request_destroy(req);
679 return NULL;
680 }
681 int wait_ms = CWIST_HTTP_TIMEOUT_MS;
682 uint64_t idle_left = CWIST_HTTP_BODY_IDLE_TIMEOUT_MS - (now - body_idle_start);
683 if (idle_left < (uint64_t)wait_ms) wait_ms = (int)idle_left;
684
685 struct pollfd pfd = { .fd = conn->fd, .events = POLLIN };
686 int pret = poll(&pfd, 1, wait_ms);
687 if (pret <= 0) {
688 cwist_free(body);
689 cwist_http_request_destroy(req);
690 return NULL;
691 }
692
693 int bytes = SSL_read(conn->ssl, body + current_body_len, (int)(req->content_length - current_body_len));
694 if (bytes <= 0) {
695 int ssl_err = SSL_get_error(conn->ssl, bytes);
696 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
697 if (cwist_ssl_wait(conn->fd, ssl_err, CWIST_HTTP_TIMEOUT_MS) != 0) {
698 cwist_free(body);
699 cwist_http_request_destroy(req);
700 return NULL;
701 }
702 continue;
703 }
704 cwist_free(body);
705 cwist_http_request_destroy(req);
706 return NULL;
707 }
708 current_body_len += (size_t)bytes;
709 body_idle_start = cwist_https_now_ms();
710 }
711 body[req->content_length] = '\0';
712 /* Adopt the filled buffer: one allocation, zero copies. */
713 cwist_sstring_adopt_len(req->body, body, req->content_length);
714
715 if (body_received > req->content_length) {
716 size_t leftover_len = body_received - req->content_length;
717 memmove(conn->read_buf, header_end + 4 + req->content_length, leftover_len);
718 conn->buf_len = leftover_len;
719 } else {
720 conn->buf_len = 0;
721 }
722 } else {
723 if (body_received > 0) {
724 memmove(conn->read_buf, header_end + 4, body_received);
725 conn->buf_len = body_received;
726 } else {
727 conn->buf_len = 0;
728 }
729 }
730 conn->read_buf[conn->buf_len] = '\0';
731
732 return req;
733}

742static int cwist_ssl_write_all(cwist_https_connection *conn, const char *buf, size_t len) {
743 size_t off = 0;
744 while (off < len) {
745 /* SSL_write takes an int; cap each call well below INT_MAX. */
746 size_t left = len - off;
747 int chunk = left > 0x3fffffff ? 0x3fffffff : (int)left;
748 int sent = SSL_write(conn->ssl, buf + off, chunk);
749 if (sent <= 0) {
750 int ssl_err = SSL_get_error(conn->ssl, sent);
751 if (ssl_err == SSL_ERROR_WANT_WRITE || ssl_err == SSL_ERROR_WANT_READ) {
752 if (cwist_ssl_wait(conn->fd, ssl_err, CWIST_HTTP_TIMEOUT_MS) != 0) {
753 return -1;
754 }
755 continue;
756 }
757 return -1;
758 }
759 off += (size_t)sent;
760 }
761 return 0;
762}

776cwist_error_t cwist_https_send_response(cwist_https_connection *conn, cwist_http_response *res) {
777 cwist_error_t err = make_error(CWIST_ERR_INT16);
778
779 if (!conn || !conn->ssl || !res) {
780 err.error.err_i16 = -1;
781 return err;
782 }
783
784 // Inject Alt-Svc when HTTP/3 is enabled so clients discover the QUIC endpoint
785 if (conn->http3_enabled) {
786 struct sockaddr_storage ss;
787 socklen_t ss_len = sizeof(ss);
788 int port = 443;
789 if (getsockname(conn->fd, (struct sockaddr *)&ss, &ss_len) == 0) {
790 if (ss.ss_family == AF_INET) {
791 port = ntohs(((struct sockaddr_in *)&ss)->sin_port);
792 } else if (ss.ss_family == AF_INET6) {
793 port = ntohs(((struct sockaddr_in6 *)&ss)->sin6_port);
794 }
795 }
796 char alt_svc[64];
797 snprintf(alt_svc, sizeof(alt_svc), "h3=\":%d\"; ma=86400", port);
798 cwist_http_header_add(&res->headers, "Alt-Svc", alt_svc);
799 }
800
801 // 1. Headers onto a stack buffer, then straight onto the wire
802 char header_buf[CWIST_HTTP_MAX_HEADER_SIZE];
803 size_t header_len = cwist_http_serialize_headers(res, header_buf, sizeof(header_buf));
804 if (cwist_ssl_write_all(conn, header_buf, header_len) != 0) {
805 return make_ssl_error("SSL header write failed");
806 }
807
808 // 2. Body streamed from its origin; no intermediate serialization
809 const char *body_ptr = NULL;
810 size_t body_len = 0;
811 if (res->is_ptr_body) {
812 body_ptr = (const char *)res->ptr_body;
813 body_len = res->ptr_body_len;
814 } else if (res->body && res->body->data) {
815 body_ptr = res->body->data;
816 body_len = res->body->size;
817 }
818 if (body_ptr && body_len > 0) {
819 if (cwist_ssl_write_all(conn, body_ptr, body_len) != 0) {
820 return make_ssl_error("SSL body write failed");
821 }
822 }
823
824 // 3. File streams cannot use sendfile() through userland TLS; chunk them
825 if (res->use_file_stream && res->file_stream_fd >= 0) {
826 char fbuf[65536];
827 size_t remaining = res->file_stream_len;
828 off_t offset = res->file_stream_offset;
829 while (remaining > 0) {
830 size_t to_read = remaining < sizeof(fbuf) ? remaining : sizeof(fbuf);
831 ssize_t n = pread(res->file_stream_fd, fbuf, to_read, offset);
832 if (n < 0) {
833 if (errno == EINTR) continue;
834 return make_ssl_error("file stream read failed");
835 }
836 if (n == 0) break;
837 if (cwist_ssl_write_all(conn, fbuf, (size_t)n) != 0) {
838 return make_ssl_error("SSL file stream write failed");
839 }
840 offset += n;
841 remaining -= (size_t)n;
842 }
843 res->file_stream_offset = offset;
844 }
845
846 err.error.err_i16 = 0;
847 return err;
848}

882cwist_error_t cwist_https_server_loop(int server_fd, cwist_https_context *ctx, void (*handler)(cwist_https_connection *, void *), void *user_ctx) {
883 cwist_error_t err = make_error(CWIST_ERR_INT16);
884 if (server_fd < 0 || !ctx || !handler) {
885 err.error.err_i16 = -1;
886 return err;
887 }
888
889 if (https_pool_init() != 0) {
890 err.error.err_i16 = -1;
891 return err;
892 }
893
894 while (atomic_load(&g_cwist_running)) {
895 struct sockaddr_in addr;
896 socklen_t len = sizeof(addr);
897 int client_fd = accept(server_fd, (struct sockaddr*)&addr, &len);
898
899 if (client_fd < 0) {
900 if (errno == EINTR) continue;
901 if (errno == EBADF || errno == EINVAL) break;
902 continue;
903 }
904
905 /* Reap vanished peers within ~2 minutes instead of the ~2h kernel
906 * default, so dead connections cannot park pool workers forever. */
907 {
908 int one = 1;
909 setsockopt(client_fd, SOL_SOCKET, SO_KEEPALIVE, &one, sizeof(one));
910#ifdef TCP_KEEPIDLE
911 int keepidle = 60;
912 setsockopt(client_fd, IPPROTO_TCP, TCP_KEEPIDLE, &keepidle, sizeof(keepidle));
913#endif
914#ifdef TCP_KEEPINTVL
915 int keepintvl = 10;
916 setsockopt(client_fd, IPPROTO_TCP, TCP_KEEPINTVL, &keepintvl, sizeof(keepintvl));
917#endif
918#ifdef TCP_KEEPCNT
919 int keepcnt = 6;
920 setsockopt(client_fd, IPPROTO_TCP, TCP_KEEPCNT, &keepcnt, sizeof(keepcnt));
921#endif
922 }
923
924 https_pool_submit(client_fd, ctx, handler, user_ctx);
925 }
926
927 https_pool_destroy();
928 return err;
929}
