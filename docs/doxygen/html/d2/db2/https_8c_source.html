118 const char *cert_path,
119 const char *key_path,
120 const cwist_https_options *options) {
121 cwist_error_t err = make_error(CWIST_ERR_INT16);
122 bool enable_http2 = options && options->enable_http2;
123
124 if (!ctx || !cert_path || !key_path) {
125 err.error.err_i16 = -1;
126 return err;
127 }
128
129 // Initialize OpenSSL
130 SSL_load_error_strings();
131 OpenSSL_add_ssl_algorithms();
132
133 const SSL_METHOD *method = TLS_server_method();
134 SSL_CTX *ssl_ctx = SSL_CTX_new(method);
135 if (!ssl_ctx) {
136 return make_ssl_error("Unable to create SSL context");
137 }
138
139 if (SSL_CTX_set_min_proto_version(ssl_ctx, TLS1_2_VERSION) != 1) {
140 SSL_CTX_free(ssl_ctx);

142 }
143 cwist_https_apply_base_tls_defaults(ssl_ctx);
144
145 if (enable_http2) {
146 err = cwist_https_apply_http2_tls_profile(ssl_ctx);
147 if (err.errtype != CWIST_ERR_INT16 || err.error.err_i16 != 0) {
148 SSL_CTX_free(ssl_ctx);
149 return err;
150 }
151 }

154 cwist_https_alpn_select_cb,
155 (void *)(uintptr_t)enable_http2);
156
157 // Load Cert and Key
158 if (SSL_CTX_use_certificate_file(ssl_ctx, cert_path, SSL_FILETYPE_PEM) <= 0) {
159 SSL_CTX_free(ssl_ctx);
160 return make_ssl_error("Unable to load certificate");
161 }
162
163 if (SSL_CTX_use_PrivateKey_file(ssl_ctx, key_path, SSL_FILETYPE_PEM) <= 0) {
164 SSL_CTX_free(ssl_ctx);
165 return make_ssl_error("Unable to load private key");
166 }
167

169 if (!SSL_CTX_check_private_key(ssl_ctx)) {
170 SSL_CTX_free(ssl_ctx);
171 return make_ssl_error("Private key does not match certificate");
172 }
173
174 *ctx = (cwist_https_context*)cwist_alloc(sizeof(cwist_https_context));
175 if (!*ctx) {
176 SSL_CTX_free(ssl_ctx);
177 err.error.err_i16 = -1;
178 return err;
179 }
180 (*ctx)->ctx = ssl_ctx;
181 (*ctx)->http2_enabled = enable_http2;
182
183 err.error.err_i16 = 0; // Success
184 return err;

324

326 size_t body_received = total_received - header_len;
327
328 if (req->content_length > 0) {

330 cwist_http_request_destroy(req);
331 return NULL;
332 }
333

335 if (!body) {

338 }

341 memcpy(body, header_end + 4, to_copy);
342 size_t current_body_len = to_copy;
343
344 while (current_body_len < req->content_length) {
345 struct pollfd pfd = { .fd = conn->fd, .events = POLLIN };
346 int pret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
347 if (pret <= 0) {

350 return NULL;
351 }

354 if (bytes <= 0) {

356 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
357 continue;
358 }
359 cwist_free(body);
360 cwist_http_request_destroy(req);
361 return NULL;
362 }
363 current_body_len += (size_t)bytes;
364 }
365 body[req->content_length] = '\0';
366 cwist_sstring_assign_len(req->body, body, req->content_length);
367 cwist_free(body);
368
369 if (body_received > req->content_length) {
370 size_t leftover_len = body_received - req->content_length;
371 memmove(conn->read_buf, header_end + 4 + req->content_length, leftover_len);
372 conn->buf_len = leftover_len;
373 } else {
374 conn->buf_len = 0;
375 }
376 } else {
377 if (body_received > 0) {
378 memmove(conn->read_buf, header_end + 4, body_received);
379 conn->buf_len = body_received;
380 } else {
381 conn->buf_len = 0;
382 }
383 }
384 conn->read_buf[conn->buf_len] = '\0';

395cwist_error_t cwist_https_send_response(cwist_https_connection *conn, cwist_http_response *res) {
396 cwist_error_t err = make_error(CWIST_ERR_INT16);
397
398 if (!conn || !conn->ssl || !res) {
399 err.error.err_i16 = -1;
400 return err;
401 }
402
403 // 1. Serialize using existing HTTP logic
404 cwist_sstring *response_str = cwist_http_stringify_response(res);

412 int left = (int)response_str->size;
413 int total_sent = 0;

416
417 while (left > 0) {

419 if (sent <= 0) {

422 continue; // Retry

425 break;
426 }
427 p += sent;
428 left -= sent;
429 total_sent += sent;
430 }
431
432 cwist_sstring_destroy(response_str);

476 cwist_error_t err = make_error(CWIST_ERR_INT16);
477 if (server_fd < 0 || !ctx || !handler) {
