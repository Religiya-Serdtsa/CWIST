130 const char *cert_path,

133 cwist_error_t err = make_error(CWIST_ERR_INT16);

135 bool enable_http2 = options && (options->enable_http2 || enable_http3);
136

138 err.error.err_i16 = -1;
139 return err;
140 }

143 SSL_load_error_strings();
144 OpenSSL_add_ssl_algorithms();
145

147 SSL_CTX *ssl_ctx = SSL_CTX_new(method);
148 if (!ssl_ctx) {

163 }
164 }

168 SSL_CTX_free(ssl_ctx);

170 }
171

173 SSL_CTX_free(ssl_ctx);
174 return make_ssl_error("Unable to load private key");
175 }
176
177 // Verify key matches cert
178 if (!SSL_CTX_check_private_key(ssl_ctx)) {
179 SSL_CTX_free(ssl_ctx);
180 return make_ssl_error("Private key does not match certificate");
181 }
182
183 *ctx = (cwist_https_context*)cwist_alloc(sizeof(cwist_https_context));
184 if (!*ctx) {
185 SSL_CTX_free(ssl_ctx);
186 err.error.err_i16 = -1;
187 return err;
188 }
189 (*ctx)->ctx = ssl_ctx;
190 (*ctx)->http2_enabled = enable_http2;
191 (*ctx)->http3_enabled = enable_http3;
192
193 SSL_CTX_set_alpn_select_cb(ssl_ctx,
194 cwist_https_alpn_select_cb,
195 (void *)*ctx);
196
197 err.error.err_i16 = 0; // Success

313cwist_http_request *cwist_https_receive_request(cwist_https_connection *conn) {
314 if (!conn || !conn->ssl || !conn->read_buf) return NULL;
315
316 size_t total_received = conn->buf_len;
317 char *header_end = NULL;
318
319 while (!(header_end = strstr(conn->read_buf, "\r\n\r\n"))) {
320 if (total_received >= CWIST_HTTP_READ_BUFFER_SIZE - 1) {
321 return NULL;
322 }
323
324 struct pollfd pfd = { .fd = conn->fd, .events = POLLIN };
325 int pret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
326 if (pret <= 0) {
327 return NULL;
328 }
329
330 int bytes = SSL_read(conn->ssl, conn->read_buf + total_received, (int)(CWIST_HTTP_READ_BUFFER_SIZE - 1 - total_received));
331 if (bytes <= 0) {
332 int ssl_err = SSL_get_error(conn->ssl, bytes);
333 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
334 continue;
335 }
336 return NULL;
337 }
338
339 total_received += (size_t)bytes;
340 conn->read_buf[total_received] = '\0';
341 }
342
343 cwist_http_request *req = cwist_http_parse_request(conn->read_buf);
344 if (!req) return NULL;
345
346 req->client_fd = conn->fd;
347
348 size_t header_len = (header_end + 4) - conn->read_buf;
349 size_t body_received = total_received - header_len;
350
351 if (req->content_length > 0) {
352 if (req->content_length > CWIST_HTTP_MAX_BODY_SIZE) {
353 cwist_http_request_destroy(req);
354 return NULL;
355 }
356
357 char *body = cwist_alloc(req->content_length + 1);
358 if (!body) {
359 cwist_http_request_destroy(req);
360 return NULL;
361 }
362
363 size_t to_copy = body_received < req->content_length ? body_received : req->content_length;
364 memcpy(body, header_end + 4, to_copy);
365 size_t current_body_len = to_copy;
366
367 while (current_body_len < req->content_length) {
368 struct pollfd pfd = { .fd = conn->fd, .events = POLLIN };
369 int pret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
370 if (pret <= 0) {
371 cwist_free(body);
372 cwist_http_request_destroy(req);
373 return NULL;
374 }
375
376 int bytes = SSL_read(conn->ssl, body + current_body_len, (int)(req->content_length - current_body_len));
377 if (bytes <= 0) {
378 int ssl_err = SSL_get_error(conn->ssl, bytes);
379 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
380 continue;
381 }
382 cwist_free(body);
383 cwist_http_request_destroy(req);
384 return NULL;
385 }
386 current_body_len += (size_t)bytes;
387 }
388 body[req->content_length] = '\0';
389 cwist_sstring_assign_len(req->body, body, req->content_length);
390 cwist_free(body);
391
392 if (body_received > req->content_length) {
393 size_t leftover_len = body_received - req->content_length;
394 memmove(conn->read_buf, header_end + 4 + req->content_length, leftover_len);
395 conn->buf_len = leftover_len;
396 } else {
397 conn->buf_len = 0;
398 }
399 } else {
400 if (body_received > 0) {
401 memmove(conn->read_buf, header_end + 4, body_received);
402 conn->buf_len = body_received;
403 } else {
404 conn->buf_len = 0;
405 }
406 }
407 conn->read_buf[conn->buf_len] = '\0';
408
409 return req;
410}

418cwist_error_t cwist_https_send_response(cwist_https_connection *conn, cwist_http_response *res) {
419 cwist_error_t err = make_error(CWIST_ERR_INT16);
420
421 if (!conn || !conn->ssl || !res) {
422 err.error.err_i16 = -1;
423 return err;
424 }
425
426 // Inject Alt-Svc when HTTP/3 is enabled so clients discover the QUIC endpoint
427 if (conn->http3_enabled) {
428 struct sockaddr_storage ss;
429 socklen_t ss_len = sizeof(ss);
430 int port = 443;
431 if (getsockname(conn->fd, (struct sockaddr *)&ss, &ss_len) == 0) {
432 if (ss.ss_family == AF_INET) {
433 port = ntohs(((struct sockaddr_in *)&ss)->sin_port);
434 } else if (ss.ss_family == AF_INET6) {
435 port = ntohs(((struct sockaddr_in6 *)&ss)->sin6_port);
436 }
437 }
438 char alt_svc[64];
439 snprintf(alt_svc, sizeof(alt_svc), "h3=\":%d\"; ma=86400", port);
440 cwist_http_header_add(&res->headers, "Alt-Svc", alt_svc);
441 }
442
443 // 1. Serialize using existing HTTP logic
444 cwist_sstring *response_str = cwist_http_stringify_response(res);
445 if (!response_str) {
446 err.error.err_i16 = -1;
447 return err;
448 }
449
450 // 2. Send over SSL
451 const char *p = response_str->data;
452 int left = (int)response_str->size;
453 int total_sent = 0;
454
455 err.error.err_i16 = 0; // Assume success initially
456
457 while (left > 0) {
458 int sent = SSL_write(conn->ssl, p, left);
459 if (sent <= 0) {
460 int ssl_err = SSL_get_error(conn->ssl, sent);
461 if (ssl_err == SSL_ERROR_WANT_WRITE || ssl_err == SSL_ERROR_WANT_READ) {
462 continue; // Retry
463 }
464 err = make_ssl_error("SSL write failed");
465 break;
466 }
467 p += sent;
468 left -= sent;
469 total_sent += sent;
470 }
471
472 cwist_sstring_destroy(response_str);
473 return err;
474}

515cwist_error_t cwist_https_server_loop(int server_fd, cwist_https_context *ctx, void (*handler)(cwist_https_connection *, void *), void *user_ctx) {
516 cwist_error_t err = make_error(CWIST_ERR_INT16);
517 if (server_fd < 0 || !ctx || !handler) {
518 err.error.err_i16 = -1;
519 return err;
520 }
521
522 while (1) {
523 struct sockaddr_in addr;
524 socklen_t len = sizeof(addr);
525 int client_fd = accept(server_fd, (struct sockaddr*)&addr, &len);
526
527 if (client_fd < 0) {
528 if (errno == EINTR) continue;
529 continue;
530 }
531
532 pthread_t thread;
533 struct https_thread_payload *payload = malloc(sizeof(*payload));
534 if (!payload) {
535 close(client_fd);
536 continue;
537 }
538 payload->client_fd = client_fd;
539 payload->ctx = ctx;
540 payload->handler = handler;
541 payload->user_ctx = user_ctx;
542
543 if (pthread_create(&thread, NULL, https_thread_handler, payload) == 0) {
