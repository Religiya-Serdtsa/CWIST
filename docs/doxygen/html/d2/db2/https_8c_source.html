62static void *https_pool_worker(void *arg) {
63 (void)arg;
64 while (1) {
65 pthread_mutex_lock(&g_https_pool.mutex);
66 while (g_https_pool.count == 0 && !g_https_pool.shutdown) {
67 pthread_cond_wait(&g_https_pool.cond_not_empty, &g_https_pool.mutex);
68 }
69 if (g_https_pool.shutdown) {
70 pthread_mutex_unlock(&g_https_pool.mutex);
71 break;
72 }
73 https_pool_task_t task = g_https_pool.queue[g_https_pool.head];
74 g_https_pool.head = (g_https_pool.head + 1) % HTTPS_TASK_QUEUE_SIZE;
75 g_https_pool.count--;
76 pthread_cond_signal(&g_https_pool.cond_not_full);
77 pthread_mutex_unlock(&g_https_pool.mutex);
78
79 // We can reuse the existing https_thread_handler logic by wrapping the task
80 struct https_thread_payload *payload = malloc(sizeof(*payload));
81 if (payload) {
82 payload->client_fd = task.client_fd;
83 payload->ctx = task.ctx;
84 payload->handler = task.handler;
85 payload->user_ctx = task.user_ctx;
86 https_thread_handler(payload);
87 } else {
88 close(task.client_fd);
89 }
90 }
91 return NULL;
92}

94int https_pool_init(void) {
95 if (g_https_pool_initialized) return 0;
96 memset(&g_https_pool, 0, sizeof(g_https_pool));
97 pthread_mutex_init(&g_https_pool.mutex, NULL);
98 pthread_cond_init(&g_https_pool.cond_not_empty, NULL);
99 pthread_cond_init(&g_https_pool.cond_not_full, NULL);
100 for (int i = 0; i < HTTPS_THREAD_POOL_SIZE; i++) {
101 if (pthread_create(&g_https_pool.threads[i], NULL, https_pool_worker, NULL) != 0) {
102 return -1;
103 }

109void https_pool_submit(int client_fd, cwist_https_context *ctx, void (*handler)(cwist_https_connection *, void *), void *user_ctx) {
110 pthread_mutex_lock(&g_https_pool.mutex);
111 while (g_https_pool.count >= HTTPS_TASK_QUEUE_SIZE && !g_https_pool.shutdown) {
112 pthread_cond_wait(&g_https_pool.cond_not_full, &g_https_pool.mutex);

115 pthread_mutex_unlock(&g_https_pool.mutex);

120 g_https_pool.queue[g_https_pool.tail].ctx = ctx;
121 g_https_pool.queue[g_https_pool.tail].handler = handler;
122 g_https_pool.queue[g_https_pool.tail].user_ctx = user_ctx;
123 g_https_pool.tail = (g_https_pool.tail + 1) % HTTPS_TASK_QUEUE_SIZE;
124 g_https_pool.count++;

129void https_pool_destroy(void) {
130 if (!g_https_pool_initialized) return;
131 pthread_mutex_lock(&g_https_pool.mutex);
132 g_https_pool.shutdown = 1;
133 pthread_cond_broadcast(&g_https_pool.cond_not_empty);
134 pthread_mutex_unlock(&g_https_pool.mutex);

136 pthread_join(g_https_pool.threads[i], NULL);

139 pthread_cond_destroy(&g_https_pool.cond_not_empty);

360 cwist_error_t err = make_error(CWIST_ERR_INT16);

453 size_t total_received = conn->buf_len;
454 char *header_end = NULL;
455
456 while (!(header_end = strstr(conn->read_buf, "\r\n\r\n"))) {
457 if (total_received >= CWIST_HTTP_READ_BUFFER_SIZE - 1) {
458 return NULL;
459 }
460
461 struct pollfd pfd = { .fd = conn->fd, .events = POLLIN };
462 int pret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
463 if (pret <= 0) {
464 return NULL;
465 }
466

468 if (bytes <= 0) {

470 if (ssl_err == SSL_ERROR_WANT_READ || ssl_err == SSL_ERROR_WANT_WRITE) {
471 continue;
472 }
473 return NULL;

477 conn->read_buf[total_received] = '\0';
478 }

486 size_t body_received = total_received - header_len;
487
488 if (req->content_length > 0) {

555cwist_error_t cwist_https_send_response(cwist_https_connection *conn, cwist_http_response *res) {
556 cwist_error_t err = make_error(CWIST_ERR_INT16);
557
558 if (!conn || !conn->ssl || !res) {
559 err.error.err_i16 = -1;
560 return err;
561 }
562
563 // Inject Alt-Svc when HTTP/3 is enabled so clients discover the QUIC endpoint
564 if (conn->http3_enabled) {
565 struct sockaddr_storage ss;
566 socklen_t ss_len = sizeof(ss);
567 int port = 443;
568 if (getsockname(conn->fd, (struct sockaddr *)&ss, &ss_len) == 0) {
569 if (ss.ss_family == AF_INET) {
570 port = ntohs(((struct sockaddr_in *)&ss)->sin_port);
571 } else if (ss.ss_family == AF_INET6) {
572 port = ntohs(((struct sockaddr_in6 *)&ss)->sin6_port);
573 }
574 }
575 char alt_svc[64];
576 snprintf(alt_svc, sizeof(alt_svc), "h3=\":%d\"; ma=86400", port);
577 cwist_http_header_add(&res->headers, "Alt-Svc", alt_svc);
578 }
579
580 // 1. Serialize using existing HTTP logic
581 cwist_sstring *response_str = cwist_http_stringify_response(res);
582 if (!response_str) {
583 err.error.err_i16 = -1;
584 return err;
585 }
586
587 // 2. Send over SSL
588 const char *p = response_str->data;
589 int left = (int)response_str->size;
590 int total_sent = 0;
591
592 err.error.err_i16 = 0; // Assume success initially
593
594 while (left > 0) {
595 int sent = SSL_write(conn->ssl, p, left);
596 if (sent <= 0) {
597 int ssl_err = SSL_get_error(conn->ssl, sent);
598 if (ssl_err == SSL_ERROR_WANT_WRITE || ssl_err == SSL_ERROR_WANT_READ) {
599 continue; // Retry
600 }
601 err = make_ssl_error("SSL write failed");
602 break;
603 }
604 p += sent;
605 left -= sent;
606 total_sent += sent;
607 }
608
609 cwist_sstring_destroy(response_str);
610 return err;
611}
