38bool cwist_seq_chunk_parse(const uint8_t *data, size_t len, cwist_seq_chunk_t *out) {
39 if (!data || !out || len < CWIST_SEQ_HEADER_SIZE) return false;
40
41 out->seq = seq_read_u16(data + 0);
42 out->total = seq_read_u16(data + 2);
43 out->payload_len = seq_read_u16(data + 4);
44 out->chunk_size = seq_read_u16(data + 6);
45 out->payload = data + CWIST_SEQ_HEADER_SIZE;
46
47 /* Reject missing or malformed sequence pairs. */
48 if (out->seq == 0 || out->total == 0 || out->total > CWIST_SEQ_MAX_CHUNKS || out->seq > out->total) return false;
49 if (out->payload_len == 0 || out->chunk_size == 0) return false;
50 if (out->payload_len > out->chunk_size) return false;
51 if (out->seq != out->total && out->payload_len != out->chunk_size) return false;
52 if (len - CWIST_SEQ_HEADER_SIZE < out->payload_len) return false;
53
54 return true;
55}

72bool cwist_seq_split(const uint8_t *data,
73 size_t len,
74 uint16_t chunk_payload_size,
75 cwist_seq_message_t *out) {
76 if (!data || len == 0 || chunk_payload_size == 0 || !out) return false;
77 memset(out, 0, sizeof(*out));
78
79 uint32_t total32 = (uint32_t)((len + chunk_payload_size - 1) / chunk_payload_size);
80 if (total32 == 0 || total32 > UINT16_MAX || total32 > CWIST_SEQ_MAX_CHUNKS || len > CWIST_SEQ_MAX_REASSEMBLED_SIZE) return false;
81 uint16_t total = (uint16_t)total32;
82
83 out->chunks = (uint8_t **)cwist_alloc_array(total, sizeof(uint8_t *));
84 out->chunk_lens = (size_t *)cwist_alloc_array(total, sizeof(size_t));
85 if (!out->chunks || !out->chunk_lens) {
86 cwist_free(out->chunks);
87 cwist_free(out->chunk_lens);
88 memset(out, 0, sizeof(*out));
89 return false;
90 }
91
92 for (uint16_t i = 0; i < total; i++) {
93 size_t offset = (size_t)i * chunk_payload_size;
94 size_t plen = len - offset;
95 if (plen > chunk_payload_size) plen = chunk_payload_size;
96 size_t chunk_len = CWIST_SEQ_HEADER_SIZE + plen;
97
98 uint8_t *chunk = (uint8_t *)cwist_alloc(chunk_len);
99 if (!chunk) {
100 for (uint16_t j = 0; j < i; j++) cwist_free(out->chunks[j]);
101 cwist_free(out->chunks);
102 cwist_free(out->chunk_lens);
103 memset(out, 0, sizeof(*out));
104 return false;
105 }
106
107 cwist_seq_chunk_build_header(chunk, i + 1, total, (uint16_t)plen, chunk_payload_size);
108 memcpy(chunk + CWIST_SEQ_HEADER_SIZE, data + offset, plen);
109 out->chunks[i] = chunk;
110 out->chunk_lens[i] = chunk_len;
111 }
112
113 out->count = total;
114 return true;
115}

159bool cwist_seq_assembler_feed(cwist_seq_assembler_t *a, const cwist_seq_chunk_t *chunk) {
160 if (!a || !chunk) return false;
161
162 /* Validate sequence pair. */
163 if (chunk->seq == 0 || chunk->total == 0 || chunk->total > CWIST_SEQ_MAX_CHUNKS || chunk->seq > chunk->total) return false;
164 if (chunk->payload_len == 0 || chunk->chunk_size == 0) return false;
165 if (chunk->payload_len > chunk->chunk_size) return false;
166 if (chunk->seq != chunk->total && chunk->payload_len != chunk->chunk_size) return false;
167 if (!chunk->payload) return false;
168
169 if (!a->have_state) {
170 a->total = chunk->total;
171 a->chunk_size = chunk->chunk_size;
172 /* The first packet may be the short final packet. Allocate for the
173 * full layout and determine total_len only after seq == total is
174 * actually received; deriving it from arrival order was the source of
175 * false protocol errors on reordered HTTP/2/3-adjacent transports. */
176 if ((size_t)chunk->total > SIZE_MAX / (size_t)chunk->chunk_size) return false;
177 a->data_cap = (size_t)chunk->total * chunk->chunk_size;
178 size_t ceiling = a->max_data_len ? a->max_data_len : CWIST_SEQ_MAX_REASSEMBLED_SIZE;
179 if (a->data_cap > ceiling) {
180 a->data_cap = 0;
181 return false;
182 }
183 a->data = (uint8_t *)cwist_alloc(a->data_cap);
184 a->received = (bool *)cwist_alloc_array(chunk->total, sizeof(bool));
185 a->payload_lens = (uint16_t *)cwist_alloc_array(chunk->total, sizeof(uint16_t));
186 if (!a->data || !a->received || !a->payload_lens) {
187 cwist_free(a->data);
188 cwist_free(a->received);
189 cwist_free(a->payload_lens);
190 memset(a, 0, sizeof(*a));
191 return false;
192 }
193 a->have_state = true;
194 } else {
195 if (chunk->total != a->total || chunk->chunk_size != a->chunk_size) return false;
196 }
197
198 size_t index = (size_t)chunk->seq - 1;
199 size_t offset = index * a->chunk_size;
200 if (a->received[index]) {
201 /* Safe duplicates are idempotent. A different duplicate is data
202 * corruption, not a retry, and must never overwrite good bytes. */
203 bool identical = a->payload_lens[index] == chunk->payload_len &&
204 memcmp(a->data + offset, chunk->payload, chunk->payload_len) == 0;
205 if (!identical) a->contaminated = true;
206 return identical;
207 }
208
209 if (offset > a->data_cap || chunk->payload_len > a->data_cap - offset) return false;
210
211 memcpy(a->data + offset, chunk->payload, chunk->payload_len);
212 a->received[index] = true;
213 a->payload_lens[index] = chunk->payload_len;
214 if (chunk->seq == chunk->total) {
215 a->total_len = offset + chunk->payload_len;
216 }
217 a->received_count++;
218 return true;
219}
