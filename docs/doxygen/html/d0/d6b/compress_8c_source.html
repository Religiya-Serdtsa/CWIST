54static int zlib_compress(void *state, const char *in, size_t in_len,
55 char *out, size_t *out_len, int flush) {
56 z_stream *zs = (z_stream *)state;
57 zs->next_in = (Bytef *)in;
58 zs->avail_in = (uInt)in_len;
59 zs->next_out = (Bytef *)out;
60 zs->avail_out = (uInt)(*out_len);
61 int rc = deflate(zs, flush ? Z_FINISH : Z_NO_FLUSH);
62 if (rc != Z_OK && rc != Z_STREAM_END && rc != Z_BUF_ERROR) return -1;
63 *out_len = (size_t)(*out_len - zs->avail_out);
64 return 0;
65}

113static int str_contains_ci(const char *haystack, const char *needle) {
114 if (!haystack || !needle) return 0;
115 size_t hlen = strlen(haystack);
116 size_t nlen = strlen(needle);
117 if (nlen == 0) return 1;
118 if (hlen < nlen) return 0;
119 for (size_t i = 0; i <= hlen - nlen; ++i) {
120 size_t j = 0;
121 for (; j < nlen; ++j) {
122 char a = haystack[i + j];
123 char b = needle[j];
124 if (a >= 'A' && a <= 'Z') a = a - 'A' + 'a';
125 if (b >= 'A' && b <= 'Z') b = b - 'A' + 'a';
126 if (a != b) break;
127 }
128 if (j == nlen) return 1;
129 }
130 return 0;
131}

133static const cwist_compress_backend *select_backend(const char *accept_encoding) {
134 if (!accept_encoding) return NULL;

136 for (int i = 0; i < g_backend_count; ++i) {
137 if (str_contains_ci(accept_encoding, g_backends[i]->encoding_name)) {

159static void cwist_mw_compress_handler(cwist_http_request *req, cwist_http_response *res, cwist_handler_func next) {
160 const char *accept = cwist_http_header_get(req->headers, "Accept-Encoding");
161 const cwist_compress_backend *backend = select_backend(accept);
162
163 next(req, res);
164
165 if (!backend) return;
166 if (!res->body || res->body->size == 0) return;
167 if (res->is_ptr_body || res->use_file_stream) return;
168 if (res->status_code < 200 || res->status_code >= 300) return;
169 if (res->body->size < g_compress_min_size) return;
170
171 void *state = NULL;
172 if (backend->init(&state) != 0) return;
173
174 size_t out_cap = res->body->size + 256;
175 char *out_buf = (char *)malloc(out_cap);
176 if (!out_buf) {
177 backend->cleanup(state);
178 return;
179 }
180
181 size_t total_out = 0;
182 int ok = 1;
183
184 /* Stream the entire body through the compressor */
185 size_t in_offset = 0;
186 while (in_offset < res->body->size && ok) {
187 size_t chunk_in = res->body->size - in_offset;
188 size_t chunk_out = out_cap - total_out;
189 if (chunk_out == 0) {
190 /* Grow output buffer */
191 size_t new_cap = out_cap * 2;
192 char *new_buf = (char *)realloc(out_buf, new_cap);
193 if (!new_buf) { ok = 0; break; }
194 out_buf = new_buf;
195 out_cap = new_cap;
196 chunk_out = out_cap - total_out;
197 }
198 int flush = (in_offset + chunk_in >= res->body->size) ? 1 : 0;
199 size_t written = chunk_out;
200 if (backend->compress(state, res->body->data + in_offset, chunk_in,
201 out_buf + total_out, &written, flush) != 0) {
202 ok = 0;
203 break;
204 }
205 total_out += written;
206 in_offset += chunk_in;
207 }
208
209 if (ok) {
210 size_t tail = out_cap - total_out;
211 if (tail == 0) {
212 size_t new_cap = out_cap + 256;
213 char *new_buf = (char *)realloc(out_buf, new_cap);
214 if (new_buf) {
215 out_buf = new_buf;
216 out_cap = new_cap;
217 tail = out_cap - total_out;
218 }
219 }
220 if (tail > 0) {
221 if (backend->finish(state, out_buf + total_out, &tail) != 0) {
222 ok = 0;
223 } else {
224 total_out += tail;
225 }
226 }
227 }
228
229 backend->cleanup(state);
230
231 if (ok && total_out > 0 && total_out < res->body->size) {
232 cwist_sstring_assign_len(res->body, out_buf, total_out);
233 cwist_http_header_add(&res->headers, "Content-Encoding", backend->encoding_name);
234 remove_header(&res->headers, "Content-Length");
235 }
236
237 free(out_buf);
238}
