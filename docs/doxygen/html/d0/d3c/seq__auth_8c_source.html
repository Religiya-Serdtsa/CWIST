28static bool seq_auth_chunk_valid(const uint8_t *data, size_t len,
29 cwist_seq_chunk_t *out) {
30 if (!data || len < CWIST_SEQ_HEADER_SIZE || !out) return false;
31 out->seq = read_u16(data);
32 out->total = read_u16(data + 2);
33 out->payload_len = read_u16(data + 4);
34 out->chunk_size = read_u16(data + 6);
35 if (!out->seq || !out->total || out->seq > out->total ||
36 !out->payload_len || !out->chunk_size ||
37 out->payload_len > out->chunk_size ||
38 (out->seq != out->total && out->payload_len != out->chunk_size) ||
39 len < (size_t)CWIST_SEQ_AUTH_HEADER_SIZE + out->payload_len) return false;
40 out->payload = data + CWIST_SEQ_AUTH_HEADER_SIZE;
41 return true;
42}

44static bool auth_tag(const cwist_seq_auth_context_t *ctx,
45 const uint8_t *header_without_tag, size_t header_len,
46 const uint8_t *payload, size_t payload_len,
47 uint8_t tag[EVP_MAX_MD_SIZE]) {
48 HMAC_CTX *h = HMAC_CTX_new();
49 unsigned int tag_len = 0;
50 bool ok = h && HMAC_Init_ex(h, ctx->key, sizeof(ctx->key), EVP_sha256(), NULL) == 1 &&
51 HMAC_Update(h, ctx->session_id, sizeof(ctx->session_id)) == 1 &&
52 HMAC_Update(h, header_without_tag, header_len) == 1 &&
53 HMAC_Update(h, payload, payload_len) == 1 &&
54 HMAC_Final(h, tag, &tag_len) == 1 && tag_len >= CWIST_SEQ_AUTH_TAG_SIZE;
55 HMAC_CTX_free(h);
56 return ok;
57}

74 const uint8_t key[CWIST_SEQ_AUTH_KEY_SIZE],
75 const uint8_t session_id[CWIST_SEQ_AUTH_SESSION_ID_SIZE], size_t replay_window) {
76 if (!key || !session_id || replay_window == 0 || replay_window > SIZE_MAX / 28) return NULL;
77 cwist_seq_auth_context_t *ctx = cwist_alloc(sizeof(*ctx));
78 if (!ctx) return NULL;
79 memset(ctx, 0, sizeof(*ctx));
80 if (pthread_mutex_init(&ctx->lock, NULL) != 0) {
81 cwist_free(ctx);
82 return NULL;
83 }
84 ctx->nonces = cwist_alloc_array(replay_window, CWIST_SEQ_AUTH_NONCE_SIZE);
85 ctx->completed_ids = cwist_alloc_array(replay_window, CWIST_SEQ_AUTH_MESSAGE_ID_SIZE);
86 if (!ctx->nonces || !ctx->completed_ids) { cwist_seq_auth_context_destroy(ctx); return NULL; }
87 memcpy(ctx->key, key, sizeof(ctx->key));
88 memcpy(ctx->session_id, session_id, sizeof(ctx->session_id));
89 ctx->capacity = replay_window;
90 return ctx;
91}
