52// @brief bsd compatible sigtimedwait
53static inline int bsd_sigtimedwait(const sigset_t *set, siginfo_t *info, const struct timespec *timeout) {
54 int kq = kqueue();
55 if (kq < 0) {
56 return -1;
57 }
58
59 struct kevent changes[NSIG];
60 int nchanges = 0;
61
62 for (int sig = 1; sig < NSIG; sig++) {
63 if (sigismember(set, sig)) {
64 EV_SET(&changes[nchanges], sig, EVFILT_SIGNAL, EV_ADD | EV_ENABLE | EV_ONESHOT, 0, 0, NULL);
65 nchanges++;
66 }
67 }
68
69 if (nchanges == 0) {
70 close(kq);
71 errno = EINVAL;
72 return -1;
73 }
74
75 struct kevent event;
76 int ret = kevent(kq, changes, nchanges, &event, 1, timeout);
77
78 int saved_errno = errno;
79 close(kq);
80
81 if (ret < 0) {
82 errno = saved_errno;
83 return -1;
84 }
85
86 if (ret == 0) {
87 errno = EAGAIN;
88 return -1;
89 }
90
91 int sig = (int)event.ident;
92
93 if (info != NULL) {
94 memset(info, 0, sizeof(siginfo_t));
95 info->si_signo = sig;
96 }
97
98 return sig;
99}

108static uint64_t nuke_estimate_required_ram(const char *disk_path) {
109 struct stat st;
110 uint64_t db_size = 0;
111 if (disk_path && stat(disk_path, &st) == 0 && st.st_size > 0) {
112 db_size = (uint64_t)st.st_size;
113 }
114 // Need room for the DB plus SQLite cache/overhead. Double the DB and add base buffer.
115 uint64_t required = (db_size * 2) + CWIST_MIB(32);
116 if (required < CWIST_MIB(64)) {
117 required = CWIST_MIB(64);
118 }
119 return required;
120}

149static bool nuke_integrity_ok(sqlite3 *db) {
150 if (!db) return false;
151 sqlite3_stmt *stmt = NULL;
152 bool healthy = false;
153 if (sqlite3_prepare_v2(db, "PRAGMA integrity_check;", -1, &stmt, NULL) == SQLITE_OK) {
154 if (sqlite3_step(stmt) == SQLITE_ROW) {
155 const unsigned char *state = sqlite3_column_text(stmt, 0);
156 healthy = (state && strcmp((const char *)state, "ok") == 0);
157 }
158 }
159 if (stmt) sqlite3_finalize(stmt);
160 return healthy;
161}

170static int nuke_backup(sqlite3 *dest, sqlite3 *source) {
171 if (!dest || !source) return -1;
172 sqlite3_backup *backup = sqlite3_backup_init(dest, "main", source, "main");
173 if (!backup) {
174 fprintf(stderr, "[NukeDB] sqlite3_backup_init failed: %s\n", sqlite3_errmsg(dest));
175 return -1;
176 }
177
178 int rc = sqlite3_backup_step(backup, -1); // Copy all pages
179 if (rc != SQLITE_DONE) {
180 fprintf(stderr, "[NukeDB] Backup step failed: %d (%s)\n", rc, sqlite3_errmsg(dest));
181 }
182
183 sqlite3_backup_finish(backup);
184 return (rc == SQLITE_DONE) ? 0 : -1;
185}

227int cwist_nuke_sync(void) {
228 pthread_mutex_lock(&g_nuke_lock);
229
230 if (g_nuke.is_disk_mode) {
231 if (g_nuke.disk_db) {
232 sqlite3_wal_checkpoint_v2(g_nuke.disk_db, "main", SQLITE_CHECKPOINT_PASSIVE, NULL, NULL);
233 pthread_mutex_unlock(&g_nuke_lock);
234 return 0;
235 }
236 pthread_mutex_unlock(&g_nuke_lock);
237 return -1;
238 }
239
240 // Safety: If initial load failed, don't overwrite disk with empty memory DB
241 if (!g_nuke.load_successful) {
242 pthread_mutex_unlock(&g_nuke_lock);
243 return -1;
244 }
245
246 if (!g_nuke.mem_db || !g_nuke.disk_db) {
247 pthread_mutex_unlock(&g_nuke_lock);
248 return -1;
249 }
250
251 int rc = nuke_backup(g_nuke.disk_db, g_nuke.mem_db);
252 pthread_mutex_unlock(&g_nuke_lock);
253 return rc;
254}

311static void *sync_thread_func(void *arg) {
312 CWIST_UNUSED(arg);
313 struct timespec timeout;
314 int signum;
315
316 while (g_running) {
317 timeout.tv_sec = g_nuke.sync_interval_ms / 1000;
318 timeout.tv_nsec = (g_nuke.sync_interval_ms % 1000) * 1000000;
319
320 // Wait for signals (INT, TERM, USR1, USR2) or Timeout
321#if defined(__FreeBSD__) || defined(__OpenBSD__) || defined(__NetBSD__) || defined(__APPLE__)
322#define sigtimedwait bsd_sigtimedwait
323#endif
324
325 signum = sigtimedwait(&g_sigset, NULL, &timeout);
326
327 if (signum > 0) {
328 if (signum == SIGUSR1) {
329 // Requested to stop via cwist_nuke_close
330 break;
331 }
332
333 if (signum == SIGUSR2) {
334 // Immediate sync requested via commit hook
335 cwist_nuke_sync();
336 continue;
337 }
338
339 // Handle termination signals (SIGINT, SIGTERM)
340 printf("\n[NukeDB] Intercepted Signal %d. Saving data...\n", signum);
341
342 cwist_nuke_sync();
343 nuke_cleanup_internal();
344 g_running = false;
345
346 // Unblock the signal first
347 sigset_t s;
348 sigemptyset(&s);
349 sigaddset(&s, signum);
350 pthread_sigmask(SIG_UNBLOCK, &s, NULL);
351
352 // Restore default handler and re-raise
353 signal(signum, SIG_DFL);
354 raise(signum);
355 return NULL;
356 } else { // Timeout (EAGAIN) or Interruption
357 if (errno == EAGAIN && g_running && g_nuke.auto_sync) {
358 if (cwist_is_ram_critical(CWIST_MIB(128))) {
359 nuke_switch_to_disk();
360 } else {
361 cwist_nuke_sync();
362 }
363 }
364 }
365 }
366 return NULL;
367}

375int cwist_nuke_init(const char *disk_path, int sync_interval_ms) {
376 if (g_running) return CWIST_NUKE_ERR_GENERIC; // Already running
377 if (!disk_path) return CWIST_NUKE_ERR_GENERIC;
378
379 uint64_t required_ram = nuke_estimate_required_ram(disk_path);
380 uint64_t available_ram = cwist_get_available_ram();
381 if (available_ram > 0 && available_ram < required_ram) {
382 fprintf(stderr,
383 "[NukeDB] Insufficient RAM for in-memory mode (have %zu bytes, need ~%zu bytes).\n",
384 (size_t)available_ram,
385 (size_t)required_ram);
386 return CWIST_NUKE_ERR_LOW_MEMORY;
387 }
388
389 g_nuke.disk_path = cwist_strdup(disk_path);
390 if (!g_nuke.disk_path) {
391 return CWIST_NUKE_ERR_GENERIC;
392 }
393 g_nuke.sync_interval_ms = sync_interval_ms > 0 ? sync_interval_ms : 1000;
394 g_nuke.auto_sync = (sync_interval_ms > 0);
395 g_nuke.is_disk_mode = false;
396 g_nuke.load_successful = false;
397
398 // 1. Open Disk DB
399 int rc = sqlite3_open(disk_path, &g_nuke.disk_db);
400 if (rc != SQLITE_OK) {
401 fprintf(stderr, "[NukeDB] Failed to open disk DB: %s\n", sqlite3_errmsg(g_nuke.disk_db));
402 nuke_cleanup_internal();
403 return CWIST_NUKE_ERR_GENERIC;
404 }
405
406 if (!nuke_integrity_ok(g_nuke.disk_db)) {
407 fprintf(stderr, "[NukeDB] Integrity check failed for '%s'. Aborting in-memory mode.\n", disk_path);
408 nuke_cleanup_internal();
409 return CWIST_NUKE_ERR_GENERIC;
410 }
411
412 sqlite3_exec(g_nuke.disk_db, "PRAGMA journal_mode=WAL;", NULL, NULL, NULL);
413 sqlite3_exec(g_nuke.disk_db, "PRAGMA synchronous=NORMAL;", NULL, NULL, NULL);
414
415 // 2. Open Memory DB
416 rc = sqlite3_open(":memory:", &g_nuke.mem_db);
417 if (rc != SQLITE_OK) {
418 fprintf(stderr, "[NukeDB] Failed to open memory DB: %s\n", sqlite3_errmsg(g_nuke.mem_db));
419 nuke_cleanup_internal();
420 return CWIST_NUKE_ERR_GENERIC;
421 }
422
423 // 3. Load Disk -> Memory
424 pthread_mutex_lock(&g_nuke_lock);
425
426 // Use sqlite3_backup as it is safer than raw deserialize for WAL/mode transitions
427 int backup_rc = nuke_backup(g_nuke.mem_db, g_nuke.disk_db);
428 if (backup_rc == 0) {
429 g_nuke.load_successful = true;
430 } else if (!nuke_disk_has_tables(g_nuke.disk_db)) {
431 g_nuke.load_successful = true;
432 } else {
433 fprintf(stderr, "[NukeDB] Initial disk->memory load failed. Falling back to disk-only mode.\n");
434 g_nuke.is_disk_mode = true;
435 sqlite3_close(g_nuke.mem_db);
436 g_nuke.mem_db = NULL;
437 }
438 pthread_mutex_unlock(&g_nuke_lock);
439
440 // Register commit hook for immediate sync. Only needed while in-memory mode is active.
441 if (g_nuke.mem_db && !g_nuke.is_disk_mode) {
442 sqlite3_commit_hook(g_nuke.mem_db, nuke_commit_hook, NULL);
443 }
444
445 g_running = true;
446
447 // 4. Setup Signal Interception
448 sigemptyset(&g_sigset);
449 sigaddset(&g_sigset, SIGINT);
450 sigaddset(&g_sigset, SIGTERM);
451 sigaddset(&g_sigset, SIGUSR1);
452 sigaddset(&g_sigset, SIGUSR2);
453
454 pthread_sigmask(SIG_BLOCK, &g_sigset, NULL);
455
456 // 5. Start Sync/Signal Thread
457 if (pthread_create(&g_sync_thread, NULL, sync_thread_func, NULL) != 0) {
458 g_running = false;
459 pthread_sigmask(SIG_UNBLOCK, &g_sigset, NULL);
460 nuke_cleanup_internal();
461 return CWIST_NUKE_ERR_GENERIC;
462 }
463
464 // 6. Register atexit for ultimate safety
465 atexit(cwist_nuke_close);
466
467 return CWIST_NUKE_OK;
468}

495int cwist_nuke_deserialize(unsigned char *data, sqlite3_int64 data_len) {
496 if (!data || data_len <= 0) return -1;
497
498 pthread_mutex_lock(&g_nuke_lock);
499
500 if (g_nuke.is_disk_mode || !g_nuke.mem_db) {
501 pthread_mutex_unlock(&g_nuke_lock);
502 return -1;
503 }
504
505 /* SQLITE_DESERIALIZE_FREEONCLOSE: SQLite will free the buffer when the
506 * database connection is closed.
507 * SQLITE_DESERIALIZE_RESIZABLE: allows the in-memory DB to grow. */
508 int rc = sqlite3_deserialize(g_nuke.mem_db, "main",
509 data, data_len, data_len,
510 SQLITE_DESERIALIZE_FREEONCLOSE |
511 SQLITE_DESERIALIZE_RESIZEABLE);
512 pthread_mutex_unlock(&g_nuke_lock);
513 return (rc == SQLITE_OK) ? 0 : -1;
