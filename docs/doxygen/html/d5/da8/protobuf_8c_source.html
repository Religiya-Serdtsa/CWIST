28int cwist_pb_writer_reserve(cwist_pb_writer *w, size_t extra) {
29 if (!w) return -1;
30 if (extra > SIZE_MAX - w->len) return -1;
31 size_t need = w->len + extra;
32 if (need <= w->cap) return 0;
33
34 size_t cap = w->cap ? w->cap : 64;
35 while (cap < need) {
36 if (cap > SIZE_MAX / 2) {
37 cap = need;
38 break;
39 }
40 cap *= 2;
41 }
42 uint8_t *next = (uint8_t *)cwist_realloc(w->data, cap);
43 if (!next) return -1;
44 w->data = next;
45 w->cap = cap;
46 return 0;
47}

105int cwist_pb_read_varint(cwist_pb_reader *r, uint64_t *out) {
106 if (!r || !out) return -1;
107 uint64_t value = 0;
108 unsigned shift = 0;
109 while (r->pos < r->len && shift <= 63) {
110 uint8_t byte = r->data[r->pos++];
111 value |= ((uint64_t)(byte & 0x7f)) << shift;
112 if ((byte & 0x80) == 0) {
113 *out = value;
114 return 0;
115 }
116 shift += 7;
117 }
118 return -1;
119}

121int cwist_pb_read_field(cwist_pb_reader *r, cwist_pb_field *field) {
122 if (!r || !field || r->pos >= r->len) return 0;
123 memset(field, 0, sizeof(*field));
124
125 uint64_t key = 0;
126 if (cwist_pb_read_varint(r, &key) != 0) return -1;
127 field->number = (uint32_t)(key >> 3);
128 field->wire_type = (cwist_pb_wire_type_t)(key & 0x07);
129 if (field->number == 0) return -1;
130
131 switch (field->wire_type) {
132 case CWIST_PB_VARINT:
133 if (cwist_pb_read_varint(r, &field->varint) != 0) return -1;
134 return 1;
135 case CWIST_PB_LEN: {
136 uint64_t len = 0;
137 if (cwist_pb_read_varint(r, &len) != 0) return -1;
138 if (len > SIZE_MAX || (size_t)len > r->len - r->pos) return -1;
139 field->bytes = r->data + r->pos;
140 field->len = (size_t)len;
141 r->pos += field->len;
142 return 1;
143 }
144 case CWIST_PB_32BIT:
145 if (r->len - r->pos < 4) return -1;
146 field->bytes = r->data + r->pos;
147 field->len = 4;
148 r->pos += 4;
149 return 1;
150 case CWIST_PB_64BIT:
151 if (r->len - r->pos < 8) return -1;
152 field->bytes = r->data + r->pos;
153 field->len = 8;
154 r->pos += 8;
155 return 1;
156 default:
157 return -1;
158 }
159}
