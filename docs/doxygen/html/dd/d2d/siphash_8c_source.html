173static void gusuryak_mix(const uint8_t in[16], uint8_t out[16]) {
174 /* Following Choe Seok-jeong (Gusuryak), treat 16 bytes as a 4x4 Latin board. */
175 uint64_t hi = 0x9e3779b185ebca87ULL;
176 uint64_t lo = 0xc2b2ae3d27d4eb4fULL;
177 for (size_t row = 0; row < 4; ++row) {
178 for (size_t col = 0; col < 4; ++col) {
179 size_t idx = row * 4 + col;
180 uint8_t base = in[idx];
181 uint8_t partner = in[((col + 1) % 4) + ((row + 1) % 4) * 4];
182 uint8_t orth = gusuryak_cell(base, partner);
183 uint64_t delta = ((uint64_t)base << 32) |
184 ((uint64_t)orth << 24) |
185 ((uint64_t)row << 12) |
186 ((uint64_t)col << 4) |
187 (uint64_t)gusuryak_cell(partner, base);
188 hi ^= rotl64(delta ^ hi, (unsigned int)((row * 13 + col * 7) & 63));
189 lo += rotl64(delta + lo, (unsigned int)((row * 11 + col * 5) & 63));
190 }
191 }
192 hi ^= rotl64(lo, 17);
193 lo ^= rotl64(hi, 41);
194 memcpy(out, &hi, 8);
195 memcpy(out + 8, &lo, 8);
196}
