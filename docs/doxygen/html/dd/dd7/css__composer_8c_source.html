15cwist_color_hsl cwist_color_rgb_to_hsl(cwist_color_rgb rgb) {
16 cwist_color_hsl hsl = {0.0f, 0.0f, 0.0f};
17 float r = rgb.r / 255.0f;
18 float g = rgb.g / 255.0f;
19 float b = rgb.b / 255.0f;
20
21 float max = MAX3(r, g, b);
22 float min = MIN3(r, g, b);
23 hsl.l = (max + min) / 2.0f;
24
25 if (max == min) {
26 hsl.h = 0.0f;
27 hsl.s = 0.0f;
28 } else {
29 float d = max - min;
30 hsl.s = hsl.l > 0.5f ? d / (2.0f - max - min) : d / (max + min);
31
32 if (max == r) {
33 hsl.h = (g - b) / d + (g < b ? 6.0f : 0.0f);
34 } else if (max == g) {
35 hsl.h = (b - r) / d + 2.0f;
36 } else {
37 hsl.h = (r - g) / d + 4.0f;
38 }
39 hsl.h /= 6.0f;
40 }
41 hsl.h *= 360.0f;
42 return hsl;
43}

45static float hue2rgb(float p, float q, float t) {
46 if (t < 0.0f) t += 1.0f;
47 if (t > 1.0f) t -= 1.0f;
48 if (t < 1.0f / 6.0f) return p + (q - p) * 6.0f * t;
49 if (t < 1.0f / 2.0f) return q;
50 if (t < 2.0f / 3.0f) return p + (q - p) * (2.0f / 3.0f - t) * 6.0f;
51 return p;
52}

54cwist_color_rgb cwist_color_hsl_to_rgb(cwist_color_hsl hsl) {
55 cwist_color_rgb rgb = {0, 0, 0};
56 float h = hsl.h / 360.0f;
57 float s = hsl.s;
58 float l = hsl.l;
59
60 if (s == 0.0f) {
61 rgb.r = rgb.g = rgb.b = (unsigned char)(l * 255.0f);
62 } else {
63 float q = l < 0.5f ? l * (1.0f + s) : l + s - l * s;
64 float p = 2.0f * l - q;
65 rgb.r = (unsigned char)(roundf(hue2rgb(p, q, h + 1.0f / 3.0f) * 255.0f));
66 rgb.g = (unsigned char)(roundf(hue2rgb(p, q, h) * 255.0f));
67 rgb.b = (unsigned char)(roundf(hue2rgb(p, q, h - 1.0f / 3.0f) * 255.0f));
68 }
69 return rgb;
70}

72cwist_color_rgb cwist_color_hex_to_rgb(const char *hex) {
73 cwist_color_rgb rgb = {0, 0, 0};
74 if (!hex) return rgb;
75 if (hex[0] == '#') hex++;
76 if (strlen(hex) == 6) {
77 int r, g, b;
78 if (sscanf(hex, "%02x%02x%02x", &r, &g, &b) == 3) {
79 rgb.r = r; rgb.g = g; rgb.b = b;
80 }
81 }
82 return rgb;
83}
