110long get_optimal_thread_count(void) {
111 if (g_http_pool_core_limit > 0) {
112 return (long)g_http_pool_core_limit;
113 }
114 const char *env = getenv("CWIST_WORKER_THREADS");
115 if (env && env[0]) {
116 long override = atol(env);
117 if (override > 0) return override;
118 }
119
120 long cores = get_cpu_cores();
121 if (cores < 1) cores = 1;
122
123 long workers = cores;
124 const char *w_env = getenv("CWIST_WORKERS");
125 if (w_env && w_env[0]) {
126 if (strcmp(w_env, "auto") != 0) {
127 long parsed = atol(w_env);
128 if (parsed > 0) workers = parsed;

131
132 if (workers == 1) {
133 long count = cores;
134 if (count < 4) count = 4;
135 if (count > 32) count = 32;
136 return count;
137 }

139 /* Dynamic thread downscaling: distribute thread budget proportionally across forked worker processes */
140 long threads_per_worker = (cores * 4) / workers;
141 if (threads_per_worker < 2) threads_per_worker = 2;
142 if (threads_per_worker > 16) threads_per_worker = 16;

202void cwist_http_pool_submit(int client_fd, void (*handler)(int, void *), void *ctx) {
203 /* Deterministic worker selection using Choi Seok-jeong's MOLS to minimize cache bouncing. */
204 uint16_t node_id = (uint16_t)(client_fd % TTAK_MOLS_NODE_COUNT);
205 uint32_t mixed = ttak_apply_mols_control(node_id, (uint32_t)g_rr_index);
206 size_t worker_idx = mixed % (size_t)g_http_thread_count;

208 g_rr_index = (g_rr_index + 1) % (size_t)g_http_thread_count;

213 if (!c) {

305cwist_http_method_t cwist_http_string_to_method_len(const char *str, size_t len) {
306 if (!str || len == 0) return CWIST_HTTP_UNKNOWN;
307
308 /* Ultra-fast path: SWAR 4-byte magic lookup for GET, POST, PUT, DELETE, HEAD */
309 if (len >= 3) {
310 uint32_t m = 0;
311 memcpy(&m, str, sizeof(uint32_t));
312 switch (m) {
313 case MAGIC_GET: return CWIST_HTTP_GET;
314 case MAGIC_POST: return CWIST_HTTP_POST;
315 case MAGIC_PUT: return CWIST_HTTP_PUT;
316 case MAGIC_DELE: if (len >= 6 && memcmp(str, "DELETE", 6) == 0) return CWIST_HTTP_DELETE; break;
317 case MAGIC_HEAD: if (len == 4) return CWIST_HTTP_HEAD; break;
318 default: break;
319 }
320 }
321
322 if (len == 3) {
323 uint32_t v = 0;
324 memcpy(&v, str, 3);
325 if ((v & 0x00FFFFFF) == 0x00544547) return CWIST_HTTP_GET;
326 if ((v & 0x00FFFFFF) == 0x00545550) return CWIST_HTTP_PUT;
327 } else if (len == 4) {
328 uint32_t v = 0;
329 memcpy(&v, str, 4);
330 if (v == 0x54534F50) return CWIST_HTTP_POST;
331 if (v == 0x44414548) return CWIST_HTTP_HEAD;
332 } else if (len == 5) {
333 if (memcmp(str, "PATCH", 5) == 0) return CWIST_HTTP_PATCH;
334 } else if (len == 6) {
335 if (memcmp(str, "DELETE", 6) == 0) return CWIST_HTTP_DELETE;
336 } else if (len == 7) {
337 if (memcmp(str, "OPTIONS", 7) == 0) return CWIST_HTTP_OPTIONS;
338 }
339 return CWIST_HTTP_UNKNOWN;
340}

666 if (now != atomic_load_explicit(&g_last_sec, memory_order_relaxed)) {
667 struct tm gmt;
668#if defined(_WIN32)
669 gmtime_s(&gmt, &now);
670#else
671 gmtime_r(&now, &gmt);
672#endif
673 strftime(g_date_str, sizeof(g_date_str), "%a, %d %b %Y %H:%M:%S GMT", &gmt);
674 atomic_store_explicit(&g_last_sec, now, memory_order_release);
675 }
676 pthread_mutex_unlock(&g_date_lock);
677 }
678 memcpy(out_buf, g_date_str, 30);
679 out_buf[29] = '\0';
680}

985static size_t serialize_headers(cwist_http_response *res, char *buf, size_t buf_size) {
986 size_t body_len = 0;
987 if (res->use_file_stream) {
988 body_len = res->file_stream_len;
989 } else if (res->is_ptr_body) {
990 body_len = res->ptr_body_len;
991 } else if (res->body) {
992 body_len = res->body->size;
993 }
994
995 /* Fast path for default HTTP/1.1 200 OK response with no custom headers */
996 if (res->status_code == 200 && !res->headers && !res->alt_svc && buf_size >= 128) {
997 /* Compact ring/line buffer optimization for high-throughput pipelining */
998 static const char status_prefix[] = "HTTP/1.1 200 OK\r\nContent-Length: ";
999 memcpy(buf, status_prefix, sizeof(status_prefix) - 1);
1000 size_t offset = sizeof(status_prefix) - 1;
1001
1002 /* Fast integer to ascii without snprintf overhead */
1003 char num_buf[20];
1004 char *p = num_buf + sizeof(num_buf);
1005 size_t tmp_len = body_len;
1006 do {
1007 *--p = '0' + (tmp_len % 10);
1008 tmp_len /= 10;
1009 } while (tmp_len > 0);
1010 size_t num_len = (num_buf + sizeof(num_buf)) - p;
1011 memcpy(buf + offset, p, num_len);
1012 offset += num_len;
1013
1014 if (res->keep_alive) {
1015 static const char conn_ka[] = "\r\nConnection: keep-alive\r\n\r\n";
1016 memcpy(buf + offset, conn_ka, sizeof(conn_ka) - 1);
1017 offset += sizeof(conn_ka) - 1;
1018 } else {
1019 static const char conn_cl[] = "\r\nConnection: close\r\n\r\n";
1020 memcpy(buf + offset, conn_cl, sizeof(conn_cl) - 1);
1021 offset += sizeof(conn_cl) - 1;
1022 }
1023 return offset;
1024 }
1025
1026 size_t offset = 0;
1027
1028 // Status Line
1029 const char *status_txt = (res->status_text && res->status_text->data)
1030 ? res->status_text->data
1031 : "OK";
1032 if (offset < buf_size) {
1033 int n = snprintf(buf + offset, buf_size - offset, "%s %d %s\r\n",
1034 res->version->data ? res->version->data : "HTTP/1.1",
1035 res->status_code,
1036 status_txt);
1037 if (n > 0) {
1038 offset += n;
1039 if (offset > buf_size) offset = buf_size;
1040 }
1041 }
1042
1043 // Headers: memcpy with known sstring lengths (no snprintf/strlen overhead),
1044 // detecting Date/Content-Length/Connection in the same single pass with a
1045 // length + first-byte filter before falling back to strcasecmp.
1046 bool have_date = false, have_clen = false, have_conn = false;
1047 cwist_http_header_node *curr = res->headers;
1048 while (curr) {
1049 if (curr->key->data && curr->value->data) {
1050 size_t klen = curr->key->size;
1051 size_t vlen = curr->value->size;
1052 if (offset + klen + 2 + vlen + 2 <= buf_size) {
1053 memcpy(buf + offset, curr->key->data, klen);
1054 buf[offset + klen] = ':';
1055 buf[offset + klen + 1] = ' ';
1056 memcpy(buf + offset + klen + 2, curr->value->data, vlen);
1057 buf[offset + klen + 2 + vlen] = '\r';
1058 buf[offset + klen + 2 + vlen + 1] = '\n';
1059 offset += klen + vlen + 4;
1060 }
1061 char k0 = curr->key->data[0];
1062 if (klen == 4 && (k0 == 'D' || k0 == 'd')) {
1063 if (strcasecmp(curr->key->data, "date") == 0) have_date = true;
1064 } else if (klen == 14 && (k0 == 'C' || k0 == 'c')) {
1065 if (strcasecmp(curr->key->data, "content-length") == 0) have_clen = true;
1066 } else if (klen == 10 && (k0 == 'C' || k0 == 'c')) {
1067 if (strcasecmp(curr->key->data, "connection") == 0) have_conn = true;
1068 }
1069 }
1070 curr = curr->next;
1071 }
1072
1073 if (!have_date && offset + 37 <= buf_size) {
1074 char date_str[36];
1075 cwist_get_cached_date_header(date_str); /* 29 bytes, NUL-terminated */
1076 memcpy(buf + offset, "Date: ", 6);
1077 memcpy(buf + offset + 6, date_str, 29);
1078 buf[offset + 35] = '\r';
1079 buf[offset + 36] = '\n';
1080 offset += 37;
1081 }
1082
1083 if (!have_clen) {
1084 /* Fast integer to ascii without snprintf overhead */
1085 char num_buf[20];
1086 char *p = num_buf + sizeof(num_buf);
1087 size_t tmp_len = body_len;
1088 do {
1089 *--p = '0' + (tmp_len % 10);
1090 tmp_len /= 10;
1091 } while (tmp_len > 0);
1092 size_t num_len = (size_t)((num_buf + sizeof(num_buf)) - p);
1093 if (offset + 16 + num_len + 2 <= buf_size) {
1094 memcpy(buf + offset, "Content-Length: ", 16);
1095 memcpy(buf + offset + 16, p, num_len);
1096 buf[offset + 16 + num_len] = '\r';
1097 buf[offset + 16 + num_len + 1] = '\n';
1098 offset += 16 + num_len + 2;
1099 }
1100 }
1101
1102 if (!have_conn) {
1103 if (res->keep_alive) {
1104 static const char conn_ka[] = "Connection: keep-alive\r\n";
1105 if (offset + sizeof(conn_ka) - 1 <= buf_size) {
1106 memcpy(buf + offset, conn_ka, sizeof(conn_ka) - 1);
1107 offset += sizeof(conn_ka) - 1;
1108 }
1109 } else {
1110 static const char conn_cl[] = "Connection: close\r\n";
1111 if (offset + sizeof(conn_cl) - 1 <= buf_size) {
1112 memcpy(buf + offset, conn_cl, sizeof(conn_cl) - 1);
1113 offset += sizeof(conn_cl) - 1;
1114 }
1115 }
1116 }
1117
1118 if (res->alt_svc) {
1119 if (offset < buf_size) {
1120 int n = snprintf(buf + offset, buf_size - offset, "Alt-Svc: %s\r\n", res->alt_svc);
1121 if (n > 0) {
1122 offset += n;
1123 if (offset > buf_size) offset = buf_size;
1124 }
1125 }
1126 }
1127
1128 if (offset + 2 <= buf_size) {
1129 buf[offset++] = '\r';
1130 buf[offset++] = '\n';
1131 }
1132 return offset;
1133}

1157static int cwist_http_sendmsg_all(int fd, struct iovec *iov, int iovcnt, int flags) {
1158 struct iovec *cur = iov;
1159 int curcnt = iovcnt;
1160 while (curcnt > 0) {
1161 struct msghdr msg = {0};
1162 msg.msg_iov = cur;
1163 msg.msg_iovlen = (size_t)curcnt;
1164 ssize_t n = sendmsg(fd, &msg, flags);
1165 if (n < 0) {
1166 if (errno == EINTR) continue;
1167 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1168 struct pollfd pfd = { .fd = fd, .events = POLLOUT };
1169 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
1170 if (ret <= 0) return -1;
1171 if (pfd.revents & (POLLERR | POLLHUP | POLLNVAL)) return -1;
1172 continue;
1173 }
1174 return -1;
1175 }
1176 if (n == 0) return -1;
1177
1178 while (curcnt > 0 && (size_t)n >= cur->iov_len) {
1179 n -= (ssize_t)cur->iov_len;
1180 cur++;
1181 curcnt--;
1182 }
1183 if (curcnt > 0) {
1184 cur->iov_base = (char *)cur->iov_base + n;
1185 cur->iov_len -= (size_t)n;
1186 }
1187 }
1188 return 0;

1197static bool cwist_http_stream_file_fast(int client_fd, cwist_http_response *res) {
1198 if (!res || !res->use_file_stream || res->file_stream_fd < 0) return false;
1199 size_t remaining = res->file_stream_len;
1200 off_t offset = res->file_stream_offset;
1201 while (remaining > 0) {
1202#if defined(__linux__)
1203 ssize_t sent = sendfile(client_fd, res->file_stream_fd, &offset, remaining);
1204 if (sent < 0) {
1205 if (errno == EINTR) continue;
1206 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1207 struct pollfd pfd = { .fd = client_fd, .events = POLLOUT };
1208 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
1209 if (ret <= 0 || (pfd.revents & (POLLERR | POLLHUP | POLLNVAL))) return false;
1210 continue;
1211 }
1212 return false;
1213 }
1214 if (sent == 0) break;
1215 remaining -= (size_t)sent;
1216#else
1217 /* Portable read+write fallback for macOS, FreeBSD, and other platforms.
1218 lseek is used to handle the offset since pread avoids modifying the
1219 file descriptor's position across calls. */
1220 char buf[65536];
1221 size_t to_read = remaining < sizeof(buf) ? remaining : sizeof(buf);
1222 ssize_t n = pread(res->file_stream_fd, buf, to_read, offset);
1223 if (n < 0) {
1224 if (errno == EINTR) continue;
1225 return false;
1226 }
1227 if (n == 0) break;
1228 ssize_t nw = 0;
1229 while (nw < n) {
1230 ssize_t w = write(client_fd, buf + nw, (size_t)(n - nw));
1231 if (w < 0) {
1232 if (errno == EINTR) continue;
1233 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1234 struct pollfd pfd = { .fd = client_fd, .events = POLLOUT };
1235 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
1236 if (ret <= 0 || (pfd.revents & (POLLERR | POLLHUP | POLLNVAL))) return false;
1237 continue;
1238 }
1239 return false;
1240 }
1241 nw += w;
1242 }
1243 offset += nw;
1244 remaining -= (size_t)nw;
1245#endif
1246 }
1247 res->file_stream_offset = offset;
1248 return remaining == 0;
1249}

1257cwist_error_t cwist_http_send_response(int client_fd, cwist_http_response *res) {
1258 cwist_error_t err = make_error(CWIST_ERR_INT16);
1259
1260 if (client_fd < 0 || !res) {
1261 err.error.err_i16 = -1;
1262 return err;
1263 }
1264
1265 // 1. Prepare Headers (On Stack)
1266 char header_buf[CWIST_HTTP_MAX_HEADER_SIZE];
1267 size_t header_len = serialize_headers(res, header_buf, sizeof(header_buf));
1268
1269 // 2. Prepare Body
1270 const void *body_ptr = NULL;
1271 size_t body_len = 0;
1272
1273 if (res->is_ptr_body) {
1274 body_ptr = res->ptr_body;
1275 body_len = res->ptr_body_len;
1276 } else if (res->body && res->body->data) {
1277 body_ptr = res->body->data;
1278 body_len = res->body->size;
1279 }
1280
1281 // 3. sendmsg (Scatter/Gather + Flags) - Zero Copy Send
1282 struct iovec iov[2];
1283 int iov_cnt = 1;
1284
1285 iov[0].iov_base = header_buf;
1286 iov[0].iov_len = header_len;
1287
1288 if (!res->use_file_stream && body_len > 0 && body_ptr) {
1289 iov[1].iov_base = (void*)body_ptr;
1290 iov[1].iov_len = body_len;
1291 iov_cnt = 2;
1292 }
1293
1294 int flags = 0;
1295 #if defined(MSG_NOSIGNAL)
1296 flags |= MSG_NOSIGNAL;
1297 #endif
1298 #if defined(MSG_DONTWAIT)
1299 flags |= MSG_DONTWAIT;
1300 #endif
1301
1302 if (cwist_http_sendmsg_all(client_fd, iov, iov_cnt, flags) != 0) {
1303 err.error.err_i16 = -1;
1304 } else {
1305 err.error.err_i16 = 0;
1306 if (res->use_file_stream) {
1307 if (!cwist_http_stream_file_fast(client_fd, res)) {
1308 err.error.err_i16 = -1;
1309 }
1310 }
1311 }
1312
1313 cwist_http_response_release_ptr_body(res);
1314 cwist_http_response_release_file_stream(res);
1315 return err;
1316}

1355 cwist_http_request *req = cwist_http_request_create();
1356 if (!req) return NULL;
1357
1358 // Fast path for root GET / HTTP/1.1\r\n
1359 if (line_start[0] == 'G' && line_start[1] == 'E' && line_start[2] == 'T' &&
1360 line_start[3] == ' ' && line_start[4] == '/' && line_start[5] == ' ' &&
1361 line_start[6] == 'H' && line_start[7] == 'T' && line_start[8] == 'T' &&
1362 line_start[9] == 'P' && line_start[10] == '/' && line_start[11] == '1' &&
1363 line_start[12] == '.' && line_start[13] == '1' && line_end == line_start + 14) {
1364 req->method = CWIST_HTTP_GET;
1365 cwist_sstring_borrow(req->path, "/", 1);
1366 cwist_sstring_borrow(req->query, "", 0);
1367 cwist_sstring_borrow(req->version, "HTTP/1.1", 8);
1368 req->keep_alive = true;
1369 } else {
1370 // 1. Request Line (Optimized: SIMD space search)
1371 const char *sp1 = cwist_simd_find_char(line_start, (size_t)(line_end - line_start), ' ');
1372 if (!sp1 || sp1 > line_end) { cwist_http_request_destroy(req); return NULL; }
1373 const char *sp2 = cwist_simd_find_char(sp1 + 1, (size_t)(line_end - (sp1 + 1)), ' ');
1374 if (!sp2 || sp2 > line_end) { cwist_http_request_destroy(req); return NULL; }
1375
1376 req->method = cwist_http_string_to_method_len(line_start, sp1 - line_start);
1377
1378 const char *path_start = sp1 + 1;
1379 const char *path_end = sp2;
1380 const char *query_sep = (const char *)memchr(path_start, '?', (size_t)(path_end - path_start));
1381
1382 if (query_sep) {
1383 cwist_http_sstring_assign_arena(req->path, (cwist_arena_t *)req->arena, path_start, (size_t)(query_sep - path_start));
1384 cwist_http_sstring_assign_arena(req->query, (cwist_arena_t *)req->arena, query_sep + 1, (size_t)(path_end - (query_sep + 1)));
1385 req->query_params = cwist_query_map_create_in_arena(req->arena);
1386 if (req->query_params) {
1387 cwist_query_map_parse(req->query_params, req->query->data);
1388 }
1389 } else {
1390 cwist_http_sstring_assign_arena(req->path, (cwist_arena_t *)req->arena, path_start, (size_t)(path_end - path_start));
1391 cwist_http_sstring_assign_arena(req->query, (cwist_arena_t *)req->arena, "", 0);
1392 }
1393
1394 cwist_http_sstring_assign_arena(req->version, (cwist_arena_t *)req->arena, sp2 + 1, (size_t)(line_end - (sp2 + 1)));
1395 if (strncmp(sp2 + 1, "HTTP/1.1", 8) == 0) {
1396 req->keep_alive = true;
1397 } else {
1398 req->keep_alive = false;
1399 }
1400 }
1401
1402 // 2. Headers (SIMD colon and CRLF scanning + SWAR)
1403 line_start = line_end + 2;
1404 while (line_start < header_end) {
1405 line_end = cwist_simd_find_crlf(line_start, (size_t)(header_end + 2 - line_start));
1406 if (!line_end || line_end == line_start) break;
1407
1408 const char *colon = cwist_simd_find_char(line_start, (size_t)(line_end - line_start), ':');
1409 if (colon) {
1410 size_t key_len = colon - line_start;
1411 const char *val_start = colon + 1;
1412 while (val_start < line_end && *val_start == ' ') val_start++;
1413 size_t val_len = line_end - val_start;
1414
1415 cwist_http_header_add_ex_len(&req->headers, (cwist_arena_t *)req->arena, line_start, key_len, val_start, val_len);
1416
1417 char k0 = line_start[0];
1418 if (key_len == 10 && (k0 == 'C' || k0 == 'c')) {
1419 uint64_t w0;
1420 memcpy(&w0, line_start, 8);
1421 if ((w0 | 0x2020202020202020ULL) == 0x697463656e6e6f63ULL) { /* "connecti" */
1422 uint16_t w1;
1423 memcpy(&w1, line_start + 8, 2);
1424 if ((w1 | 0x2020) == 0x6e6f) { /* "on" */
1425 if (val_len == 5 && (val_start[0] == 'c' || val_start[0] == 'C')) {
1426 if (strncasecmp(val_start, "close", 5) == 0) req->keep_alive = false;
1427 } else if (val_len == 10 && (val_start[0] == 'k' || val_start[0] == 'K')) {
1428 if (strncasecmp(val_start, "keep-alive", 10) == 0) req->keep_alive = true;
1429 }
1430 }
1431 }
1432 } else if (key_len == 14 && (k0 == 'C' || k0 == 'c')) {
1433 if (strncasecmp(line_start, "Content-Length", 14) == 0) {
1434 size_t len = 0;
1435 for (size_t i = 0; i < val_len; i++) {
1436 if (val_start[i] >= '0' && val_start[i] <= '9') {
1437 len = len * 10 + (val_start[i] - '0');
1438 } else {
1439 break;
1440 }
1441 }
1442 req->content_length = len;
1443 }
1444 }
1445 }
1446 line_start = line_end + 2;
1447 }
1448
1449 const char *body_start = header_end + 4;
1450 if (*body_start != '\0') {
1451 size_t available = strlen(body_start);
1452 size_t to_take = (req->content_length > 0 && req->content_length < available)
1453 ? req->content_length
1454 : available;
1455 cwist_http_sstring_assign_arena(req->body, (cwist_arena_t *)req->arena, body_start, to_take);
1456 }
1457
1458 return req;
1459}

1475static int http_parse_chunk_size(const char *line, size_t len, size_t *out_size) {
1476 size_t value = 0, digits = 0;
1477 for (size_t i = 0; i < len && line[i] != ';'; ++i) {
1478 unsigned char c = (unsigned char)line[i];
1479 unsigned int digit;
1480 if (c >= '0' && c <= '9') digit = c - '0';
1481 else if (c >= 'a' && c <= 'f') digit = c - 'a' + 10;
1482 else if (c >= 'A' && c <= 'F') digit = c - 'A' + 10;
1483 else return -1;
1484 if (value > (CWIST_HTTP_MAX_BODY_SIZE - digit) / 16) return -1;
1485 value = value * 16 + digit;
1486 ++digits;
1487 }
1488 if (!digits) return -1;
1489 *out_size = value;
1490 return 0;
1491}

1493static int http_read_chunked_body(int client_fd, char *buf, size_t *avail, size_t buf_cap, cwist_sstring *out) {
1494 size_t offset = 0;
1495
1496 while (1) {
1497 char *crlf = memmem(buf + offset, *avail - offset, "\r\n", 2);
1498 while (!crlf) {
1499 if (*avail >= buf_cap - 1) return -1;
1500 ssize_t bytes = recv(client_fd, buf + *avail, buf_cap - 1 - *avail, 0);
1501 if (bytes < 0) {
1502 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1503 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
1504 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
1505 if (ret <= 0) return -1;
1506 continue;
1507 }
1508 if (errno == EINTR) continue;
1509 return -1;
1510 }
1511 if (bytes == 0) return -1;
1512 *avail += (size_t)bytes;
1513 buf[*avail] = '\0';
1514 crlf = memmem(buf + offset, *avail - offset, "\r\n", 2);
1515 }
1516
1517 size_t line_len = (size_t)(crlf - (buf + offset)) + 2;
1518 size_t chunk_size = 0;
1519 /* Strict hexadecimal parsing prevents accepting contaminated framing
1520 * such as `4junk` or signed/overflowed chunk lengths. */
1521 if (http_parse_chunk_size(buf + offset, line_len - 2, &chunk_size) != 0) return -1;
1522 offset += line_len;
1523
1524 if (chunk_size == 0) {
1525 /* Consume optional trailers until empty line */
1526 while (offset + 1 < *avail && !(buf[offset] == '\r' && buf[offset + 1] == '\n')) {
1527 char *trailer_crlf = memmem(buf + offset, *avail - offset, "\r\n", 2);
1528 if (!trailer_crlf) {
1529 if (*avail >= buf_cap - 1) return -1;
1530 ssize_t bytes = recv(client_fd, buf + *avail, buf_cap - 1 - *avail, 0);
1531 if (bytes < 0) {
1532 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1533 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
1534 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
1535 if (ret <= 0) return -1;
1536 continue;
1537 }
1538 if (errno == EINTR) continue;
1539 return -1;
1540 }
1541 if (bytes == 0) return -1;
1542 *avail += (size_t)bytes;
1543 buf[*avail] = '\0';
1544 trailer_crlf = memmem(buf + offset, *avail - offset, "\r\n", 2);
1545 }
1546 if (trailer_crlf) {
1547 offset = (size_t)(trailer_crlf - buf) + 2;
1548 }
1549 }
1550 if (offset + 1 < *avail && buf[offset] == '\r' && buf[offset + 1] == '\n') {
1551 offset += 2;
1552 }
1553 break;
1554 }
1555
1556 if (chunk_size > CWIST_HTTP_MAX_BODY_SIZE || out->size + chunk_size > CWIST_HTTP_MAX_BODY_SIZE) return -1;
1557
1558 while (*avail - offset < chunk_size + 2) {
1559 if (*avail >= buf_cap - 1) return -1;
1560 ssize_t bytes = recv(client_fd, buf + *avail, buf_cap - 1 - *avail, 0);
1561 if (bytes < 0) {
1562 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1563 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
1564 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
1565 if (ret <= 0) return -1;
1566 continue;
1567 }
1568 if (errno == EINTR) continue;
1569 return -1;
1570 }
1571 if (bytes == 0) return -1;
1572 *avail += (size_t)bytes;
1573 buf[*avail] = '\0';
1574 }
1575
1576 if (buf[offset + chunk_size] != '\r' || buf[offset + chunk_size + 1] != '\n') return -1;
1577 if (cwist_sstring_append_len(out, buf + offset, chunk_size).error.err_i8 != 0) return -1;
1578 offset += chunk_size + 2;
1579 }
1580
1581 size_t leftover = *avail - offset;
1582 if (leftover > 0) {
1583 memmove(buf, buf + offset, leftover);
1584 }
1585 *avail = leftover;
1586 return 0;
1587}

1589cwist_http_request *cwist_http_receive_request(int client_fd, char *read_buf, size_t buf_size, size_t *buf_len) {
1590 size_t total_received = *buf_len;
1591 char *header_end = NULL;
1592
1593 // 1. Read until headers are complete
1594 while (!(header_end = (char *)cwist_simd_find_crlfcrlf(read_buf, total_received))) {
1595 if (total_received >= buf_size - 1) {
1596 return NULL;
1597 }
1598
1599 ssize_t bytes = recv(client_fd, read_buf + total_received, buf_size - 1 - total_received, 0);
1600 if (bytes <= 0) {
1601 if (bytes < 0 && errno == EINTR) continue;
1602 return NULL;
1603 }
1604 total_received += (size_t)bytes;
1605 read_buf[total_received] = '\0';
1606 }
1607
1608 cwist_http_request *req = cwist_http_parse_request_with_header_end(read_buf, header_end);
1609 if (!req) return NULL;
1610
1611 size_t header_len = (size_t)(header_end + 4 - read_buf);
1612 size_t body_received = total_received - header_len;
1613
1614 // 2. Read body based on Content-Length or Transfer-Encoding
1615 if (req->content_length > 0) {
1616 if (req->content_length > CWIST_HTTP_MAX_BODY_SIZE) {
1617 cwist_http_request_destroy(req);
1618 return NULL;
1619 }
1620
1621 // Allocate body
1622 char *body = cwist_alloc(req->content_length + 1);
1623 if (!body) {
1624 cwist_http_request_destroy(req);
1625 return NULL;
1626 }
1627
1628 size_t to_copy = (body_received < (size_t)req->content_length) ? body_received : (size_t)req->content_length;
1629 memcpy(body, header_end + 4, to_copy);
1630 size_t current_body_len = to_copy;
1631
1632 while (current_body_len < (size_t)req->content_length) {
1633 ssize_t bytes = recv(client_fd, body + current_body_len, (size_t)req->content_length - current_body_len, 0);
1634 if (bytes < 0) {
1635 if (errno == EAGAIN || errno == EWOULDBLOCK) {
1636 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
1637 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
1638 if (ret <= 0) {
1639 cwist_free(body);
1640 cwist_http_request_destroy(req);
1641 return NULL;
1642 }
1643 continue;
1644 }
1645 if (errno == EINTR) continue;
1646 cwist_free(body);
1647 cwist_http_request_destroy(req);
1648 return NULL;
1649 }
1650 if (bytes == 0) {
1651 cwist_free(body);
1652 cwist_http_request_destroy(req);
1653 return NULL;
1654 }
1655 current_body_len += (size_t)bytes;
1656 }
1657 body[req->content_length] = '\0';
1658 /* Adopt the filled buffer: one allocation, zero copies. The partial
1659 * body the parser staged in the arena is simply superseded. */
1660 cwist_sstring_adopt_len(req->body, body, (size_t)req->content_length);
1661
1662 // Calculate leftovers
1663 if (body_received > (size_t)req->content_length) {
1664 size_t leftover_len = body_received - (size_t)req->content_length;
1665 memmove(read_buf, header_end + 4 + req->content_length, leftover_len);
1666 *buf_len = leftover_len;
1667 } else {
1668 *buf_len = 0;
1669 }
1670 } else {
1671 const char *te = cwist_http_header_get(req->headers, "Transfer-Encoding");
1672 if (te && strcasecmp(te, "chunked") == 0) {
1673 if (body_received > 0) {
1674 memmove(read_buf, header_end + 4, body_received);
1675 }
1676 *buf_len = body_received;
1677 read_buf[*buf_len] = '\0';
1678
1679 cwist_sstring *chunked = cwist_sstring_create();
1680 if (!chunked) {
1681 cwist_http_request_destroy(req);
1682 return NULL;
1683 }
1684 if (http_read_chunked_body(client_fd, read_buf, buf_len, buf_size, chunked) != 0) {
1685 cwist_sstring_destroy(chunked);
1686 cwist_http_request_destroy(req);
1687 return NULL;
1688 }
1689 /* Move the assembled buffer into the request body instead of
1690 * copying it a second time. */
1691 char *chunked_data = chunked->data;
1692 size_t chunked_len = chunked->size;
1693 chunked->data = NULL;
1694 chunked->size = 0;
1695 cwist_sstring_destroy(chunked);
1696 cwist_sstring_adopt_len(req->body, chunked_data, chunked_len);
1697 } else {
1698 // No body, leftovers are everything after headers
1699 if (body_received > 0) {
1700 memmove(read_buf, header_end + 4, body_received);
1701 *buf_len = body_received;
1702 } else {
1703 *buf_len = 0;
1704 }
1705 }
1706 }
1707 read_buf[*buf_len] = '\0';
1708
1709 return req;
1710}

1756cwist_error_t cwist_http_response_send_file(cwist_http_response *res, const char *file_path, const char *content_type_hint, size_t *out_size) {
1757 cwist_error_t err = make_error(CWIST_ERR_INT16);
1758 if (!res || !file_path) {
1759 err.error.err_i16 = -EINVAL;
1760 return err;
1761 }
1762
1763 cwist_http_response_release_file_stream(res);
1764 cwist_http_response_release_ptr_body(res);
1765
1766 int fd = open(file_path, O_RDONLY);
1767 if (fd < 0) {
1768 err.error.err_i16 = -errno;
1769 return err;
1770 }
1771
1772 struct stat st;
1773 if (fstat(fd, &st) < 0) {
1774 err.error.err_i16 = -errno;
1775 close(fd);
1776 return err;
1777 }
1778
1779 if (!S_ISREG(st.st_mode)) {
1780 close(fd);
1781 err.error.err_i16 = -EISDIR;
1782 return err;
1783 }
1784
1785 bool endpoint_file = cwist_endpoint_has(res->endpoint_opts, CWIST_ENDPOINT_FILE);
1786
1787 if (!endpoint_file && (size_t)st.st_size > CWIST_HTTP_MAX_BODY_SIZE) {
1788 close(fd);
1789 err.error.err_i16 = -EFBIG;
1790 return err;
1791 }
1792
1793 size_t file_size = (size_t)st.st_size;
1794 bool use_fast_stream = false;
1795
1796 if (file_size > 0 && endpoint_file) {
1797#if defined(__linux__) || defined(__APPLE__) || defined(__FreeBSD__)
1798 res->use_file_stream = true;
1799 res->file_stream_fd = fd;
1800 res->file_stream_len = file_size;
1801 res->file_stream_offset = 0;
1802 res->file_stream_auto_close = true;
1803 use_fast_stream = true;
1804#endif
1805 }
1806
1807 char *buffer = NULL;
1808
1809 if (!use_fast_stream && file_size > 0) {
1810 buffer = (char *)cwist_alloc(file_size + 1);
1811 if (!buffer) {
1812 close(fd);
1813 err.error.err_i16 = -ENOMEM;
1814 return err;
1815 }
1816 }
1817
1818 if (!use_fast_stream) {
1819 size_t total_read = 0;
1820 while (total_read < file_size) {
1821 ssize_t bytes = read(fd, buffer + total_read, file_size - total_read);
1822 if (bytes < 0) {
1823 if (errno == EINTR) continue;
1824 err.error.err_i16 = -errno;
1825 cwist_free(buffer);
1826 close(fd);
1827 return err;
1828 }
1829 if (bytes == 0) {
1830 err.error.err_i16 = -EIO;
1831 cwist_free(buffer);
1832 close(fd);
1833 return err;
1834 }
1835 total_read += (size_t)bytes;
1836 }
1837 close(fd);
1838
1839 if (file_size > 0) {
1840 /* Adopt the read buffer as the body: no second file-size copy. */
1841 buffer[file_size] = '\0';
1842 cwist_sstring_adopt_len(res->body, buffer, file_size);
1843 } else {
1844 cwist_sstring_borrow(res->body, "", 0);
1845 }
1846 } else {
1847 cwist_sstring_borrow(res->body, "", 0);
1848 }
1849
1850 const char *mime = content_type_hint ? content_type_hint : cwist_guess_mime(file_path);
1851 if (mime && !cwist_http_header_get(res->headers, "Content-Type")) {
1852 cwist_http_header_add(&res->headers, "Content-Type", mime);
1853 }
1854
1855 if (out_size) {
1856 *out_size = file_size;
1857 }
1858
1859 res->status_code = CWIST_HTTP_OK;
1860 err.error.err_i16 = 0;
1861 return err;
1862}
