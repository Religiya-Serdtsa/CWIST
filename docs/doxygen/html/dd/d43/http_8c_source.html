65static void *http_pool_worker(void *arg) {
66 http_thread_worker_t *w = (http_thread_worker_t *)arg;
67 /* Bind this thread to the lattice worker ID for deterministic slot selection. */
68 ttak_net_lattice_set_worker_id(w->worker_id);
69
70 while (1) {
71 pthread_mutex_lock(&w->mutex);
72 while (w->count == 0 && !w->shutdown) {
73 pthread_cond_wait(&w->cond_not_empty, &w->mutex);
74 }
75 if (w->shutdown && w->count == 0) {
76 pthread_mutex_unlock(&w->mutex);
77 break;
78 }
79 http_pool_task_t task = w->queue[w->head];
80 w->head = (w->head + 1) % HTTP_TASKS_PER_THREAD;
81 w->count--;
82 pthread_cond_signal(&w->cond_not_full);
83 pthread_mutex_unlock(&w->mutex);
84
85 task.handler_func(task.client_fd, task.ctx);
86 }
87 return NULL;
88}

108static void http_pool_submit(int client_fd, void (*handler)(int, void *), void *ctx) {
109 /* Deterministic worker selection using Choi Seok-jeong's MOLS to minimize cache bouncing. */
110 uint16_t node_id = (uint16_t)(client_fd % TTAK_MOLS_NODE_COUNT);
111 uint32_t mixed = ttak_apply_mols_control(node_id, (uint32_t)g_rr_index);
112 size_t worker_idx = mixed % HTTP_THREAD_POOL_SIZE;
113
114 g_rr_index = (g_rr_index + 1) % HTTP_THREAD_POOL_SIZE;
115
116 http_thread_worker_t *w = &g_workers[worker_idx];
117
118 pthread_mutex_lock(&w->mutex);
119 while (w->count >= HTTP_TASKS_PER_THREAD && !w->shutdown) {
120 pthread_cond_wait(&w->cond_not_full, &w->mutex);
121 }
122 if (w->shutdown) {
123 pthread_mutex_unlock(&w->mutex);
124 close(client_fd);
125 return;
126 }
127 w->queue[w->tail].client_fd = client_fd;
128 w->queue[w->tail].handler_func = handler;
129 w->queue[w->tail].ctx = ctx;
130 w->tail = (w->tail + 1) % HTTP_TASKS_PER_THREAD;
131 w->count++;
132 pthread_cond_signal(&w->cond_not_empty);
133 pthread_mutex_unlock(&w->mutex);
134}

599static size_t serialize_headers(cwist_http_response *res, char *buf, size_t buf_size) {
600 size_t body_len = 0;
601 if (res->use_file_stream) {
602 body_len = res->file_stream_len;
603 } else if (res->is_ptr_body) {
604 body_len = res->ptr_body_len;
605 } else if (res->body) {
606 body_len = res->body->size;
607 }
608 int offset = 0;
609
610 // Status Line
611 offset += snprintf(buf + offset, buf_size - offset, "%s %d %s\r\n",
612 res->version->data ? res->version->data : "HTTP/1.1",
613 res->status_code,
614 res->status_text->data ? res->status_text->data : "OK");
615
616 // Headers
617 cwist_http_header_node *curr = res->headers;
618 while (curr) {
619 if (curr->key->data && curr->value->data) {
620 offset += snprintf(buf + offset, buf_size - offset, "%s: %s\r\n", curr->key->data, curr->value->data);
621 }
622 curr = curr->next;
623 }
624
625 if (!headers_have_content_length(res->headers)) {
626 offset += snprintf(buf + offset, buf_size - offset, "Content-Length: %zu\r\n", body_len);
627 }
628
629 if (!headers_have_connection(res->headers)) {
630 if (res->keep_alive) {
631 offset += snprintf(buf + offset, buf_size - offset, "Connection: keep-alive\r\n");
632 } else {
633 offset += snprintf(buf + offset, buf_size - offset, "Connection: close\r\n");
634 }
635 }
636
637 if (res->alt_svc) {
638 offset += snprintf(buf + offset, buf_size - offset, "Alt-Svc: %s\r\n", res->alt_svc);
639 }
640
641 offset += snprintf(buf + offset, buf_size - offset, "\r\n");
642 return offset;
643}

716cwist_error_t cwist_http_send_response(int client_fd, cwist_http_response *res) {
717 cwist_error_t err = make_error(CWIST_ERR_INT16);
718
719 if (client_fd < 0 || !res) {
720 err.error.err_i16 = -1;
721 return err;
722 }
723
724 // 1. Prepare Headers (On Stack)
725 char header_buf[CWIST_HTTP_MAX_HEADER_SIZE];
726 size_t header_len = serialize_headers(res, header_buf, sizeof(header_buf));
727
728 // 2. Prepare Body
729 const void *body_ptr = NULL;
730 size_t body_len = 0;
731
732 if (res->is_ptr_body) {
733 body_ptr = res->ptr_body;
734 body_len = res->ptr_body_len;
735 } else if (res->body && res->body->data) {
736 body_ptr = res->body->data;
737 body_len = res->body->size;
738 }
739
740 // 3. sendmsg (Scatter/Gather + Flags) - Zero Copy Send
741 struct msghdr msg = {0};
742 struct iovec iov[2];
743 int iov_cnt = 1;
744
745 iov[0].iov_base = header_buf;
746 iov[0].iov_len = header_len;
747
748 if (!res->use_file_stream && body_len > 0 && body_ptr) {
749 iov[1].iov_base = (void*)body_ptr;
750 iov[1].iov_len = body_len;
751 iov_cnt = 2;
752 }
753
754 msg.msg_iov = iov;
755 msg.msg_iovlen = iov_cnt;

804cwist_http_request *cwist_http_parse_request(const char *raw_request) {
805 if (!raw_request) return NULL;
806
807 cwist_http_request *req = cwist_http_request_create();
808 if (!req) return NULL;
809
810 const char *line_start = raw_request;
811 const char *header_end = strstr(raw_request, "\r\n\r\n");
812 if (!header_end) {
813 cwist_http_request_destroy(req);
814 return NULL;
815 }
816
817 const char *line_end = strstr(line_start, "\r\n");
818 if (!line_end || line_end > header_end) {
819 cwist_http_request_destroy(req);
820 return NULL;
821 }
822
823 // 1. Request Line (Optimized: No intermediate allocation)
824 const char *sp1 = strchr(line_start, ' ');
825 if (!sp1 || sp1 > line_end) { cwist_http_request_destroy(req); return NULL; }
826 const char *sp2 = strchr(sp1 + 1, ' ');
827 if (!sp2 || sp2 > line_end) { cwist_http_request_destroy(req); return NULL; }
828
829 req->method = cwist_http_string_to_method_len(line_start, sp1 - line_start);
830
831 const char *path_start = sp1 + 1;
832 const char *path_end = sp2;
833 const char *query_sep = memchr(path_start, '?', path_end - path_start);
834
835 if (query_sep) {
836 cwist_sstring_assign_len(req->path, path_start, query_sep - path_start);
837 cwist_sstring_assign_len(req->query, query_sep + 1, path_end - (query_sep + 1));
838 cwist_query_map_parse(req->query_params, req->query->data);
839 } else {
840 cwist_sstring_assign_len(req->path, path_start, path_end - path_start);
841 cwist_sstring_assign_len(req->query, "", 0);
842 }
843
844 cwist_sstring_assign_len(req->version, sp2 + 1, line_end - (sp2 + 1));
845 if (strncmp(sp2 + 1, "HTTP/1.1", 8) == 0) {
846 req->keep_alive = true;
847 } else {
848 req->keep_alive = false;
849 }
850
851 // 2. Headers (Optimized: Minimal copies)
852 line_start = line_end + 2;
853 while (line_start < header_end) {
854 line_end = strstr(line_start, "\r\n");
855 if (!line_end || line_end == line_start) break;
856
857 const char *colon = memchr(line_start, ':', line_end - line_start);
858 if (colon) {
859 int key_len = colon - line_start;
860 const char *val_start = colon + 1;
861 while (val_start < line_end && *val_start == ' ') val_start++;
862 int val_len = line_end - val_start;
863
864 // Temporary NUL termination for legacy header_add
865 char key_tmp[256];
866 char val_tmp[1024];
867 if (key_len < 256 && val_len < 1024) {
868 memcpy(key_tmp, line_start, key_len); key_tmp[key_len] = '\0';
869 memcpy(val_tmp, val_start, val_len); val_tmp[val_len] = '\0';
870
871 cwist_http_header_add(&req->headers, key_tmp, val_tmp);
872
873 if (strcasecmp(key_tmp, "Connection") == 0) {
874 if (strcasecmp(val_tmp, "close") == 0) req->keep_alive = false;
875 else if (strcasecmp(val_tmp, "keep-alive") == 0) req->keep_alive = true;
876 } else if (strcasecmp(key_tmp, "Content-Length") == 0) {
877 req->content_length = (size_t)atoll(val_tmp);
878 }
879 }
880 }
881 line_start = line_end + 2;
882 }
883
884 const char *body_start = header_end + 4;
885 if (*body_start != '\0') {
886 cwist_sstring_assign(req->body, (char*)body_start);
887 }
888
889 return req;
890}

902cwist_http_request *cwist_http_receive_request(int client_fd, char *read_buf, size_t buf_size, size_t *buf_len) {
903 size_t total_received = *buf_len;
904 char *header_end = NULL;
905
906 // 1. Read until headers are complete
907 while (!(header_end = strstr(read_buf, "\r\n\r\n"))) {
908 if (total_received >= buf_size - 1) {
909 // Buffer full, but headers not complete
910 return NULL;
911 }
912
913 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
914 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
915 if (ret <= 0) return NULL; // Timeout or error
916
917 ssize_t bytes = recv(client_fd, read_buf + total_received, buf_size - 1 - total_received, 0);
918 if (bytes <= 0) return NULL;
919 total_received += (size_t)bytes;
920 read_buf[total_received] = '\0';
921 }
922
923 cwist_http_request *req = cwist_http_parse_request(read_buf);
924 if (!req) return NULL;
925
926 size_t header_len = (header_end + 4) - read_buf;
927 size_t body_received = total_received - header_len;
928
929 // 2. Read body based on Content-Length
930 if (req->content_length > 0) {
931 if (req->content_length > CWIST_HTTP_MAX_BODY_SIZE) {
932 cwist_http_request_destroy(req);
933 return NULL;
934 }
935
936 // Allocate body
937 char *body = cwist_alloc(req->content_length + 1);
938 if (!body) {
939 cwist_http_request_destroy(req);
940 return NULL;
941 }
942
943 size_t to_copy = (body_received < req->content_length) ? body_received : req->content_length;
944 memcpy(body, header_end + 4, to_copy);
945 size_t current_body_len = to_copy;
946
947 while (current_body_len < req->content_length) {
948 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
949 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
950 if (ret <= 0) {
951 cwist_free(body);
952 cwist_http_request_destroy(req);
953 return NULL;
954 }
955
956 ssize_t bytes = recv(client_fd, body + current_body_len, req->content_length - current_body_len, 0);
957 if (bytes <= 0) {
958 cwist_free(body);
959 cwist_http_request_destroy(req);
960 return NULL;
961 }
962 current_body_len += (size_t)bytes;
963 }
964 body[req->content_length] = '\0';
965 cwist_sstring_assign_len(req->body, body, req->content_length);
966 cwist_free(body);
967
968 // Calculate leftovers
969 if (body_received > req->content_length) {
970 size_t leftover_len = body_received - req->content_length;
971 memmove(read_buf, header_end + 4 + req->content_length, leftover_len);
972 *buf_len = leftover_len;
973 } else {
974 *buf_len = 0;
975 }
976 } else {
977 // No body, leftovers are everything after headers
978 if (body_received > 0) {
979 memmove(read_buf, header_end + 4, body_received);
980 *buf_len = body_received;
981 } else {
982 *buf_len = 0;
983 }
984 }
985 read_buf[*buf_len] = '\0';
986
987 return req;
988}

1119 cwist_free(buffer);
1120 } else {
1121 cwist_sstring_assign(res->body, "");

1129 cwist_http_header_add(&res->headers, "Content-Type", mime);

1131
1132 if (out_size) {
1133 *out_size = file_size;
1134 }
1135

1341cwist_error_t cwist_http_server_loop(int server_fd, cwist_server_config *config, void (*handler)(int, void *), void *ctx) {
1342 cwist_error_t err = make_error(CWIST_ERR_INT16);
1343 if (!config || server_fd < 0 || !handler) {
1344 err.error.err_i16 = -1;
1345 return err;
1346 }
1347
1348 if (config->use_forking) {
1349 while (atomic_load(&g_cwist_running)) {
1350 int client_fd = accept(server_fd, NULL, NULL);
1351 if (client_fd < 0) {
1352 int accept_err = errno;
1353 if (accept_err == EBADF || accept_err == EINVAL) break;
1354 if (cwist_accept_error_should_retry(accept_err)) {
1355 cwist_accept_error_backoff(accept_err);
1356 continue;
1357 }
1358 err.error.err_i16 = -1;

1361 handle_client_forking(client_fd, handler, ctx);

1363 }

1367 err.error.err_i16 = -1;
1368 return err;
1369 }
1370 while (atomic_load(&g_cwist_running)) {
1371 int client_fd = accept(server_fd, NULL, NULL);
1372 if (client_fd < 0) {
1373 int accept_err = errno;
1374 if (accept_err == EBADF || accept_err == EINVAL) break;

1376 cwist_accept_error_backoff(accept_err);
1377 continue;
1378 }
1379 err.error.err_i16 = -1;

1425 if (accept_err == EINTR) continue;

1427 cwist_accept_error_backoff(accept_err);

1435 }
1436 }
