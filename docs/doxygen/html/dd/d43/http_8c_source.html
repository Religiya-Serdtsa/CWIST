65static void *http_pool_worker(void *arg) {
66 http_thread_worker_t *w = (http_thread_worker_t *)arg;
67 /* Bind this thread to the lattice worker ID for deterministic slot selection. */
68 ttak_net_lattice_set_worker_id(w->worker_id);
69
70 while (1) {
71 pthread_mutex_lock(&w->mutex);
72 while (w->count == 0 && !w->shutdown) {
73 pthread_cond_wait(&w->cond_not_empty, &w->mutex);
74 }
75 if (w->shutdown && w->count == 0) {
76 pthread_mutex_unlock(&w->mutex);
77 break;
78 }
79 http_pool_task_t task = w->queue[w->head];
80 w->head = (w->head + 1) % HTTP_TASKS_PER_THREAD;
81 w->count--;
82 pthread_cond_signal(&w->cond_not_full);
83 pthread_mutex_unlock(&w->mutex);
84
85 task.handler_func(task.client_fd, task.ctx);
86 }
87 return NULL;
88}

91 g_rr_index = 0;
92 memset(g_workers, 0, sizeof(g_workers));
93 for (int i = 0; i < HTTP_THREAD_POOL_SIZE; i++) {
94 g_workers[i].queue = cwist_alloc(HTTP_TASKS_PER_THREAD * sizeof(http_pool_task_t));

108static void http_pool_submit(int client_fd, void (*handler)(int, void *), void *ctx) {
109 /* Deterministic worker selection using Choi Seok-jeong's MOLS to minimize cache bouncing. */
110 uint16_t node_id = (uint16_t)(client_fd % TTAK_MOLS_NODE_COUNT);
111 uint32_t mixed = ttak_apply_mols_control(node_id, (uint32_t)g_rr_index);
112 size_t worker_idx = mixed % HTTP_THREAD_POOL_SIZE;
113
114 g_rr_index = (g_rr_index + 1) % HTTP_THREAD_POOL_SIZE;
115
116 http_thread_worker_t *w = &g_workers[worker_idx];
117
118 pthread_mutex_lock(&w->mutex);
119 while (w->count >= HTTP_TASKS_PER_THREAD && !w->shutdown) {
120 pthread_cond_wait(&w->cond_not_full, &w->mutex);
121 }
122 if (w->shutdown) {
123 pthread_mutex_unlock(&w->mutex);
124 close(client_fd);
125 return;
126 }
127 w->queue[w->tail].client_fd = client_fd;
128 w->queue[w->tail].handler_func = handler;
129 w->queue[w->tail].ctx = ctx;
130 w->tail = (w->tail + 1) % HTTP_TASKS_PER_THREAD;
131 w->count++;
132 pthread_cond_signal(&w->cond_not_empty);
133 pthread_mutex_unlock(&w->mutex);
134}
135static void http_pool_destroy(void) {
136 for (int i = 0; i < HTTP_THREAD_POOL_SIZE; i++) {
137 pthread_mutex_lock(&g_workers[i].mutex);
138 g_workers[i].shutdown = 1;
139 pthread_cond_broadcast(&g_workers[i].cond_not_empty);
140 pthread_mutex_unlock(&g_workers[i].mutex);

143 pthread_join(g_workers[i].thread, NULL);

148 cwist_free(g_workers[i].queue);

433 // If unavailable, return localhost

635static size_t serialize_headers(cwist_http_response *res, char *buf, size_t buf_size) {
636 size_t body_len = 0;
637 if (res->use_file_stream) {
638 body_len = res->file_stream_len;
639 } else if (res->is_ptr_body) {
640 body_len = res->ptr_body_len;
641 } else if (res->body) {
642 body_len = res->body->size;
643 }
644 int offset = 0;
645
646 // Status Line
647 offset += snprintf(buf + offset, buf_size - offset, "%s %d %s\r\n",
648 res->version->data ? res->version->data : "HTTP/1.1",
649 res->status_code,
650 res->status_text->data ? res->status_text->data : "OK");
651
652 // Headers
653 cwist_http_header_node *curr = res->headers;
654 while (curr) {
655 if (curr->key->data && curr->value->data) {
656 offset += snprintf(buf + offset, buf_size - offset, "%s: %s\r\n", curr->key->data, curr->value->data);
657 }
658 curr = curr->next;
659 }
660
661 if (!headers_have_content_length(res->headers)) {
662 offset += snprintf(buf + offset, buf_size - offset, "Content-Length: %zu\r\n", body_len);
663 }
664
665 if (!headers_have_connection(res->headers)) {
666 if (res->keep_alive) {
667 offset += snprintf(buf + offset, buf_size - offset, "Connection: keep-alive\r\n");
668 } else {
669 offset += snprintf(buf + offset, buf_size - offset, "Connection: close\r\n");
670 }
671 }
672
673 if (res->alt_svc) {
674 offset += snprintf(buf + offset, buf_size - offset, "Alt-Svc: %s\r\n", res->alt_svc);
675 }
676
677 offset += snprintf(buf + offset, buf_size - offset, "\r\n");
678 return offset;
679}

689static bool cwist_http_stream_file_fast(int client_fd, cwist_http_response *res) {
690#if defined(__linux__) || defined(__APPLE__) || defined(__FreeBSD__)
691 if (!res || !res->use_file_stream || res->file_stream_fd < 0) return false;
692 size_t remaining = res->file_stream_len;
693 off_t offset = res->file_stream_offset;
694 while (remaining > 0) {
695#if defined(__linux__)
696 ssize_t sent = sendfile(client_fd, res->file_stream_fd, &offset, remaining);
697 if (sent < 0) {
698 if (errno == EINTR || errno == EAGAIN) continue;
699 return false;
700 }
701 if (sent == 0) break;
702 remaining -= (size_t)sent;
703#elif defined(__APPLE__)
704 off_t chunk = (off_t)remaining;
705 int rc = sendfile(res->file_stream_fd, client_fd, offset, &chunk, NULL, 0);
706 if (rc == -1) {
707 if (errno == EINTR || errno == EAGAIN) {
708 if (chunk == 0) continue;
709 offset += chunk;
710 remaining -= (size_t)chunk;
711 continue;
712 }
713 return false;
714 }
715 if (chunk == 0) break;
716 offset += chunk;
717 remaining -= (size_t)chunk;
718#elif defined(__FreeBSD__)
719 off_t sent = 0;
720 size_t chunk = remaining;
721 int rc = sendfile(res->file_stream_fd, client_fd, offset, chunk, NULL, &sent, 0);
722 if (rc == -1) {
723 if (errno == EINTR || errno == EAGAIN) {
724 if (sent == 0) continue;
725 offset += sent;
726 remaining -= (size_t)sent;
727 continue;
728 }

730 }
731 if (sent == 0) break;
732 offset += sent;
733 remaining -= (size_t)sent;
734#endif

738#else

752cwist_error_t cwist_http_send_response(int client_fd, cwist_http_response *res) {
753 cwist_error_t err = make_error(CWIST_ERR_INT16);
754
755 if (client_fd < 0 || !res) {
756 err.error.err_i16 = -1;
757 return err;
758 }
759
760 // 1. Prepare Headers (On Stack)
761 char header_buf[CWIST_HTTP_MAX_HEADER_SIZE];
762 size_t header_len = serialize_headers(res, header_buf, sizeof(header_buf));
763
764 // 2. Prepare Body
765 const void *body_ptr = NULL;
766 size_t body_len = 0;
767
768 if (res->is_ptr_body) {
769 body_ptr = res->ptr_body;
770 body_len = res->ptr_body_len;
771 } else if (res->body && res->body->data) {
772 body_ptr = res->body->data;
773 body_len = res->body->size;
774 }
775
776 // 3. sendmsg (Scatter/Gather + Flags) - Zero Copy Send
777 struct msghdr msg = {0};
778 struct iovec iov[2];
779 int iov_cnt = 1;
780
781 iov[0].iov_base = header_buf;
782 iov[0].iov_len = header_len;
783
784 if (!res->use_file_stream && body_len > 0 && body_ptr) {
785 iov[1].iov_base = (void*)body_ptr;
786 iov[1].iov_len = body_len;
787 iov_cnt = 2;
788 }
789
790 msg.msg_iov = iov;
791 msg.msg_iovlen = iov_cnt;
792
793 int flags = 0;
794 #if defined(MSG_NOSIGNAL)
795 flags = MSG_NOSIGNAL;
796 #endif
797
798 ssize_t written = sendmsg(client_fd, &msg, flags);
799 if (written < 0) {
800 err.error.err_i16 = -1;
801 } else {
802 err.error.err_i16 = 0;
803 if (res->use_file_stream) {
804 if (!cwist_http_stream_file_fast(client_fd, res)) {
805 err.error.err_i16 = -1;
806 }
807 }
808 }
809
810 cwist_http_response_release_ptr_body(res);
811 cwist_http_response_release_file_stream(res);
812 return err;
813}

840cwist_http_request *cwist_http_parse_request(const char *raw_request) {
841 if (!raw_request) return NULL;
842
843 cwist_http_request *req = cwist_http_request_create();
844 if (!req) return NULL;
845
846 const char *line_start = raw_request;
847 const char *header_end = strstr(raw_request, "\r\n\r\n");
848 if (!header_end) {
849 cwist_http_request_destroy(req);
850 return NULL;
851 }
852
853 const char *line_end = strstr(line_start, "\r\n");
854 if (!line_end || line_end > header_end) {
855 cwist_http_request_destroy(req);
856 return NULL;
857 }
858
859 // 1. Request Line (Optimized: No intermediate allocation)
860 const char *sp1 = strchr(line_start, ' ');
861 if (!sp1 || sp1 > line_end) { cwist_http_request_destroy(req); return NULL; }
862 const char *sp2 = strchr(sp1 + 1, ' ');
863 if (!sp2 || sp2 > line_end) { cwist_http_request_destroy(req); return NULL; }
864
865 req->method = cwist_http_string_to_method_len(line_start, sp1 - line_start);
866
867 const char *path_start = sp1 + 1;
868 const char *path_end = sp2;
869 const char *query_sep = memchr(path_start, '?', path_end - path_start);
870
871 if (query_sep) {
872 cwist_sstring_assign_len(req->path, path_start, query_sep - path_start);
873 cwist_sstring_assign_len(req->query, query_sep + 1, path_end - (query_sep + 1));
874 cwist_query_map_parse(req->query_params, req->query->data);
875 } else {
876 cwist_sstring_assign_len(req->path, path_start, path_end - path_start);
877 cwist_sstring_assign_len(req->query, "", 0);
878 }
879
880 cwist_sstring_assign_len(req->version, sp2 + 1, line_end - (sp2 + 1));
881 if (strncmp(sp2 + 1, "HTTP/1.1", 8) == 0) {
882 req->keep_alive = true;
883 } else {
884 req->keep_alive = false;
885 }
886
887 // 2. Headers (Optimized: Minimal copies)
888 line_start = line_end + 2;
889 while (line_start < header_end) {
890 line_end = strstr(line_start, "\r\n");
891 if (!line_end || line_end == line_start) break;
892
893 const char *colon = memchr(line_start, ':', line_end - line_start);
894 if (colon) {
895 int key_len = colon - line_start;
896 const char *val_start = colon + 1;
897 while (val_start < line_end && *val_start == ' ') val_start++;
898 int val_len = line_end - val_start;
899
900 // Temporary NUL termination for legacy header_add
901 char key_tmp[256];
902 char val_tmp[1024];
903 if (key_len < 256 && val_len < 1024) {
904 memcpy(key_tmp, line_start, key_len); key_tmp[key_len] = '\0';
905 memcpy(val_tmp, val_start, val_len); val_tmp[val_len] = '\0';
906
907 cwist_http_header_add(&req->headers, key_tmp, val_tmp);
908
909 if (strcasecmp(key_tmp, "Connection") == 0) {
910 if (strcasecmp(val_tmp, "close") == 0) req->keep_alive = false;
911 else if (strcasecmp(val_tmp, "keep-alive") == 0) req->keep_alive = true;
912 } else if (strcasecmp(key_tmp, "Content-Length") == 0) {
913 req->content_length = (size_t)atoll(val_tmp);
914 }
915 }
916 }
917 line_start = line_end + 2;
918 }
919
920 const char *body_start = header_end + 4;
921 if (*body_start != '\0') {
922 cwist_sstring_assign(req->body, (char*)body_start);
923 }
924
925 return req;
926}

938cwist_http_request *cwist_http_receive_request(int client_fd, char *read_buf, size_t buf_size, size_t *buf_len) {
939 size_t total_received = *buf_len;
940 char *header_end = NULL;
941
942 // 1. Read until headers are complete
943 while (!(header_end = strstr(read_buf, "\r\n\r\n"))) {
944 if (total_received >= buf_size - 1) {
945 // Buffer full, but headers not complete
946 return NULL;
947 }
948
949 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
950 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
951 if (ret <= 0) return NULL; // Timeout or error
952
953 ssize_t bytes = recv(client_fd, read_buf + total_received, buf_size - 1 - total_received, 0);
954 if (bytes <= 0) return NULL;
955 total_received += (size_t)bytes;
956 read_buf[total_received] = '\0';
957 }
958
959 cwist_http_request *req = cwist_http_parse_request(read_buf);
960 if (!req) return NULL;
961
962 size_t header_len = (header_end + 4) - read_buf;
963 size_t body_received = total_received - header_len;
964
965 // 2. Read body based on Content-Length
966 if (req->content_length > 0) {
967 if (req->content_length > CWIST_HTTP_MAX_BODY_SIZE) {
968 cwist_http_request_destroy(req);
969 return NULL;
970 }
971
972 // Allocate body
973 char *body = cwist_alloc(req->content_length + 1);
974 if (!body) {
975 cwist_http_request_destroy(req);
976 return NULL;
977 }
978
979 size_t to_copy = (body_received < req->content_length) ? body_received : req->content_length;
980 memcpy(body, header_end + 4, to_copy);
981 size_t current_body_len = to_copy;
982
983 while (current_body_len < req->content_length) {
984 struct pollfd pfd = { .fd = client_fd, .events = POLLIN };
985 int ret = poll(&pfd, 1, CWIST_HTTP_TIMEOUT_MS);
986 if (ret <= 0) {
987 cwist_free(body);
988 cwist_http_request_destroy(req);
989 return NULL;
990 }
991
992 ssize_t bytes = recv(client_fd, body + current_body_len, req->content_length - current_body_len, 0);
993 if (bytes <= 0) {
994 cwist_free(body);
995 cwist_http_request_destroy(req);
996 return NULL;
997 }
998 current_body_len += (size_t)bytes;
999 }
1000 body[req->content_length] = '\0';
1001 cwist_sstring_assign_len(req->body, body, req->content_length);
1002 cwist_free(body);
1003
1004 // Calculate leftovers
1005 if (body_received > req->content_length) {
1006 size_t leftover_len = body_received - req->content_length;
1007 memmove(read_buf, header_end + 4 + req->content_length, leftover_len);
1008 *buf_len = leftover_len;
1009 } else {
1010 *buf_len = 0;
1011 }
1012 } else {
1013 // No body, leftovers are everything after headers
1014 if (body_received > 0) {
1015 memmove(read_buf, header_end + 4, body_received);
1016 *buf_len = body_received;
1017 } else {
1018 *buf_len = 0;
1019 }
1020 }
1021 read_buf[*buf_len] = '\0';
1022
1023 return req;
1024}

1070cwist_error_t cwist_http_response_send_file(cwist_http_response *res, const char *file_path, const char *content_type_hint, size_t *out_size) {
1071 cwist_error_t err = make_error(CWIST_ERR_INT16);
1072 if (!res || !file_path) {
1073 err.error.err_i16 = -EINVAL;
1074 return err;
1075 }
1076
1077 cwist_http_response_release_file_stream(res);
1078 cwist_http_response_release_ptr_body(res);
1079
1080 int fd = open(file_path, O_RDONLY);
1081 if (fd < 0) {
1082 err.error.err_i16 = -errno;
1083 return err;
1084 }
1085
1086 struct stat st;
1087 if (fstat(fd, &st) < 0) {
1088 err.error.err_i16 = -errno;
1089 close(fd);
1090 return err;
1091 }
1092
1093 if (!S_ISREG(st.st_mode)) {
1094 close(fd);
1095 err.error.err_i16 = -EISDIR;
1096 return err;
1097 }
1098
1099 bool endpoint_file = cwist_endpoint_has(res->endpoint_opts, CWIST_ENDPOINT_FILE);
1100
1101 if (!endpoint_file && (size_t)st.st_size > CWIST_HTTP_MAX_BODY_SIZE) {
1102 close(fd);
1103 err.error.err_i16 = -EFBIG;
1104 return err;
1105 }
1106
1107 size_t file_size = (size_t)st.st_size;
1108 bool use_fast_stream = false;
1109
1110 if (file_size > 0 && endpoint_file) {
1111#if defined(__linux__) || defined(__APPLE__) || defined(__FreeBSD__)
1112 res->use_file_stream = true;
1113 res->file_stream_fd = fd;
1114 res->file_stream_len = file_size;
1115 res->file_stream_offset = 0;
1116 res->file_stream_auto_close = true;
1117 use_fast_stream = true;
1118#endif
1119 }
1120
1121 char *buffer = NULL;
1122
1123 if (!use_fast_stream && file_size > 0) {
1124 buffer = (char *)cwist_alloc(file_size);
1125 if (!buffer) {
1126 close(fd);
1127 err.error.err_i16 = -ENOMEM;
1128 return err;
1129 }
1130 }
1131
1132 if (!use_fast_stream) {
1133 size_t total_read = 0;
1134 while (total_read < file_size) {
1135 ssize_t bytes = read(fd, buffer + total_read, file_size - total_read);
1136 if (bytes < 0) {
1137 if (errno == EINTR) continue;
1138 err.error.err_i16 = -errno;
1139 cwist_free(buffer);
1140 close(fd);
1141 return err;
1142 }
1143 if (bytes == 0) {
1144 err.error.err_i16 = -EIO;
1145 cwist_free(buffer);
1146 close(fd);
1147 return err;
1148 }
1149 total_read += (size_t)bytes;
1150 }
1151 close(fd);
1152
1153 if (file_size > 0) {
1154 cwist_sstring_assign_len(res->body, buffer, file_size);
1155 cwist_free(buffer);
1156 } else {
1157 cwist_sstring_assign(res->body, "");
1158 }
1159 } else {
1160 cwist_sstring_assign(res->body, "");
1161 }
1162
1163 const char *mime = content_type_hint ? content_type_hint : cwist_guess_mime(file_path);
1164 if (mime && !cwist_http_header_get(res->headers, "Content-Type")) {
1165 cwist_http_header_add(&res->headers, "Content-Type", mime);
1166 }
1167
1168 if (out_size) {
1169 *out_size = file_size;
1170 }
1171
1172 res->status_code = CWIST_HTTP_OK;
1173 err.error.err_i16 = 0;
1174 return err;
1175}

1377cwist_error_t cwist_http_server_loop(int server_fd, cwist_server_config *config, void (*handler)(int, void *), void *ctx) {
1378 cwist_error_t err = make_error(CWIST_ERR_INT16);
1379 if (!config || server_fd < 0 || !handler) {
1380 err.error.err_i16 = -1;
1381 return err;
1382 }
1383
1384 if (config->use_forking) {
1385 while (atomic_load(&g_cwist_running)) {
1386 int client_fd = accept(server_fd, NULL, NULL);
1387 if (client_fd < 0) {
1388 int accept_err = errno;
1389 if (accept_err == EBADF || accept_err == EINVAL) break;
1390 if (cwist_accept_error_should_retry(accept_err)) {
1391 cwist_accept_error_backoff(accept_err);
1392 continue;
1393 }
1394 err.error.err_i16 = -1;
1395 return err;
1396 }
1397 handle_client_forking(client_fd, handler, ctx);
1398 }
1399 }
1400
1401 if (config->use_threading) {
1402 if (http_pool_init() != 0) {
1403 err.error.err_i16 = -1;
1404 return err;
1405 }
1406 while (atomic_load(&g_cwist_running)) {
1407 int client_fd = accept(server_fd, NULL, NULL);
1408 if (client_fd < 0) {
1409 int accept_err = errno;
1410 if (accept_err == EBADF || accept_err == EINVAL) break;
1411 if (cwist_accept_error_should_retry(accept_err)) {
1412 cwist_accept_error_backoff(accept_err);
1413 continue;
1414 }
1415 err.error.err_i16 = -1;
1416 http_pool_destroy();
1417 return err;
1418 }
1419 http_pool_submit(client_fd, handler, ctx);
1420 }
1421 http_pool_destroy();
1422 }

1424#ifdef __linux__
1425 if (config->use_epoll) {
1426 int flags = fcntl(server_fd, F_GETFL, 0);
1427 if (flags >= 0) fcntl(server_fd, F_SETFL, flags | O_NONBLOCK);
1428
1429 int epoll_fd = epoll_create1(0);
1430 if (epoll_fd < 0) {
1431 err.error.err_i16 = -1;

1433 }
