83static void heap_push(cwist_scheduler_t *s, scheduled_job_t job) {
84 if (heap_grow(s) != 0) return;
85 size_t i = s->heap_count++;
86 s->heap[i] = job;
87 while (i > 0) {
88 size_t parent = (i - 1) / 2;
89 if (s->heap[parent].deadline_ms <= s->heap[i].deadline_ms) break;
90 heap_swap(&s->heap[parent], &s->heap[i]);
91 i = parent;
92 }
93}

95static scheduled_job_t heap_pop(cwist_scheduler_t *s) {
96 scheduled_job_t root = s->heap[0];
97 s->heap[0] = s->heap[--s->heap_count];
98 size_t i = 0;
99 while (1) {
100 size_t left = i * 2 + 1;
101 size_t right = left + 1;
102 size_t smallest = i;
103 if (left < s->heap_count && s->heap[left].deadline_ms < s->heap[smallest].deadline_ms) {
104 smallest = left;
105 }
106 if (right < s->heap_count && s->heap[right].deadline_ms < s->heap[smallest].deadline_ms) {
107 smallest = right;
108 }
109 if (smallest == i) break;
110 heap_swap(&s->heap[i], &s->heap[smallest]);
111 i = smallest;
112 }
113 return root;
114}

124static void *scheduler_timer_thread(void *arg) {
125 cwist_scheduler_t *s = (cwist_scheduler_t *)arg;
126
127 pthread_mutex_lock(&s->delayed_mtx);
128 while (s->running) {
129 if (s->heap_count == 0) {
130 pthread_cond_wait(&s->delayed_cond, &s->delayed_mtx);
131 continue;
132 }
133
134 uint64_t current_ms = now_ms();
135 if (s->heap[0].deadline_ms <= current_ms) {
136 scheduled_job_t job = heap_pop(s);
137 pthread_mutex_unlock(&s->delayed_mtx);
138 cwist_io_queue_submit(s->queue, job.func, job.arg);
139 pthread_mutex_lock(&s->delayed_mtx);
140 continue;
141 }
142
143 uint64_t wait_ms = s->heap[0].deadline_ms - current_ms;
144 struct timespec deadline;
145 ms_to_timespec(current_ms + wait_ms, &deadline);
146 pthread_cond_timedwait(&s->delayed_cond, &s->delayed_mtx, &deadline);
147 }
148 pthread_mutex_unlock(&s->delayed_mtx);
149 return NULL;
150}

162cwist_scheduler_t *cwist_scheduler_create(size_t worker_count, size_t queue_capacity) {
163 if (worker_count == 0) worker_count = 1;
164
165 cwist_scheduler_t *s = (cwist_scheduler_t *)cwist_alloc(sizeof(*s));
166 if (!s) return NULL;
167 memset(s, 0, sizeof(*s));
168
169 s->queue = cwist_io_queue_create(queue_capacity);
170 if (!s->queue) {
171 cwist_free(s);
172 return NULL;
173 }
174
175 s->workers = (pthread_t *)cwist_alloc_array(worker_count, sizeof(pthread_t));
176 if (!s->workers) {
177 cwist_io_queue_destroy(s->queue);
178 cwist_free(s);
179 return NULL;
180 }
181 s->worker_count = worker_count;
182
183 pthread_mutex_init(&s->delayed_mtx, NULL);
184 cond_init_monotonic(&s->delayed_cond);
185 s->running = true;
186
187 for (size_t i = 0; i < worker_count; i++) {
188 if (pthread_create(&s->workers[i], NULL, scheduler_worker_thread, s) != 0) {
189 scheduler_cleanup_threads(s, i);
190 pthread_mutex_destroy(&s->delayed_mtx);
191 pthread_cond_destroy(&s->delayed_cond);
192 cwist_free(s->workers);
193 cwist_io_queue_destroy(s->queue);
194 cwist_free(s);
195 return NULL;
196 }
197 }
198
199 if (pthread_create(&s->timer_thread, NULL, scheduler_timer_thread, s) != 0) {
200 scheduler_cleanup_threads(s, worker_count);
201 pthread_mutex_destroy(&s->delayed_mtx);
202 pthread_cond_destroy(&s->delayed_cond);
203 cwist_free(s->workers);
204 cwist_io_queue_destroy(s->queue);
205 cwist_free(s);
206 return NULL;
207 }
208
209 return s;
210}

240bool cwist_scheduler_schedule(cwist_scheduler_t *s, cwist_job_func func, void *arg,
241 uint64_t delay_ms) {
242 if (!s || !func || !s->running) return false;
243 if (delay_ms == 0) {
244 return cwist_io_queue_submit(s->queue, func, arg);
245 }
246
247 scheduled_job_t job;
248 job.deadline_ms = now_ms() + delay_ms;
249 job.func = func;
250 job.arg = arg;
251
252 pthread_mutex_lock(&s->delayed_mtx);
253 size_t before = s->heap_count;
254 heap_push(s, job);
255 if (s->heap_count != before + 1) {
256 pthread_mutex_unlock(&s->delayed_mtx);
257 return false;
258 }
259 pthread_cond_signal(&s->delayed_cond);
260 pthread_mutex_unlock(&s->delayed_mtx);
261 return true;
262}
