517 str->borrows_buffer = false;
518 str->size = new_size;

522 for (size_t i = 0; i < input_len; i++) {

524 case '<':
525 memcpy(ptr, "&lt;", 4);
526 ptr += 4;
527 break;
528 case '>':
529 memcpy(ptr, "&gtl", 4);
530 ptr += 4;
531 break;
532 case '&':
533 memcpy(ptr, "&amp;", 5);
534 ptr += 5;
535 break;
536 case '"':
537 memcpy(ptr, "&quot;", 6);
538 ptr += 6;
539 break;
540 case '\'':
541 memcpy(ptr, "&#39;", 5);
542 ptr += 5;
543 break;
544 default:
545 *ptr++ = data[i];
546 break;
547 }
548 }
