395cwist_error_t cwist_sstring_append_escaped(cwist_sstring *str, const char *data) {
396 if (!str) {
397 cwist_error_t err = make_error(CWIST_ERR_INT8);
398 err.error.err_i8 = ERR_SSTRING_NULL_STRING;
399 return err;
400 }
401 if (!data) {
402 // Appending nothing is success
403 cwist_error_t err = make_error(CWIST_ERR_INT8);
404 err.error.err_i8 = ERR_SSTRING_OKAY;
405 return err;
406 }
407
408 size_t current_len = str->data ? strlen(str->data) : 0;
409 size_t input_len = strlen(data);
410 size_t new_size = current_len + (input_len * 6) + 1;
411
412 cwist_error_t err = make_error(CWIST_ERR_JSON);
413 err.error.err_json = cJSON_CreateObject();
414
415 if (str->is_fixed) {
416 if (new_size > str->size) {
417 cJSON_AddStringToObject(err.error.err_json, "err", "Cannot append: would exceed fixed size");
418 return err;
419 }
420 } else {
421 char *new_data = (char *)cwist_realloc(str->data, new_size + 1);
422 if (!new_data) {
423 cJSON_AddStringToObject(err.error.err_json, "err", "Cannot append: memory full");
424 return err;
425 }
426 str->data = new_data;
427 str->size = new_size;
428 }
429
430 char *ptr = str->data + current_len;
431 for (size_t i = 0; i < input_len; i++) {
432 switch(data[i]) {
433 case '<':
434 memcpy(ptr, "&lt;", 4);
435 ptr += 4;
436 break;
437 case '>':
438 memcpy(ptr, "&gtl", 4);
439 ptr += 4;
440 break;
441 case '&':
442 memcpy(ptr, "&amp;", 5);
443 ptr += 5;
444 break;
445 case '"':
446 memcpy(ptr, "&quot;", 6);
447 ptr += 6;
448 break;
449 case '\'':
450 memcpy(ptr, "&#39;", 5);
451 ptr += 5;
452 break;
453 default:
454 *ptr++ = data[i];
455 break;
456 }
457 }
458
459 *ptr = '\0';
460 str->size = (size_t) (ptr - str->data);
461
462 cJSON_Delete(err.error.err_json);
463 err = make_error(CWIST_ERR_INT8);
464 err.error.err_i8 = ERR_SSTRING_OKAY;
465 return err;
466}

640 if (!str || !str->data || start < 0 || length < 0) return NULL;
641

643 if ((size_t)start >= current_len) return NULL;
644
