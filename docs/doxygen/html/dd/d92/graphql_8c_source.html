129/* Parse GraphQL arguments: ( arg1: "val", arg2: 123, arg3: $varName ) */
130static const char *parse_arguments(const char *p, const cJSON *variables, cJSON **out_args, cJSON *errors) {
131 *out_args = NULL;
132 p = skip_ws_comments(p);
133 if (*p != '(') return p;
134 p++; /* skip '(' */
135
136 cJSON *args = cJSON_CreateObject();
137 if (!args) {
138 gql_add_error(errors, "Internal memory error parsing arguments");
139 return p;
140 }
141
142 while (*p) {
143 p = skip_ws_comments(p);
144 if (*p == ')' || *p == '\0') {
145 if (*p == ')') p++;
146 break;
147 }
148
149 /* Read argument name */
150 const char *name_start = p;
151 if (!gql_name_char(*p, true)) {
152 gql_add_error(errors, "Invalid argument name");
153 cJSON_Delete(args);
154 return p;
155 }
156 while (*p && gql_name_char(*p, false)) p++;
157 size_t name_len = (size_t)(p - name_start);
158 char arg_name[128];
159 if (name_len >= sizeof(arg_name)) name_len = sizeof(arg_name) - 1;
160 memcpy(arg_name, name_start, name_len);
161 arg_name[name_len] = '\0';
162
163 p = skip_ws_comments(p);
164 if (*p != ':') {
165 gql_add_error(errors, "Expected ':' after argument name");
166 cJSON_Delete(args);
167 return p;
168 }
169 p++; /* skip ':' */
170 p = skip_ws_comments(p);
171
172 /* Parse argument value */
173 cJSON *val = NULL;
174 if (*p == '"') {
175 p++;
176 const char *str_start = p;
177 while (*p && *p != '"') {
178 if (*p == '\\' && *(p + 1)) p += 2;
179 else p++;
180 }
181 size_t str_len = (size_t)(p - str_start);
182 char *str_val = cwist_alloc(str_len + 1);
183 if (str_val) {
184 memcpy(str_val, str_start, str_len);
185 str_val[str_len] = '\0';
186 val = cJSON_CreateString(str_val);
187 cwist_free(str_val);
188 }
189 if (*p == '"') p++;
190 } else if (*p == '$') { /* Variable reference */
191 p++;
192 const char *var_start = p;
193 while (*p && gql_name_char(*p, false)) p++;
194 size_t var_len = (size_t)(p - var_start);
195 char var_name[128];
196 if (var_len >= sizeof(var_name)) var_len = sizeof(var_name) - 1;
197 memcpy(var_name, var_start, var_len);
198 var_name[var_len] = '\0';
199
200 if (variables && cJSON_IsObject(variables)) {
201 cJSON *v = cJSON_GetObjectItemCaseSensitive(variables, var_name);
202 if (v) val = cJSON_Duplicate(v, true);
203 }
204 if (!val) val = cJSON_CreateNull();
205 } else if (isdigit((unsigned char)*p) || *p == '-') {
206 char *endptr = NULL;
207 double num = strtod(p, &endptr);
208 val = cJSON_CreateNumber(num);
209 p = endptr;
210 } else if (strncmp(p, "true", 4) == 0 && !gql_name_char(p[4], false)) {
211 val = cJSON_CreateBool(true);
212 p += 4;
213 } else if (strncmp(p, "false", 5) == 0 && !gql_name_char(p[5], false)) {
214 val = cJSON_CreateBool(false);
215 p += 5;
216 } else if (strncmp(p, "null", 4) == 0 && !gql_name_char(p[4], false)) {
217 val = cJSON_CreateNull();
218 p += 4;
219 } else {
220 gql_add_error(errors, "Unsupported argument value format");
221 cJSON_Delete(args);
222 return p;
223 }
224
225 if (val) {
226 cJSON_AddItemToObject(args, arg_name, val);
227 }
228 }
229
230 *out_args = args;
231 return p;
232}

234/* Parse and execute field selections recursively */
235static const char *parse_selection_set(const char *p, graphql_field_t *schema_fields, const cJSON *variables, cJSON *parent_data, cJSON *errors, unsigned int depth) {
236 if (depth > CWIST_GRAPHQL_MAX_DEPTH) {
237 gql_add_error(errors, "Query selection depth exceeds maximum allowed");
238 return p;
239 }
240
241 p = skip_ws_comments(p);
242 if (*p != '{') return p;
243 p++; /* skip '{' */
244
245 size_t field_count = 0;
246 while (*p) {
247 p = skip_ws_comments(p);
248 if (*p == '}' || *p == '\0') {
249 if (*p == '}') p++;
250 break;
251 }
252
253 if (++field_count > CWIST_GRAPHQL_MAX_FIELDS) {
254 gql_add_error(errors, "Field count limit exceeded");
255 break;
256 }
257
258 /* Read Alias or Field Name */
259 const char *name_start = p;
260 if (!gql_name_char(*p, true)) {
261 gql_add_error(errors, "Invalid field name syntax");
262 break;
263 }
264 while (*p && gql_name_char(*p, false)) p++;
265 size_t name_len = (size_t)(p - name_start);
266 char token1[128];
267 if (name_len >= sizeof(token1)) name_len = sizeof(token1) - 1;
268 memcpy(token1, name_start, name_len);
269 token1[name_len] = '\0';
270
271 char field_name[128];
272 char output_key[128];
273 strcpy(output_key, token1);
274 strcpy(field_name, token1);
275
276 p = skip_ws_comments(p);
277 if (*p == ':') { /* Alias: output_key: field_name */
278 p++;
279 p = skip_ws_comments(p);
280 const char *field_start = p;
281 if (!gql_name_char(*p, true)) {
282 gql_add_error(errors, "Invalid field name after alias");
283 break;
284 }
285 while (*p && gql_name_char(*p, false)) p++;
286 size_t f_len = (size_t)(p - field_start);
287 if (f_len >= sizeof(field_name)) f_len = sizeof(field_name) - 1;
288 memcpy(field_name, field_start, f_len);
289 field_name[f_len] = '\0';
290 }
291
292 /* Parse Field Arguments if present */
293 cJSON *args = NULL;
294 p = parse_arguments(p, variables, &args, errors);
295
296 /* Resolve Field Value */
297 cJSON *val = NULL;
298 graphql_field_t *f = gql_lookup(schema_fields, field_name);
299 if (!f) {
300 char err_buf[256];
301 snprintf(err_buf, sizeof(err_buf), "Cannot query field '%s' on schema", field_name);
302 gql_add_error(errors, err_buf);
303 val = cJSON_CreateNull();
304 } else {
305 val = f->fn(args, variables, f->ctx);
306 if (!val) {
307 val = cJSON_CreateNull();
308 }
309 }
310 if (args) cJSON_Delete(args);
311
312 /* Parse nested selection set if present and value is object */
313 p = skip_ws_comments(p);
314 if (*p == '{') {
315 if (cJSON_IsObject(val)) {
316 p = parse_selection_set(p, schema_fields, variables, val, errors, depth + 1);
317 } else {
318 /* Skip sub-selection if non-object or null */
319 int brace_count = 0;
320 while (*p) {
321 if (*p == '{') brace_count++;
322 else if (*p == '}') {
323 brace_count--;
324 if (brace_count == 0) { p++; break; }
325 }
326 p++;
327 }
328 }
329 }
330
331 cJSON_AddItemToObject(parent_data, output_key, val);
332 }
333
334 return p;
335}

337cwist_error_t cwist_graphql_execute(cwist_graphql_schema_t *schema, const char *request_json, cwist_sstring **out_json) {
338 if (!schema || !request_json || !out_json) return gql_error(-1);
339 *out_json = NULL;
340
341 cJSON *request = cJSON_Parse(request_json);
342 cJSON *root = cJSON_CreateObject();
343 cJSON *data = cJSON_CreateObject();
344 cJSON *errors = cJSON_CreateArray();
345
346 if (!request || !root || !data || !errors) {
347 cJSON_Delete(request); cJSON_Delete(root); cJSON_Delete(data); cJSON_Delete(errors);
348 return gql_error(-1);
349 }
350 cJSON_AddItemToObject(root, "data", data);
351 cJSON_AddItemToObject(root, "errors", errors);
352
353 cJSON *query = cJSON_GetObjectItemCaseSensitive(request, "query");
354 cJSON *variables = cJSON_GetObjectItemCaseSensitive(request, "variables");
355
356 if (!cJSON_IsString(query) || !query->valuestring) {
357 gql_add_error(errors, "GraphQL request must contain a string 'query'");
358 } else if (strlen(query->valuestring) > CWIST_GRAPHQL_MAX_QUERY_SIZE) {
359 gql_add_error(errors, "Query string exceeds maximum allowed size");
360 } else {
361 const char *p = skip_ws_comments(query->valuestring);
362 gql_op_type_t op_type = GQL_OP_QUERY;
363
364 if (strncmp(p, "mutation", 8) == 0 && !gql_name_char(p[8], false)) {
365 op_type = GQL_OP_MUTATION;
366 p += 8;
367 p = skip_ws_comments(p);
368 /* Skip optional mutation name / variable defs until '{' */
369 while (*p && *p != '{') p++;
370 } else if (strncmp(p, "query", 5) == 0 && !gql_name_char(p[5], false)) {
371 op_type = GQL_OP_QUERY;
372 p += 5;
373 p = skip_ws_comments(p);
374 while (*p && *p != '{') p++;
375 }
376
377 graphql_field_t *target_schema = (op_type == GQL_OP_MUTATION) ? schema->mutations : schema->queries;
378 parse_selection_set(p, target_schema, variables, data, errors, 1);
379 }
380
381 if (cJSON_GetArraySize(errors) == 0) {
382 cJSON_DeleteItemFromObjectCaseSensitive(root, "errors");
383 }
384
385 char *printed = cJSON_PrintUnformatted(root);
386 cJSON_Delete(request);
387 cJSON_Delete(root);
388
389 if (!printed) return gql_error(-1);
390
391 *out_json = cwist_sstring_create();
392 if (!*out_json) {
393 free(printed);
394 return gql_error(-1);
395 }
396 cwist_sstring_assign(*out_json, printed);
397 free(printed);
398
399 return gql_error(0);
400}
