29static cwist_mux_signature cwist_mux_signature_from_path(cwist_http_method_t method, const char *path) {
30 uint64_t hi = 14695981039346656037ULL;
31 uint64_t lo = 0xcbf29ce484222325ULL;
32
33 const uint64_t prime_hi = 1099511628211ULL; /* FNV Prime 64 */
34 const uint64_t prime_lo = 0x9e3779b185ebca87ULL; /* SplitMix Prime */
35
36 /* 1. Method hashing (single pass) */
37 const uint8_t *m_ptr = (const uint8_t *)&method;
38 for (size_t i = 0; i < sizeof(method); ++i) {
39 hi = (hi ^ m_ptr[i]) * prime_hi;
40 lo = (lo ^ m_ptr[i]) * prime_lo;
41 }
42
43 /* 2. Path normalization and single-pass hashing */
44 if (!path) {
45 path = "/";
46 }
47
48 size_t seg_count = 0;
49 const char *cursor = path;
50
51 while (*cursor) {
52 while (*cursor == '/') {
53 cursor++;
54 }
55 if (!*cursor) {
56 break;
57 }
58
59 const char *start = cursor;
60 while (*cursor && *cursor != '/') {
61 cursor++;
62 }
63
64 /* Inject normalized segment delimiter '/' */
65 hi = (hi ^ '/') * prime_hi;
66 lo = (lo ^ '/') * prime_lo;
67
68 /* Hash segment bytes */
69 for (const char *p = start; p < cursor; ++p) {
70 uint8_t b = (uint8_t)*p;
71 hi = (hi ^ b) * prime_hi;
72 lo = (lo ^ b) * prime_lo;
73 }
74 seg_count++;
75 }
76
77 /* Handle root path ('/') or empty path */
78 if (seg_count == 0) {
79 hi = (hi ^ '/') * prime_hi;
80 lo = (lo ^ '/') * prime_lo;
81 }
82
83 return (cwist_mux_signature) { .hi = hi, .lo = lo };
84}

92static size_t cwist_mux_bucket_index(const cwist_mux_router *router, const cwist_mux_signature *sig) {
93 uint64_t hash = sig->hi ^ (sig->lo + 0x9e3779b97f4a7c15ULL);
94
95 /* SplitMix64-style finalizer to maximize avalanche effect. */
96 hash ^= hash >> 30;
97 hash *= 0xbf58476d1ce4e5b9ULL;
98 hash ^= hash >> 27;
99 hash *= 0x94d049bb133111ebULL;
100 hash ^= hash >> 31;
101
102 return (size_t)(hash % router->bucket_count);
103}

195static bool match_parametric_route(const char *route_tmpl, const char *req_path, cwist_query_map **out_params) {
196 const char *t = route_tmpl;
197 const char *p = req_path;
198 cwist_query_map *params = NULL;
199
200 while (*t && *p) {
201 if (*t == ':') {
202 t++;
203 const char *t_next = strchr(t, '/');
204 size_t t_len = t_next ? (size_t)(t_next - t) : strlen(t);
205 char param_name[256] = {0};
206 if (t_len < sizeof(param_name)) memcpy(param_name, t, t_len);
207
208 const char *p_next = strchr(p, '/');
209 size_t p_len = p_next ? (size_t)(p_next - p) : strlen(p);
210 char param_value[256] = {0};
211 if (p_len < sizeof(param_value)) memcpy(param_value, p, p_len);
212
213 if (!params) params = cwist_query_map_create();
214 cwist_query_map_set(params, param_name, param_value);
215
216 t += t_len;
217 p += p_len;
218 } else if (*t == *p) {
219 t++;
220 p++;
221 } else {
222 if (params) cwist_query_map_destroy(params);
223 return false;
224 }
225 }
226
227 // Ignore trailing slashes
228 if (*t == '/' && *(t + 1) == '\0') t++;
229 if (*p == '/' && *(p + 1) == '\0') p++;
230
231 if (*t == '\0' && *p == '\0') {
232 *out_params = params;
233 return true;
234 }
235
236 if (params) cwist_query_map_destroy(params);
237 return false;
238}

357bool cwist_mux_serve(cwist_mux_router *router, cwist_http_request *req, cwist_http_response *res) {
358 if (!router || !req || !res) return false;
359
360 const char *path = (req->path && req->path->data) ? req->path->data : "/";
361 cwist_mux_signature signature = cwist_mux_signature_from_path(req->method, path);
362 size_t idx = cwist_mux_bucket_index(router, &signature);
363 cwist_mux_route *curr = router->buckets[idx];
364
365 // 1. Try exact match (O(1))
366 while (curr) {
367 if (curr->method == req->method &&
368 curr->signature_hi == signature.hi &&
369 curr->signature_lo == signature.lo &&
370 curr->path && curr->path->data &&
371 strcmp(curr->path->data, path) == 0) {
372 if (curr->middleware) {
373 run_middleware_chain(curr->middleware, req, res, curr->handler);
374 } else {
375 curr->handler(req, res);
376 }
377 return true;
378 }
379 curr = curr->bucket_next;
380 }
381
382 // 2. Try parametric matching (linear search)
383 curr = router->param_routes;
384 while (curr) {
385 if (curr->method == req->method && curr->path && curr->path->data) {
386 cwist_query_map *extracted_params = NULL;
387 if (match_parametric_route(curr->path->data, path, &extracted_params)) {
388 if (extracted_params) {
389 if (req->path_params) {
390 cwist_query_map_destroy(req->path_params);
391 }
392 req->path_params = extracted_params;
393 }
394 if (curr->middleware) {
395 run_middleware_chain(curr->middleware, req, res, curr->handler);
396 } else {
397 curr->handler(req, res);
398 }
399 return true;
400 }
401 }
402 curr = curr->param_next;
403 }
404
405 // 3. Try wildcard matching (linear search)
406 curr = router->wildcard_routes;
407 while (curr) {
408 if (curr->method == req->method && curr->path && curr->path->data) {
409 if (match_wildcard_route(curr->path->data, path)) {
410 if (curr->middleware) {
411 run_middleware_chain(curr->middleware, req, res, curr->handler);
412 } else {
413 curr->handler(req, res);
414 }
415 return true;
416 }
417 }
418 curr = curr->param_next;
419 }
420
421 return false;
