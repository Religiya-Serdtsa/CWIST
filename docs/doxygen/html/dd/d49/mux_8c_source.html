68static uint8_t mux_byte_permute(uint8_t lhs, uint8_t rhs) {
69 uint8_t row = lhs & 0x3;
70 uint8_t col = rhs & 0x3;
71 uint8_t a = MUX_PERM_TABLE_A[row][col];
72 uint8_t b = MUX_PERM_TABLE_B[col][row];
73 return (uint8_t)((a << 2) | b);
74}

83static void cwist_mux_mix_segment(cwist_mux_signature *sig, const char *segment, size_t len, size_t seg_idx) {
84 uint64_t acc = 0xA0761D6478BD642FULL ^ ((uint64_t)len << (seg_idx & 15));
85 for (size_t i = 0; i < len; ++i) {
86 uint8_t ch = (uint8_t)segment[i];
87 uint8_t latin = mux_byte_permute(ch, (uint8_t)(i + seg_idx));
88 uint64_t delta = ((uint64_t)latin << 48) |
89 ((uint64_t)ch << 24) |
90 ((uint64_t)(len - i) << 8) |
91 (uint64_t)seg_idx;
92 acc ^= mux_rotl64(delta, (unsigned int)(((i * 11) + seg_idx * 5) & 63));
93 }
94 sig->hi ^= mux_rotl64(acc ^ sig->lo, (unsigned int)(((seg_idx * 13) + 7) & 63));
95 sig->lo += mux_rotl64(acc + sig->hi, (unsigned int)(((seg_idx * 17) + 3) & 63));
96}

107static bool mux_solve_linear3(const double matrix[3][3], const double rhs[3], double out[3]) {
108 double aug[3][4];
109 for (size_t r = 0; r < 3; ++r) {
110 for (size_t c = 0; c < 3; ++c) {
111 aug[r][c] = matrix[r][c];
112 }
113 aug[r][3] = rhs[r];
114 }
115
116 for (size_t pivot = 0; pivot < 3; ++pivot) {
117 size_t best = pivot;
118 double best_abs = fabs(aug[pivot][pivot]);
119 for (size_t r = pivot + 1; r < 3; ++r) {
120 double val = fabs(aug[r][pivot]);
121 if (val > best_abs) {
122 best_abs = val;
123 best = r;
124 }
125 }
126 if (best_abs < 1e-9) {
127 return false;
128 }
129 if (best != pivot) {
130 for (size_t c = pivot; c < 4; ++c) {
131 double tmp = aug[pivot][c];
132 aug[pivot][c] = aug[best][c];
133 aug[best][c] = tmp;
134 }
135 }
136 double inv = 1.0 / aug[pivot][pivot];
137 for (size_t c = pivot; c < 4; ++c) {
138 aug[pivot][c] *= inv;
139 }
140 for (size_t r = 0; r < 3; ++r) {
141 if (r == pivot) continue;
142 double factor = aug[r][pivot];
143 for (size_t c = pivot; c < 4; ++c) {
144 aug[r][c] -= factor * aug[pivot][c];
145 }
146 }
147 }
148
149 for (size_t i = 0; i < 3; ++i) {
150 out[i] = aug[i][3];
151 }
152 return true;
153}

186static double mux_refine_magnitude(double guess, double target) {
187 double g = fabs(guess) + 1.0;
188 double t = fabs(target) + 1.0;
189 for (int i = 0; i < 2; ++i) {
190 g = 0.5 * (g + t / g);
191 }
192 return g;
193}

209static cwist_mux_signature cwist_mux_signature_from_path(cwist_http_method_t method, const char *path) {
210 cwist_mux_signature sig = {
211 .hi = 0x6a09e667f3bcc909ULL ^ ((uint64_t)method * 0x9e3779b97f4a7c15ULL),
212 .lo = 0xbb67ae8584caa73bULL ^ (((uint64_t)method << 40) | 0x100000001b3ULL)
213 };
214
215 if (!path) {
216 cwist_mux_mix_segment(&sig, "", 0, 0);
217 } else {
218 size_t seg_idx = 0;
219 const char *cursor = path;
220 while (*cursor) {
221 while (*cursor == '/') cursor++;
222 const char *start = cursor;
223 while (*cursor && *cursor != '/') cursor++;
224 size_t len = (size_t)(cursor - start);
225 cwist_mux_mix_segment(&sig, start, len, seg_idx++);
226 if (!*cursor) break;
227 }
228 if (seg_idx == 0) {
229 cwist_mux_mix_segment(&sig, "", 0, 0);
230 }
231 }
232 sig.hi ^= mux_rotl64(sig.lo, 29);
233 sig.lo ^= mux_rotl64(sig.hi, 19);
234 return sig;
235}

243static size_t cwist_mux_bucket_index(const cwist_mux_router *router, const cwist_mux_signature *sig) {
244 uint8_t hash_slot = cwist_routing_hash_slot(sig);
245 double coeffs[3];
246 mux_derive_coeffs(sig, coeffs);
247
248 double ratio_mix = coeffs[0] * 7.0 + coeffs[1] * 5.0 + coeffs[2] * 3.0;
249 double refined = mux_refine_magnitude(ratio_mix, (double)(sig->hi | 1ULL));
250 double slope = ((double)((sig->hi >> 8ULL) & 0xFFULL) - (double)((sig->lo >> 8ULL) & 0xFFULL)) / 64.0;
251 double predicted = mux_predict_state(refined, slope);
252 if (predicted < 0.0) {
253 predicted = -predicted + (double)hash_slot;
254 }
255
256 uint64_t mix = ((uint64_t)predicted) ^ mux_rotl64(sig->hi + sig->lo, hash_slot % 61U);
257 mix ^= mux_rotl64(((uint64_t)hash_slot * 0x9e3779b185ebca87ULL), hash_slot % 31U);
258 return (size_t)(mix % router->bucket_count);
259}

351static bool match_parametric_route(const char *route_tmpl, const char *req_path, cwist_query_map **out_params) {
352 const char *t = route_tmpl;
353 const char *p = req_path;
354 cwist_query_map *params = NULL;
355
356 while (*t && *p) {
357 if (*t == ':') {
358 t++;
359 const char *t_next = strchr(t, '/');
360 size_t t_len = t_next ? (size_t)(t_next - t) : strlen(t);
361 char param_name[256] = {0};
362 if (t_len < sizeof(param_name)) memcpy(param_name, t, t_len);
363
364 const char *p_next = strchr(p, '/');
365 size_t p_len = p_next ? (size_t)(p_next - p) : strlen(p);
366 char param_value[256] = {0};
367 if (p_len < sizeof(param_value)) memcpy(param_value, p, p_len);
368
369 if (!params) params = cwist_query_map_create();
370 cwist_query_map_set(params, param_name, param_value);
371
372 t += t_len;
373 p += p_len;
374 } else if (*t == *p) {
375 t++;
376 p++;
377 } else {
378 if (params) cwist_query_map_destroy(params);
379 return false;
380 }
381 }
382
383 // Ignore trailing slashes
384 if (*t == '/' && *(t + 1) == '\0') t++;
385 if (*p == '/' && *(p + 1) == '\0') p++;
386
387 if (*t == '\0' && *p == '\0') {
388 *out_params = params;
389 return true;
390 }
391
392 if (params) cwist_query_map_destroy(params);
393 return false;
394}

513bool cwist_mux_serve(cwist_mux_router *router, cwist_http_request *req, cwist_http_response *res) {
514 if (!router || !req || !res) return false;
515
516 const char *path = (req->path && req->path->data) ? req->path->data : "/";
517 cwist_mux_signature signature = cwist_mux_signature_from_path(req->method, path);
518 size_t idx = cwist_mux_bucket_index(router, &signature);
519 cwist_mux_route *curr = router->buckets[idx];
520
521 // 1. Try exact match (O(1))
522 while (curr) {
523 if (curr->method == req->method &&
524 curr->signature_hi == signature.hi &&
525 curr->signature_lo == signature.lo &&
526 curr->path && curr->path->data &&
527 strcmp(curr->path->data, path) == 0) {
528 if (curr->middleware) {
529 run_middleware_chain(curr->middleware, req, res, curr->handler);
530 } else {
531 curr->handler(req, res);
532 }
533 return true;
534 }
535 curr = curr->bucket_next;
536 }
537
538 // 2. Try parametric matching (linear search)
539 curr = router->param_routes;
540 while (curr) {
541 if (curr->method == req->method && curr->path && curr->path->data) {
542 cwist_query_map *extracted_params = NULL;
543 if (match_parametric_route(curr->path->data, path, &extracted_params)) {
544 if (extracted_params) {

547 }
548 req->path_params = extracted_params;
549 }
550 if (curr->middleware) {
551 run_middleware_chain(curr->middleware, req, res, curr->handler);
552 } else {
553 curr->handler(req, res);
554 }
555 return true;
556 }
557 }
558 curr = curr->param_next;
559 }
560
561 // 3. Try wildcard matching (linear search)

564 if (curr->method == req->method && curr->path && curr->path->data) {
565 if (match_wildcard_route(curr->path->data, path)) {

568 } else {

575 }
