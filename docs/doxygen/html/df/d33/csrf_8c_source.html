19static char *csrf_generate_token(void) {
20 unsigned char bytes[CWIST_CSRF_TOKEN_BYTES];
21 int fd = open("/dev/urandom", O_RDONLY | O_CLOEXEC);
22 if (fd < 0) return NULL;
23 size_t got = 0;
24 while (got < sizeof(bytes)) {
25 ssize_t n = read(fd, bytes + got, sizeof(bytes) - got);
26 if (n > 0) { got += (size_t)n; continue; }
27 if (n < 0 && errno == EINTR) continue;
28 close(fd);
29 return NULL;
30 }
31 close(fd);
32 static const char hex[] = "0123456789abcdef";
33 char *token = cwist_alloc(CWIST_CSRF_TOKEN_CHARS + 1);
34 if (!token) return NULL;
35 for (size_t i = 0; i < sizeof(bytes); ++i) {
36 token[2 * i] = hex[bytes[i] >> 4];
37 token[2 * i + 1] = hex[bytes[i] & 15U];
38 }
39 token[CWIST_CSRF_TOKEN_CHARS] = '\0';
40 return token;
41}

43static bool csrf_equal(const char *a, const char *b) {
44 if (!a || !b) return false;
45 size_t alen = strlen(a), blen = strlen(b);
46 size_t diff = alen ^ blen;
47 size_t limit = alen > blen ? alen : blen;
48 for (size_t i = 0; i < limit; ++i) {
49 unsigned char ac = i < alen ? (unsigned char)a[i] : 0;
50 unsigned char bc = i < blen ? (unsigned char)b[i] : 0;
51 diff |= (size_t)(ac ^ bc);
52 }
53 return diff == 0;
54}

56static int hex_value(unsigned char c) {
57 if (c >= '0' && c <= '9') return c - '0';
58 if (c >= 'a' && c <= 'f') return c - 'a' + 10;
59 if (c >= 'A' && c <= 'F') return c - 'A' + 10;
60 return -1;
61}

63static char *csrf_form_value(const cwist_http_request *req) {
64 if (!req || !req->body || !req->body->data) return NULL;
65 const char *body = req->body->data;
66 size_t len = req->body->size;
67 size_t pos = 0;
68 while (pos < len) {
69 size_t field_start = pos;
70 while (pos < len && body[pos] != '&') ++pos;
71 size_t field_end = pos++;
72 const char *eq = memchr(body + field_start, '=', field_end - field_start);
73 size_t key_len = eq ? (size_t)(eq - (body + field_start)) : field_end - field_start;
74 if (!((key_len == 5 && memcmp(body + field_start, "_csrf", 5) == 0) ||
75 (key_len == 10 && memcmp(body + field_start, "csrf_token", 10) == 0))) continue;
76 const char *value = eq ? eq + 1 : "";
77 size_t value_len = eq ? field_end - (size_t)(value - body) : 0;
78 char *decoded = cwist_alloc(value_len + 1);
79 if (!decoded) return NULL;
80 size_t out = 0;
81 for (size_t i = 0; i < value_len; ++i) {
82 unsigned char c = (unsigned char)value[i];
83 if (c == '+') decoded[out++] = ' ';
84 else if (c == '%' && i + 2 < value_len) {
85 int hi = hex_value((unsigned char)value[i + 1]);
86 int lo = hex_value((unsigned char)value[i + 2]);
87 if (hi < 0 || lo < 0) { cwist_free(decoded); return NULL; }
88 decoded[out++] = (char)((hi << 4) | lo);
89 i += 2;
90 } else decoded[out++] = (char)c;
91 }
92 decoded[out] = '\0';
93 return decoded;
94 }
95 return NULL;
96}

108static void csrf_handler(cwist_http_request *req, cwist_http_response *res, cwist_handler_func next) {
109 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
110 cwist_query_map *cookies = cwist_query_map_create();
111 const char *cookie = cookies ? cwist_cookie_get(cookies, CWIST_CSRF_COOKIE_NAME) : NULL;
112 if (cookies) {
113 cwist_cookie_parse(cookies, cookie_header);
114 cookie = cwist_cookie_get(cookies, CWIST_CSRF_COOKIE_NAME);
115 }
116 if (req->csrf_token) { cwist_free(req->csrf_token); req->csrf_token = NULL; }
117 if (!cookie) {
118 char *token = csrf_generate_token();
119 if (!token) { if (cookies) cwist_query_map_destroy(cookies); res->status_code = CWIST_HTTP_INTERNAL_ERROR; return; }
120 req->csrf_token = token;
121 csrf_issue_cookie(req, res, token);
122 if (cookies) cwist_query_map_destroy(cookies);
123 if (csrf_safe_method(req->method)) next(req, res);
124 else { res->status_code = CWIST_HTTP_FORBIDDEN; cwist_sstring_assign(res->body, "CSRF token missing"); }
125 return;
126 }
127 req->csrf_token = cwist_strdup(cookie);
128 if (csrf_safe_method(req->method)) { cwist_query_map_destroy(cookies); next(req, res); return; }
129 const char *header = cwist_http_header_get(req->headers, "X-CSRF-Token");
130 char *form = header ? NULL : csrf_form_value(req);
131 bool valid = csrf_equal(header ? header : form, cookie);
132 if (form) cwist_free(form);
133 cwist_query_map_destroy(cookies);
134 if (!valid) { res->status_code = CWIST_HTTP_FORBIDDEN; cwist_sstring_assign(res->body, "CSRF token invalid"); return; }
135 next(req, res);
136}
