352static size_t cwist_route_hash(cwist_http_method_t method, const char *path, size_t bucket_count) {
353 const unsigned long long FNV_OFFSET = 1469598103934665603ULL;
354 const unsigned long long FNV_PRIME = 1099511628211ULL;
355 unsigned long long hash = FNV_OFFSET ^ (unsigned long long)method;
356 const unsigned char *ptr = (const unsigned char *)path;
357 while (ptr && *ptr) {
358 hash ^= (unsigned long long)(*ptr++);
359 hash *= FNV_PRIME;
360 }
361 return (size_t)(hash % bucket_count);
362}

468 if (!table || !path) return NULL;

470 cwist_route_entry *curr = table->buckets[idx];

472 // Tiny string optimization: if length <= 8, cast to uint64 and compare in one shot
473 // Note: We need to handle potential access beyond string end safely.
474 // However, simplest heuristic is checking length first.
475 // Actually, we can just check length.
476
477 size_t path_len = strlen(path);
478 uint64_t path_u64 = 0;
479 bool use_fast_path = (path_len <= 8);
480 if (use_fast_path) {
481 memcpy(&path_u64, path, path_len); // Safe copy
482 }
483
484 while (curr) {
485 if (curr->method == method) {
486 if (use_fast_path) {
487 // Fast path check
488 size_t curr_len = strlen(curr->path);
489 if (curr_len == path_len) {
490 uint64_t curr_u64 = 0;
491 memcpy(&curr_u64, curr->path, curr_len);
492 if (path_u64 == curr_u64) return curr;
493 }
494 } else {
495 if (strcmp(curr->path, path) == 0) {
496 return curr;
497 }
498 }
499 }
500 curr = curr->next;
501 }

745static char *cwist_normalize_prefix(const char *prefix) {
746 if (!prefix || prefix[0] == '\0') {
747 return cwist_strdup("/");
748 }
749
750 size_t len = strlen(prefix);
751 bool needs_leading_slash = prefix[0] != '/';
752 char *buffer = (char *)cwist_alloc(len + needs_leading_slash + 1);
753 if (!buffer) {
754 return NULL;
755 }
756
757 if (needs_leading_slash) {
758 buffer[0] = '/';
759 memcpy(buffer + 1, prefix, len + 1);
760 len += 1;
761 } else {
762 memcpy(buffer, prefix, len + 1);
763 }
764
765 while (len > 1 && buffer[len - 1] == '/') {
766 buffer[len - 1] = '\0';
767 len--;
768 }
769
770 return buffer;
771}

773static char *cwist_normalize_directory(const char *directory) {
774 if (!directory || directory[0] == '\0') {
775 directory = ".";
776 }
777 char *resolved = realpath(directory, NULL);
778 if (resolved) {
779 return resolved;
780 }
781 /* Fallback when the path does not yet exist: strip trailing slashes */
782 size_t len = strlen(directory);
783 while (len > 1 && directory[len - 1] == '/') {
784 len--;
785 }
786 char *copy = (char *)cwist_alloc(len + 1);
787 if (!copy) return NULL;
788 memcpy(copy, directory, len);
789 copy[len] = '\0';
790 return copy;
791}

813static void cwist_static_handler(cwist_http_request *req, cwist_http_response *res) {
814 mw_executor_ctx *ctx = (mw_executor_ctx *)req->private_data;
815 cwist_static_request_info *info = ctx ? (cwist_static_request_info *)ctx->handler_data : NULL;
816 if (!info || !info->mapping) {
817 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
818 cwist_sstring_assign(res->body, "Static handler misconfigured");
819 return;
820 }
821
822 char relative_buf[PATH_MAX];
823 if (info->use_index || !info->relative_ptr || info->relative_ptr[0] == '\0') {
824 snprintf(relative_buf, sizeof(relative_buf), "index.html");
825 } else {
826 snprintf(relative_buf, sizeof(relative_buf), "%s", info->relative_ptr);
827 }
828 relative_buf[PATH_MAX - 1] = '\0';
829
830 char decoded_relative[PATH_MAX];
831 cwist_url_decode(relative_buf, decoded_relative, sizeof(decoded_relative));
832
833 if (cwist_path_has_parent_ref(decoded_relative)) {
834 res->status_code = CWIST_HTTP_FORBIDDEN;
835 cwist_sstring_assign(res->body, "Directory traversal blocked");
836 return;
837 }
838
839 char fs_path[PATH_MAX];
840 int written = snprintf(fs_path, sizeof(fs_path), "%s/%s", info->mapping->fs_root, decoded_relative);
841 if (written < 0 || written >= (int)sizeof(fs_path)) {
842 res->status_code = CWIST_HTTP_BAD_REQUEST;
843 cwist_sstring_assign(res->body, "Static path too long");
844 return;
845 }
846
847 char canonical[PATH_MAX];
848 if (!realpath(fs_path, canonical)) {
849 res->status_code = CWIST_HTTP_NOT_FOUND;
850 cwist_sstring_assign(res->body, "Not Found");
851 return;
852 }
853
854 /* Ensure the resolved path stays inside the configured static root */
855 size_t root_len = strlen(info->mapping->fs_root);
856 if (strncmp(canonical, info->mapping->fs_root, root_len) != 0 ||
857 (canonical[root_len] != '/' && canonical[root_len] != '\0')) {
858 res->status_code = CWIST_HTTP_FORBIDDEN;
859 cwist_sstring_assign(res->body, "Directory traversal blocked");
860 return;
861 }
862
863 cwist_app *app = req->app;
864 if (!app || !app->mem_manager) {
865 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
866 cwist_sstring_assign(res->body, "Server memory not initialized");
867 return;
868 }
869
870 cwist_fix_server_mem *mem = app->mem_manager;
871 pthread_mutex_lock(&mem->lock);
872 cwist_file_t *file = cwist_mem_get_file(mem, canonical);
873
874 if (file) {
875 // Simple MIME guess
876 const char *dot = strrchr(canonical, '.');
877 const char *mime = "application/octet-stream";
878 if (dot) {
879 if (strcasecmp(dot, ".html") == 0) mime = "text/html; charset=utf-8";
880 else if (strcasecmp(dot, ".css") == 0) mime = "text/css; charset=utf-8";
881 else if (strcasecmp(dot, ".js") == 0) mime = "application/javascript";
882 else if (strcasecmp(dot, ".json") == 0) mime = "application/json";
883 else if (strcasecmp(dot, ".png") == 0) mime = "image/png";
884 else if (strcasecmp(dot, ".jpg") == 0 || strcasecmp(dot, ".jpeg") == 0) mime = "image/jpeg";
885 else if (strcasecmp(dot, ".gif") == 0) mime = "image/gif";
886 else if (strcasecmp(dot, ".svg") == 0) mime = "image/svg+xml";
887 else if (strcasecmp(dot, ".txt") == 0) mime = "text/plain; charset=utf-8";
888 }
889
890 // Generate cache headers
891 char etag[64];
892 snprintf(etag, sizeof(etag), "\"%lx-%lx\"", (unsigned long)file->last_mod, (unsigned long)file->size);
893 char last_mod_buf[64];
894 cwist_http_format_date(file->last_mod, last_mod_buf, sizeof(last_mod_buf));
895
896 // Check conditional requests
897 bool not_modified = false;
898 const char *if_none_match = cwist_http_header_get(req->headers, "If-None-Match");
899 if (if_none_match && strcmp(if_none_match, etag) == 0) {
900 not_modified = true;
901 } else {
902 const char *if_modified_since = cwist_http_header_get(req->headers, "If-Modified-Since");
903 if (if_modified_since) {
904 time_t ims = cwist_http_parse_date(if_modified_since);
905 if (ims != (time_t)-1 && file->last_mod <= ims) {
906 not_modified = true;
907 }
908 }
909 }
910
911 if (not_modified) {
912 res->status_code = CWIST_HTTP_NOT_MODIFIED; // 304
913 cwist_http_header_add(&res->headers, "ETag", etag);
914 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
915 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
916 cwist_http_header_add(&res->headers, "Cache-Control", cc);
917 cwist_sstring_assign(res->body, "");
918 } else if (req->method == CWIST_HTTP_HEAD) {
919 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
920 char len_buf[32];
921 snprintf(len_buf, sizeof(len_buf), "%zu", file->size);
922 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
923 cwist_http_header_add(&res->headers, "Content-Type", mime);
924 cwist_http_header_add(&res->headers, "ETag", etag);
925 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
926 cwist_http_header_add(&res->headers, "Cache-Control", cc);
927 cwist_sstring_assign(res->body, "");
928 } else {
929 size_t send_offset = 0;
930 size_t send_len = file->size;
931
932 /* --- Range Request Handling --- */
933 const char *range_hdr = cwist_http_header_get(req->headers, "Range");
934 if (range_hdr && strncmp(range_hdr, "bytes=", 6) == 0) {
935 const char *p = range_hdr + 6;
936 size_t range_start = 0, range_end = file->size - 1;
937 bool range_valid = true;
938
939 if (p[0] == '-') {
940 size_t suffix = (size_t)atoi(p + 1);
941 range_start = (file->size > suffix) ? file->size - suffix : 0;
942 range_end = file->size - 1;
943 } else {
944 range_start = (size_t)atoi(p);
945 char *dash = strchr(p, '-');
946 if (dash && dash[1] != '\0') {
947 range_end = (size_t)atoi(dash + 1);
948 } else {
949 range_end = file->size - 1;
950 }
951 }
952
953 if (range_start > range_end || range_start >= file->size) {
954 range_valid = false;
955 res->status_code = CWIST_HTTP_RANGE_NOT_SATISFIABLE;
956 char cr[64];
957 snprintf(cr, sizeof(cr), "bytes */%zu", file->size);
958 cwist_http_header_add(&res->headers, "Content-Range", cr);
959 cwist_sstring_assign(res->body, "");
960 } else {
961 if (range_end >= file->size) range_end = file->size - 1;
962 send_offset = range_start;
963 send_len = range_end - range_start + 1;
964 res->status_code = CWIST_HTTP_PARTIAL_CONTENT;
965 char cr[64];
966 snprintf(cr, sizeof(cr), "bytes %zu-%zu/%zu", range_start, range_end, file->size);
967 cwist_http_header_add(&res->headers, "Content-Range", cr);
968 }
969 }
970
971 if (res->status_code != CWIST_HTTP_RANGE_NOT_SATISFIABLE) {
972 if (file->data && file->node) {
973 ttak_mem_node_acquire(file->node);
974 cwist_http_response_set_body_ptr_managed(res, (char *)file->data + send_offset, send_len, cwist_static_release_body, file->node);
975 } else {
976 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
977 cwist_sstring_assign(res->body, "Static buffer missing");
978 }
979 }
980
981 if (res->status_code != CWIST_HTTP_INTERNAL_ERROR && res->status_code != CWIST_HTTP_RANGE_NOT_SATISFIABLE) {
982 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
983 char len_buf[32];
984 snprintf(len_buf, sizeof(len_buf), "%zu", send_len);
985 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
986 cwist_http_header_add(&res->headers, "Content-Type", mime);
987 cwist_http_header_add(&res->headers, "ETag", etag);
988 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
989 cwist_http_header_add(&res->headers, "Cache-Control", cc);
990 cwist_http_header_add(&res->headers, "Accept-Ranges", "bytes");
991 if (res->status_code != CWIST_HTTP_PARTIAL_CONTENT) {
992 res->status_code = CWIST_HTTP_OK;
993 }
994 }
995 }
996 } else {
997 res->status_code = CWIST_HTTP_NOT_FOUND;
998 cwist_sstring_assign(res->body, "Not Found");
999 }
1000 pthread_mutex_unlock(&mem->lock);
1001}

1260 cwist_free(app->mem_manager);
1261 }
1262
1263 if (app->bdr_ctx) {
1264 cwist_bdr_destroy(app->bdr_ctx);
1265 }
1266
1267 if (app->nuke_enabled) {
1268 cwist_nuke_close();
1269 }
1270
1271 if (app->db) {
1272 // If nuke was used, app->db->conn was shared with nuke.
1273 // But nuke_close already closed its handles.
1274 // However, cwist_db_close will try to close it again if we are not careful.
1275 // Actually, NukeDB is a global singleton currently, so it's a bit messy.
1276 // If nuke_enabled is true, we should probably NOT call cwist_db_close(app->db)

1282 } else {
1283 cwist_db_close(app->db);
1284 }
1285 app->db = NULL;
1286 }
1287 if (app->db_path) {
1288 cwist_free(app->db_path);
1289 }
1290
1291 cwist_free(app);
1292}

1779static bool match_path(const char *pattern, const char *actual, cwist_query_map *params) {
1780 char p[256], a[256];
1781 strncpy(p, pattern, 255);
1782 strncpy(a, actual, 255);
1783 p[255] = a[255] = '\0';
1784
1785 if (params) {
1786 cwist_query_map_clear(params);
1787 }
1788
1789 char *saveptr_p, *saveptr_a;
1790 char *tok_p = strtok_r(p, "/", &saveptr_p);
1791 char *tok_a = strtok_r(a, "/", &saveptr_a);
1792
1793 while (tok_p && tok_a) {
1794 if (tok_p[0] == ':') {
1795 // Path Parameter
1796 cwist_query_map_set(params, tok_p + 1, tok_a);
1797 } else if (strcmp(tok_p, tok_a) != 0) {
1798 return false;
1799 }
1800 tok_p = strtok_r(NULL, "/", &saveptr_p);
1801 tok_a = strtok_r(NULL, "/", &saveptr_a);
1802 }
1803
1804 return tok_p == NULL && tok_a == NULL;
1805}

1807static cwist_route_entry *cwist_route_table_find_by_name(cwist_route_table *table, const char *name) {
1808 if (!table || !name) return NULL;
1809 for (size_t i = 0; i < table->bucket_count; i++) {
1810 cwist_route_entry *curr = table->buckets[i];
1811 while (curr) {
1812 if (curr->name && strcmp(curr->name, name) == 0) return curr;
1813 curr = curr->next;
1814 }
1815 }
1816 cwist_route_entry *curr = table->param_routes;
1817 while (curr) {
1818 if (curr->name && strcmp(curr->name, name) == 0) return curr;
1819 curr = curr->next;
1820 }
1821 return NULL;
1822}

1824char *cwist_url_for(cwist_app *app, const char *name, cwist_query_map *params) {
1825 if (!app || !app->router || !name) return NULL;
1826 cwist_route_entry *entry = cwist_route_table_find_by_name(app->router, name);
1827 if (!entry) return NULL;
1828 const char *path = entry->path;
1829 if (!params || !entry->has_params) {
1830 return cwist_strdup(path);
1831 }
1832 // Build URL by replacing :param segments
1833 cwist_sstring *result = cwist_sstring_create();
1834 if (!result) return NULL;
1835 char path_copy[512];
1836 strncpy(path_copy, path, sizeof(path_copy) - 1);
1837 path_copy[sizeof(path_copy) - 1] = '\0';
1838 char *saveptr;
1839 char *tok = strtok_r(path_copy, "/", &saveptr);
1840 int first = 1;
1841 while (tok) {
1842 if (!first) cwist_sstring_append(result, "/");
1843 first = 0;
1844 if (tok[0] == ':') {
1845 const char *val = cwist_query_map_get(params, tok + 1);
1846 if (val) {
1847 cwist_sstring_append(result, val);
1848 } else {
1849 cwist_sstring_append(result, tok);
1850 }
1851 } else {
1852 cwist_sstring_append(result, tok);
1853 }
1854 tok = strtok_r(NULL, "/", &saveptr);
1855 }
1856 char *out = cwist_strdup(result->data);
1857 cwist_sstring_destroy(result);
1858 return out;
1859}

1872static void internal_route_handler(cwist_app *app, cwist_http_request *req, cwist_http_response *res) {
1873 if (!req || !app || !app->router) return;
1874
1875 req->endpoint_opts = CWIST_ENDPOINT_DEFAULT;
1876 if (res) {
1877 res->endpoint_opts = req->endpoint_opts;
1878 }
1879
1880 cwist_static_request_info static_info = {0};
1881 if (cwist_prepare_static(app, req, &static_info)) {
1882 req->endpoint_opts = CWIST_ENDPOINT_FILE;
1883 if (res) res->endpoint_opts = req->endpoint_opts;
1884 execute_chain(app, req, res, cwist_static_handler, &static_info);
1885 return;
1886 }
1887
1888 const char *path = (req->path && req->path->data) ? req->path->data : "/";
1889 cwist_route_entry *found_route = cwist_route_table_lookup(app->router, req->method, path);
1890
1891 if (found_route && req->path_params) {
1892 cwist_query_map_clear(req->path_params);
1893 }
1894
1895 if (!found_route) {
1896 found_route = cwist_route_table_match_params(app->router, req);
1897 }
1898
1899 if (found_route) {
1900 req->endpoint_opts = found_route->opts ? found_route->opts : CWIST_ENDPOINT_DEFAULT;
1901 if (res) res->endpoint_opts = req->endpoint_opts;
1902 if (found_route->ws_handler) {
1903 if (req->client_fd >= 0) {
1904 cwist_websocket *ws = cwist_websocket_upgrade(req, req->client_fd);
1905 if (ws) {
1906 found_route->ws_handler(ws);
1907 cwist_websocket_destroy(ws);
1908 } else {
1909 res->status_code = CWIST_HTTP_BAD_REQUEST;
1910 cwist_sstring_assign(res->body, "WebSocket Upgrade Failed");
1911 }
1912 }
1913 } else {
1914 execute_chain(app, req, res, found_route->handler, NULL);
1915 }
1916 } else {
1917 cwist_error_handler_func eh = cwist_app_find_error_handler(app, CWIST_HTTP_NOT_FOUND);
1918 if (eh) {
1919 eh(req, res, CWIST_HTTP_NOT_FOUND);
1920 } else {
1921 res->status_code = CWIST_HTTP_NOT_FOUND;
1922 cwist_sstring_assign(res->body, "404 Not Found");
1923 }
1924 }
1925}
