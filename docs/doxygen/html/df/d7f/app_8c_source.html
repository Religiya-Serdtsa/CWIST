351static size_t cwist_route_hash(cwist_http_method_t method, const char *path, size_t bucket_count) {
352 const unsigned long long FNV_OFFSET = 1469598103934665603ULL;
353 const unsigned long long FNV_PRIME = 1099511628211ULL;
354 unsigned long long hash = FNV_OFFSET ^ (unsigned long long)method;
355 const unsigned char *ptr = (const unsigned char *)path;
356 while (ptr && *ptr) {
357 hash ^= (unsigned long long)(*ptr++);
358 hash *= FNV_PRIME;
359 }
360 return (size_t)(hash % bucket_count);
361}

466static cwist_route_entry *cwist_route_table_lookup(cwist_route_table *table, cwist_http_method_t method, const char *path) {
467 if (!table || !path) return NULL;
468 size_t idx = cwist_route_hash(method, path, table->bucket_count);
469 cwist_route_entry *curr = table->buckets[idx];
470
471 // Tiny string optimization: if length <= 8, cast to uint64 and compare in one shot
472 // Note: We need to handle potential access beyond string end safely.
473 // However, simplest heuristic is checking length first.
474 // Actually, we can just check length.
475
476 size_t path_len = strlen(path);
477 uint64_t path_u64 = 0;
478 bool use_fast_path = (path_len <= 8);
479 if (use_fast_path) {
480 memcpy(&path_u64, path, path_len); // Safe copy
481 }
482
483 while (curr) {
484 if (curr->method == method) {
485 if (use_fast_path) {
486 // Fast path check
487 size_t curr_len = strlen(curr->path);
488 if (curr_len == path_len) {
489 uint64_t curr_u64 = 0;
490 memcpy(&curr_u64, curr->path, curr_len);
491 if (path_u64 == curr_u64) return curr;
492 }
493 } else {
494 if (strcmp(curr->path, path) == 0) {
495 return curr;
496 }
497 }
498 }
499 curr = curr->next;
500 }
501 return NULL;
502}

621 if (!d) return;
622

626

628 if (strcmp(dir->d_name, ".") == 0 || strcmp(dir->d_name, "..") == 0) continue;

642 }
643 }
644 }

722static char *cwist_normalize_prefix(const char *prefix) {
723 if (!prefix || prefix[0] == '\0') {
724 return cwist_strdup("/");
725 }
726
727 size_t len = strlen(prefix);
728 bool needs_leading_slash = prefix[0] != '/';
729 char *buffer = (char *)cwist_alloc(len + needs_leading_slash + 1);
730 if (!buffer) {
731 return NULL;
732 }
733
734 if (needs_leading_slash) {
735 buffer[0] = '/';
736 memcpy(buffer + 1, prefix, len + 1);
737 len += 1;
738 } else {
739 memcpy(buffer, prefix, len + 1);
740 }
741
742 while (len > 1 && buffer[len - 1] == '/') {
743 buffer[len - 1] = '\0';
744 len--;
745 }
746
747 return buffer;
748}

750static char *cwist_normalize_directory(const char *directory) {
751 if (!directory || directory[0] == '\0') {
752 return cwist_strdup(".");
753 }
754 size_t len = strlen(directory);
755 while (len > 1 && directory[len - 1] == '/') {
756 len--;
757 }
758 char *copy = (char *)cwist_alloc(len + 1);
759 if (!copy) return NULL;
760 memcpy(copy, directory, len);
761 copy[len] = '\0';
762 return copy;
763}

785static void cwist_static_handler(cwist_http_request *req, cwist_http_response *res) {
786 mw_executor_ctx *ctx = (mw_executor_ctx *)req->private_data;
787 cwist_static_request_info *info = ctx ? (cwist_static_request_info *)ctx->handler_data : NULL;
788 if (!info || !info->mapping) {
789 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
790 cwist_sstring_assign(res->body, "Static handler misconfigured");
791 return;
792 }
793
794 char relative_buf[PATH_MAX];
795 if (info->use_index || !info->relative_ptr || info->relative_ptr[0] == '\0') {
796 snprintf(relative_buf, sizeof(relative_buf), "index.html");
797 } else {
798 snprintf(relative_buf, sizeof(relative_buf), "%s", info->relative_ptr);
799 }
800 relative_buf[PATH_MAX - 1] = '\0';
801
802 if (cwist_path_has_parent_ref(relative_buf)) {
803 res->status_code = CWIST_HTTP_FORBIDDEN;
804 cwist_sstring_assign(res->body, "Directory traversal blocked");
805 return;
806 }
807
808 char fs_path[PATH_MAX];
809 int written = snprintf(fs_path, sizeof(fs_path), "%s/%s", info->mapping->fs_root, relative_buf);
810 if (written < 0 || written >= (int)sizeof(fs_path)) {
811 res->status_code = CWIST_HTTP_BAD_REQUEST;
812 cwist_sstring_assign(res->body, "Static path too long");
813 return;
814 }
815
816 cwist_app *app = req->app;
817 if (!app || !app->mem_manager) {
818 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
819 cwist_sstring_assign(res->body, "Server memory not initialized");
820 return;
821 }
822
823 cwist_fix_server_mem *mem = app->mem_manager;
824 pthread_mutex_lock(&mem->lock);
825 cwist_file_t *file = cwist_mem_get_file(mem, fs_path);
826
827 if (file) {
828 // Simple MIME guess
829 const char *dot = strrchr(fs_path, '.');
830 const char *mime = "application/octet-stream";
831 if (dot) {
832 if (strcasecmp(dot, ".html") == 0) mime = "text/html; charset=utf-8";
833 else if (strcasecmp(dot, ".css") == 0) mime = "text/css; charset=utf-8";
834 else if (strcasecmp(dot, ".js") == 0) mime = "application/javascript";
835 else if (strcasecmp(dot, ".json") == 0) mime = "application/json";
836 else if (strcasecmp(dot, ".png") == 0) mime = "image/png";
837 else if (strcasecmp(dot, ".jpg") == 0 || strcasecmp(dot, ".jpeg") == 0) mime = "image/jpeg";
838 else if (strcasecmp(dot, ".gif") == 0) mime = "image/gif";
839 else if (strcasecmp(dot, ".svg") == 0) mime = "image/svg+xml";
840 else if (strcasecmp(dot, ".txt") == 0) mime = "text/plain; charset=utf-8";
841 }
842
843 // Generate cache headers
844 char etag[64];
845 snprintf(etag, sizeof(etag), "\"%lx-%lx\"", (unsigned long)file->last_mod, (unsigned long)file->size);
846 char last_mod_buf[64];
847 cwist_http_format_date(file->last_mod, last_mod_buf, sizeof(last_mod_buf));
848
849 // Check conditional requests
850 bool not_modified = false;
851 const char *if_none_match = cwist_http_header_get(req->headers, "If-None-Match");
852 if (if_none_match && strcmp(if_none_match, etag) == 0) {
853 not_modified = true;
854 } else {
855 const char *if_modified_since = cwist_http_header_get(req->headers, "If-Modified-Since");
856 if (if_modified_since) {
857 time_t ims = cwist_http_parse_date(if_modified_since);
858 if (ims != (time_t)-1 && file->last_mod <= ims) {
859 not_modified = true;
860 }
861 }
862 }
863
864 if (not_modified) {
865 res->status_code = CWIST_HTTP_NOT_MODIFIED; // 304
866 cwist_http_header_add(&res->headers, "ETag", etag);
867 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
868 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
869 cwist_http_header_add(&res->headers, "Cache-Control", cc);
870 cwist_sstring_assign(res->body, "");
871 } else if (req->method == CWIST_HTTP_HEAD) {
872 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
873 char len_buf[32];
874 snprintf(len_buf, sizeof(len_buf), "%zu", file->size);
875 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
876 cwist_http_header_add(&res->headers, "Content-Type", mime);
877 cwist_http_header_add(&res->headers, "ETag", etag);
878 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
879 cwist_http_header_add(&res->headers, "Cache-Control", cc);
880 cwist_sstring_assign(res->body, "");
881 } else if (file->data && file->node) {
882 // ZERO COPY
883 ttak_mem_node_acquire(file->node);
884 cwist_http_response_set_body_ptr_managed(res, file->data, file->size, cwist_static_release_body, file->node);
885
886 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
887 char len_buf[32];
888 snprintf(len_buf, sizeof(len_buf), "%zu", file->size);
889 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
890 cwist_http_header_add(&res->headers, "Content-Type", mime);
891 cwist_http_header_add(&res->headers, "ETag", etag);
892 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
893 cwist_http_header_add(&res->headers, "Cache-Control", cc);
894 } else {
895 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
896 cwist_sstring_assign(res->body, "Static buffer missing");
897 }
898 if (!not_modified) {
899 res->status_code = CWIST_HTTP_OK;
900 }
901 } else {
902 res->status_code = CWIST_HTTP_NOT_FOUND;
903 cwist_sstring_assign(res->body, "Not Found");
904 }
905 pthread_mutex_unlock(&mem->lock);
906}

1100void cwist_app_destroy(cwist_app *app) {
1101 if (!app) return;
1102 cwist_multiport_destroy_owned_subapps(app);
1103 cwist_multiport_unlink_app(app);
1104 if (app->cert_path) cwist_free(app->cert_path);
1105 if (app->key_path) cwist_free(app->key_path);
1106 if (app->tls_groups) cwist_free(app->tls_groups);
1107 if (app->ssl_ctx) cwist_https_destroy_context(app->ssl_ctx);
1108 if (app->h3_ctx) cwist_http3_destroy_context(app->h3_ctx);
1109
1110 cwist_route_table_destroy(app->router);
1111
1112 cwist_middleware_node *curr_m = app->middlewares;
1113 while (curr_m) {
1114 cwist_middleware_node *next = curr_m->next;
1115 cwist_free(curr_m);
1116 curr_m = next;
1117 }
1118
1119 cwist_error_handler_entry *curr_e = app->error_handlers;
1120 while (curr_e) {
1121 cwist_error_handler_entry *next = curr_e->next;
1122 cwist_free(curr_e);
1123 curr_e = next;
1124 }
1125
1126 if (app->config) cwist_config_destroy(app->config);
1127 if (app->logger) cwist_logger_destroy(app->logger);
1128 if (app->rdbms) {
1129 cwist_rdbms_runtime *rt = app->rdbms;
1130 if (rt->host) cwist_free(rt->host);
1131 cwist_free(rt);
1132 }
1133
1134 cwist_static_dir *curr_s = app->static_dirs;
1135 while (curr_s) {
1136 cwist_static_dir *next = curr_s->next;
1137 cwist_free(curr_s->url_prefix);
1138 cwist_free(curr_s->fs_root);
1139 cwist_free(curr_s->cache_control);
1140 cwist_free(curr_s);
1141 curr_s = next;
1142 }
1143
1144 if (app->mem_manager) {
1145 app->mem_manager->watcher_running = false;
1146 // If thread was started, join it.
1147 // Note: In simple destroy we might not have started it or might be crashing, but try join.
1148 if (app->mem_manager->watcher_thread) {
1149 pthread_join(app->mem_manager->watcher_thread, NULL);
1150 }
1151 pthread_mutex_destroy(&app->mem_manager->lock);
1152
1153 for (size_t i = 0; i < app->mem_manager->file_count; i++) {
1154 cwist_free(app->mem_manager->files[i].path);
1155 cwist_free(app->mem_manager->files[i].fs_path);
1156 }
1157 for (size_t i = 0; i < app->mem_manager->file_count; i++) {
1158 if (app->mem_manager->files[i].node) {
1159 ttak_mem_tree_remove(&app->mem_manager->file_tree, app->mem_manager->files[i].node);
1160 app->mem_manager->files[i].node = NULL;
1161 }
1162 }
1163 cwist_free(app->mem_manager->files);
1164 ttak_mem_tree_destroy(&app->mem_manager->file_tree);
1165 cwist_free(app->mem_manager);
1166 }
1167
1168 if (app->bdr_ctx) {
1169 cwist_bdr_destroy(app->bdr_ctx);
1170 }
1171
1172 if (app->nuke_enabled) {
1173 cwist_nuke_close();
1174 }
1175
1176 if (app->db) {
1177 // If nuke was used, app->db->conn was shared with nuke.
1178 // But nuke_close already closed its handles.
1179 // However, cwist_db_close will try to close it again if we are not careful.
1180 // Actually, NukeDB is a global singleton currently, so it's a bit messy.
1181 // If nuke_enabled is true, we should probably NOT call cwist_db_close(app->db)
1182 // OR we should make sure it's safe.
1183 // Based on cwist_db_close implementation, it calls sqlite3_close.
1184 if (app->nuke_enabled) {
1185 // Just free the wrapper, don't close the conn as Nuke already did it.
1186 ttak_mem_free(app->db);
1187 } else {
1188 cwist_db_close(app->db);
1189 }
1190 app->db = NULL;
1191 }
1192 if (app->db_path) {
1193 cwist_free(app->db_path);
1194 }
1195
1196 cwist_free(app);
1197}

1684static bool match_path(const char *pattern, const char *actual, cwist_query_map *params) {
1685 char p[256], a[256];
1686 strncpy(p, pattern, 255);
1687 strncpy(a, actual, 255);
1688 p[255] = a[255] = '\0';
1689
1690 if (params) {
1691 cwist_query_map_clear(params);
1692 }
1693
1694 char *saveptr_p, *saveptr_a;
1695 char *tok_p = strtok_r(p, "/", &saveptr_p);
1696 char *tok_a = strtok_r(a, "/", &saveptr_a);
1697
1698 while (tok_p && tok_a) {
1699 if (tok_p[0] == ':') {
1700 // Path Parameter
1701 cwist_query_map_set(params, tok_p + 1, tok_a);
1702 } else if (strcmp(tok_p, tok_a) != 0) {
1703 return false;
1704 }
1705 tok_p = strtok_r(NULL, "/", &saveptr_p);
1706 tok_a = strtok_r(NULL, "/", &saveptr_a);
1707 }
1708
1709 return tok_p == NULL && tok_a == NULL;
1710}

1729char *cwist_url_for(cwist_app *app, const char *name, cwist_query_map *params) {
1730 if (!app || !app->router || !name) return NULL;
1731 cwist_route_entry *entry = cwist_route_table_find_by_name(app->router, name);
1732 if (!entry) return NULL;
1733 const char *path = entry->path;
1734 if (!params || !entry->has_params) {
1735 return cwist_strdup(path);
1736 }
1737 // Build URL by replacing :param segments
1738 cwist_sstring *result = cwist_sstring_create();
1739 if (!result) return NULL;
1740 char path_copy[512];
1741 strncpy(path_copy, path, sizeof(path_copy) - 1);
1742 path_copy[sizeof(path_copy) - 1] = '\0';
1743 char *saveptr;
1744 char *tok = strtok_r(path_copy, "/", &saveptr);
1745 int first = 1;
1746 while (tok) {
1747 if (!first) cwist_sstring_append(result, "/");
1748 first = 0;
1749 if (tok[0] == ':') {
1750 const char *val = cwist_query_map_get(params, tok + 1);
1751 if (val) {
1752 cwist_sstring_append(result, val);
1753 } else {
1754 cwist_sstring_append(result, tok);
1755 }
1756 } else {
1757 cwist_sstring_append(result, tok);
1758 }
1759 tok = strtok_r(NULL, "/", &saveptr);
1760 }
1761 char *out = cwist_strdup(result->data);
1762 cwist_sstring_destroy(result);
1763 return out;
1764}

1777static void internal_route_handler(cwist_app *app, cwist_http_request *req, cwist_http_response *res) {
1778 if (!req || !app || !app->router) return;
1779
1780 req->endpoint_opts = CWIST_ENDPOINT_DEFAULT;
1781 if (res) {
1782 res->endpoint_opts = req->endpoint_opts;
1783 }
1784
1785 cwist_static_request_info static_info = {0};
1786 if (cwist_prepare_static(app, req, &static_info)) {
1787 req->endpoint_opts = CWIST_ENDPOINT_FILE;
1788 if (res) res->endpoint_opts = req->endpoint_opts;
1789 execute_chain(app, req, res, cwist_static_handler, &static_info);
1790 return;
1791 }
1792
1793 const char *path = (req->path && req->path->data) ? req->path->data : "/";
1794 cwist_route_entry *found_route = cwist_route_table_lookup(app->router, req->method, path);
1795
1796 if (found_route && req->path_params) {
1797 cwist_query_map_clear(req->path_params);
1798 }
1799
1800 if (!found_route) {
1801 found_route = cwist_route_table_match_params(app->router, req);
1802 }
1803
1804 if (found_route) {
1805 req->endpoint_opts = found_route->opts ? found_route->opts : CWIST_ENDPOINT_DEFAULT;
1806 if (res) res->endpoint_opts = req->endpoint_opts;
1807 if (found_route->ws_handler) {
1808 if (req->client_fd >= 0) {
1809 cwist_websocket *ws = cwist_websocket_upgrade(req, req->client_fd);
1810 if (ws) {
1811 found_route->ws_handler(ws);
1812 cwist_websocket_destroy(ws);
1813 } else {
1814 res->status_code = CWIST_HTTP_BAD_REQUEST;
1815 cwist_sstring_assign(res->body, "WebSocket Upgrade Failed");
1816 }
1817 }
1818 } else {
1819 execute_chain(app, req, res, found_route->handler, NULL);
1820 }
1821 } else {
1822 cwist_error_handler_func eh = cwist_app_find_error_handler(app, CWIST_HTTP_NOT_FOUND);
1823 if (eh) {
1824 eh(req, res, CWIST_HTTP_NOT_FOUND);
1825 } else {
1826 res->status_code = CWIST_HTTP_NOT_FOUND;
1827 cwist_sstring_assign(res->body, "404 Not Found");
1828 }
1829 }
1830}

1888static void static_http_handler(int client_fd, void *ctx) {
1889 cwist_app *app = (cwist_app *)ctx;
1890
1891 if (app->use_http2) {
1892 char peek_buf[24];
1893 ssize_t peeked = recv(client_fd, peek_buf, 24, MSG_PEEK);
1894 if (peeked >= 24 && memcmp(peek_buf, "PRI * HTTP/2.0\r\n\r\nSM\r\n\r\n", 24) == 0) {
1895 cwist_https_connection conn = { .fd = client_fd, .ssl = NULL, .negotiated_http2 = true, .negotiated_protocol = CWIST_HTTPS_PROTOCOL_HTTP2 };
1896 cwist_http2_serve_connection(&conn, app, static_http2_route_bridge);
1897 close(client_fd);
1898 return;
1899 }
1900 }
1901
1902 /* Apply Choi Seok-jeong's Lattice (Sanpan) for priority scaling. */
1903 uint32_t tid = ttak_net_lattice_get_worker_id();
1904 uint16_t node_id = (uint16_t)(client_fd % TTAK_MOLS_NODE_COUNT);
1905 uint32_t mixed_seed = ttak_apply_mols_control(node_id, tid);
1906
1907 /* Jeungseung Gaebang Scaling for priority weighting. */
1908 uint32_t priority_weight = ((mixed_seed * 16777619U) >> 8) % 100;
1909
1910 /* Use stack-allocated buffer for zero-allocation ingress path.
1911 * Aligned to cache line to optimize lattice-friendly access and prevent buffer loss. */
1912 _Alignas(64) char read_buf[CWIST_HTTP_READ_BUFFER_SIZE];
1913 size_t buf_len = 0;
1914 read_buf[0] = '\0';
1915
1916 while (true) {
1917 cwist_http_request *req = cwist_http_receive_request(client_fd, read_buf, sizeof(read_buf), &buf_len);
1918 if (!req) {
1919 break;
1920 }
1921 req->client_fd = client_fd;
1922 req->app = app;
1923 req->db = app->db;
1924
1925 // --- Big Dumb Reply (Read) ---
1926 if (app->bdr_ctx && req->method == CWIST_HTTP_GET) {
1927 size_t cached_len = 0;
1928 const void *cached_blob = cwist_bdr_get(app->bdr_ctx, "GET", req->path->data, &cached_len);
1929 if (cached_blob) {
1930 // BDR Hit! Blast it out.
1931 send(client_fd, cached_blob, cached_len, 0); // Flags handled by socket opt ideally or just 0
1932
1933 // Cleanup and Loop
1934 bool keep_alive = req->keep_alive;
1935 cwist_http_request_destroy(req);
1936 if (!keep_alive) break;
1937 continue;
1938 }
1939 }
1940 // -----------------------------
1941
1942 cwist_http_response *res = cwist_http_response_create();
1943 if (!res) {
1944 cwist_http_request_destroy(req);
1945 break;
1946 }
1947
1948 struct timespec start, end;
1949 clock_gettime(CLOCK_MONOTONIC, &start);
1950
1951 internal_route_handler(app, req, res);
1952
1953 clock_gettime(CLOCK_MONOTONIC, &end);
1954 uint64_t duration_ms = (end.tv_sec - start.tv_sec) * 1000 + (end.tv_nsec - start.tv_nsec) / 1000000;
1955
1956 bool keep_alive = req->keep_alive && res->keep_alive;
1957 bool upgraded = req->upgraded;
1958
1959 if (!req->upgraded) {
1960 if (cwist_http_send_response(client_fd, res).error.err_i16 < 0) {
1961 cwist_http_response_destroy(res);
1962 cwist_http_request_destroy(req);
1963 break;
1964 }
1965
1966 // --- Big Dumb Reply (Learn) ---
1967 bool endpoint_fixed = cwist_endpoint_has(req->endpoint_opts, CWIST_ENDPOINT_FIXED);
1968 bool endpoint_file = cwist_endpoint_has(req->endpoint_opts, CWIST_ENDPOINT_FILE);
1969
1970 /* Scale latency threshold using Lattice priority_weight (Jeungseung Gaebang) */
1971 uint64_t scaled_threshold = (uint64_t)app->bdr_ctx->latency_threshold_ms;
1972 if (priority_weight > 50) {
1973 scaled_threshold = scaled_threshold * (100 - priority_weight) / 100;
1974 }
1975
1976 if (app->bdr_ctx &&
1977 req->method == CWIST_HTTP_GET &&
1978 !endpoint_file &&
1979 (endpoint_fixed || duration_ms > scaled_threshold)) {
1980 // Too slow! Cache it.
1981 // We need to serialize the response we just sent.
1982 // Note: This duplicates serialization work (once in send_response, once here).
1983 // Optimization: send_response could return the blob, or we serialize first then send.
1984 // For now, re-serialize for BDR.
1985 cwist_sstring *serialized = cwist_http_stringify_response(res);
1986 if (serialized) {
1987 cwist_bdr_put(app->bdr_ctx, "GET", req->path->data, serialized->data, serialized->size);
1988 cwist_sstring_destroy(serialized);
1989 }
1990 }
1991 // ------------------------------
1992 }
1993
1994 cwist_http_response_destroy(res);
1995 cwist_http_request_destroy(req);
1996
1997 if (!keep_alive || upgraded) {
1998 break;
1999 }
2000 }
2001
2002 close(client_fd);
2003}
