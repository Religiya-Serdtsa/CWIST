46static bool cwist_mem_has_capacity(cwist_fix_server_mem *mem, size_t incoming, size_t reclaimable) {
47 if (!mem || mem->total_capacity == 0) {
48 return true;
49 }
50 if (incoming > mem->total_capacity) {
51 return false;
52 }
53 size_t used = mem->current_used;
54 if (reclaimable > used) {
55 reclaimable = used;
56 }
57 size_t projected = used - reclaimable + incoming;
58 return projected <= mem->total_capacity;
59}

237 mem->current_used = 0;

316static size_t cwist_route_hash(cwist_http_method_t method, const char *path, size_t bucket_count) {
317 const unsigned long long FNV_OFFSET = 1469598103934665603ULL;
318 const unsigned long long FNV_PRIME = 1099511628211ULL;
319 unsigned long long hash = FNV_OFFSET ^ (unsigned long long)method;
320 const unsigned char *ptr = (const unsigned char *)path;
321 while (ptr && *ptr) {
322 hash ^= (unsigned long long)(*ptr++);
323 hash *= FNV_PRIME;
324 }
325 return (size_t)(hash % bucket_count);
326}

396static void cwist_route_table_insert(cwist_route_table *table,
397 const char *path,
398 const char *name,
399 cwist_http_method_t method,
400 cwist_handler_func handler,
401 cwist_ws_handler_func ws_handler,
402 cwist_endpoint_opt_t opts) {
403 if (!table || !path) return;
404 cwist_route_entry *entry = cwist_route_entry_create(path, name, method, handler, ws_handler, opts);
405 if (!entry) return;

420 curr->opts = opts;
421 cwist_route_entry_free(entry);
422 return;
423 }
424 curr = curr->next;
425 }
426
427 entry->next = *bucket;
428 *bucket = entry;

431static cwist_route_entry *cwist_route_table_lookup(cwist_route_table *table, cwist_http_method_t method, const char *path) {
432 if (!table || !path) return NULL;
433 size_t idx = cwist_route_hash(method, path, table->bucket_count);
434 cwist_route_entry *curr = table->buckets[idx];
435
436 // Tiny string optimization: if length <= 8, cast to uint64 and compare in one shot
437 // Note: We need to handle potential access beyond string end safely.
438 // However, simplest heuristic is checking length first.
439 // Actually, we can just check length.
440
441 size_t path_len = strlen(path);
442 uint64_t path_u64 = 0;
443 bool use_fast_path = (path_len <= 8);
444 if (use_fast_path) {
445 memcpy(&path_u64, path, path_len); // Safe copy
446 }
447
448 while (curr) {
449 if (curr->method == method) {
450 if (use_fast_path) {
451 // Fast path check
452 size_t curr_len = strlen(curr->path);
453 if (curr_len == path_len) {
454 uint64_t curr_u64 = 0;
455 memcpy(&curr_u64, curr->path, curr_len);
456 if (path_u64 == curr_u64) return curr;
457 }
458 } else {
459 if (strcmp(curr->path, path) == 0) {
460 return curr;
461 }
462 }
463 }
464 curr = curr->next;
465 }
466 return NULL;
467}

513static bool cwist_static_match_entry(const cwist_static_dir *entry, const char *req_path, const char **relative_ptr, bool *use_index) {
514 if (!entry || !req_path || req_path[0] == '\0') return false;
515 size_t prefix_len = strlen(entry->url_prefix);
516 if (prefix_len == 0) return false;
517 if (prefix_len == 1 && entry->url_prefix[0] == '/') {
518 if (req_path[0] != '/') return false;
519 if (req_path[1] == '\0') {
520 if (use_index) *use_index = true;
521 if (relative_ptr) *relative_ptr = NULL;
522 } else {
523 if (use_index) *use_index = false;
524 if (relative_ptr) *relative_ptr = req_path + 1;
525 }
526 return true;
527 }
528
529 if (strncmp(req_path, entry->url_prefix, prefix_len) != 0) {
530 return false;
531 }
532
533 char separator = req_path[prefix_len];
534 if (separator == '\0') {
535 if (use_index) *use_index = true;
536 if (relative_ptr) *relative_ptr = NULL;
537 return true;
538 }
539 if (separator != '/') {
540 return false;
541 }
542 if (use_index) *use_index = false;
543 if (relative_ptr) *relative_ptr = req_path + prefix_len + 1;
544 return true;
545}

584static void cwist_scan_recursive(const char *fs_root, size_t *total_size, cwist_fix_server_mem *mem, bool dry_run) {
585 DIR *d = opendir(fs_root);
586 if (!d) return;
587
588 struct dirent *dir;
589 char full_path[PATH_MAX];
590 struct stat st;
591
592 while ((dir = readdir(d)) != NULL) {
593 if (strcmp(dir->d_name, ".") == 0 || strcmp(dir->d_name, "..") == 0) continue;
594
595 snprintf(full_path, sizeof(full_path), "%s/%s", fs_root, dir->d_name);
596 if (stat(full_path, &st) == -1) continue;
597
598 if (S_ISDIR(st.st_mode)) {
599 cwist_scan_recursive(full_path, total_size, mem, dry_run);
600 } else if (S_ISREG(st.st_mode)) {
601 if (dry_run) {
602 if (total_size) *total_size += st.st_size;
603 } else if (mem) {
604 if (!cwist_mem_register_file(mem, full_path, &st)) {
605 fprintf(stderr, "[StaticMem] Failed to load %s\n", full_path);
606 }
607 }
608 }
609 }
610 closedir(d);
611}

687static char *cwist_normalize_prefix(const char *prefix) {
688 if (!prefix || prefix[0] == '\0') {
689 return cwist_strdup("/");
690 }
691
692 size_t len = strlen(prefix);
693 bool needs_leading_slash = prefix[0] != '/';
694 char *buffer = (char *)cwist_alloc(len + needs_leading_slash + 1);
695 if (!buffer) {
696 return NULL;
697 }
698
699 if (needs_leading_slash) {
700 buffer[0] = '/';
701 memcpy(buffer + 1, prefix, len + 1);
702 len += 1;
703 } else {
704 memcpy(buffer, prefix, len + 1);
705 }
706
707 while (len > 1 && buffer[len - 1] == '/') {
708 buffer[len - 1] = '\0';
709 len--;
710 }
711
712 return buffer;
713}

750static void cwist_static_handler(cwist_http_request *req, cwist_http_response *res) {
751 mw_executor_ctx *ctx = (mw_executor_ctx *)req->private_data;
752 cwist_static_request_info *info = ctx ? (cwist_static_request_info *)ctx->handler_data : NULL;
753 if (!info || !info->mapping) {
754 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
755 cwist_sstring_assign(res->body, "Static handler misconfigured");
756 return;
757 }
758
759 char relative_buf[PATH_MAX];
760 if (info->use_index || !info->relative_ptr || info->relative_ptr[0] == '\0') {
761 snprintf(relative_buf, sizeof(relative_buf), "index.html");
762 } else {
763 snprintf(relative_buf, sizeof(relative_buf), "%s", info->relative_ptr);
764 }
765 relative_buf[PATH_MAX - 1] = '\0';
766
767 if (cwist_path_has_parent_ref(relative_buf)) {
768 res->status_code = CWIST_HTTP_FORBIDDEN;
769 cwist_sstring_assign(res->body, "Directory traversal blocked");
770 return;
771 }
772
773 char fs_path[PATH_MAX];
774 int written = snprintf(fs_path, sizeof(fs_path), "%s/%s", info->mapping->fs_root, relative_buf);
775 if (written < 0 || written >= (int)sizeof(fs_path)) {
776 res->status_code = CWIST_HTTP_BAD_REQUEST;
777 cwist_sstring_assign(res->body, "Static path too long");
778 return;
779 }
780
781 cwist_app *app = req->app;
782 if (!app || !app->mem_manager) {
783 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
784 cwist_sstring_assign(res->body, "Server memory not initialized");
785 return;
786 }
787
788 cwist_fix_server_mem *mem = app->mem_manager;
789 pthread_mutex_lock(&mem->lock);
790 cwist_file_t *file = cwist_mem_get_file(mem, fs_path);
791
792 if (file) {
793 // Simple MIME guess
794 const char *dot = strrchr(fs_path, '.');
795 const char *mime = "application/octet-stream";
796 if (dot) {
797 if (strcasecmp(dot, ".html") == 0) mime = "text/html; charset=utf-8";
798 else if (strcasecmp(dot, ".css") == 0) mime = "text/css; charset=utf-8";
799 else if (strcasecmp(dot, ".js") == 0) mime = "application/javascript";
800 else if (strcasecmp(dot, ".json") == 0) mime = "application/json";
801 else if (strcasecmp(dot, ".png") == 0) mime = "image/png";
802 else if (strcasecmp(dot, ".jpg") == 0 || strcasecmp(dot, ".jpeg") == 0) mime = "image/jpeg";
803 else if (strcasecmp(dot, ".gif") == 0) mime = "image/gif";
804 else if (strcasecmp(dot, ".svg") == 0) mime = "image/svg+xml";
805 else if (strcasecmp(dot, ".txt") == 0) mime = "text/plain; charset=utf-8";
806 }
807
808 if (req->method == CWIST_HTTP_HEAD) {
809 char len_buf[32];
810 snprintf(len_buf, sizeof(len_buf), "%zu", file->size);
811 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
812 cwist_http_header_add(&res->headers, "Content-Type", mime);
813 cwist_sstring_assign(res->body, "");
814 } else if (file->data && file->node) {
815 // ZERO COPY
816 ttak_mem_node_acquire(file->node);
817 cwist_http_response_set_body_ptr_managed(res, file->data, file->size, cwist_static_release_body, file->node);
818
819 char len_buf[32];
820 snprintf(len_buf, sizeof(len_buf), "%zu", file->size);
821 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
822 cwist_http_header_add(&res->headers, "Content-Type", mime);
823 } else {
824 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
825 cwist_sstring_assign(res->body, "Static buffer missing");
826 }
827 res->status_code = CWIST_HTTP_OK;
828 } else {
829 res->status_code = CWIST_HTTP_NOT_FOUND;
830 cwist_sstring_assign(res->body, "Not Found");
831 }
832 pthread_mutex_unlock(&mem->lock);
833}

1020void cwist_app_destroy(cwist_app *app) {
1021 if (!app) return;
1022 if (app->cert_path) cwist_free(app->cert_path);
1023 if (app->key_path) cwist_free(app->key_path);
1024 if (app->ssl_ctx) cwist_https_destroy_context(app->ssl_ctx);
1025 if (app->h3_ctx) cwist_http3_destroy_context(app->h3_ctx);
1026
1027 cwist_route_table_destroy(app->router);
1028
1029 cwist_middleware_node *curr_m = app->middlewares;
1030 while (curr_m) {
1031 cwist_middleware_node *next = curr_m->next;
1032 cwist_free(curr_m);
1033 curr_m = next;
1034 }
1035
1036 cwist_error_handler_entry *curr_e = app->error_handlers;
1037 while (curr_e) {
1038 cwist_error_handler_entry *next = curr_e->next;
1039 cwist_free(curr_e);
1040 curr_e = next;
1041 }
1042
1043 if (app->config) cwist_config_destroy(app->config);
1044 if (app->logger) cwist_logger_destroy(app->logger);
1045
1046 cwist_static_dir *curr_s = app->static_dirs;
1047 while (curr_s) {
1048 cwist_static_dir *next = curr_s->next;
1049 cwist_free(curr_s->url_prefix);
1050 cwist_free(curr_s->fs_root);
1051 cwist_free(curr_s);
1052 curr_s = next;
1053 }
1054
1055 if (app->mem_manager) {
1056 app->mem_manager->watcher_running = false;
1057 // If thread was started, join it.
1058 // Note: In simple destroy we might not have started it or might be crashing, but try join.
1059 if (app->mem_manager->watcher_thread) {
1060 pthread_join(app->mem_manager->watcher_thread, NULL);
1061 }
1062 pthread_mutex_destroy(&app->mem_manager->lock);
1063
1064 for (size_t i = 0; i < app->mem_manager->file_count; i++) {
1065 cwist_free(app->mem_manager->files[i].path);
1066 cwist_free(app->mem_manager->files[i].fs_path);
1067 }
1068 for (size_t i = 0; i < app->mem_manager->file_count; i++) {
1069 if (app->mem_manager->files[i].node) {
1070 ttak_mem_tree_remove(&app->mem_manager->file_tree, app->mem_manager->files[i].node);
1071 app->mem_manager->files[i].node = NULL;
1072 }
1073 }
1074 cwist_free(app->mem_manager->files);
1075 ttak_mem_tree_destroy(&app->mem_manager->file_tree);
1076 cwist_free(app->mem_manager);
1077 }
1078
1079 if (app->bdr_ctx) {
1080 cwist_bdr_destroy(app->bdr_ctx);
1081 }
1082
1083 if (app->nuke_enabled) {
1084 cwist_nuke_close();
1085 }
1086
1087 if (app->db) {
1088 // If nuke was used, app->db->conn was shared with nuke.
1089 // But nuke_close already closed its handles.
1090 // However, cwist_db_close will try to close it again if we are not careful.
1091 // Actually, NukeDB is a global singleton currently, so it's a bit messy.
1092 // If nuke_enabled is true, we should probably NOT call cwist_db_close(app->db)
1093 // OR we should make sure it's safe.
1094 // Based on cwist_db_close implementation, it calls sqlite3_close.
1095 if (app->nuke_enabled) {
1096 // Just free the wrapper, don't close the conn as Nuke already did it.
1097 ttak_mem_free(app->db);
1098 } else {
1099 cwist_db_close(app->db);
1100 }
1101 app->db = NULL;
1102 }
1103 if (app->db_path) {
1104 cwist_free(app->db_path);
1105 }
1106
1107 cwist_free(app);
1108}

1482static bool match_path(const char *pattern, const char *actual, cwist_query_map *params) {

1485 strncpy(a, actual, 255);

1488 if (params) {
1489 cwist_query_map_clear(params);
1490 }

1493 char *tok_p = strtok_r(p, "/", &saveptr_p);
1494 char *tok_a = strtok_r(a, "/", &saveptr_a);
1495
1496 while (tok_p && tok_a) {
1497 if (tok_p[0] == ':') {

1499 cwist_query_map_set(params, tok_p + 1, tok_a);
1500 } else if (strcmp(tok_p, tok_a) != 0) {

1502 }
1503 tok_p = strtok_r(NULL, "/", &saveptr_p);
1504 tok_a = strtok_r(NULL, "/", &saveptr_a);
1505 }

1686static void static_http_handler(int client_fd, void *ctx) {
1687 cwist_app *app = (cwist_app *)ctx;
1688
1689 if (app->use_http2) {
1690 char peek_buf[24];
1691 ssize_t peeked = recv(client_fd, peek_buf, 24, MSG_PEEK);
1692 if (peeked >= 24 && memcmp(peek_buf, "PRI * HTTP/2.0\r\n\r\nSM\r\n\r\n", 24) == 0) {
1693 cwist_https_connection conn = { .fd = client_fd, .ssl = NULL, .negotiated_http2 = true, .negotiated_protocol = CWIST_HTTPS_PROTOCOL_HTTP2 };
1694 cwist_http2_serve_connection(&conn, app, static_http2_route_bridge);
1695 close(client_fd);
1696 return;
1697 }
1698 }
1699
1700 char *read_buf = cwist_alloc(CWIST_HTTP_READ_BUFFER_SIZE);
1701 if (!read_buf) {
1702 close(client_fd);
1703 return;
1704 }
1705 size_t buf_len = 0;
1706 read_buf[0] = '\0';
1707
1708 while (true) {
1709 cwist_http_request *req = cwist_http_receive_request(client_fd, read_buf, CWIST_HTTP_READ_BUFFER_SIZE, &buf_len);
1710 if (!req) {
1711 break;
1712 }
1713 req->client_fd = client_fd;
1714 req->app = app;
1715 req->db = app->db;
1716
1717 // --- Big Dumb Reply (Read) ---
1718 if (app->bdr_ctx && req->method == CWIST_HTTP_GET) {
1719 size_t cached_len = 0;
1720 const void *cached_blob = cwist_bdr_get(app->bdr_ctx, "GET", req->path->data, &cached_len);
1721 if (cached_blob) {
1722 // BDR Hit! Blast it out.
1723 send(client_fd, cached_blob, cached_len, 0); // Flags handled by socket opt ideally or just 0
1724
1725 // Cleanup and Loop
1726 bool keep_alive = req->keep_alive;
1727 cwist_http_request_destroy(req);
1728 if (!keep_alive) break;
1729 continue;
1730 }
1731 }
1732 // -----------------------------
1733
1734 cwist_http_response *res = cwist_http_response_create();
1735 if (!res) {
1736 cwist_http_request_destroy(req);
1737 break;
1738 }
1739
1740 struct timespec start, end;
1741 clock_gettime(CLOCK_MONOTONIC, &start);
1742
1743 internal_route_handler(app, req, res);
1744
1745 clock_gettime(CLOCK_MONOTONIC, &end);
1746 uint64_t duration_ms = (end.tv_sec - start.tv_sec) * 1000 + (end.tv_nsec - start.tv_nsec) / 1000000;
1747
1748 bool keep_alive = req->keep_alive && res->keep_alive;
1749 bool upgraded = req->upgraded;
1750
1751 if (!req->upgraded) {
1752 if (cwist_http_send_response(client_fd, res).error.err_i16 < 0) {
1753 cwist_http_response_destroy(res);
1754 cwist_http_request_destroy(req);
1755 break;
1756 }
1757
1758 // --- Big Dumb Reply (Learn) ---
1759 bool endpoint_fixed = cwist_endpoint_has(req->endpoint_opts, CWIST_ENDPOINT_FIXED);
1760 bool endpoint_file = cwist_endpoint_has(req->endpoint_opts, CWIST_ENDPOINT_FILE);
1761 if (app->bdr_ctx &&
1762 req->method == CWIST_HTTP_GET &&
1763 !endpoint_file &&
1764 (endpoint_fixed || duration_ms > (uint64_t)app->bdr_ctx->latency_threshold_ms)) {
1765 // Too slow! Cache it.
1766 // We need to serialize the response we just sent.
1767 // Note: This duplicates serialization work (once in send_response, once here).
1768 // Optimization: send_response could return the blob, or we serialize first then send.
1769 // For now, re-serialize for BDR.
1770 cwist_sstring *serialized = cwist_http_stringify_response(res);
1771 if (serialized) {
1772 cwist_bdr_put(app->bdr_ctx, "GET", req->path->data, serialized->data, serialized->size);
1773 cwist_sstring_destroy(serialized);
1774 }
1775 }
1776 // ------------------------------
1777 }
1778
1779 cwist_http_response_destroy(res);
1780 cwist_http_request_destroy(req);
1781
1782 if (!keep_alive || upgraded) {
1783 break;
1784 }
1785 }
1786
1787 cwist_free(read_buf);
1788 close(client_fd);
1789}
