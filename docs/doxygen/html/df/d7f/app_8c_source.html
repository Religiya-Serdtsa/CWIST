356static size_t cwist_route_hash(cwist_http_method_t method, const char *path, size_t bucket_count) {
357 const unsigned long long FNV_OFFSET = 1469598103934665603ULL;
358 const unsigned long long FNV_PRIME = 1099511628211ULL;
359 unsigned long long hash = FNV_OFFSET ^ (unsigned long long)method;
360 const unsigned char *ptr = (const unsigned char *)path;
361 while (ptr && *ptr) {
362 hash ^= (unsigned long long)(*ptr++);
363 hash *= FNV_PRIME;
364 }
365 return (size_t)(hash % bucket_count);
366}

472 if (!table || !path) return NULL;

476 // Tiny string optimization: if length <= 8, cast to uint64 and compare in one shot
477 // Note: We need to handle potential access beyond string end safely.

479 // Actually, we can just check length.
480
481 size_t path_len = strlen(path);

485 memcpy(&path_u64, path, path_len); // Safe copy

494 uint64_t curr_u64 = 0;

499 if (strcmp(curr->path, path) == 0) {
500 return curr;
501 }
502 }
503 }
504 curr = curr->next;
505 }

736 }
737 pthread_mutex_unlock(&mem->lock);
738 }

752static char *cwist_normalize_prefix(const char *prefix) {
753 if (!prefix || prefix[0] == '\0') {
754 return cwist_strdup("/");
755 }
756
757 size_t len = strlen(prefix);
758 bool needs_leading_slash = prefix[0] != '/';
759 char *buffer = (char *)cwist_alloc(len + needs_leading_slash + 1);
760 if (!buffer) {
761 return NULL;
762 }
763
764 if (needs_leading_slash) {
765 buffer[0] = '/';
766 memcpy(buffer + 1, prefix, len + 1);
767 len += 1;
768 } else {
769 memcpy(buffer, prefix, len + 1);
770 }
771
772 while (len > 1 && buffer[len - 1] == '/') {
773 buffer[len - 1] = '\0';
774 len--;
775 }
776
777 return buffer;
778}

780static char *cwist_normalize_directory(const char *directory) {
781 if (!directory || directory[0] == '\0') {
782 directory = ".";
783 }
784 char *resolved = realpath(directory, NULL);
785 if (resolved) {
786 return resolved;
787 }
788 /* Fallback when the path does not yet exist: strip trailing slashes */
789 size_t len = strlen(directory);
790 while (len > 1 && directory[len - 1] == '/') {
791 len--;
792 }
793 char *copy = (char *)cwist_alloc(len + 1);
794 if (!copy) return NULL;
795 memcpy(copy, directory, len);
796 copy[len] = '\0';
797 return copy;
798}

820static void cwist_static_handler(cwist_http_request *req, cwist_http_response *res) {
821 mw_executor_ctx *ctx = (mw_executor_ctx *)req->private_data;
822 cwist_static_request_info *info = ctx ? (cwist_static_request_info *)ctx->handler_data : NULL;
823 if (!info || !info->mapping) {
824 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
825 cwist_sstring_assign(res->body, "Static handler misconfigured");
826 return;
827 }
828
829 char relative_buf[PATH_MAX];
830 if (info->use_index || !info->relative_ptr || info->relative_ptr[0] == '\0') {
831 snprintf(relative_buf, sizeof(relative_buf), "index.html");
832 } else {
833 snprintf(relative_buf, sizeof(relative_buf), "%s", info->relative_ptr);
834 }
835 relative_buf[PATH_MAX - 1] = '\0';
836
837 char decoded_relative[PATH_MAX];
838 cwist_url_decode(relative_buf, decoded_relative, sizeof(decoded_relative));
839
840 if (cwist_path_has_parent_ref(decoded_relative)) {
841 res->status_code = CWIST_HTTP_FORBIDDEN;
842 cwist_sstring_assign(res->body, "Directory traversal blocked");
843 return;
844 }
845
846 char fs_path[PATH_MAX];
847 int written = snprintf(fs_path, sizeof(fs_path), "%s/%s", info->mapping->fs_root, decoded_relative);
848 if (written < 0 || written >= (int)sizeof(fs_path)) {
849 res->status_code = CWIST_HTTP_BAD_REQUEST;
850 cwist_sstring_assign(res->body, "Static path too long");
851 return;
852 }
853
854 char canonical[PATH_MAX];
855 if (!realpath(fs_path, canonical)) {
856 res->status_code = CWIST_HTTP_NOT_FOUND;
857 cwist_sstring_assign(res->body, "Not Found");
858 return;
859 }
860
861 /* Ensure the resolved path stays inside the configured static root */
862 size_t root_len = strlen(info->mapping->fs_root);
863 if (strncmp(canonical, info->mapping->fs_root, root_len) != 0 ||
864 (canonical[root_len] != '/' && canonical[root_len] != '\0')) {
865 res->status_code = CWIST_HTTP_FORBIDDEN;
866 cwist_sstring_assign(res->body, "Directory traversal blocked");
867 return;
868 }
869
870 cwist_app *app = req->app;
871 if (!app || !app->mem_manager) {
872 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
873 cwist_sstring_assign(res->body, "Server memory not initialized");
874 return;
875 }
876
877 cwist_fix_server_mem *mem = app->mem_manager;
878 pthread_mutex_lock(&mem->lock);
879 cwist_file_t *file = cwist_mem_get_file(mem, canonical);
880
881 if (file) {
882 // Simple MIME guess
883 const char *dot = strrchr(canonical, '.');
884 const char *mime = "application/octet-stream";
885 if (dot) {
886 if (strcasecmp(dot, ".html") == 0) mime = "text/html; charset=utf-8";
887 else if (strcasecmp(dot, ".css") == 0) mime = "text/css; charset=utf-8";
888 else if (strcasecmp(dot, ".js") == 0) mime = "application/javascript";
889 else if (strcasecmp(dot, ".json") == 0) mime = "application/json";
890 else if (strcasecmp(dot, ".png") == 0) mime = "image/png";
891 else if (strcasecmp(dot, ".jpg") == 0 || strcasecmp(dot, ".jpeg") == 0) mime = "image/jpeg";
892 else if (strcasecmp(dot, ".gif") == 0) mime = "image/gif";
893 else if (strcasecmp(dot, ".svg") == 0) mime = "image/svg+xml";
894 else if (strcasecmp(dot, ".txt") == 0) mime = "text/plain; charset=utf-8";
895 }
896
897 // Generate cache headers
898 char etag[64];
899 snprintf(etag, sizeof(etag), "\"%lx-%lx\"", (unsigned long)file->last_mod, (unsigned long)file->size);
900 char last_mod_buf[64];
901 cwist_http_format_date(file->last_mod, last_mod_buf, sizeof(last_mod_buf));
902
903 // Check conditional requests
904 bool not_modified = false;
905 const char *if_none_match = cwist_http_header_get(req->headers, "If-None-Match");
906 if (if_none_match && strcmp(if_none_match, etag) == 0) {
907 not_modified = true;
908 } else {
909 const char *if_modified_since = cwist_http_header_get(req->headers, "If-Modified-Since");
910 if (if_modified_since) {
911 time_t ims = cwist_http_parse_date(if_modified_since);
912 if (ims != (time_t)-1 && file->last_mod <= ims) {
913 not_modified = true;
914 }
915 }
916 }
917
918 if (not_modified) {
919 res->status_code = CWIST_HTTP_NOT_MODIFIED; // 304
920 cwist_http_header_add(&res->headers, "ETag", etag);
921 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
922 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
923 cwist_http_header_add(&res->headers, "Cache-Control", cc);
924 cwist_sstring_assign(res->body, "");
925 } else if (req->method == CWIST_HTTP_HEAD) {
926 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
927 char len_buf[32];
928 snprintf(len_buf, sizeof(len_buf), "%zu", file->size);
929 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
930 cwist_http_header_add(&res->headers, "Content-Type", mime);
931 cwist_http_header_add(&res->headers, "ETag", etag);
932 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
933 cwist_http_header_add(&res->headers, "Cache-Control", cc);
934 cwist_sstring_assign(res->body, "");
935 } else {
936 size_t send_offset = 0;
937 size_t send_len = file->size;
938
939 /* --- Range Request Handling --- */
940 const char *range_hdr = cwist_http_header_get(req->headers, "Range");
941 if (range_hdr && strncmp(range_hdr, "bytes=", 6) == 0) {
942 const char *p = range_hdr + 6;
943 size_t range_start = 0, range_end = file->size - 1;
944 bool range_valid = true;
945
946 if (p[0] == '-') {
947 size_t suffix = (size_t)atoi(p + 1);
948 range_start = (file->size > suffix) ? file->size - suffix : 0;
949 range_end = file->size - 1;
950 } else {
951 range_start = (size_t)atoi(p);
952 char *dash = strchr(p, '-');
953 if (dash && dash[1] != '\0') {
954 range_end = (size_t)atoi(dash + 1);
955 } else {
956 range_end = file->size - 1;
957 }
958 }
959
960 if (range_start > range_end || range_start >= file->size) {
961 (void)range_valid;
962 res->status_code = CWIST_HTTP_RANGE_NOT_SATISFIABLE;
963 char cr[128];
964 snprintf(cr, sizeof(cr), "bytes */%zu", file->size);
965 cwist_http_header_add(&res->headers, "Content-Range", cr);
966 cwist_sstring_assign(res->body, "");
967 } else {
968 if (range_end >= file->size) range_end = file->size - 1;
969 send_offset = range_start;
970 send_len = range_end - range_start + 1;
971 res->status_code = CWIST_HTTP_PARTIAL_CONTENT;
972 char cr[128];
973 snprintf(cr, sizeof(cr), "bytes %zu-%zu/%zu", range_start, range_end, file->size);
974 cwist_http_header_add(&res->headers, "Content-Range", cr);
975 }
976 }
977
978 if (res->status_code != CWIST_HTTP_RANGE_NOT_SATISFIABLE) {
979 if (file->data && file->node) {
980 ttak_mem_node_acquire(file->node);
981 cwist_http_response_set_body_ptr_managed(res, (char *)file->data + send_offset, send_len, cwist_static_release_body, file->node);
982 } else {
983 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
984 cwist_sstring_assign(res->body, "Static buffer missing");
985 }
986 }
987
988 if (res->status_code != CWIST_HTTP_INTERNAL_ERROR && res->status_code != CWIST_HTTP_RANGE_NOT_SATISFIABLE) {
989 const char *cc = info->mapping->cache_control ? info->mapping->cache_control : "public, max-age=3600";
990 char len_buf[32];
991 snprintf(len_buf, sizeof(len_buf), "%zu", send_len);
992 cwist_http_header_add(&res->headers, "Content-Length", len_buf);
993 cwist_http_header_add(&res->headers, "Content-Type", mime);
994 cwist_http_header_add(&res->headers, "ETag", etag);
995 cwist_http_header_add(&res->headers, "Last-Modified", last_mod_buf);
996 cwist_http_header_add(&res->headers, "Cache-Control", cc);
997 cwist_http_header_add(&res->headers, "Accept-Ranges", "bytes");
998 if (res->status_code != CWIST_HTTP_PARTIAL_CONTENT) {
999 res->status_code = CWIST_HTTP_OK;
1000 }
1001 }
1002 }
1003 } else {
1004 res->status_code = CWIST_HTTP_NOT_FOUND;
1005 cwist_sstring_assign(res->body, "Not Found");
1006 }
1007 pthread_mutex_unlock(&mem->lock);
1008}

1332
1333 if (ctx->current_mw_node) {
1334 cwist_middleware_node *node = ctx->current_mw_node;
1335 // Advance the chain for the next "next" call
1336 ctx->current_mw_node = node->next;
1337 node->func(req, res, mw_next_wrapper);
1338 } else if (ctx->final_handler) {
1339 ctx->final_handler(req, res);
1340 }
1341}

1353 if (final_handler) final_handler(req, res);
1354 return;
1355 }
1356 mw_executor_ctx ctx = { app->middlewares, final_handler, handler_data };
1357 req->private_data = &ctx;
1358 mw_next_wrapper(req, res);
1359 req->private_data = NULL;
1360}

1922static bool match_path(const char *pattern, const char *actual, cwist_query_map *params) {
1923 char p[256], a[256];
1924 strncpy(p, pattern, 255);
1925 strncpy(a, actual, 255);
1926 p[255] = a[255] = '\0';
1927
1928 if (params) {
1929 cwist_query_map_clear(params);
1930 }
1931
1932 char *saveptr_p, *saveptr_a;
1933 char *tok_p = strtok_r(p, "/", &saveptr_p);
1934 char *tok_a = strtok_r(a, "/", &saveptr_a);
1935
1936 while (tok_p && tok_a) {
1937 if (tok_p[0] == ':') {
1938 // Path Parameter
1939 cwist_query_map_set(params, tok_p + 1, tok_a);
1940 } else if (strcmp(tok_p, tok_a) != 0) {
1941 return false;
1942 }
1943 tok_p = strtok_r(NULL, "/", &saveptr_p);
1944 tok_a = strtok_r(NULL, "/", &saveptr_a);
1945 }
1946
1947 return tok_p == NULL && tok_a == NULL;
1948}

1950static cwist_route_entry *cwist_route_table_find_by_name(cwist_route_table *table, const char *name) {
1951 if (!table || !name) return NULL;
1952 for (size_t i = 0; i < table->bucket_count; i++) {
1953 cwist_route_entry *curr = table->buckets[i];
1954 while (curr) {
1955 if (curr->name && strcmp(curr->name, name) == 0) return curr;
1956 curr = curr->next;
1957 }
1958 }
1959 cwist_route_entry *curr = table->param_routes;
1960 while (curr) {
1961 if (curr->name && strcmp(curr->name, name) == 0) return curr;
1962 curr = curr->next;
1963 }
1964 return NULL;
1965}

1967char *cwist_url_for(cwist_app *app, const char *name, cwist_query_map *params) {
1968 if (!app || !app->router || !name) return NULL;
1969 cwist_route_entry *entry = cwist_route_table_find_by_name(app->router, name);
1970 if (!entry) return NULL;
1971 const char *path = entry->path;
1972 if (!params || !entry->has_params) {
1973 return cwist_strdup(path);
1974 }
1975 // Build URL by replacing :param segments
1976 cwist_sstring *result = cwist_sstring_create();
1977 if (!result) return NULL;
1978 char path_copy[512];
1979 strncpy(path_copy, path, sizeof(path_copy) - 1);
1980 path_copy[sizeof(path_copy) - 1] = '\0';
1981 char *saveptr;
1982 char *tok = strtok_r(path_copy, "/", &saveptr);
1983 int first = 1;
1984 while (tok) {
1985 if (!first) cwist_sstring_append(result, "/");
1986 first = 0;
1987 if (tok[0] == ':') {
1988 const char *val = cwist_query_map_get(params, tok + 1);
1989 if (val) {
1990 cwist_sstring_append(result, val);
1991 } else {
1992 cwist_sstring_append(result, tok);

3269 perror("Failed to bind UDP port for HTTP/3");
3270 close(udp_fd);
3271 udp_fd = -1;
3272 }
3273 }
3274 }

3276 // Fork worker processes before any threads are created.
3277 int workers = 1;
3278 const char *workers_env = getenv("CWIST_WORKERS");
3279 if (workers_env) {
3280 if (strcmp(workers_env, "auto") == 0) {
3281 long cores = get_cpu_cores();
3282 workers = (cores > 0) ? (int)cores : 1;
3283 } else {
3284 workers = atoi(workers_env);
3285 }
3286 } else {
3287 // Default to auto (number of online CPU cores) to maximize performance on multi-core systems out of the box.
3288 long cores = get_cpu_cores();
3289 workers = (cores > 0) ? (int)cores : 1;
3290 }
3291 if (workers < 1) workers = 1;
3292 bool is_worker_child = false;
3293 pid_t worker_pids[workers > 1 ? workers - 1 : 1];
3294 size_t worker_count = 0;
3295 for (int i = 1; i < workers; i++) {
3296 pid_t pid = fork();
3297 if (pid == 0) {
3298 is_worker_child = true;
3299 break;
3300 } else if (pid < 0) {
3301 perror("fork worker failed");
3302 break;
3303 } else {
3304 worker_pids[worker_count++] = pid;
3305 }
3306 }
3307
3308 if (is_worker_child) {
3309 close(server_fd);
3310 server_fd = cwist_make_socket_ipv4(&addr, "0.0.0.0", port, 32768);
3311 if (server_fd >= 0) {
3312 g_cwist_listen_fd = server_fd;
3313 }
3314 }
3315
3316 // Per-process threads start here. Each worker gets its own watcher and
3317 // HTTP/3 thread, so fork-after-thread deadlock is avoided.
3318 if (app->mem_manager) {
3319 app->mem_manager->watcher_running = true;
3320 pthread_create(&app->mem_manager->watcher_thread, NULL, cwist_mem_watcher, app);
3321 }
3322

3324 struct h3_thread_payload *h3_p = malloc(sizeof(*h3_p));
3325 if (h3_p) {
3326 h3_p->udp_fd = udp_fd;
3327 h3_p->app = app;
3328 pthread_t h3_tid;
3329 if (pthread_create(&h3_tid, NULL, h3_server_thread_func, h3_p) == 0) {
3330 pthread_detach(h3_tid);
3331 g_cwist_udp_fd = udp_fd;
3332 printf("HTTP/3 (QUIC) enabled on UDP port %d\n", port);
3333 } else {
3334 free(h3_p);
3335 close(udp_fd);
3336 udp_fd = -1;
3337 }
3338 } else {
3339 close(udp_fd);
3340 udp_fd = -1;
3341 }
3342 }
3343
3344 printf("CWIST App running on port %d (SSL: %s) [Event-driven, workers=%d, pid=%d]\n",
3345 port, app->use_ssl ? "On" : "Off", workers, (int)getpid());
3346
3347 // Check config for non-blocking scale mode (default enabled)
3348 const char *c1m = getenv("CWIST_C1M_MODE");
3349 bool use_c1m = true;
3350 if (c1m) {
3351 if (c1m[0] == '0' || strcmp(c1m, "false") == 0) {
3352 use_c1m = false;
3353 }
3354 }
3355 if (use_c1m) {
3356 cwist_async_server_loop(server_fd, app);
3357 } else {
3358 if (app->use_ssl) {
3359 if (!app->ssl_ctx) {
3360 fprintf(stderr, "SSL enabled but context not initialized.\n");
3361 g_cwist_listen_fd = -1;
3362 return -1;
3363 }
3364 cwist_https_server_loop(server_fd, app->ssl_ctx, static_ssl_handler, app);
3365 } else {
3366 cwist_server_config config = { .use_forking = false, .use_threading = true, .use_epoll = false };
3367 cwist_http_server_loop(server_fd, &config, cwist_app_http_handler, app);
3368 }
3369 }
3370
3371 /* Graceful shutdown cleanup */
3372 g_cwist_listen_fd = -1;
3373 g_cwist_udp_fd = -1;
3374
3375 if (app->h3_ctx) {
3376 app->h3_ctx->running = 0;
3377 }
3378
3379 if (app->mem_manager) {
3380 app->mem_manager->watcher_running = false;
3381 }
3382
3383 printf("[CWIST] Draining connections for %d seconds...\n", g_cwist_drain_timeout_sec);
3384 if (is_worker_child || workers == 1) {
3385 sleep(g_cwist_drain_timeout_sec);
3386 }
3387
3388 /* Parent process reaps worker children so they do not become zombies. */
3389 if (!is_worker_child && workers > 1) {
3390 /* SIGTERM is delivered to the supervisor only. Ask every worker to
3391 * leave its inherited accept loop before waiting for it; otherwise a
3392 * supervisor shutdown can block indefinitely. */
3393 for (size_t i = 0; i < worker_count; i++) {
3394 kill(worker_pids[i], SIGTERM);
3395 }
3396 for (size_t i = 0; i < worker_count; i++) {
3397 int status;
3398 wait(&status);
3399 }
3400 }
3401
3402 printf("[CWIST] Shutdown complete.\n");
3403
3404 return 0;
3405}
