250static void cwist_mw_rate_limit_ip_handler(cwist_http_request *req, cwist_http_response *res, cwist_handler_func next) {
251 rate_limit_ctx_t *ctx = (rate_limit_ctx_t *)req->private_data;
252 int rpm = (ctx && ctx->rpm > 0) ? ctx->rpm : 60;
253
254 cwist_sstring *ip = cwist_get_client_ip_from_fd(req->client_fd);
255 if (!ip) {
256 next(req, res);
257 return;
258 }
259
260 pthread_mutex_lock(&rate_mutex);
261 ip_bucket_t *found = NULL;
262 for (int i = 0; i < ip_bucket_count; i++) {
263 if (ip_buckets[i].active && strcmp(ip_buckets[i].ip, ip->data) == 0) {
264 found = &ip_buckets[i];
265 break;
266 }
267 }
268
269 if (!found && ip_bucket_count < MAX_IP_BUCKETS) {
270 found = &ip_buckets[ip_bucket_count++];
271 strncpy(found->ip, ip->data, sizeof(found->ip) - 1);
272 found->ip[sizeof(found->ip) - 1] = '\0';
273 found->active = true;
274 double rate = rpm / 60.0;
275 ttak_token_bucket_init(&found->bucket, rate, (double)rpm);
276 }
277
278 bool allowed = false;
279 if (found) {
280 allowed = ttak_token_bucket_consume(&found->bucket, 1.0);
281 }
282 cwist_sstring_destroy(ip);
283 pthread_mutex_unlock(&rate_mutex);
284
285 if (!allowed) {
286 res->status_code = 429;
287 cwist_sstring_assign(res->body, "Too Many Requests");
288 char retry_buf[16];
289 snprintf(retry_buf, sizeof(retry_buf), "%d", 60 / rpm > 0 ? 60 / rpm : 1);
290 cwist_http_header_add(&res->headers, "Retry-After", retry_buf);
291 return;
292 }
293
294 next(req, res);
295}

400/* --- JWT Authentication Middleware --- */
401
402/*
403 * We use req->private_data to carry the decoded claims across the middleware
404 * boundary into the downstream handler. The previous value of private_data
405 * is saved/restored so other middleware can also use that field without
406 * conflict.
407 *
408 * The context is stack-allocated inside each wrapper function. It is only
409 * valid during the synchronous execution of the middleware chain; no pointer
410 * to the context escapes to asynchronous code.
411 */
412
413#define CWIST_JWT_CTX_MAGIC 0x4A574354UL /* "JWCT" */
414
415typedef struct {
416 unsigned long magic;
417 const char *secret;
418 cwist_jwt_claims *claims;
419 void *prev_private_data;
420} cwist_jwt_ctx_t;
421
428void cwist_mw_jwt_auth_handler(cwist_http_request *req, cwist_http_response *res, cwist_handler_func next) {
429 /* Retrieve the secret stored in the context tag */
430 cwist_jwt_ctx_t *ctx = (cwist_jwt_ctx_t *)req->private_data;
431 if (!ctx) {
432 /* Should not happen if the middleware was set up correctly */
433 res->status_code = CWIST_HTTP_INTERNAL_ERROR;
434 cwist_sstring_assign(res->body, "{\"error\":\"JWT middleware misconfigured\"}");
435 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
436 return;
437 }
438
439 const char *secret = ctx->secret;
440
441 /* Extract Bearer token from Authorization header */
442 char *auth_header = cwist_http_header_get(req->headers, "Authorization");
443 if (!auth_header) {
444 res->status_code = CWIST_HTTP_UNAUTHORIZED;
445 cwist_sstring_assign(res->body, "{\"error\":\"Missing Authorization header\"}");
446 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
447 return;
448 }
449
450 /* Expect "Bearer <token>" */
451 if (strncmp(auth_header, "Bearer ", 7) != 0) {
452 res->status_code = CWIST_HTTP_UNAUTHORIZED;
453 cwist_sstring_assign(res->body, "{\"error\":\"Invalid Authorization scheme\"}");
454 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
455 return;
456 }
457
458 const char *token = auth_header + 7;
459
460 cwist_jwt_claims *claims = cwist_jwt_verify(token, secret);
461 if (!claims) {
462 res->status_code = CWIST_HTTP_UNAUTHORIZED;
463 cwist_sstring_assign(res->body, "{\"error\":\"Invalid or expired token\"}");
464 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
465 return;
466 }
467
468 /* Stash claims so downstream handlers can retrieve them */
469 ctx->claims = claims;
470
471 next(req, res);
472
473 /* Clean up after the chain returns */
474 cwist_jwt_claims_destroy(claims);
475 ctx->claims = NULL;
476}

478/*
479 * Factory — we use a small heap-allocated context to bind the secret to the
480 * handler. Because cwist_middleware_func is a plain function pointer we cannot
481 * capture the secret in a closure, so we embed it in the req->private_data
482 * field before dispatching.
483 *
484 * The returned function pointer is cwist_mw_jwt_auth_handler. The secret is
485 * stored in a per-request cwist_jwt_ctx_t that is pushed/popped around the
486 * call so it does not trample any existing private_data.
487 */
488
489typedef struct {
490 const char *secret;
491} cwist_jwt_mw_cfg_t;
492
493/* We keep a small static table of registered secrets. A server rarely needs
494 * more than a handful of distinct signing keys. 8 slots cover the common case
495 * (e.g. one per audience / service) without dynamic allocation. If more are
496 * needed, increase CWIST_JWT_MAX_SECRETS and add the corresponding wrapper. */
497#define CWIST_JWT_MAX_SECRETS 8
498
499static cwist_jwt_mw_cfg_t s_jwt_cfgs[CWIST_JWT_MAX_SECRETS];
500static int s_jwt_cfg_count = 0;
501static pthread_mutex_t s_jwt_cfg_mutex = PTHREAD_MUTEX_INITIALIZER;
502
503/* One wrapper function per registered secret slot.
504 * The ctx is stack-allocated; its lifetime is confined to this function frame. */
505#define CWIST_JWT_DEFINE_WRAPPER(N) \
506static void cwist_mw_jwt_wrap_##N(cwist_http_request *req, cwist_http_response *res, cwist_handler_func next) { \
507 cwist_jwt_ctx_t ctx = { \
508 .magic = CWIST_JWT_CTX_MAGIC, \
509 .secret = s_jwt_cfgs[N].secret, \
510 .claims = NULL, \
511 .prev_private_data = req->private_data \
512 }; \
513 req->private_data = &ctx; \
514 cwist_mw_jwt_auth_handler(req, res, next); \
515 req->private_data = ctx.prev_private_data; \
516}
