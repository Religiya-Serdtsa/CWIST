24static void
25cwist_http2_flow_control_adjust(cwist_http2_flow_control *flow_control)
26{
27 uint64_t delay_us;
28 uint64_t multiplier;
29
30 if (!flow_control->rtt_initialized) {
31 return;
32 }
33
34 delay_us = flow_control->srtt_us + (4U * flow_control->rttvar_us);
35 if (flow_control->rttvar_us >= flow_control->srtt_us / 2U) {
36 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_UNSTABLE;
37 } else if (delay_us >= 200000U) {
38 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_HIGH_RTT;
39 } else {
40 flow_control->network_quality = CWIST_HTTP2_NETWORK_QUALITY_NORMAL;
41 }
42
43 multiplier = delay_us / 200000U;
44 if (multiplier < 1U) {
45 multiplier = 1U;
46 }
47 if (flow_control->network_quality != CWIST_HTTP2_NETWORK_QUALITY_NORMAL) {
48 ++multiplier;
49 }
50
51 flow_control->target_window = cwist_http2_window_clamp(
52 (uint64_t)flow_control->min_window * multiplier,
53 flow_control->min_window,
54 flow_control->max_window);
55 flow_control->pacing_rate_bytes_per_sec =
56 ((uint64_t)flow_control->target_window * 1000000U) /
57 (flow_control->srtt_us ? flow_control->srtt_us : 1U);
58}

60static int
61cwist_http2_write_window_update(uint8_t *buffer, size_t buffer_len,
62 size_t *written, uint32_t stream_id,
63 uint32_t increment)
64{
65 if (written != NULL) {
66 *written = 0;
67 }
68 if (buffer == NULL || written == NULL || buffer_len < 9U || increment == 0U) {
69 return -1;
70 }
71
72 buffer[0] = 0;
73 buffer[1] = 0;
74 buffer[2] = 4;
75 buffer[3] = 0x08;
76 buffer[4] = 0;
77 buffer[5] = (uint8_t)((stream_id >> 24) & 0x7fU);
78 buffer[6] = (uint8_t)(stream_id >> 16);
79 buffer[7] = (uint8_t)(stream_id >> 8);
80 buffer[8] = (uint8_t)stream_id;
81 buffer[9] = (uint8_t)(increment >> 24);
82 buffer[10] = (uint8_t)(increment >> 16);
83 buffer[11] = (uint8_t)(increment >> 8);
84 buffer[12] = (uint8_t)increment;
85 *written = 13U;
86 return 0;
87}

146void
147cwist_http2_flow_control_update_rtt(cwist_http2_flow_control *flow_control,
148 uint64_t rtt_sample_us)
149{
150 uint64_t difference;
151
152 if (flow_control == NULL || rtt_sample_us == 0U) {
153 return;
154 }
155 if (!flow_control->rtt_initialized) {
156 flow_control->srtt_us = rtt_sample_us;
157 flow_control->rttvar_us = rtt_sample_us / 2U;
158 flow_control->rtt_initialized = true;
159 } else {
160 difference = cwist_http2_abs_diff(flow_control->srtt_us, rtt_sample_us);
161 flow_control->rttvar_us = (3U * flow_control->rttvar_us + difference) / 4U;
162 flow_control->srtt_us = (7U * flow_control->srtt_us + rtt_sample_us) / 8U;
163 }
164 cwist_http2_flow_control_adjust(flow_control);
165}

294 const cwist_http2_stream_flow_control *stream_flow_control,
295 size_t requested, uint64_t now_us)
296{
297 uint64_t elapsed;
298 uint64_t added;
299 size_t allowed;
300

302 return 0;

307 elapsed = now_us - connection_flow_control->pacing_last_us;

309 connection_flow_control->pacing_tokens += added;
310 if (connection_flow_control->pacing_tokens > connection_flow_control->target_window) {
311 connection_flow_control->pacing_tokens = connection_flow_control->target_window;
312 }
313 connection_flow_control->pacing_last_us = now_us;
314 }
315
316 allowed = requested;
317 if (allowed > connection_flow_control->send_window) {
318 allowed = connection_flow_control->send_window;
319 }
320 if (allowed > stream_flow_control->send_window) {
321 allowed = stream_flow_control->send_window;
322 }
323 if (allowed > connection_flow_control->pacing_tokens) {
324 allowed = (size_t)connection_flow_control->pacing_tokens;
