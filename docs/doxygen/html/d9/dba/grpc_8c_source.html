102 if (!req || !res || !req->app || !req->path || !req->path->data) return;
103
104 cwist_grpc_route *route = grpc_find_route(req->app, req->path->data);
105 if (!route || !route->stream_handler) {
106 cwist_grpc_set_error(res, CWIST_GRPC_UNIMPLEMENTED, "gRPC stream method not registered");
107 return;
108 }
109 if (grpc_validate_request(req, res) != 0) return;
110
111 size_t offset = 0;
112 size_t cap = 4;
113 size_t count = 0;
114 cwist_grpc_message *messages = (cwist_grpc_message *)cwist_alloc(cap * sizeof(*messages));
115 if (!messages) {
116 cwist_grpc_set_error(res, CWIST_GRPC_RESOURCE_EXHAUSTED, "failed to allocate gRPC stream messages");
117 return;
118 }
119
120 while (offset < req->body->size) {
121 if (count == cap) {
122 cap *= 2;

124 if (!next) {

127 return;
128 }
129 messages = next;
130 }
131 if (cwist_grpc_decode_next_message(req->body->data, req->body->size,
132 &offset, &messages[count]) != 0) {
133 cwist_free(messages);
134 cwist_grpc_set_error(res, CWIST_GRPC_INVALID_ARGUMENT, "malformed gRPC stream frame");
135 return;
136 }

138 cwist_free(messages);
139 cwist_grpc_set_error(res, CWIST_GRPC_UNIMPLEMENTED, "compressed gRPC messages are not supported");
140 return;
141 }
142 count++;
143 }
144
145 res->status_code = CWIST_HTTP_OK;
146 cwist_http_header_add(&res->headers, "content-type", "application/grpc");
147 cwist_http_header_add(&res->headers, "grpc-accept-encoding", "identity");

152 .res = res,
153 .messages = messages,

155 .status = CWIST_GRPC_OK,
156 .status_message = NULL,
157 .closed = 0,
158 };
159 route->stream_handler(&stream, route->user_ctx);

164int cwist_grpc_decode_message(const void *frame,
165 size_t frame_len,
166 cwist_grpc_message *out) {
167 if (!frame || !out || frame_len < 5) return -1;
168 const uint8_t *p = (const uint8_t *)frame;
169 uint32_t len = ((uint32_t)p[1] << 24) |
170 ((uint32_t)p[2] << 16) |
171 ((uint32_t)p[3] << 8) |
172 (uint32_t)p[4];
173 if ((size_t)len > frame_len - 5) return -1;
174 if ((size_t)len != frame_len - 5) return -1;
175
176 out->compressed = p[0];
177 out->data = p + 5;
178 out->len = (size_t)len;
179 return 0;
180}

182int cwist_grpc_decode_next_message(const void *frames,
183 size_t frames_len,
184 size_t *offset,
185 cwist_grpc_message *out) {
186 if (!frames || !offset || !out || *offset > frames_len) return -1;
187 size_t pos = *offset;
188 if (frames_len - pos < 5) return -1;
189
190 const uint8_t *p = (const uint8_t *)frames + pos;
191 uint32_t len = ((uint32_t)p[1] << 24) |
192 ((uint32_t)p[2] << 16) |
193 ((uint32_t)p[3] << 8) |
194 (uint32_t)p[4];
195 if ((size_t)len > frames_len - pos - 5) return -1;
196
197 out->compressed = p[0];
198 out->data = p + 5;
199 out->len = (size_t)len;
200 *offset = pos + 5 + (size_t)len;
201 return 0;

204int cwist_grpc_encode_message(const void *payload,
205 size_t payload_len,
206 uint8_t compressed,
207 uint8_t **out,
208 size_t *out_len) {
209 if (!out || !out_len || (payload_len > 0 && !payload)) return -1;
210 if (payload_len > UINT32_MAX) return -1;
211
212 uint8_t *buf = (uint8_t *)cwist_alloc(payload_len + 5);
213 if (!buf) return -1;
214 buf[0] = compressed ? 1 : 0;
215 buf[1] = (uint8_t)((payload_len >> 24) & 0xff);
216 buf[2] = (uint8_t)((payload_len >> 16) & 0xff);
217 buf[3] = (uint8_t)((payload_len >> 8) & 0xff);
218 buf[4] = (uint8_t)(payload_len & 0xff);
219 if (payload_len > 0) memcpy(buf + 5, payload, payload_len);
220
221 *out = buf;
222 *out_len = payload_len + 5;
223 return 0;
224}

238int cwist_grpc_decoder_feed(cwist_grpc_decoder *decoder, const void *data, size_t len,
239 cwist_grpc_message_callback callback, void *ctx) {
240 if (!decoder || (!data && len) || !callback) return -1;
241 const uint8_t *input = data;
242 while (len) {
243 if (decoder->header_len < sizeof(decoder->header)) {
244 size_t take = sizeof(decoder->header) - decoder->header_len;
245 if (take > len) take = len;
246 memcpy(decoder->header + decoder->header_len, input, take);
247 decoder->header_len += take;
248 input += take;
249 len -= take;
250 if (decoder->header_len < sizeof(decoder->header)) continue;
251 decoder->compressed = decoder->header[0];
252 decoder->payload_len = ((size_t)decoder->header[1] << 24) |
253 ((size_t)decoder->header[2] << 16) |
254 ((size_t)decoder->header[3] << 8) |
255 (size_t)decoder->header[4];
256 if (decoder->compressed > 1 || decoder->payload_len > decoder->max_message_size)
257 return -1;
258 if (decoder->payload_len) {
259 decoder->payload = cwist_alloc(decoder->payload_len);
260 if (!decoder->payload) return -1;
261 }
262 }
263 size_t take = decoder->payload_len - decoder->payload_used;
264 if (take > len) take = len;
265 if (take) memcpy(decoder->payload + decoder->payload_used, input, take);
266 decoder->payload_used += take;
267 input += take;
268 len -= take;
269 if (decoder->payload_used != decoder->payload_len) continue;
270 cwist_grpc_message message = { decoder->compressed, decoder->payload, decoder->payload_len };
271 if (callback(ctx, &message) != 0) return -1;
272 cwist_free(decoder->payload);
273 decoder->payload = NULL;
274 decoder->payload_len = decoder->payload_used = decoder->header_len = 0;
275 }
276 return 0;
277}

317 const void *payload,
318 size_t payload_len) {
319 if (!stream || !stream->res || !stream->res->body) return -1;

334 cwist_free(frame);

337 stream->status_message = "failed to append gRPC stream message";

484static int grpc_register_route(cwist_app *app,
485 const char *service,
486 const char *method,
487 int streaming,
488 cwist_grpc_unary_handler_func unary_handler,
489 cwist_grpc_stream_handler_func stream_handler,
490 void *user_ctx) {
491 if (!app || !service || !method) return -1;
492 if ((!streaming && !unary_handler) || (streaming && !stream_handler)) return -1;
493 size_t service_len = strlen(service);
494 size_t method_len = strlen(method);
495 if (service_len == 0 || method_len == 0) return -1;
496
497 size_t path_len = service_len + method_len + 3;
498 char *path = (char *)cwist_alloc(path_len);
499 if (!path) return -1;
500 snprintf(path, path_len, "/%s/%s", service, method);
501
502 cwist_grpc_route *existing = grpc_find_route(app, path);
503 if (existing) {
504 existing->streaming = streaming;
505 existing->handler = unary_handler;
506 existing->stream_handler = stream_handler;
507 existing->user_ctx = user_ctx;
508 cwist_free(path);
509 return 0;
510 }
511
512 cwist_grpc_route *route = (cwist_grpc_route *)cwist_alloc(sizeof(*route));
513 if (!route) {
514 cwist_free(path);
515 return -1;
516 }
517 route->path = path;
518 route->streaming = streaming;
519 route->handler = unary_handler;
520 route->stream_handler = stream_handler;
521 route->user_ctx = user_ctx;
522 route->next = (cwist_grpc_route *)app->grpc_routes;
523 app->grpc_routes = route;
524
525 cwist_app_post(app, path, streaming ? grpc_dispatch_stream : grpc_dispatch_unary);
526 return 0;
527}

550 if (route->builtin == 1 && route->user_ctx) {
551 cwist_grpc_health_state *state = route->user_ctx;
552 while (state) {
553 cwist_grpc_health_state *state_next = state->next;
554 cwist_free(state->service);
555 cwist_free(state);
556 state = state_next;

562 }

566int cwist_grpc_routes_clone(cwist_app *dst, const cwist_app *src) {
567 if (!dst || !src) return -1;
568 const cwist_grpc_route *route = (const cwist_grpc_route *)src->grpc_routes;
569 while (route) {
570 if (route->builtin == 1) {
571 if (cwist_app_grpc_health(dst) != 0) return -1;
572 const cwist_grpc_health_state *state = route->user_ctx;
573 for (const cwist_grpc_health_state *it = state; it; it = it->next)
574 if (it->service && cwist_app_grpc_health_set_status(dst, it->service, it->serving) != 0)
575 return -1;
576 route = route->next;
577 continue;
578 }
579 if (route->builtin == 2) { route = route->next; continue; }
580 if (route->builtin == 3) {
581 if (cwist_app_grpc_reflection(dst) != 0) return -1;
582 route = route->next;
583 continue;
584 }
585 const char *path = route->path;
586 if (!path || path[0] != '/') return -1;
587 const char *method_sep = strrchr(path + 1, '/');
588 if (!method_sep || method_sep == path + 1 || method_sep[1] == '\0') return -1;
589
590 size_t service_len = (size_t)(method_sep - (path + 1));
591 char *service = (char *)cwist_alloc(service_len + 1);
592 if (!service) return -1;
593 memcpy(service, path + 1, service_len);
594 service[service_len] = '\0';
595
596 int rc = route->streaming
597 ? cwist_app_grpc_stream(dst, service, method_sep + 1,
598 route->stream_handler, route->user_ctx)
599 : cwist_app_grpc_unary(dst, service, method_sep + 1,
600 route->handler, route->user_ctx);
601 cwist_free(service);
602 if (rc != 0) return -1;
603 route = route->next;
604 }
605 return 0;
606}
