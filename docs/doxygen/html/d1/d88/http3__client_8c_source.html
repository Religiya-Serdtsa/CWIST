105static int h3c_packets_out(void *ctx, const struct lsquic_out_spec *specs,
106 unsigned n_specs) {
107 cwist_http3_client *client = ctx;
108 unsigned i;
109 for (i = 0; i < n_specs; ++i) {
110 const struct lsquic_out_spec *spec = &specs[i];
111 struct msghdr msg = {0};
112 msg.msg_name = (void *)spec->dest_sa;
113 msg.msg_namelen = (spec->dest_sa && spec->dest_sa->sa_family == AF_INET)
114 ? sizeof(struct sockaddr_in)
115 : sizeof(struct sockaddr_in6);
116 msg.msg_iov = (struct iovec *)spec->iov;
117 msg.msg_iovlen = spec->iovlen;
118 ssize_t nw = sendmsg(client->udp_fd, &msg, MSG_DONTWAIT);
119 if (nw < 0) {
120 if (errno == EAGAIN || errno == EWOULDBLOCK)
121 break;
122 return -1;
123 }
124 }
125 return (int)i;
126}

151static struct lsxpack_header *
152h3c_hsi_prepare(void *hset_p, struct lsxpack_header *xhdr, size_t req_space) {
153 h3c_hset_t *hset = hset_p;
154 if (xhdr) {
155 size_t name_len = xhdr->buf + xhdr->name_offset - (hset->decode_buf + hset->decode_off);
156 size_t value_len = xhdr->buf + xhdr->val_offset - (hset->decode_buf + hset->decode_off);
157 size_t total = name_len + value_len + 2;
158 if (total > sizeof(hset->decode_buf) - hset->decode_off)
159 total = sizeof(hset->decode_buf) - hset->decode_off;
160 hset->decode_off += total;
161 if (hset->count < H3C_MAX_HEADERS)
162 hset->count++;
163 }
164 if (hset->count >= H3C_MAX_HEADERS)
165 return NULL;
166 if (req_space > sizeof(hset->decode_buf) - hset->decode_off)
167 return NULL;
168 lsxpack_header_prepare_decode(&hset->headers[hset->count],
169 hset->decode_buf, hset->decode_off,
170 sizeof(hset->decode_buf) - hset->decode_off);
171 return &hset->headers[hset->count];
172}

220static void h3c_on_read(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
221 h3c_stream_ctx_t *st = (h3c_stream_ctx_t *)st_h;
222 if (!st) return;
223
224 unsigned char buf[8192];
225 ssize_t nread;
226
227 if (!st->headers_done) {
228 void *hset = lsquic_stream_get_hset(stream);
229 if (hset) {
230 h3c_hset_t *hs = hset;
231 st->res = cwist_http_response_create();
232 size_t i;
233 for (i = 0; i < hs->count; ++i) {
234 const char *name = lsxpack_header_get_name(&hs->headers[i]);
235 const char *value = lsxpack_header_get_value(&hs->headers[i]);
236 if (!name || !value) continue;
237 if (strcmp(name, ":status") == 0) {
238 st->res->status_code = (cwist_http_status_t)atoi(value);
239 } else {
240 cwist_http_header_add(&st->res->headers, name, value);
241 }
242 }
243 st->headers_done = 1;
244 }
245 }
246
247 while ((nread = lsquic_stream_read(stream, buf, sizeof(buf))) > 0) {
248 size_t need = st->resp_body_len + (size_t)nread;
249 if (need > st->resp_body_cap) {
250 size_t new_cap = st->resp_body_cap ? st->resp_body_cap * 2 : 4096;
251 while (new_cap < need) new_cap *= 2;
252 char *tmp = realloc(st->resp_body, new_cap);
253 if (!tmp) {
254 lsquic_stream_close(stream);
255 return;
256 }
257 st->resp_body = tmp;
258 st->resp_body_cap = new_cap;
259 }
260 memcpy(st->resp_body + st->resp_body_len, buf, (size_t)nread);
261 st->resp_body_len += (size_t)nread;
262 }
263
264 if (nread == 0) {
265 if (st->resp_body_len > 0 && st->res && st->res->body) {
266 cwist_sstring_assign_len(st->res->body, st->resp_body, st->resp_body_len);
267 }
268 st->response_ready = 1;
269 lsquic_stream_wantread(stream, 0);
270 }
271}

273static void h3c_on_write(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
274 h3c_stream_ctx_t *st = (h3c_stream_ctx_t *)st_h;
275 if (!st || st->write_done) return;
276
277 /* Send request headers */
278 struct lsxpack_header headers_arr[64];
279 char hbuf[8192];
280 size_t hbuf_off = 0;
281 size_t hdr_count = 0;
282
283 /* :method */
284 const char *method_str = cwist_http_method_to_string(st->req_method);
285 if (!method_str || !*method_str) method_str = "GET";
286 size_t mlen = strlen(method_str);
287 if (hbuf_off + 7 + 2 + mlen <= sizeof(hbuf)) {
288 memcpy(hbuf + hbuf_off, ":method", 7);
289 memcpy(hbuf + hbuf_off + 9, method_str, mlen);
290 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
291 0, 7, 9, mlen);
292 hbuf_off += 9 + mlen;
293 hdr_count++;
294 }
295
296 /* :path */
297 const char *path = st->req_path ? st->req_path : "/";
298 size_t plen = strlen(path);
299 if (hbuf_off + 5 + 2 + plen <= sizeof(hbuf)) {
300 memcpy(hbuf + hbuf_off, ":path", 5);
301 memcpy(hbuf + hbuf_off + 7, path, plen);
302 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
303 0, 5, 7, plen);
304 hbuf_off += 7 + plen;
305 hdr_count++;
306 }
307
308 /* :authority */
309 cwist_http3_client *client = (cwist_http3_client *)lsquic_conn_get_ctx(lsquic_stream_conn(stream));
310 const char *authority = client && client->host ? client->host : "localhost";
311 size_t alen = strlen(authority);
312 if (hbuf_off + 10 + 2 + alen <= sizeof(hbuf)) {
313 memcpy(hbuf + hbuf_off, ":authority", 10);
314 memcpy(hbuf + hbuf_off + 12, authority, alen);
315 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
316 0, 10, 12, alen);
317 hbuf_off += 12 + alen;
318 hdr_count++;
319 }
320
321 /* :scheme */
322 const char *scheme = "https";
323 size_t slen = strlen(scheme);
324 if (hbuf_off + 7 + 2 + slen <= sizeof(hbuf)) {
325 memcpy(hbuf + hbuf_off, ":scheme", 7);
326 memcpy(hbuf + hbuf_off + 9, scheme, slen);
327 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
328 0, 7, 9, slen);
329 hbuf_off += 9 + slen;
330 hdr_count++;
331 }
332
333 cwist_http_header_node *node = st->req_headers;
334 while (node && hdr_count < 64) {
335 if (!node->key || !node->key->data || !node->value || !node->value->data) {
336 node = node->next;
337 continue;
338 }
339 const char *name = node->key->data;
340 const char *value = node->value->data;
341 size_t nlen = node->key->size;
342 size_t vlen = node->value->size;
343 if ((nlen == 7 && strncasecmp(name, ":method", nlen) == 0) ||
344 (nlen == 5 && strncasecmp(name, ":path", nlen) == 0) ||
345 (nlen == 10 && strncasecmp(name, ":authority", nlen) == 0) ||
346 (nlen == 7 && strncasecmp(name, ":scheme", nlen) == 0)) {
347 node = node->next;
348 continue;
349 }
350 if (hbuf_off + nlen + 2 + vlen <= sizeof(hbuf)) {
351 memcpy(hbuf + hbuf_off, name, nlen);
352 memcpy(hbuf + hbuf_off + nlen + 2, value, vlen);
353 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
354 0, nlen, nlen + 2, vlen);
355 hbuf_off += nlen + 2 + vlen;
356 hdr_count++;
357 }
358 node = node->next;
359 }
360
361 lsquic_http_headers_t headers = {
362 .count = (unsigned)hdr_count,
363 .headers = headers_arr,
364 };
365
366 int has_body = (st->req_body && st->req_body_len > 0);
367 if (lsquic_stream_send_headers(stream, &headers, !has_body) != 0) {
368 lsquic_stream_close(stream);
369 return;
370 }
371
372 if (has_body) {
373 ssize_t n = lsquic_stream_write(stream, st->req_body + st->req_body_sent,
374 st->req_body_len - st->req_body_sent);
375 if (n < 0) {
376 lsquic_stream_close(stream);
377 return;
378 }
379 st->req_body_sent += (size_t)n;
380 if (st->req_body_sent >= st->req_body_len) {
381 lsquic_stream_shutdown(stream, 1);
382 st->write_done = 1;
383 lsquic_stream_wantwrite(stream, 0);
384 lsquic_stream_wantread(stream, 1);
385 }
386 } else {
387 st->write_done = 1;
388 lsquic_stream_wantwrite(stream, 0);
389 lsquic_stream_wantread(stream, 1);
390 }
391}

393static void h3c_on_close(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
394 h3c_stream_ctx_t *st = (h3c_stream_ctx_t *)st_h;
395 if (!st) return;
396 cwist_http3_client *client = (cwist_http3_client *)lsquic_conn_get_ctx(lsquic_stream_conn(stream));
397 pthread_mutex_lock(&client->mtx);
398 st->response_ready = 1;
399 if (client->active_stream == st) {
400 client->active_stream = NULL;
401 }
402 pthread_cond_broadcast(&client->cond);
403 pthread_mutex_unlock(&client->mtx);
404 free(st->req_path);
405 free(st->req_body);
406 free(st->resp_body);
407 free(st);
408}

423static void h3c_process_io(cwist_http3_client *client, int timeout_ms) {
424 lsquic_engine_t *engine = client->engine;
425 int diff = timeout_ms * 1000; /* microseconds */
426 if (diff <= 0) diff = 1000;
427
428 if (lsquic_engine_earliest_adv_tick(engine, &diff)) {
429 if (diff <= 0) diff = 0;
430 else if (diff > 1000000) diff = 1000000;
431 }
432
433 struct pollfd pfd = { .fd = client->udp_fd, .events = POLLIN };
434 int pret = poll(&pfd, 1, diff / 1000);
435
436 if (pret > 0 && (pfd.revents & POLLIN)) {
437 unsigned char buf[65535];
438 struct sockaddr_storage peer_addr;
439 socklen_t peer_addr_len = sizeof(peer_addr);
440 ssize_t nr = recvfrom(client->udp_fd, buf, sizeof(buf), 0,
441 (struct sockaddr *)&peer_addr, &peer_addr_len);
442 if (nr > 0) {
443 lsquic_engine_packet_in(engine, buf, (size_t)nr,
444 NULL, (struct sockaddr *)&peer_addr,
445 NULL, 0);
446 }
447 }
448
449 lsquic_engine_process_conns(engine);
450}

456cwist_http3_client *cwist_http3_client_create(void) {
457 h3c_global_init();
458
459 cwist_http3_client *client = calloc(1, sizeof(*client));
460 if (!client) {
461 h3c_global_cleanup();
462 return NULL;
463 }
464
465 client->udp_fd = -1;
466 client->timeout_ms = 30000;
467 client->max_retries = 0;
468 client->retry_delay_ms = 1000;
469 client->conn_timeout_ms = 5000;
470 pthread_mutex_init(&client->mtx, NULL);
471 pthread_cond_init(&client->cond, NULL);
472
473 const SSL_METHOD *method = TLS_method();
474 client->ssl_ctx = SSL_CTX_new(method);
475 if (!client->ssl_ctx) {
476 free(client);
477 h3c_global_cleanup();
478 return NULL;
479 }
480
481 SSL_CTX_set_min_proto_version(client->ssl_ctx, TLS1_3_VERSION);
482 SSL_CTX_set_max_proto_version(client->ssl_ctx, TLS1_3_VERSION);
483 SSL_CTX_set_default_verify_paths(client->ssl_ctx);
484
485 struct lsquic_engine_settings settings;
486 lsquic_engine_init_settings(&settings, LSENG_HTTP);
487 settings.es_versions = (1 << LSQVER_I001) | (1 << LSQVER_I002);
488 settings.es_ping_period = 15;
489 settings.es_noprogress_timeout = 30;
490 settings.es_ecn = 1;
491 settings.es_pace_packets = 1;
492 settings.es_optimistic_nat = 1;
493
494 char err_buf[256];
495 if (lsquic_engine_check_settings(&settings, LSENG_HTTP,
496 err_buf, sizeof(err_buf)) != 0) {
497 fprintf(stderr, "[HTTP/3-CLIENT] Invalid engine settings: %s\n", err_buf);
498 SSL_CTX_free(client->ssl_ctx);
499 pthread_mutex_destroy(&client->mtx);
500 pthread_cond_destroy(&client->cond);
501 free(client);
502 h3c_global_cleanup();
503 return NULL;
504 }

584/* ------------------------------------------------------------------ */
585/* Synchronous request */
586/* ------------------------------------------------------------------ */
587
588cwist_error_t cwist_http3_client_request(cwist_http3_client *client,
589 const char *path,
590 cwist_http_method_t method,
591 cwist_http_header_node *headers,
592 const char *body,
593 size_t body_len,
594 cwist_http_response **out_response) {
595 cwist_error_t err = make_error(CWIST_ERR_INT16);
596 if (!client || !client->engine || !client->host || !path || !out_response) {
597 err.error.err_i16 = -1;
598 return err;
599 }
600
601 *out_response = NULL;
602
603 /* Create UDP socket if needed */
604 if (client->udp_fd < 0) {
605 client->udp_fd = socket(AF_INET, SOCK_DGRAM, 0);
606 if (client->udp_fd < 0) {
607 err.error.err_i16 = -1;
608 return err;
609 }
610 int flags = fcntl(client->udp_fd, F_GETFL, 0);
611 if (flags >= 0) fcntl(client->udp_fd, F_SETFL, flags | O_NONBLOCK);
612 }
613
614 /* Resolve host if not already done */
615 if (client->peer_addr_len == 0) {
616 struct addrinfo hints = {0}, *res = NULL;
617 hints.ai_family = AF_INET;
618 hints.ai_socktype = SOCK_DGRAM;
619 char port_str[8];
620 snprintf(port_str, sizeof(port_str), "%u", client->port);
621 if (getaddrinfo(client->host, port_str, &hints, &res) != 0 || !res) {
622 err.error.err_i16 = -1;
623 return err;
624 }
625 memcpy(&client->peer_addr, res->ai_addr, res->ai_addrlen);
626 client->peer_addr_len = res->ai_addrlen;
627 freeaddrinfo(res);
628 }
629
630 /* ---------------------------------------------------------------- */
631 /* Retry loop with exponential backoff */
632 /* ---------------------------------------------------------------- */
633 int attempt = 0;
634 int max_attempts = 1 + client->max_retries;
635 while (attempt < max_attempts) {
636 /* Ensure connection is healthy */
637 if (client->conn) {
638 enum LSQUIC_CONN_STATUS st_status = lsquic_conn_status(client->conn, NULL, 0);
639 if (st_status == LSCONN_ST_CLOSED || st_status == LSCONN_ST_TIMED_OUT ||
640 st_status == LSCONN_ST_ERROR || st_status == LSCONN_ST_RESET ||
641 st_status == LSCONN_ST_HSK_FAILURE) {
642 client->conn = NULL; /* lsquic will clean it up via engine */
643 }
644 }
645
646 if (!client->conn) {
647 client->conn = lsquic_engine_connect(client->engine, N_LSQVER,
648 NULL,
649 (struct sockaddr *)&client->peer_addr,
650 client, NULL,
651 client->host, 0,
652 NULL, 0, NULL, 0);
653 if (!client->conn) {
654 err.error.err_i16 = -1;
655 goto retry_backoff;
656 }
657 }
658
659 /* Request a new stream */
660 pthread_mutex_lock(&client->mtx);
661 client->active_stream = NULL;
662 pthread_mutex_unlock(&client->mtx);
663 lsquic_conn_make_stream(client->conn);
664
665 /* I/O loop until stream is created */
666 struct timespec stream_deadline;
667 clock_gettime(CLOCK_REALTIME, &stream_deadline);
668 stream_deadline.tv_sec += client->conn_timeout_ms / 1000;
669 stream_deadline.tv_nsec += (client->conn_timeout_ms % 1000) * 1000000;
670 if (stream_deadline.tv_nsec >= 1000000000) {
671 stream_deadline.tv_sec++;
672 stream_deadline.tv_nsec -= 1000000000;
673 }
674 h3c_stream_ctx_t *st = NULL;
675 while (!st) {
676 h3c_process_io(client, 100);
677 pthread_mutex_lock(&client->mtx);
678 st = client->active_stream;
679 pthread_mutex_unlock(&client->mtx);
680 struct timespec now;
681 clock_gettime(CLOCK_REALTIME, &now);
682 if (now.tv_sec > stream_deadline.tv_sec ||
683 (now.tv_sec == stream_deadline.tv_sec && now.tv_nsec >= stream_deadline.tv_nsec)) {
684 break;
685 }
686 }
687 if (!st) {
688 err.error.err_i16 = -1;
689 goto retry_backoff;
690 }
691
692 st->req_path = strdup(path);
693 if (!st->req_path) {
694 err.error.err_i16 = -1;
695 goto retry_backoff;
696 }
697 st->req_method = method;
698 st->req_headers = headers;
699
700 /* Store request body */
701 if (body && body_len > 0) {
702 st->req_body = malloc(body_len);
703 if (!st->req_body) {
704 err.error.err_i16 = -1;
705 goto retry_backoff;
706 }
707 memcpy(st->req_body, body, body_len);
708 st->req_body_len = body_len;
709 }
710
711 /* I/O loop until response is ready or timeout */
712 struct timespec deadline;
713 clock_gettime(CLOCK_REALTIME, &deadline);
714 deadline.tv_sec += client->timeout_ms / 1000;
715 deadline.tv_nsec += (client->timeout_ms % 1000) * 1000000;
716 if (deadline.tv_nsec >= 1000000000) {
717 deadline.tv_sec++;
718 deadline.tv_nsec -= 1000000000;
719 }
720
721 pthread_mutex_lock(&client->mtx);
722 while (!st->response_ready) {
723 pthread_mutex_unlock(&client->mtx);
724 h3c_process_io(client, 100);
725 pthread_mutex_lock(&client->mtx);
726 struct timespec now;
727 clock_gettime(CLOCK_REALTIME, &now);
728 if (now.tv_sec > deadline.tv_sec ||
729 (now.tv_sec == deadline.tv_sec && now.tv_nsec >= deadline.tv_nsec)) {
730 break;
731 }
732 }
733 pthread_mutex_unlock(&client->mtx);
734
735 if (st->response_ready) {
736 *out_response = st->res;
737 st->res = NULL; /* Transfer ownership */
738 err.error.err_i16 = 0;
739 return err;
740 }
741
742 /* Timeout on this attempt – clean up stream state and retry */
743 err.error.err_i16 = -1;
744 pthread_mutex_lock(&client->mtx);
745 client->active_stream = NULL;
746 pthread_mutex_unlock(&client->mtx);
747 free(st->req_body);
748 free(st->req_path);
749 st->req_body = NULL;
750 st->req_path = NULL;
751 free(st);
752
753 retry_backoff:
754 attempt++;
755 if (attempt < max_attempts && client->retry_delay_ms > 0) {
756 int backoff = client->retry_delay_ms * (1 << (attempt - 1));
757 if (backoff > 30000) backoff = 30000; /* cap at 30s */
758 usleep((useconds_t)backoff * 1000);
759 }
760 }
761
762 return err;
763}
