43void cwist_logger_log(cwist_logger *log, cwist_log_level_t level, const char *fmt, ...) {
44 if (!log || level < log->level) return;
45 FILE *out = level_file(level);
46 time_t now = time(NULL);
47 struct tm *tm_info = localtime(&now);
48 char time_buf[32];
49 strftime(time_buf, sizeof(time_buf), "%Y-%m-%d %H:%M:%S", tm_info);
50 if (log->use_colors) {
51 fprintf(out, "%s[%s]%s [%s] %s: ", level_colors[level], time_buf, color_reset, level_strings[level], log->name);
52 } else {
53 fprintf(out, "[%s] [%s] %s: ", time_buf, level_strings[level], log->name);
54 }
55 va_list args;
56 va_start(args, fmt);
57 vfprintf(out, fmt, args);
58 va_end(args);
59 fprintf(out, "\n");
60 fflush(out);
