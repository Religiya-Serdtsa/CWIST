22static void bind_add_error(cwist_bind_result_t *r, const char *key, const char *msg) {
23 if (!r) return;
24 if (r->error_count >= CWIST_BIND_MAX_ERRORS) return;
25 cwist_bind_error_t *e = &r->errors[r->error_count++];
26 snprintf(e->key, sizeof(e->key), "%s", key ? key : "(unknown)");
27 snprintf(e->message, sizeof(e->message), "%s", msg ? msg : "validation failed");
28 r->ok = false;
29}

35static bool is_email_like(const char *s, size_t len) {
36 if (!s || len < 5) return false;
37 const char *at = strchr(s, '@');
38 if (!at) return false;
39 if (at == s) return false;
40 const char *dot = strrchr(at, '.');
41 if (!dot || dot <= at + 1) return false;
42 if (dot + 1 == s + len) return false; /* trailing dot */
43 return true;
44}

50static bool check_rule(const cwist_bind_rule_t *rule,
51 const char *value,
52 size_t len,
53 cwist_bind_result_t *r,
54 const char *key) {
55 char buf[CWIST_BIND_MSG_MAX];
56
57 switch (rule->type) {
58 case CWIST_BIND_RULE_REQUIRED:
59 if (!value || len == 0) {
60 snprintf(buf, sizeof(buf), "%s",
61 rule->error_message ? rule->error_message : "field is required");
62 bind_add_error(r, key, buf);
63 return false;
64 }
65 break;
66
67 case CWIST_BIND_RULE_MIN_LEN:
68 if (!value || len < rule->u.min_len) {
69 snprintf(buf, sizeof(buf), "%s: minimum length is %zu",
70 rule->error_message ? rule->error_message : "too short",
71 rule->u.min_len);
72 bind_add_error(r, key, buf);
73 return false;
74 }
75 break;
76
77 case CWIST_BIND_RULE_MAX_LEN:
78 if (value && len > rule->u.max_len) {
79 snprintf(buf, sizeof(buf), "%s: maximum length is %zu",
80 rule->error_message ? rule->error_message : "too long",
81 rule->u.max_len);
82 bind_add_error(r, key, buf);
83 return false;
84 }
85 break;
86
87 case CWIST_BIND_RULE_MIN_VAL: {
88 long double v = 0.0L;
89 if (value) v = strtold(value, NULL);
90 if (!value || v < rule->u.min_val) {
91 snprintf(buf, sizeof(buf), "%s: minimum value is %Lg",
92 rule->error_message ? rule->error_message : "value too small",
93 rule->u.min_val);
94 bind_add_error(r, key, buf);
95 return false;
96 }
97 break;
98 }
99
100 case CWIST_BIND_RULE_MAX_VAL: {
101 long double v = 0.0L;
102 if (value) v = strtold(value, NULL);
103 if (!value || v > rule->u.max_val) {
104 snprintf(buf, sizeof(buf), "%s: maximum value is %Lg",
105 rule->error_message ? rule->error_message : "value too large",
106 rule->u.max_val);
107 bind_add_error(r, key, buf);
108 return false;
109 }
110 break;
111 }
112
113 case CWIST_BIND_RULE_REGEX: {
114 if (!value) {
115 bind_add_error(r, key, "missing value for regex check");
116 return false;
117 }
118 regex_t re;
119 int rc = regcomp(&re, rule->u.pattern, REG_EXTENDED | REG_NOSUB);
120 if (rc != 0) {
121 bind_add_error(r, key, "invalid regex pattern");
122 return false;
123 }
124 rc = regexec(&re, value, 0, NULL, 0);
125 regfree(&re);
126 if (rc != 0) {
127 snprintf(buf, sizeof(buf), "%s",
128 rule->error_message ? rule->error_message : "value does not match pattern");
129 bind_add_error(r, key, buf);
130 return false;
131 }
132 break;
133 }
134
135 case CWIST_BIND_RULE_EMAIL:
136 if (!value || !is_email_like(value, len)) {
137 snprintf(buf, sizeof(buf), "%s",
138 rule->error_message ? rule->error_message : "invalid email format");
139 bind_add_error(r, key, buf);
140 return false;
141 }
142 break;
143
144 case CWIST_BIND_RULE_CUSTOM:
145 if (!value || !rule->u.custom.fn(value, len, rule->u.custom.ctx)) {
146 snprintf(buf, sizeof(buf), "%s",
147 rule->error_message ? rule->error_message : "custom validation failed");
148 bind_add_error(r, key, buf);
149 return false;
150 }
151 break;
152
153 default:
154 break;
155 }
156 return true;
157}

162static bool run_rules(const cwist_bind_field_t *f,
163 const char *value,
164 size_t len,
165 cwist_bind_result_t *r) {
166 if (!f->rules) return true;
167 bool all_ok = true;
168 for (size_t i = 0; i < CWIST_BIND_MAX_ERRORS; ++i) {
169 if (f->rules[i].type == (cwist_bind_rule_type_t)-1) break;
170 if (!check_rule(&f->rules[i], value, len, r, f->json_key))
171 all_ok = false;
172 }
173 return all_ok;
174}

179static bool write_value(const cwist_bind_field_t *f,
180 const char *value,
181 void *out,
182 cwist_bind_result_t *r) {
183 void *dest = (char *)out + f->target_offset;
184
185 switch (f->target_type) {
186 case CWIST_BIND_BOOL: {
187 bool v = false;
188 if (value) {
189 if (strcmp(value, "true") == 0 || strcmp(value, "1") == 0) v = true;
190 else if (strcmp(value, "false") == 0 || strcmp(value, "0") == 0) v = false;
191 else { bind_add_error(r, f->json_key, "expected boolean"); return false; }
192 }
193 memcpy(dest, &v, sizeof(v));
194 break;
195 }
196 case CWIST_BIND_INT: {
197 long long v = 0;
198 if (value) {
199 char *end = NULL;
200 errno = 0;
201 v = strtoll(value, &end, 10);
202 if (errno == ERANGE || end == value || *end != '\0') {
203 bind_add_error(r, f->json_key, "expected integer");
204 return false;
205 }
206 }
207 memcpy(dest, &v, f->target_size <= sizeof(v) ? f->target_size : sizeof(v));
208 break;
209 }
210 case CWIST_BIND_UINT: {
211 unsigned long long v = 0;
212 if (value) {
213 char *end = NULL;
214 errno = 0;
215 v = strtoull(value, &end, 10);
216 if (errno == ERANGE || end == value || *end != '\0') {
217 bind_add_error(r, f->json_key, "expected unsigned integer");
218 return false;
219 }
220 }
221 memcpy(dest, &v, f->target_size <= sizeof(v) ? f->target_size : sizeof(v));
222 break;
223 }
224 case CWIST_BIND_FLOAT: {
225 float v = 0.0f;
226 if (value) {
227 char *end = NULL;
228 v = strtof(value, &end);
229 if (end == value || *end != '\0') {
230 bind_add_error(r, f->json_key, "expected float");
231 return false;
232 }
233 }
234 memcpy(dest, &v, sizeof(v));
235 break;
236 }
237 case CWIST_BIND_DOUBLE: {
238 double v = 0.0;
239 if (value) {
240 char *end = NULL;
241 v = strtod(value, &end);
242 if (end == value || *end != '\0') {
243 bind_add_error(r, f->json_key, "expected double");
244 return false;
245 }
246 }
247 memcpy(dest, &v, sizeof(v));
248 break;
249 }
250 case CWIST_BIND_STRING: {
251 if (f->target_size == 0) {
252 bind_add_error(r, f->json_key, "bad target size");
253 return false;
254 }
255 if (value) {
256 size_t copy = strlen(value);
257 if (copy >= f->target_size) copy = f->target_size - 1;
258 memcpy(dest, value, copy);
259 ((char *)dest)[copy] = '\0';
260 } else {
261 ((char *)dest)[0] = '\0';
262 }
263 break;
264 }
265 case CWIST_BIND_SSTRING: {
266 cwist_sstring *ss = *(cwist_sstring **)dest;
267 if (!ss) {
268 bind_add_error(r, f->json_key, "null sstring target");
269 return false;
270 }
271 if (value) cwist_sstring_assign(ss, (char *)value);
272 else cwist_sstring_assign(ss, "");
273 break;
274 }
275 case CWIST_BIND_JSON_OBJECT: {
276 cJSON **pp = (cJSON **)dest;
277 if (!pp) {
278 bind_add_error(r, f->json_key, "null cJSON target");
279 return false;
280 }
281 if (*pp) { cJSON_Delete(*pp); *pp = NULL; }
282 if (value) {
283 *pp = cJSON_Parse(value);
284 if (!*pp) {
285 bind_add_error(r, f->json_key, "invalid nested JSON");
286 return false;
287 }
288 }
289 break;
290 }
291 default:
292 bind_add_error(r, f->json_key, "unsupported bind target type");
293 return false;
294 }
295 return true;
296}

301static bool cwist_bind_generic(const cwist_bind_schema_t *schema,
302 const char *payload,
303 bool is_json,
304 void *out,
305 cwist_bind_result_t *result) {
306 if (!schema || !out || !result) return false;
307 memset(result, 0, sizeof(*result));
308 result->ok = true;
309
310 if (!payload) {
311 bind_add_error(result, "(root)", "missing request body");
312 return false;
313 }
314
315 cJSON *root = NULL;
316 if (is_json) {
317 root = cJSON_Parse(payload);
318 if (!root) {
319 bind_add_error(result, "(root)", "JSON parse error");
320 return false;
321 }
322 }
323
324 for (size_t i = 0; i < schema->field_count; ++i) {
325 const cwist_bind_field_t *f = &schema->fields[i];
326 const char *raw_value = NULL;
327
328 if (is_json && root) {
329 const cJSON *item = cJSON_GetObjectItemCaseSensitive(root, f->json_key);
330 if (item) {
331 if (cJSON_IsString(item)) {
332 raw_value = item->valuestring;
333 } else if (cJSON_IsNumber(item)) {
334 /* Keep number as string for uniform validation then convert */
335 static __thread char numbuf[64];
336 double intpart;
337 if (modf(item->valuedouble, &intpart) == 0.0) {
338 snprintf(numbuf, sizeof(numbuf), "%lld", (long long)item->valuedouble);
339 } else {
340 snprintf(numbuf, sizeof(numbuf), "%g", item->valuedouble);
341 }
342 raw_value = numbuf;
343 } else if (cJSON_IsBool(item)) {
344 raw_value = cJSON_IsTrue(item) ? "true" : "false";
345 } else if (cJSON_IsNull(item)) {
346 raw_value = NULL;
347 } else if (cJSON_IsObject(item) || cJSON_IsArray(item)) {
348 /* For JSON_OBJECT targets, serialise the subtree */
349 if (f->target_type == CWIST_BIND_JSON_OBJECT) {
350 char *printed = cJSON_PrintUnformatted(item);
351 raw_value = printed; /* transient; written below */
352 write_value(f, raw_value, out, result);
353 free(printed);
354 continue;
355 } else {
356 raw_value = cJSON_PrintUnformatted(item);
357 /* fall through to string validation then cleanup */
358 }
359 }
360 }
361 } else {
362 /* Form parsing: simple key=value scan (naïve, no unescaping) */
363 const char *cursor = payload;
364 size_t key_len = strlen(f->json_key);
365 while (cursor && *cursor) {
366 if (strncmp(cursor, f->json_key, key_len) == 0 && cursor[key_len] == '=') {
367 const char *start = cursor + key_len + 1;
368 const char *end = strchr(start, '&');
369 size_t vlen = end ? (size_t)(end - start) : strlen(start);
370 /* Zero-copy reference into payload; safe because req->body is stable */
371 raw_value = start;
372 /* Temporarily null-terminate for validation (destructive to body). */
373 /* We make a bounded copy instead to be safe. */
374 if (vlen > 0) {
375 static __thread char formbuf[4096];
376 if (vlen >= sizeof(formbuf)) vlen = sizeof(formbuf) - 1;
377 memcpy(formbuf, start, vlen);
378 formbuf[vlen] = '\0';
379 raw_value = formbuf;
380 } else {
381 raw_value = "";
382 }
383 break;
384 }
385 const char *next = strchr(cursor, '&');
386 cursor = next ? next + 1 : NULL;
387 }
388 }
389
390 size_t val_len = raw_value ? strlen(raw_value) : 0;
391
392 if (!run_rules(f, raw_value, val_len, result)) {
393 /* continue to collect more errors */
394 }
395
396 if (result->error_count == 0 || result->ok) {
397 if (!write_value(f, raw_value, out, result)) {
398 result->ok = false;
399 }
400 }
401
402 /* Clean up temporary printed JSON if we allocated one above */
403 if (!is_json) { /* already handled */ }
404 }
405
406 if (root) cJSON_Delete(root);
407 result->ok = (result->error_count == 0);
408 return result->ok;
409}

431void cwist_bind_write_error_response(cwist_http_response *res,
432 const cwist_bind_result_t *result) {
433 if (!res || !result) return;
434 res->status_code = CWIST_HTTP_BAD_REQUEST;
435 cwist_sstring_assign(res->status_text, "Bad Request");
436
437 cwist_json_builder *jb = cwist_json_builder_create();
438 cwist_json_begin_object(jb);
439 cwist_json_add_bool(jb, "success", false);
440 cwist_json_begin_array(jb, "errors");
441 for (size_t i = 0; i < result->error_count; ++i) {
442 cwist_json_begin_object(jb);
443 cwist_json_add_string(jb, "field", result->errors[i].key);
444 cwist_json_add_string(jb, "message", result->errors[i].message);
445 cwist_json_end_object(jb);
446 }
447 cwist_json_end_array(jb);
448 cwist_json_end_object(jb);
449
450 const char *json = cwist_json_get_raw(jb);
451 cwist_sstring_assign(res->body, (char *)json);
452 cwist_http_header_add(&res->headers, "Content-Type", "application/json");
453 cwist_json_builder_destroy(jb);
454}
