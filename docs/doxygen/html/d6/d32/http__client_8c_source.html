34static size_t write_callback(void *contents, size_t size, size_t nmemb, void *userp) {
35 size_t total = size * nmemb;
36 response_body_t *body = (response_body_t *)userp;
37 size_t need = body->size + total;
38 if (need > body->cap) {
39 size_t new_cap = body->cap ? body->cap * 2 : 4096;
40 while (new_cap < need) new_cap *= 2;
41 char *tmp = realloc(body->data, new_cap);
42 if (!tmp) return 0;
43 body->data = tmp;
44 body->cap = new_cap;
45 }
46 memcpy(body->data + body->size, contents, total);
47 body->size += total;
48 return total;
49}

51static size_t header_callback(char *ptr, size_t size, size_t nmemb, void *userp) {
52 size_t total = size * nmemb;
53 response_headers_t *rh = (response_headers_t *)userp;
54
55 /* libcurl sends headers as "Name: value\r\n" or "HTTP/1.1 200 OK\r\n" */
56 if (total > 5 && strncasecmp(ptr, "HTTP/", 5) == 0) {
57 /* Status line: parse status code */
58 char *p = ptr;
59 while (*p && !isspace((unsigned char)*p)) p++;
60 while (*p && isspace((unsigned char)*p)) p++;
61 rh->status_code = strtol(p, NULL, 10);
62 return total;
63 }
64
65 /* Find colon separator */
66 char *colon = memchr(ptr, ':', total);
67 if (!colon) return total;
68
69 size_t name_len = (size_t)(colon - ptr);
70 if (name_len == 0) return total;
71
72 char *value = colon + 1;
73 size_t value_len = total - name_len - 1;
74
75 /* Trim leading spaces in value */
76 while (value_len > 0 && isspace((unsigned char)*value)) {
77 value++;
78 value_len--;
79 }
80 /* Trim trailing whitespace (\r\n) */
81 while (value_len > 0 && isspace((unsigned char)value[value_len - 1])) {
82 value_len--;
83 }
84
85 if (name_len > 0 && value_len > 0) {
86 char *name = cwist_alloc(name_len + 1);
87 if (!name) return total;
88 memcpy(name, ptr, name_len);
89 name[name_len] = '\0';
90
91 char *val = cwist_alloc(value_len + 1);
92 if (!val) {
93 cwist_free(name);
94 return total;
95 }
96 memcpy(val, value, value_len);
97 val[value_len] = '\0';
98
99 cwist_http_header_node *node = cwist_alloc(sizeof(*node));
100 if (!node) {
101 cwist_free(name);
102 cwist_free(val);
103 return total;
104 }
105 node->key = cwist_sstring_create();
106 node->value = cwist_sstring_create();
107 if (!node->key || !node->value) {
108 cwist_sstring_destroy(node->key);
109 cwist_sstring_destroy(node->value);
110 cwist_free(node);
111 cwist_free(name);
112 cwist_free(val);
113 return total;
114 }
115 cwist_sstring_assign(node->key, name);
116 cwist_sstring_assign(node->value, val);
117 node->next = rh->headers;
118 rh->headers = node;
119
120 cwist_free(name);
121 cwist_free(val);
122 }
123
124 return total;
125}

140cwist_http_client *cwist_http_client_create(void) {
141 cwist_curl_global_acquire();
142
143 cwist_http_client *client = cwist_alloc(sizeof(*client));
144 if (!client) {
145 cwist_curl_global_release();
146 return NULL;
147 }
148
149 client->curl = curl_easy_init();
150 if (!client->curl) {
151 cwist_free(client);
152 cwist_curl_global_release();
153 return NULL;
154 }
155
156 client->follow_redirects = 1;
157 client->timeout_ms = 30000;
158 client->ca_bundle = NULL;
159 client->altsvc_enabled = 0;
160 client->altsvc_db = NULL;
161
162 curl_easy_setopt(client->curl, CURLOPT_FOLLOWLOCATION, 1L);
163 curl_easy_setopt(client->curl, CURLOPT_MAXREDIRS, 10L);
164 curl_easy_setopt(client->curl, CURLOPT_TIMEOUT_MS, 30000L);
165 curl_easy_setopt(client->curl, CURLOPT_TCP_KEEPALIVE, 1L);
166 curl_easy_setopt(client->curl, CURLOPT_ACCEPT_ENCODING, "");
167 curl_easy_setopt(client->curl, CURLOPT_USERAGENT, "CWIST-HTTP-Client/1.0");
168
169 return client;
170}

220cwist_error_t cwist_http_client_request(cwist_http_client *client,
221 const char *url,
222 cwist_http_method_t method,
223 cwist_http_header_node *headers,
224 const char *body,
225 size_t body_len,
226 cwist_http_response **out_response) {
227 cwist_error_t err = make_error(CWIST_ERR_INT16);
228 if (!client || !client->curl || !url || !out_response) {
229 err.error.err_i16 = -1;
230 return err;
231 }
232
233 *out_response = NULL;
234
235 CURL *curl = client->curl;
236
237 /* Reset handle for reuse */
238 curl_easy_reset(curl);
239
240 /* Re-apply persistent settings after reset */
241 curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, client->follow_redirects ? 1L : 0L);
242 curl_easy_setopt(curl, CURLOPT_TIMEOUT_MS, (long)client->timeout_ms);
243 curl_easy_setopt(curl, CURLOPT_TCP_KEEPALIVE, 1L);
244 curl_easy_setopt(curl, CURLOPT_ACCEPT_ENCODING, "");
245 curl_easy_setopt(curl, CURLOPT_USERAGENT, "CWIST-HTTP-Client/1.0");
246 if (client->ca_bundle) {
247 curl_easy_setopt(curl, CURLOPT_CAINFO, client->ca_bundle);
248 }
249 if (client->altsvc_enabled) {
250 curl_easy_setopt(curl, CURLOPT_ALTSVC_CTRL,
251 (long)(CURLALTSVC_H1 | CURLALTSVC_H2 | CURLALTSVC_H3));
252 curl_easy_setopt(curl, CURLOPT_ALTSVC, client->altsvc_db);
253 }
254
255 /* URL */
256 curl_easy_setopt(curl, CURLOPT_URL, url);
257
258 /* Method */
259 const char *method_str = cwist_http_method_to_string(method);
260 if (strcmp(method_str, "GET") == 0) {
261 curl_easy_setopt(curl, CURLOPT_HTTPGET, 1L);
262 } else if (strcmp(method_str, "POST") == 0) {
263 curl_easy_setopt(curl, CURLOPT_POST, 1L);
264 } else if (strcmp(method_str, "PUT") == 0) {
265 curl_easy_setopt(curl, CURLOPT_UPLOAD, 0L);
266 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PUT");
267 } else if (strcmp(method_str, "DELETE") == 0) {
268 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "DELETE");
269 } else if (strcmp(method_str, "PATCH") == 0) {
270 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PATCH");
271 } else if (strcmp(method_str, "HEAD") == 0) {
272 curl_easy_setopt(curl, CURLOPT_NOBODY, 1L);
273 } else if (strcmp(method_str, "OPTIONS") == 0) {
274 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "OPTIONS");
275 } else {
276 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, method_str);
277 }
278
279 /* Request headers */
280 struct curl_slist *curl_headers = NULL;
281 cwist_http_header_node *node = headers;
282 while (node) {
283 if (node->key && node->key->data && node->value && node->value->data) {
284 char header_line[4096];
285 int n = snprintf(header_line, sizeof(header_line), "%s: %s",
286 node->key->data, node->value->data);
287 if (n > 0 && (size_t)n < sizeof(header_line)) {
288 curl_headers = curl_slist_append(curl_headers, header_line);
289 }
290 }
291 node = node->next;
292 }
293 if (curl_headers) {
294 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, curl_headers);
295 }
296
297 /* Request body */
298 if (body && body_len > 0) {
299 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE_LARGE, (curl_off_t)body_len);
300 curl_easy_setopt(curl, CURLOPT_COPYPOSTFIELDS, body);
301 }
302
303 /* Response accumulators */
304 response_body_t resp_body = {0};
305 response_headers_t resp_hdrs = {0};
306
307 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_callback);
308 curl_easy_setopt(curl, CURLOPT_WRITEDATA, &resp_body);
309 curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, header_callback);
310 curl_easy_setopt(curl, CURLOPT_HEADERDATA, &resp_hdrs);
311
312 /* Prefer HTTP/2, allow downgrade to HTTP/1.1 */
313 curl_easy_setopt(curl, CURLOPT_HTTP_VERSION, CURL_HTTP_VERSION_2_0);
314
315 /* Perform */
316 CURLcode res = curl_easy_perform(curl);
317
318 /* Clean up request headers */
319 if (curl_headers) {
320 curl_slist_free_all(curl_headers);
321 }
322
323 if (res != CURLE_OK) {
324 free(resp_body.data);
325 cwist_http_header_free_all(resp_hdrs.headers);
326 err.error.err_i16 = -1;
327 return err;
328 }
329
330 /* Build response object */
331 cwist_http_response *response = cwist_http_response_create();
332 if (!response) {
333 free(resp_body.data);
334 cwist_http_header_free_all(resp_hdrs.headers);
335 err.error.err_i16 = -1;
336 return err;
337 }
338
339 response->status_code = (cwist_http_status_t)resp_hdrs.status_code;
340 if (resp_body.data && resp_body.size > 0) {
341 cwist_sstring_assign_len(response->body, resp_body.data, resp_body.size);
342 }
343
344 /* Transfer collected headers */
345 response->headers = resp_hdrs.headers;
346
347 /* Populate status text based on code */
348 switch (response->status_code) {
349 case CWIST_HTTP_OK: cwist_sstring_assign(response->status_text, "OK"); break;
350 case CWIST_HTTP_CREATED: cwist_sstring_assign(response->status_text, "Created"); break;
351 case CWIST_HTTP_NO_CONTENT: cwist_sstring_assign(response->status_text, "No Content"); break;
352 case CWIST_HTTP_BAD_REQUEST: cwist_sstring_assign(response->status_text, "Bad Request"); break;
353 case CWIST_HTTP_UNAUTHORIZED: cwist_sstring_assign(response->status_text, "Unauthorized"); break;
354 case CWIST_HTTP_FORBIDDEN: cwist_sstring_assign(response->status_text, "Forbidden"); break;
355 case CWIST_HTTP_NOT_FOUND: cwist_sstring_assign(response->status_text, "Not Found"); break;
356 case CWIST_HTTP_INTERNAL_ERROR: cwist_sstring_assign(response->status_text, "Internal Server Error"); break;
357 case CWIST_HTTP_NOT_IMPLEMENTED: cwist_sstring_assign(response->status_text, "Not Implemented"); break;
358 case CWIST_HTTP_SERVICE_UNAVAILABLE: cwist_sstring_assign(response->status_text, "Service Unavailable"); break;
359 default: cwist_sstring_assign(response->status_text, ""); break;
360 }
361
362 free(resp_body.data);
363 *out_response = response;
364 err.error.err_i16 = 0;
365 return err;
366}
