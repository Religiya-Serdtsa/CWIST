61static size_t write_callback(void *contents, size_t size, size_t nmemb, void *userp) {
62 size_t total = size * nmemb;
63 response_body_t *body = (response_body_t *)userp;
64 size_t need = body->size + total;
65 if (need > body->cap) {
66 size_t new_cap = body->cap ? body->cap * 2 : 4096;
67 while (new_cap < need) new_cap *= 2;
68 char *tmp = realloc(body->data, new_cap);
69 if (!tmp) return 0;
70 body->data = tmp;
71 body->cap = new_cap;
72 }
73 memcpy(body->data + body->size, contents, total);
74 body->size += total;
75 return total;
76}

78static size_t header_callback(char *ptr, size_t size, size_t nmemb, void *userp) {
79 size_t total = size * nmemb;
80 response_headers_t *rh = (response_headers_t *)userp;
81
82 /* libcurl sends headers as "Name: value\r\n" or "HTTP/1.1 200 OK\r\n" */
83 if (total > 5 && strncasecmp(ptr, "HTTP/", 5) == 0) {
84 /* Status line: parse status code */
85 char *p = ptr;
86 while (*p && !isspace((unsigned char)*p)) p++;
87 while (*p && isspace((unsigned char)*p)) p++;
88 rh->status_code = strtol(p, NULL, 10);
89 return total;
90 }
91
92 /* Find colon separator */
93 char *colon = memchr(ptr, ':', total);
94 if (!colon) return total;
95
96 size_t name_len = (size_t)(colon - ptr);
97 if (name_len == 0) return total;
98
99 char *value = colon + 1;
100 size_t value_len = total - name_len - 1;
101
102 /* Trim leading spaces in value */
103 while (value_len > 0 && isspace((unsigned char)*value)) {
104 value++;
105 value_len--;
106 }
107 /* Trim trailing whitespace (\r\n) */
108 while (value_len > 0 && isspace((unsigned char)value[value_len - 1])) {
109 value_len--;
110 }
111
112 if (name_len > 0 && value_len > 0) {
113 char *name = cwist_alloc(name_len + 1);
114 if (!name) return total;
115 memcpy(name, ptr, name_len);
116 name[name_len] = '\0';
117
118 char *val = cwist_alloc(value_len + 1);
119 if (!val) {
120 cwist_free(name);
121 return total;
122 }
123 memcpy(val, value, value_len);
124 val[value_len] = '\0';
125
126 cwist_http_header_node *node = cwist_alloc(sizeof(*node));
127 if (!node) {
128 cwist_free(name);
129 cwist_free(val);
130 return total;
131 }
132 node->key = cwist_sstring_create();
133 node->value = cwist_sstring_create();
134 if (!node->key || !node->value) {
135 cwist_sstring_destroy(node->key);
136 cwist_sstring_destroy(node->value);
137 cwist_free(node);
138 cwist_free(name);
139 cwist_free(val);
140 return total;
141 }
142 cwist_sstring_assign(node->key, name);
143 cwist_sstring_assign(node->value, val);
144 node->next = rh->headers;
145 rh->headers = node;
146
147 cwist_free(name);
148 cwist_free(val);
149 }
150
151 return total;
152}

167cwist_http_client *cwist_http_client_create(void) {
168 curl_global_init_ref();

170 cwist_http_client *client = cwist_alloc(sizeof(*client));

176 client->curl = curl_easy_init();

183 client->follow_redirects = 1;

186 client->altsvc_enabled = 0;
187 client->altsvc_db = NULL;

189 curl_easy_setopt(client->curl, CURLOPT_FOLLOWLOCATION, 1L);
190 curl_easy_setopt(client->curl, CURLOPT_MAXREDIRS, 10L);
191 curl_easy_setopt(client->curl, CURLOPT_TIMEOUT_MS, 30000L);
192 curl_easy_setopt(client->curl, CURLOPT_TCP_KEEPALIVE, 1L);
193 curl_easy_setopt(client->curl, CURLOPT_ACCEPT_ENCODING, "");
194 curl_easy_setopt(client->curl, CURLOPT_USERAGENT, "CWIST-HTTP-Client/1.0");

254 cwist_error_t err = make_error(CWIST_ERR_INT16);
255 if (!client || !client->curl || !url || !out_response) {
256 err.error.err_i16 = -1;
257 return err;
258 }
259
260 *out_response = NULL;
261
262 CURL *curl = client->curl;
263
264 /* Reset handle for reuse */
265 curl_easy_reset(curl);
266
267 /* Re-apply persistent settings after reset */
268 curl_easy_setopt(curl, CURLOPT_FOLLOWLOCATION, client->follow_redirects ? 1L : 0L);
269 curl_easy_setopt(curl, CURLOPT_TIMEOUT_MS, (long)client->timeout_ms);
270 curl_easy_setopt(curl, CURLOPT_TCP_KEEPALIVE, 1L);
271 curl_easy_setopt(curl, CURLOPT_ACCEPT_ENCODING, "");
272 curl_easy_setopt(curl, CURLOPT_USERAGENT, "CWIST-HTTP-Client/1.0");
273 if (client->ca_bundle) {
274 curl_easy_setopt(curl, CURLOPT_CAINFO, client->ca_bundle);
275 }
276 if (client->altsvc_enabled) {
277 curl_easy_setopt(curl, CURLOPT_ALTSVC_CTRL,
278 (long)(CURLALTSVC_H1 | CURLALTSVC_H2 | CURLALTSVC_H3));
279 curl_easy_setopt(curl, CURLOPT_ALTSVC, client->altsvc_db);
280 }
281
282 /* URL */
283 curl_easy_setopt(curl, CURLOPT_URL, url);
284
285 /* Method */
286 const char *method_str = cwist_http_method_to_string(method);
287 if (strcmp(method_str, "GET") == 0) {
288 curl_easy_setopt(curl, CURLOPT_HTTPGET, 1L);
289 } else if (strcmp(method_str, "POST") == 0) {
290 curl_easy_setopt(curl, CURLOPT_POST, 1L);
291 } else if (strcmp(method_str, "PUT") == 0) {
292 curl_easy_setopt(curl, CURLOPT_UPLOAD, 0L);
293 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PUT");
294 } else if (strcmp(method_str, "DELETE") == 0) {
295 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "DELETE");
296 } else if (strcmp(method_str, "PATCH") == 0) {
297 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "PATCH");
298 } else if (strcmp(method_str, "HEAD") == 0) {
299 curl_easy_setopt(curl, CURLOPT_NOBODY, 1L);
300 } else if (strcmp(method_str, "OPTIONS") == 0) {
301 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, "OPTIONS");
302 } else {
303 curl_easy_setopt(curl, CURLOPT_CUSTOMREQUEST, method_str);
304 }
305
306 /* Request headers */
307 struct curl_slist *curl_headers = NULL;
308 cwist_http_header_node *node = headers;
309 while (node) {
310 if (node->key && node->key->data && node->value && node->value->data) {
311 char header_line[4096];
312 int n = snprintf(header_line, sizeof(header_line), "%s: %s",
313 node->key->data, node->value->data);
314 if (n > 0 && (size_t)n < sizeof(header_line)) {
315 curl_headers = curl_slist_append(curl_headers, header_line);
316 }
317 }
318 node = node->next;
319 }
320 if (curl_headers) {
321 curl_easy_setopt(curl, CURLOPT_HTTPHEADER, curl_headers);
322 }
323
324 /* Request body */
325 if (body && body_len > 0) {
326 curl_easy_setopt(curl, CURLOPT_POSTFIELDSIZE_LARGE, (curl_off_t)body_len);
327 curl_easy_setopt(curl, CURLOPT_COPYPOSTFIELDS, body);
328 }
329
330 /* Response accumulators */
331 response_body_t resp_body = {0};
332 response_headers_t resp_hdrs = {0};
333
334 curl_easy_setopt(curl, CURLOPT_WRITEFUNCTION, write_callback);
335 curl_easy_setopt(curl, CURLOPT_WRITEDATA, &resp_body);
336 curl_easy_setopt(curl, CURLOPT_HEADERFUNCTION, header_callback);
337 curl_easy_setopt(curl, CURLOPT_HEADERDATA, &resp_hdrs);
338
339 /* Prefer HTTP/2, allow downgrade to HTTP/1.1 */
340 curl_easy_setopt(curl, CURLOPT_HTTP_VERSION, CURL_HTTP_VERSION_2_0);
341
342 /* Perform */
343 CURLcode res = curl_easy_perform(curl);
344
345 /* Clean up request headers */
346 if (curl_headers) {
347 curl_slist_free_all(curl_headers);
348 }
349
350 if (res != CURLE_OK) {
351 free(resp_body.data);
352 cwist_http_header_free_all(resp_hdrs.headers);
353 err.error.err_i16 = -1;
354 return err;
355 }
356
357 /* Build response object */
358 cwist_http_response *response = cwist_http_response_create();
359 if (!response) {
360 free(resp_body.data);
361 cwist_http_header_free_all(resp_hdrs.headers);
362 err.error.err_i16 = -1;
363 return err;
364 }
365
366 response->status_code = (cwist_http_status_t)resp_hdrs.status_code;
367 if (resp_body.data && resp_body.size > 0) {
368 cwist_sstring_assign_len(response->body, resp_body.data, resp_body.size);
369 }
370
371 /* Transfer collected headers */
372 response->headers = resp_hdrs.headers;
373
374 /* Populate status text based on code */
375 switch (response->status_code) {
376 case CWIST_HTTP_OK: cwist_sstring_assign(response->status_text, "OK"); break;
377 case CWIST_HTTP_CREATED: cwist_sstring_assign(response->status_text, "Created"); break;
378 case CWIST_HTTP_NO_CONTENT: cwist_sstring_assign(response->status_text, "No Content"); break;
379 case CWIST_HTTP_BAD_REQUEST: cwist_sstring_assign(response->status_text, "Bad Request"); break;
380 case CWIST_HTTP_UNAUTHORIZED: cwist_sstring_assign(response->status_text, "Unauthorized"); break;
381 case CWIST_HTTP_FORBIDDEN: cwist_sstring_assign(response->status_text, "Forbidden"); break;
382 case CWIST_HTTP_NOT_FOUND: cwist_sstring_assign(response->status_text, "Not Found"); break;
383 case CWIST_HTTP_INTERNAL_ERROR: cwist_sstring_assign(response->status_text, "Internal Server Error"); break;
384 case CWIST_HTTP_NOT_IMPLEMENTED: cwist_sstring_assign(response->status_text, "Not Implemented"); break;
385 case CWIST_HTTP_SERVICE_UNAVAILABLE: cwist_sstring_assign(response->status_text, "Service Unavailable"); break;
386 default: cwist_sstring_assign(response->status_text, ""); break;
387 }
388
389 free(resp_body.data);
390 *out_response = response;
391 err.error.err_i16 = 0;
392 return err;
