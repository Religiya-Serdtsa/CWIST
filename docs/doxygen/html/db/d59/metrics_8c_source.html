38cwist_metrics_registry_t *cwist_metrics_registry(void) {
39 static cwist_metrics_registry_t *g_reg = NULL;
40 if (__builtin_expect(g_reg != NULL, 1)) return g_reg;
41
42 cwist_metrics_registry_t *alloc = cwist_alloc(sizeof(*alloc));
43 if (!alloc) return NULL;
44 memcpy(alloc->metrics, metric_defaults, sizeof(metric_defaults));
45 for (int i = 0; i < CWIST_METRIC_COUNT; ++i) {
46 atomic_init(&alloc->metrics[i].value.raw, 0);
47 atomic_init(&alloc->metrics[i].count, 0);
48 atomic_init(&alloc->metrics[i].sum, 0);
49 }
50 g_reg = alloc;
51 return g_reg;
52}

111char *cwist_metrics_render_prometheus(const cwist_metrics_registry_t *reg) {
112 if (!reg) return NULL;
113
114 size_t cap = 4096;
115 char *buf = malloc(cap);
116 if (!buf) return NULL;
117 size_t len = 0;
118
119 for (int i = 0; i < CWIST_METRIC_COUNT; ++i) {
120 const cwist_metric_t *m = &reg->metrics[i];
121 uintmax_t raw = atomic_load_explicit(&m->value.raw, memory_order_acquire);
122 uintmax_t cnt = atomic_load_explicit(&m->count, memory_order_acquire);
123
124 int need = snprintf(NULL, 0,
125 "# HELP %s %s\n"
126 "# TYPE %s %s\n"
127 "%s %lu.%03lu\n"
128 "%s_count %lu\n\n",
129 m->name, m->help,
130 m->name, type_str(m->type),
131 m->name,
132 (unsigned long)(raw / 1000),
133 (unsigned long)(raw % 1000),
134 m->name,
135 (unsigned long)cnt);
136
137 if (len + (size_t)need + 1 > cap) {
138 cap = (len + (size_t)need + 1) * 2;
139 char *nb = realloc(buf, cap);
140 if (!nb) { free(buf); return NULL; }
141 buf = nb;
142 }
143
144 len += (size_t)snprintf(buf + len, cap - len,
145 "# HELP %s %s\n"
146 "# TYPE %s %s\n"
147 "%s %lu.%03lu\n"
148 "%s_count %lu\n\n",
149 m->name, m->help,
150 m->name, type_str(m->type),
151 m->name,
152 (unsigned long)(raw / 1000),
153 (unsigned long)(raw % 1000),
154 m->name,
155 (unsigned long)cnt);
156 }
157
158 buf[len] = '\0';
159 return buf;
160}
