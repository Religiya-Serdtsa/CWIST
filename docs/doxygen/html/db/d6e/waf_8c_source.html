24bool cwist_waf_is_safe(const char *input, size_t length) {
25 if (!input) return true;
26 static const waf_signature signatures[] = {
27 { "<script", 7 }, { "</script", 8 }, { "javascript:", 11 }, { "vbscript:", 9 },
28 { "union select", 12 }, { "drop table", 10 }, { "insert into", 11 }, { "delete from", 11 },
29 { " or 1=1", 7 }, { " and 1=1", 8 }, { "--", 2 }, { "/*", 2 }, { "*/", 2 }
30 };
31 for (size_t i = 0; i < length; ++i) {
32 unsigned char c = (unsigned char)input[i];
33 if (c == 0 || (c < 0x20U && c != '\t' && c != '\n' && c != '\r')) return false;
34 c = ascii_lower(c);
35 for (size_t rule = 0; rule < sizeof(signatures) / sizeof(signatures[0]); ++rule) {
36 if (c == (unsigned char)signatures[rule].text[0] && ascii_equal_ci(input, length, i, &signatures[rule])) return false;
37 }
38 }
39 return true;
