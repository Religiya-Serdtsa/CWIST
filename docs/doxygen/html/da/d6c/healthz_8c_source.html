75void cwist_healthz_run(cwist_health_probe_t *out_probes,
76 size_t max_probes,
77 size_t *out_count,
78 cwist_health_status_t *out_overall) {
79 size_t count = 0;
80 cwist_health_status_t overall = CWIST_HEALTH_OK;
81
82 for (int i = 0; i < g_entry_count && count < max_probes; ++i) {
83 if (!g_entries[i].active) continue;
84 cwist_health_probe_t r = g_entries[i].fn(g_entries[i].ctx);
85 out_probes[count++] = r;
86 if (r.status == CWIST_HEALTH_FAIL) overall = CWIST_HEALTH_FAIL;
87 else if (r.status == CWIST_HEALTH_DEGRADED && overall == CWIST_HEALTH_OK)
88 overall = CWIST_HEALTH_DEGRADED;
89 }
90
91 if (out_count) *out_count = count;
92 if (out_overall) *out_overall = overall;
93}
