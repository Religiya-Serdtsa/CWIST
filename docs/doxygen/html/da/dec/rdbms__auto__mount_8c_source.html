24static bool probe_postgresql(int sock)
25{
26 char startup[64];
27 memset(startup, 0, sizeof(startup));
28
29 /* "user\0postgres\0\0" payload after version */
30 const char *user_key = "user";
31 const char *user_val = "postgres";
32 size_t payload = strlen(user_key) + 1 + strlen(user_val) + 1 + 1; /* +1 terminator */
33 uint32_t len = htonl((uint32_t)(4 + 4 + payload));
34 uint32_t version = htonl(196608); /* 3.0 */
35
36 memcpy(startup, &len, 4);
37 memcpy(startup + 4, &version, 4);
38 strcpy(startup + 8, user_key);
39 strcpy(startup + 8 + strlen(user_key) + 1, user_val);
40 /* trailing null already present from memset */
41
42 size_t total = 8 + payload;
43 if (send(sock, startup, total, 0) != (ssize_t)total)
44 return false;
45
46 char resp = 0;
47 ssize_t n = recv(sock, &resp, 1, 0);
48 return (n == 1 && resp == 'R');
49}

58static cwist_rdbms_provider_t classify_mysql(const char *buf, size_t len)
59{
60 if (len < 5)
61 return CWIST_RDBMS_NONE;
62
63 unsigned char protocol = (unsigned char)buf[4];
64 if (protocol != 0x0a)
65 return CWIST_RDBMS_NONE;
66
67 /* Look for "MariaDB" in the server version string that follows */
68 for (size_t i = 5; i + 6 < len; ++i) {
69 if (memcmp(&buf[i], "MariaDB", 7) == 0)
70 return CWIST_RDBMS_MARIADB;
71 }
72 return CWIST_RDBMS_MYSQL;
73}

75cwist_rdbms_provider_t cwist_rdbms_probe_port(int port)
76{
77 int sock = socket(AF_INET, SOCK_STREAM, 0);
78 if (sock < 0) {
79 CWIST_LOG_ERROR("RDBMS probe: socket creation failed: %s", strerror(errno));
80 return CWIST_RDBMS_NONE;
81 }
82
83 struct timeval tv;
84 tv.tv_sec = 2;
85 tv.tv_usec = 0;
86 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
87 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
88
89 struct sockaddr_in addr;
90 memset(&addr, 0, sizeof(addr));
91 addr.sin_family = AF_INET;
92 addr.sin_port = htons((uint16_t)port);
93 inet_pton(AF_INET, "127.0.0.1", &addr.sin_addr);
94
95 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
96 CWIST_LOG_ERROR("RDBMS probe: connect to 127.0.0.1:%d failed: %s", port, strerror(errno));
97 close(sock);
98 return CWIST_RDBMS_NONE;
99 }
100
101 char buf[512];
102 ssize_t n = recv(sock, buf, sizeof(buf), 0);
103 if (n > 0) {
104 cwist_rdbms_provider_t p = classify_mysql(buf, (size_t)n);
105 if (p != CWIST_RDBMS_NONE) {
106 close(sock);
107 return p;
108 }
109 /* Data was received but did not look like MySQL; could be noise.
110 * Close and re-probe as PostgreSQL below. */
111 close(sock);
112 sock = socket(AF_INET, SOCK_STREAM, 0);
113 if (sock < 0)
114 return CWIST_RDBMS_NONE;
115 setsockopt(sock, SOL_SOCKET, SO_SNDTIMEO, &tv, sizeof(tv));
116 setsockopt(sock, SOL_SOCKET, SO_RCVTIMEO, &tv, sizeof(tv));
117 if (connect(sock, (struct sockaddr *)&addr, sizeof(addr)) < 0) {
118 close(sock);
119 return CWIST_RDBMS_NONE;
120 }
121 } else if (n < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
122 CWIST_LOG_ERROR("RDBMS probe: recv error on port %d: %s", port, strerror(errno));
123 close(sock);
124 return CWIST_RDBMS_NONE;
125 }
126
127 /* PostgreSQL speaks only after the client sends a StartupMessage */
128 if (probe_postgresql(sock)) {
129 close(sock);
130 return CWIST_RDBMS_POSTGRES;
131 }
132
133 close(sock);
134 return CWIST_RDBMS_NONE;
135}
