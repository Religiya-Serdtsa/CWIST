124 if (!f) return;
125 char line[1024];
126 while (fgets(line, sizeof(line), f)) {
127 char *p = line;
128 while (isspace((unsigned char)*p)) p++;
129 if (*p == '#' || *p == '\0') continue;
130 char *eq = strchr(p, '=');
131 if (!eq) continue;
132 *eq = '\0';
133 char *key = p;
134 char *value = eq + 1;
135 // trim key
136 char *ke = key + strlen(key) - 1;
137 while (ke > key && isspace((unsigned char)*ke)) *ke-- = '\0';
138 // trim value

140 char *ve = value + strlen(value) - 1;

142 // unquote value
