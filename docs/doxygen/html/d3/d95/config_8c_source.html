51void cwist_config_set(cwist_config *cfg, const char *key, const char *value) {
52 if (!cfg || !key) return;
53 size_t idx = cwist_config_hash(key) % cfg->bucket_count;
54 cwist_config_bucket *curr = cfg->buckets[idx];

103 if (!cfg) return;

105 size_t prefix_len = prefix ? strlen(prefix) : 0;
106 for (char **env = environ; *env; env++) {
107 char *eq = strchr(*env, '=');

122void cwist_config_load_file(cwist_config *cfg, const char *path) {
123 if (!cfg || !path) return;
124 FILE *f = fopen(path, "r");
125 if (!f) return;
126 char line[1024];
127 while (fgets(line, sizeof(line), f)) {
128 char *p = line;
129 while (isspace((unsigned char)*p)) p++;
130 if (*p == '#' || *p == '\0') continue;
131 char *eq = strchr(p, '=');
132 if (!eq) continue;
133 *eq = '\0';
134 char *key = p;
135 char *value = eq + 1;
136 // trim key
137 char *ke = key + strlen(key) - 1;
138 while (ke > key && isspace((unsigned char)*ke)) *ke-- = '\0';
139 // trim value
140 while (isspace((unsigned char)*value)) value++;
141 char *ve = value + strlen(value) - 1;
142 while (ve > value && isspace((unsigned char)*ve)) *ve-- = '\0';
143 // unquote value
144 if (value[0] == '"' && ve == value + strlen(value) - 1 && *ve == '"') {
145 value++;
146 *ve = '\0';
147 }
148 cwist_config_set(cfg, key, value);
149 }
150 fclose(f);
