55static void mp_parse_headers(mp_parse_ctx *ctx) {
56 if (ctx->header_field_len >= sizeof(ctx->header_field))

60 if (ctx->header_value_len >= sizeof(ctx->header_value))
61 ctx->header_value_len = sizeof(ctx->header_value) - 1;
62 ctx->header_value[ctx->header_value_len] = '\0';

124static int mp_on_part_data(multipart_parser *p, const char *at, size_t len) {
125 mp_parse_ctx *ctx = (mp_parse_ctx *)multipart_parser_get_data(p);
126 if (len == 0) return 0;
127 if (ctx->data_len + len > ctx->data_cap) {
128 size_t new_cap = (ctx->data_len + len) * 2;
129 if (new_cap < 256) new_cap = 256;
130 char *new_data = (char *)cwist_alloc(new_cap);
131 if (!new_data) return 1;
132 if (ctx->data) {
133 memcpy(new_data, ctx->data, ctx->data_len);
134 cwist_free(ctx->data);
135 }
136 ctx->data = new_data;
137 ctx->data_cap = new_cap;
138 }
139 memcpy(ctx->data + ctx->data_len, at, len);
140 ctx->data_len += len;
141 return 0;

231char *cwist_multipart_extract_boundary(const char *content_type) {
232 if (!content_type) return NULL;
233 const char *p = content_type;
234 while (*p) {
235 while (*p && (isspace((unsigned char)*p) || *p == ';')) p++;
236 if (strncasecmp(p, "boundary", 8) == 0) {
237 p += 8;
238 while (*p && isspace((unsigned char)*p)) p++;
239 if (*p == '=') {
240 p++;
241 while (*p && isspace((unsigned char)*p)) p++;
242 const char *start = p;
243 if (*p == '"') {
244 start = ++p;
245 while (*p && *p != '"') p++;
246 } else {
247 while (*p && *p != ';' && !isspace((unsigned char)*p)) p++;
248 }
249 size_t len = (size_t)(p - start);
250 char *boundary = (char *)cwist_alloc(len + 1);
251 if (boundary) {
252 memcpy(boundary, start, len);
253 boundary[len] = '\0';

259 if (*p == ';') p++;
260 }
