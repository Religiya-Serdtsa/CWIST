41static size_t cwist_cert_fetch_write_cb(void *ptr, size_t size, size_t nmemb, void *userdata) {
42 if (size != 0 && nmemb > SIZE_MAX / size) return 0;
43
44 size_t total = size * nmemb;
45 cwist_cert_fetch_buffer *buf = (cwist_cert_fetch_buffer *)userdata;
46 if (total > CWIST_AIA_MAX_RESPONSE_BYTES || buf->len > CWIST_AIA_MAX_RESPONSE_BYTES - total) {
47 return 0;
48 }
49
50 size_t need = buf->len + total;
51 if (need > buf->cap) {
52 size_t new_cap = buf->cap ? buf->cap * 2 : 4096;
53 while (new_cap < need) {
54 if (new_cap > CWIST_AIA_MAX_RESPONSE_BYTES / 2) {
55 new_cap = CWIST_AIA_MAX_RESPONSE_BYTES;
56 break;
57 }
58 new_cap *= 2;
59 }
60
61 uint8_t *tmp = (uint8_t *)realloc(buf->data, new_cap);
62 if (!tmp) return 0;
63 buf->data = tmp;
64 buf->cap = new_cap;
65 }
66
67 memcpy(buf->data + buf->len, ptr, total);
68 buf->len += total;
69 return total;
70}
