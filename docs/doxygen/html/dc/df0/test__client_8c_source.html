76static void cookie_jar_update(cwist_test_client *client, cwist_http_response *res) {
77 if (!client || !res) return;

81 if (!buf) continue;

101 if (c) {

140static cwist_http_response *do_request(cwist_test_client *client, cwist_http_method_t method,
141 const char *path, const cwist_test_client_request_options *opts) {
142 if (!client || !client->app || !path) return NULL;
143 cwist_http_request *req = cwist_http_request_create();
144 if (!req) return NULL;
145 req->method = method;
146
147 const char *query = opts && opts->query_string ? opts->query_string : NULL;
148 const char *hash = strchr(path, '?');
149 if (hash) {
150 size_t path_len = (size_t)(hash - path);
151 char *path_only = (char *)cwist_alloc(path_len + 1);
152 if (!path_only) { cwist_http_request_destroy(req); return NULL; }
153 memcpy(path_only, path, path_len);
154 path_only[path_len] = '\0';
155 cwist_sstring_assign(req->path, path_only);
156 cwist_free(path_only);
157 query = hash + 1;
158 } else {
159 cwist_sstring_assign(req->path, (char *)path);
160 }
161 if (query) {
162 cwist_sstring_assign(req->query, (char *)query);
163 if (!req->query_params) req->query_params = cwist_query_map_create();
164 cwist_query_map_clear(req->query_params);
165 cwist_query_map_parse(req->query_params, query);
166 }
167
168 const char *body = opts ? opts->body : NULL;
169 const char *content_type = opts ? opts->content_type : NULL;
170
171 if (body) {
172 if (opts && opts->body_len > 0) {
173 cwist_sstring_assign_len(req->body, (char *)body, opts->body_len);
174 } else {
175 cwist_sstring_assign(req->body, (char *)body);
176 }
177 }
178 if (content_type) cwist_http_header_add(&req->headers, "Content-Type", content_type);
179
180 if (opts && opts->headers) {
181 for (size_t i = 0; i < opts->header_count; i++) {
182 if (opts->headers[i].key && opts->headers[i].value)
183 cwist_http_header_add(&req->headers, opts->headers[i].key, opts->headers[i].value);
184 }
185 }
186
187 cookie_jar_apply(client, req, opts ? opts->cookies : NULL, opts ? opts->cookie_count : 0);
188
189 cwist_http_response *res = cwist_http_response_create();
190 if (!res) {
191 cwist_http_request_destroy(req);
192 return NULL;
193 }
194 cwist_app_dispatch(client->app, req, res);
195 cookie_jar_update(client, res);
196 cwist_http_request_destroy(req);
197 return res;
198}
