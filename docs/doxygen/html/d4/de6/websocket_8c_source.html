32static char *ws_strcasestr(const char *haystack, const char *needle) {
33 if (!*needle) return (char *)haystack;
34 for (; *haystack; haystack++) {
35 const char *h = haystack;
36 const char *n = needle;
37 while (*h && *n &&
38 tolower((unsigned char)*h) == tolower((unsigned char)*n)) {
39 h++;
40 n++;
41 }
42 if (!*n) return (char *)haystack;
43 }
44 return NULL;
45}

58cwist_websocket *cwist_websocket_upgrade(cwist_http_request *req, int client_fd) {
59 if (!req || client_fd < 0) return NULL;
60
61 // Validate Headers
62 char *connection = cwist_http_header_get(req->headers, "Connection");
63 char *upgrade = cwist_http_header_get(req->headers, "Upgrade");
64 char *key = cwist_http_header_get(req->headers, "Sec-WebSocket-Key");
65
66 if (!connection || !upgrade || !key) return NULL;
67 if (ws_strcasestr(connection, "Upgrade") == NULL) return NULL;
68 if (strcasecmp(upgrade, "websocket") != 0) return NULL;
69
70 // Handshake Key Generation
71 char combined_key[512];
72 snprintf(combined_key, sizeof(combined_key), "%s%s", key, WS_GUID);
73
74 uint8_t hash[20];
75 sha1((uint8_t *)combined_key, strlen(combined_key), hash);
76
77 size_t accept_len;
78 char *accept_key = base64_encode(hash, 20, &accept_len);
79 if (!accept_key) return NULL;
80
81 // Send Response
82 cwist_http_response *res = cwist_http_response_create();
83 res->status_code = 101;
84 cwist_sstring_assign(res->status_text, "Switching Protocols");
85 cwist_http_header_add(&res->headers, "Upgrade", "websocket");
86 cwist_http_header_add(&res->headers, "Connection", "Upgrade");
87 cwist_http_header_add(&res->headers, "Sec-WebSocket-Accept", accept_key);
88
89 cwist_error_t err = cwist_http_send_response(client_fd, res);
90
91 cwist_http_response_destroy(res);
92 cwist_free(accept_key);
93
94 if (err.error.err_i16 != 0) return NULL;
95
96 req->upgraded = true;
97
98 cwist_websocket *ws = (cwist_websocket *)cwist_alloc(sizeof(cwist_websocket));
99 ws->fd = client_fd;
100 ws->is_closed = false;
101 return ws;
102}

131cwist_ws_frame *cwist_websocket_receive(cwist_websocket *ws) {
132 if (!ws || ws->is_closed) return NULL;
133
134 uint8_t head[2];
135 if (read_exact(ws->fd, head, 2) < 0) return NULL;
136
137 bool fin = (head[0] & 0x80) != 0;
138 cwist_ws_opcode_t opcode = head[0] & 0x0F;
139 bool masked = (head[1] & 0x80) != 0;
140 uint64_t payload_len = head[1] & 0x7F;
141
142 if (!masked) {
143 // Client-to-server frames must be masked per spec
144 // We can choose to strict close or allow. Strict is better.
145 // For now, let's just return error.
146 return NULL;
147 }
148
149 if (payload_len == 126) {
150 uint16_t len16;
151 if (read_exact(ws->fd, &len16, 2) < 0) return NULL;
152 payload_len = ntohs(len16);
153 } else if (payload_len == 127) {
154 uint64_t len64;
155 if (read_exact(ws->fd, &len64, 8) < 0) return NULL;
156 // manually swap if no be64toh
157 // assuming be64toh or similar exists, or manual
158 // Linux usually has be64toh in <endian.h>
159 // Let's implement manual swap to be portable
160 uint8_t *p = (uint8_t *)&len64;
161 payload_len = ((uint64_t)p[0] << 56) | ((uint64_t)p[1] << 48) |
162 ((uint64_t)p[2] << 40) | ((uint64_t)p[3] << 32) |
163 ((uint64_t)p[4] << 24) | ((uint64_t)p[5] << 16) |
164 ((uint64_t)p[6] << 8) | ((uint64_t)p[7]);
165 }
166
167 uint8_t masking_key[4];
168 if (read_exact(ws->fd, masking_key, 4) < 0) return NULL;
169
170 uint8_t *payload = NULL;
171 if (payload_len > 0) {
172 payload = (uint8_t *)cwist_alloc(payload_len + 1); // +1 for safety null term if text
173 if (!payload) return NULL;
174 if (read_exact(ws->fd, payload, payload_len) < 0) {
175 cwist_free(payload);
176 return NULL;
177 }
178
179 // Unmask
180 for (uint64_t i = 0; i < payload_len; i++) {
181 payload[i] ^= masking_key[i % 4];
182 }
183 payload[payload_len] = '\0'; // Null terminate for convenience if text
184 }
185
186 if (opcode == CWIST_WS_FRAME_CLOSE) {
187 ws->is_closed = true;
188 }
189
190 cwist_ws_frame *frame = (cwist_ws_frame *)cwist_alloc(sizeof(cwist_ws_frame));
191 frame->fin = fin;
192 frame->opcode = opcode;
193 frame->payload = payload;
194 frame->payload_len = payload_len;
195
196 return frame;
197}

207int cwist_websocket_send(cwist_websocket *ws, cwist_ws_opcode_t opcode, const uint8_t *data, size_t len) {
208 if (!ws || ws->is_closed) return -1;
209
210 uint8_t head[10]; // Max header size (2 + 8)
211 size_t head_len = 2;
212
213 head[0] = 0x80 | (opcode & 0x0F); // FIN=1
214
215 if (len < 126) {
216 head[1] = len;
217 } else if (len < 65536) {
218 head[1] = 126;
219 head[2] = (len >> 8) & 0xFF;
220 head[3] = len & 0xFF;
221 head_len += 2;
222 } else {
223 head[1] = 127;
224 for (int i = 0; i < 8; i++) {
225 head[2 + i] = (len >> ((7 - i) * 8)) & 0xFF;
226 }
227 head_len += 8;
228 }
229
230 // Server does not mask frames
231
232 if (send(ws->fd, head, head_len, 0) < 0) return -1;
233 if (len > 0) {
234 if (send(ws->fd, data, len, 0) < 0) return -1;
235 }
236
237 return 0;
238}

254int cwist_websocket_send_sequenced(cwist_websocket *ws,
255 const uint8_t *data,
256 size_t len,
257 uint16_t chunk_payload_size) {
258 if (!ws || ws->is_closed || !data || len == 0 || chunk_payload_size == 0) return -1;
259
260 cwist_seq_message_t msg;
261 if (!cwist_seq_split(data, len, chunk_payload_size, &msg)) return -1;
262
263 int rc = 0;
264 for (size_t i = 0; i < msg.count && rc == 0; i++) {
265 if (cwist_websocket_send(ws, CWIST_WS_FRAME_BINARY, msg.chunks[i], msg.chunk_lens[i]) != 0) {
266 rc = -1;
267 }
268 }
269
270 cwist_seq_message_free(&msg);
271 return rc;
272}

277uint8_t *cwist_websocket_receive_sequenced(cwist_websocket *ws, size_t *out_len) {
278 if (!out_len) return NULL;
279 *out_len = 0;
280 if (!ws || ws->is_closed) return NULL;
281
282 cwist_seq_assembler_t *a = cwist_seq_assembler_create();
283 if (!a) return NULL;
284
285 uint8_t *result = NULL;
286 while (!cwist_seq_assembler_is_complete(a)) {
287 cwist_ws_frame *frame = cwist_websocket_receive(ws);
288 if (!frame) break;
289
290 if (frame->opcode == CWIST_WS_FRAME_BINARY && frame->payload && frame->payload_len > 0) {
291 cwist_seq_chunk_t chunk;
292 if (cwist_seq_chunk_parse(frame->payload, frame->payload_len, &chunk)) {
293 cwist_seq_assembler_feed(a, &chunk);
294 }
295 }
296
297 bool is_close = (frame->opcode == CWIST_WS_FRAME_CLOSE);
298 cwist_websocket_frame_destroy(frame);
299 if (is_close) break;
300 }
301
302 const uint8_t *assembled = NULL;
303 size_t assembled_len = 0;
304 if (cwist_seq_assembler_get_data(a, &assembled, &assembled_len) && assembled_len > 0) {
305 result = (uint8_t *)cwist_alloc(assembled_len + 1);
306 if (result) {
307 memcpy(result, assembled, assembled_len);
308 result[assembled_len] = '\0';
309 *out_len = assembled_len;
310 }
311 }
312
313 cwist_seq_assembler_destroy(a);
314 return result;
315}
