36static bool pool_cond_init(pthread_cond_t *cond) {
37#if defined(__FreeBSD__) || defined(__NetBSD__) || defined(__OpenBSD__) || defined(__APPLE__)
38 return pthread_cond_init(cond, NULL) == 0;
39#else
40 pthread_condattr_t attr;

42 bool ok = pthread_condattr_setclock(&attr, CLOCK_MONOTONIC) == 0 &&

49static void pool_deadline_after_ms(struct timespec *deadline, int timeout_ms) {
50 clock_gettime(pool_cond_clock(), deadline);
51 deadline->tv_sec += timeout_ms / 1000;
52 deadline->tv_nsec += (long)(timeout_ms % 1000) * 1000000L;
53 if (deadline->tv_nsec >= 1000000000L) {
54 ++deadline->tv_sec;
55 deadline->tv_nsec -= 1000000000L;

131cwist_db *cwist_db_pool_acquire_timeout(cwist_db_pool_t *pool, int timeout_ms) {
132 if (!pool || timeout_ms < 0) return NULL;
133 struct timespec deadline;
134 pool_deadline_after_ms(&deadline, timeout_ms);
135 pthread_mutex_lock(&pool->mtx);
136 while (!pool->closing && pool->idle_count == 0) {
137 if (pthread_cond_timedwait(&pool->cond, &pool->mtx, &deadline) != 0) break;
138 }
139 cwist_db *db = pool_acquire_locked(pool);
140 pthread_mutex_unlock(&pool->mtx);

144void cwist_db_pool_release(cwist_db_pool_t *pool, cwist_db *conn) {
145 if (!pool || !conn) return;
146 pthread_mutex_lock(&pool->mtx);
147 size_t slot = conn->pool_slot;
148 if (slot < pool->max_conns && pool->conns[slot] == conn && pool->leased[slot]) {
149 pool->leased[slot] = false;
150 pool->idle_slots[pool->idle_count++] = slot;
151 --pool->in_use;
152 pthread_cond_signal(&pool->cond);
153 }
154 pthread_mutex_unlock(&pool->mtx);
155}

163bool cwist_db_pool_destroy_timeout(cwist_db_pool_t *pool, int timeout_ms) {
164 if (!pool || timeout_ms < -1) return false;
165 pthread_mutex_lock(&pool->mtx);
166 pool->closing = true;
167 pthread_cond_broadcast(&pool->cond);
168 if (timeout_ms < 0) {
169 while (pool->in_use != 0) pthread_cond_wait(&pool->cond, &pool->mtx);
170 } else {
171 struct timespec deadline;
172 pool_deadline_after_ms(&deadline, timeout_ms);
173 while (pool->in_use != 0) {
174 int rc = pthread_cond_timedwait(&pool->cond, &pool->mtx, &deadline);
175 if (rc != 0 && pool->in_use != 0) {
176 pthread_mutex_unlock(&pool->mtx);
177 return false;
178 }
179 }
180 }
181 pthread_mutex_unlock(&pool->mtx);
182 for (size_t i = 0; i < pool->max_conns; ++i) cwist_db_close(pool->conns[i]);
183 pthread_cond_destroy(&pool->cond); pthread_mutex_destroy(&pool->mtx);
184 cwist_free(pool->leased); cwist_free(pool->idle_slots); cwist_free(pool->conns); cwist_free(pool->open_path); cwist_free(pool->path); cwist_free(pool);
185 return true;
186}
