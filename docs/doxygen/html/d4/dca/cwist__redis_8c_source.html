46static int redis_open_socket(const char *host, int port) {
47 struct addrinfo hints = {0}, *res = NULL;
48 hints.ai_family = AF_INET;
49 hints.ai_socktype = SOCK_STREAM;
50 char port_str[16];
51 snprintf(port_str, sizeof(port_str), "%d", port);
52 if (getaddrinfo(host, port_str, &hints, &res) != 0 || !res) return -1;
53
54 int fd = socket(res->ai_family, res->ai_socktype, res->ai_protocol);
55 if (fd < 0) {
56 freeaddrinfo(res);
57 return -1;
58 }
59 if (connect(fd, res->ai_addr, res->ai_addrlen) != 0) {
60 close(fd);
61 freeaddrinfo(res);
62 return -1;
63 }
64 freeaddrinfo(res);
65 return fd;
66}

69static int redis_send_all(int fd, const char *buf, size_t len) {
70 size_t sent = 0;
71 while (sent < len) {
72 /* RESP speaks plaintext TCP by design (same as redis-cli without
73 * --tls). Credentials must be protected at the deployment layer:
74 * loopback/Unix sockets, private network, or a TLS-terminating proxy
75 * such as stunnel/spiped in front of Redis. */
76 ssize_t n = send(fd, buf + sent, len - sent, 0); // codeql[cpp/cleartext-transmission]
77 if (n < 0) {
78 if (errno == EINTR) continue;
79 return -1;
80 }
81 if (n == 0) return -1;
82 sent += (size_t)n;
83 }
84 return 0;
85}

87static int redis_recv_line(cwist_redis_t *r, char *line, size_t line_len) {
88 size_t i = 0;
89 while (1) {
90 if (r->recv_len > 0) {
91 while (r->recv_len > 0 && i < line_len - 1) {
92 char c = r->recv_buf[0];
93 /* simple shift */
94 memmove(r->recv_buf, r->recv_buf + 1, r->recv_len - 1);
95 r->recv_len--;
96 if (c == '\n') {
97 if (i > 0 && line[i - 1] == '\r') i--;
98 line[i] = '\0';
99 return 0;
100 }
101 line[i++] = c;
102 }
103 if (i >= line_len - 1) return -1;
104 }
105 ssize_t n = recv(r->fd, r->recv_buf + r->recv_len,
106 CWIST_REDIS_BUF_SIZE - r->recv_len - 1, 0);
107 if (n < 0) {
108 if (errno == EINTR) continue;
109 return -1;
110 }
111 if (n == 0) return -1;
112 r->recv_len += (size_t)n;
113 }
114}

116static int redis_recv_bytes(cwist_redis_t *r, char *out, size_t len) {
117 size_t got = 0;
118 while (got < len) {
119 if (r->recv_len > 0) {
120 size_t take = r->recv_len < (len - got) ? r->recv_len : (len - got);
121 memcpy(out + got, r->recv_buf, take);
122 memmove(r->recv_buf, r->recv_buf + take, r->recv_len - take);
123 r->recv_len -= take;
124 got += take;
125 continue;
126 }
127 ssize_t n = recv(r->fd, r->recv_buf + r->recv_len,
128 CWIST_REDIS_BUF_SIZE - r->recv_len - 1, 0);
129 if (n < 0) {
130 if (errno == EINTR) continue;
131 return -1;
132 }
133 if (n == 0) return -1;
134 r->recv_len += (size_t)n;
135 }
136 return 0;
137}

151static cwist_error_t read_reply(cwist_redis_t *r, char **out_value, size_t *out_len, char *out_type) {
152 char line[CWIST_REDIS_LINE_MAX];
153 if (out_len) *out_len = 0;
154 if (redis_recv_line(r, line, sizeof(line)) != 0) return make_error(CWIST_ERR_INT16);
155 if (out_type) *out_type = line[0];
156
157 switch (line[0]) {
158 case '+': /* simple string */
159 case '-': /* error */
160 if (out_value) {
161 if (line[0] == '+') {
162 *out_value = cwist_strdup(line + 1);
163 if (out_len) *out_len = strlen(line + 1);
164 } else {
165 *out_value = NULL;
166 }
167 }
168 return (line[0] == '-') ? make_error(CWIST_ERR_INT16) : (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
169
170 case ':': /* integer */
171 if (out_value) *out_value = cwist_strdup(line + 1);
172 if (out_len) *out_len = strlen(line + 1);
173 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
174
175 case '$': { /* bulk string */
176 long long blen = atoll(line + 1);
177 if (blen < 0) {
178 if (out_value) *out_value = NULL;
179 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
180 }
181 char *buf = cwist_alloc((size_t)blen + 1);
182 if (!buf) return make_error(CWIST_ERR_INT16);
183 if (redis_recv_bytes(r, buf, (size_t)blen) != 0) {
184 cwist_free(buf);
185 return make_error(CWIST_ERR_INT16);
186 }
187 buf[blen] = '\0';
188 /* consume trailing \r\n */
189 char crlf[2];
190 if (redis_recv_bytes(r, crlf, 2) != 0) {
191 cwist_free(buf);
192 return make_error(CWIST_ERR_INT16);
193 }
194 if (out_value) *out_value = buf;
195 else cwist_free(buf);
196 if (out_len) *out_len = (size_t)blen;
197 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
198 }
199
200 case '*': { /* array - skip for simple client */
201 long long count = atoll(line + 1);
202 for (long long i = 0; i < count; i++) {
203 char *tmp = NULL;
204 cwist_error_t e = read_reply(r, &tmp, NULL, NULL);
205 cwist_free(tmp);
206 if (e.error.err_i16 != 0) return e;
207 }
208 if (out_value) *out_value = NULL;
209 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
210 }
211
212 default:
213 return make_error(CWIST_ERR_INT16);
214 }

217static cwist_error_t redis_command_frame(cwist_redis_t *r, const char *frame, size_t frame_len,
218 char **out_value, size_t *out_len) {
219 if (!r || !frame) return make_error(CWIST_ERR_INT16);
220 if (out_value) *out_value = NULL;
221 /* A request/reply pair must be serialized as one critical section. Keeping
222 * send and receive separate allows two callers to consume each other's
223 * replies on the same connection. */
224 pthread_mutex_lock(&r->mtx);
225 cwist_error_t err = redis_send_all(r->fd, frame, frame_len) == 0
226 ? read_reply(r, out_value, out_len, NULL) : make_error(CWIST_ERR_INT16);
227 pthread_mutex_unlock(&r->mtx);
228 return err;
229}

251void cwist_redis_close(cwist_redis_t *r) {
252 if (!r) return;
253 /* shutdown wakes a subscriber blocked in recv before the descriptor is
254 * released. This is also the documented way to stop subscribe(). */
255 if (r->fd >= 0) {

258 r->fd = -1;
259 }
260 cwist_free(r->host);
261 pthread_mutex_destroy(&r->mtx);
262 cwist_free(r);

435cwist_error_t cwist_redis_subscribe(cwist_redis_t *r,
436 const char **channels,
437 cwist_redis_msg_cb cb,
438 void *ctx) {
439 if (!r || !channels || !channels[0] || !cb) return make_error(CWIST_ERR_INT16);
440
441 cwist_sstring *cmd = cwist_sstring_create();
442 if (!cmd) return make_error(CWIST_ERR_INT16);
443 size_t count = 0;
444 while (channels[count]) count++;
445 char prefix[32];
446 snprintf(prefix, sizeof(prefix), "*%zu\r\n", count + 1);
447 cwist_sstring_append(cmd, prefix);
448 write_bulk_string(cmd, "SUBSCRIBE");
449 for (size_t i = 0; i < count; i++) {
450 write_bulk_string(cmd, channels[i]);
451 }
452 cwist_error_t err = send_command(r, cmd->data);
453 cwist_sstring_destroy(cmd);
454 if (err.error.err_i16 != 0) return err;
455
456 /* Consume subscription confirmations. */
457 for (size_t i = 0; i < count; i++) {
458 char *out = NULL;
459 err = recv_reply(r, &out);
460 cwist_free(out);
461 if (err.error.err_i16 != 0) return err;
462 }
463
464 /* Blocking read loop for messages. */
465 while (1) {
466 char line[CWIST_REDIS_LINE_MAX];
467 if (redis_recv_line(r, line, sizeof(line)) != 0) break;
468 if (line[0] != '*') continue;
469 long long arr_count = atoll(line + 1);
470 if (arr_count < 3) continue;
471
472 char *type = NULL;
473 if (read_reply(r, &type, NULL, NULL).error.err_i16 != 0) break;
474
475 char *channel = NULL;
476 if (read_reply(r, &channel, NULL, NULL).error.err_i16 != 0) {
477 cwist_free(type);
478 break;
479 }
480
481 char *message = NULL;
482 if (read_reply(r, &message, NULL, NULL).error.err_i16 != 0) {
483 cwist_free(type);
484 cwist_free(channel);
485 break;
486 }
487
488 if (type && strcmp(type, "message") == 0) {
489 cb(channel, message, ctx);
490 }
491 cwist_free(type);
492 cwist_free(channel);
493 cwist_free(message);
494 }
495 return (cwist_error_t){.errtype = CWIST_ERR_INT16, .error.err_i16 = 0};
496}

500cwist_redis_pool_t *cwist_redis_pool_create(const char *host, int port, size_t max_conns) {
501 if (!host || port <= 0 || max_conns == 0) return NULL;
502
503 cwist_redis_pool_t *pool = cwist_alloc(sizeof(*pool));
504 if (!pool) return NULL;
505 pool->host = cwist_strdup(host);
506 pool->port = port;
507 pool->max_conns = max_conns;
508 pool->conns = cwist_alloc_array(max_conns, sizeof(cwist_redis_t *));
509 pool->available = cwist_alloc_array(max_conns, sizeof(bool));
510 if (!pool->host || !pool->conns || !pool->available) goto fail;
511
512 for (size_t i = 0; i < max_conns; i++) {
513 pool->conns[i] = cwist_redis_connect(host, port);
514 if (!pool->conns[i]) goto fail;
515 pool->available[i] = true;
516 }
517
518 pthread_mutex_init(&pool->mtx, NULL);
519 pthread_cond_init(&pool->cond, NULL);
520 return pool;
521
522fail:
523 if (pool) {
524 for (size_t i = 0; i < max_conns && pool->conns; i++) {
525 if (pool->conns[i]) cwist_redis_close(pool->conns[i]);
526 }
527 cwist_free(pool->conns);
528 cwist_free(pool->available);
529 cwist_free(pool->host);
530 cwist_free(pool);
531 }
532 return NULL;
533}

550static cwist_redis_t *redis_pool_acquire(cwist_redis_pool_t *pool) {
551 if (!pool) return NULL;
