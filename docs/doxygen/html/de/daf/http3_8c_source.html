113static struct lsxpack_header *
114cwist_h3_hsi_prepare(void *hset_p, struct lsxpack_header *xhdr, size_t req_space) {
115 cwist_h3_hset_t *hset = hset_p;
116 if (xhdr) {
117 /* Previous header now has its final size known; advance offset. */
118 size_t name_len = xhdr->buf + xhdr->name_offset - (hset->decode_buf + hset->decode_off);
119 size_t value_len = xhdr->buf + xhdr->val_offset - (hset->decode_buf + hset->decode_off);
120 /* The header structure itself was prepared at decode_off. After
121 * decoding finishes lsquic tells us the used space via xhdr.
122 * We bump decode_off by the total occupied bytes. */
123 size_t total = name_len + value_len + 2; /* +2 for separator/padding */
124 if (total > sizeof(hset->decode_buf) - hset->decode_off)
125 total = sizeof(hset->decode_buf) - hset->decode_off;
126 hset->decode_off += total;
127 if (hset->count < H3_MAX_HEADERS)
128 hset->count++;
129 }
130 if (hset->count >= H3_MAX_HEADERS)
131 return NULL;
132 if (req_space > sizeof(hset->decode_buf) - hset->decode_off)
133 return NULL;
134 lsxpack_header_prepare_decode(&hset->headers[hset->count],
135 hset->decode_buf, hset->decode_off,
136 sizeof(hset->decode_buf) - hset->decode_off);
137 return &hset->headers[hset->count];
138}

172static int cwist_h3_packets_out(void *ctx,
173 const struct lsquic_out_spec *specs,
174 unsigned n_specs) {
175 int udp_fd = *(int *)ctx;
176 unsigned i;
177 for (i = 0; i < n_specs; ++i) {
178 const struct lsquic_out_spec *spec = &specs[i];
179 struct msghdr msg = {0};
180 msg.msg_name = (void *)spec->dest_sa;
181 msg.msg_namelen = (spec->dest_sa && spec->dest_sa->sa_family == AF_INET)
182 ? sizeof(struct sockaddr_in)
183 : sizeof(struct sockaddr_in6);
184 msg.msg_iov = (struct iovec *)spec->iov;
185 msg.msg_iovlen = spec->iovlen;
186 ssize_t nw = sendmsg(udp_fd, &msg, MSG_DONTWAIT);
187 if (nw < 0) {
188 if (errno == EAGAIN || errno == EWOULDBLOCK)
189 break;
190 /* Non-fatal errors: log and continue if possible */
191 if (errno == ECONNREFUSED || errno == ENETUNREACH ||
192 errno == EHOSTUNREACH || errno == EMSGSIZE)
193 continue;
194 return -1;
195 }
196 }
197 return (int)i;
198}

204static int cwist_h3_alpn_select_cb(SSL *ssl, const uint8_t **out, uint8_t *outlen,
205 const uint8_t *in, unsigned inlen, void *arg) {
206 (void)ssl;
207 (void)arg;
208 static const char *const protos[] = { "h3", "h3-29" };
209 for (size_t p = 0; p < sizeof(protos) / sizeof(protos[0]); ++p) {
210 const char *proto = protos[p];
211 size_t plen = strlen(proto);
212 const uint8_t *ptr = in;
213 const uint8_t *end = in + inlen;
214 while (ptr < end) {
215 uint8_t len = *ptr++;
216 if (ptr + len > end) break;
217 if (len == plen && memcmp(ptr, proto, plen) == 0) {
218 *out = ptr;
219 *outlen = (uint8_t)plen;
220 return SSL_TLSEXT_ERR_OK;
221 }
222 ptr += len;
223 }
224 }
225 return SSL_TLSEXT_ERR_NOACK;
226}

290static void h3_apply_header(cwist_http_request *req,
291 const char *name, const char *value) {
292 if (strcmp(name, ":method") == 0) {
293 req->method = cwist_http_string_to_method(value);
294 } else if (strcmp(name, ":path") == 0) {
295 h3_parse_path(req, value);
296 } else if (strcmp(name, ":authority") == 0 || strcmp(name, "host") == 0) {
297 cwist_http_header_add(&req->headers, "host", value);
298 } else if (strcmp(name, ":scheme") == 0) {
299 /* RFC 9114: silently ignore pseudo-headers we don't need to expose */
300 } else if (strcmp(name, "content-length") == 0) {
301 char *endptr = NULL;
302 unsigned long long cl = strtoull(value, &endptr, 10);
303 if (endptr && *endptr == '\0') {
304 req->content_length = (size_t)cl;
305 }
306 cwist_http_header_add(&req->headers, name, value);
307 } else if (strcmp(name, "content-type") == 0) {
308 cwist_http_header_add(&req->headers, name, value);
309 } else if (strcmp(name, "cookie") == 0) {
310 cwist_http_header_add(&req->headers, name, value);
311 } else if (strcmp(name, "authorization") == 0) {
312 cwist_http_header_add(&req->headers, name, value);
313 } else if (strcmp(name, "accept") == 0) {
314 cwist_http_header_add(&req->headers, name, value);
315 } else if (strcmp(name, "user-agent") == 0) {
316 cwist_http_header_add(&req->headers, name, value);
317 } else if (strcmp(name, "accept-encoding") == 0) {
318 cwist_http_header_add(&req->headers, name, value);
319 } else if (strcmp(name, "accept-language") == 0) {
320 cwist_http_header_add(&req->headers, name, value);
321 } else if (strcmp(name, "referer") == 0 || strcmp(name, "referrer") == 0) {
322 cwist_http_header_add(&req->headers, "referer", value);
323 } else if (strcmp(name, "origin") == 0) {
324 cwist_http_header_add(&req->headers, name, value);
325 } else if (strcmp(name, "x-requested-with") == 0) {
326 cwist_http_header_add(&req->headers, name, value);
327 } else if (strcmp(name, "priority") == 0) {
328 /* RFC 9218 Extensible Priorities: u=urgency, i=incremental */
329 cwist_http_header_add(&req->headers, name, value);
330 /* Parse urgency value (u=N) where N is 0-7 */
331 const char *u = strstr(value, "u=");
332 if (u) {
333 int urgency = atoi(u + 2);
334 if (urgency >= 0 && urgency <= 7) {
335 /* Map HTTP urgency (0=high, 7=low) to lsquic priority (1=high, 256=low) */
336 unsigned pri = 1 + (unsigned)(urgency * 36);
337 if (pri > 256) pri = 256;
338 if (req->private_data) {
339 lsquic_stream_set_priority((lsquic_stream_t *)req->private_data, pri);
340 }
341 }
342 }
343 } else if (name[0] != ':') {
344 /* Any other non-pseudo header */
345 cwist_http_header_add(&req->headers, name, value);
346 }
347}

349static void cwist_h3_on_read(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
350 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
351 if (!st) return;
352
353 unsigned char buf[8192];
354 ssize_t nread;
355
356 if (!st->headers_done) {
357 void *hset = lsquic_stream_get_hset(stream);
358 if (hset) {
359 cwist_h3_hset_t *hs = hset;
360 size_t i;
361 for (i = 0; i < hs->count; ++i) {
362 const char *name = lsxpack_header_get_name(&hs->headers[i]);
363 const char *value = lsxpack_header_get_value(&hs->headers[i]);
364 if (name && value) {
365 h3_apply_header(st->req, name, value);
366 if (strcmp(name, ":protocol") == 0 && strcmp(value, "webtransport") == 0) {
367 st->is_webtransport = 1;
368 }
369 }
370 }
371 st->headers_done = 1;
372 }
373 }
374
375 while ((nread = lsquic_stream_read(stream, buf, sizeof(buf))) > 0) {
376 size_t need = st->body_len + (size_t)nread;
377 if (need > CWIST_HTTP_MAX_BODY_SIZE) {
378 /* Body too large: abort stream */
379 lsquic_stream_close(stream);
380 return;
381 }
382 if (need > st->body_cap) {
383 size_t new_cap = st->body_cap ? st->body_cap * 2 : 4096;
384 while (new_cap < need) new_cap *= 2;
385 char *tmp = realloc(st->body, new_cap);
386 if (!tmp) {
387 lsquic_stream_close(stream);
388 return;
389 }
390 st->body = tmp;
391 st->body_cap = new_cap;
392 }
393 memcpy(st->body + st->body_len, buf, (size_t)nread);
394 st->body_len += (size_t)nread;
395 }
396
397 if (nread == 0) {
398 /* End of stream (FIN received) */
399 if (!st->headers_done) {
400 /* Malformed request: no headers before FIN */
401 st->res = cwist_http_response_create();
402 if (st->res) {
403 st->res->status_code = CWIST_HTTP_BAD_REQUEST;
404 }
405 st->response_ready = 1;
406 lsquic_stream_wantread(stream, 0);
407 lsquic_stream_wantwrite(stream, 1);
408 return;
409 }
410
411 if (st->body_len > 0 && st->req && st->req->body) {
412 cwist_sstring_assign_len(st->req->body, st->body, st->body_len);
413 }
414
415 st->res = cwist_http_response_create();
416 if (st->res && st->req) {
417 cwist_http3_context *h3_ctx = (cwist_http3_context *)
418 lsquic_conn_get_ctx(lsquic_stream_conn(stream));
419 if (st->is_webtransport && h3_ctx && h3_ctx->wt_handler) {
420 h3_ctx->wt_handler(st->req, st->res, stream);
421 } else if (h3_ctx && h3_ctx->handler) {
422 h3_ctx->handler(h3_ctx->user_ctx, st->req, st->res);
423 }
424 }
425 st->response_ready = 1;
426 lsquic_stream_wantread(stream, 0);
427 lsquic_stream_wantwrite(stream, 1);
428 } else if (nread < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
429 lsquic_stream_close(stream);
430 }
431}

433static void cwist_h3_on_write(lsquic_stream_t *stream, lsquic_stream_ctx_t *st_h) {
434 h3_stream_ctx_t *st = (h3_stream_ctx_t *)st_h;
435 if (!st || !st->response_ready) return;
436
437 if (st->write_state == 0) {
438 struct lsxpack_header headers_arr[64];
439 char hbuf[8192];
440 size_t hbuf_off = 0;
441 size_t hdr_count = 0;
442
443 /* :status */
444 char status_str[16];
445 snprintf(status_str, sizeof(status_str), "%d", st->res->status_code);
446 size_t slen = strlen(status_str);
447 if (hdr_count < 64 && hbuf_off + 7 + 2 + slen <= sizeof(hbuf)) {
448 memcpy(hbuf + hbuf_off, ":status", 7);
449 memcpy(hbuf + hbuf_off + 9, status_str, slen);
450 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
451 0, 7, 9, slen);
452 hbuf_off += 9 + slen;
453 hdr_count++;
454 }
455
456 /* content-length */
457 size_t body_len = 0;
458 if (st->res->use_file_stream) body_len = st->res->file_stream_len;
459 else if (st->res->is_ptr_body) body_len = st->res->ptr_body_len;
460 else if (st->res->body) body_len = st->res->body->size;
461
462 if (body_len > 0 && hdr_count < 64) {
463 char cl_str[32];
464 snprintf(cl_str, sizeof(cl_str), "%zu", body_len);
465 size_t cl_name_len = strlen("content-length");
466 size_t cl_val_len = strlen(cl_str);
467 size_t total = cl_name_len + 2 + cl_val_len;
468 if (hbuf_off + total <= sizeof(hbuf)) {
469 memcpy(hbuf + hbuf_off, "content-length", cl_name_len);
470 memcpy(hbuf + hbuf_off + cl_name_len + 2, cl_str, cl_val_len);
471 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
472 0, cl_name_len, cl_name_len + 2, cl_val_len);
473 hbuf_off += total;
474 hdr_count++;
475 }
476 }
477
478 /* content-type (if present) */
479 if (st->res->headers) {
480 char *ct = cwist_http_header_get(st->res->headers, "content-type");
481 if (ct && hdr_count < 64) {
482 size_t klen = strlen("content-type");
483 size_t vlen = strlen(ct);
484 if (hbuf_off + klen + 2 + vlen <= sizeof(hbuf)) {
485 memcpy(hbuf + hbuf_off, "content-type", klen);
486 memcpy(hbuf + hbuf_off + klen + 2, ct, vlen);
487 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
488 0, klen, klen + 2, vlen);
489 hbuf_off += klen + 2 + vlen;
490 hdr_count++;
491 }
492 }
493 }
494
495 /* user headers (skip content-length/content-type already handled) */
496 cwist_http_header_node *node = st->res->headers;
497 while (node && hdr_count < 64) {
498 if (node->key && node->key->data && node->value && node->value->data) {
499 /* Skip pseudo-headers and duplicates we already sent */
500 if (node->key->data[0] == ':') {
501 node = node->next;
502 continue;
503 }
504 if (strncasecmp(node->key->data, "content-length", node->key->size) == 0 ||
505 strncasecmp(node->key->data, "content-type", node->key->size) == 0) {
506 node = node->next;
507 continue;
508 }
509 size_t klen = node->key->size;
510 size_t vlen = node->value->size;
511 if (hbuf_off + klen + 2 + vlen <= sizeof(hbuf)) {
512 memcpy(hbuf + hbuf_off, node->key->data, klen);
513 memcpy(hbuf + hbuf_off + klen + 2, node->value->data, vlen);
514 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
515 0, klen, klen + 2, vlen);
516 hbuf_off += klen + 2 + vlen;
517 hdr_count++;
518 }
519 }
520 node = node->next;
521 }
522
523 lsquic_http_headers_t headers = {
524 .count = (unsigned)hdr_count,
525 .headers = headers_arr,
526 };
527 if (lsquic_stream_send_headers(stream, &headers, body_len == 0) != 0) {
528 lsquic_stream_close(stream);
529 return;
530 }
531 st->write_state = 1;
532 if (body_len == 0) {
533 st->write_state = 2;
534 lsquic_stream_wantwrite(stream, 0);
535 return;
536 }
537 }
538
539 if (st->write_state == 1 && st->res) {
540 size_t body_len = 0;
541 const char *body_data = NULL;
542
543 if (st->res->use_file_stream) {
544 /* Read file chunk into temporary buffer and write */
545 if (st->res->file_stream_fd >= 0 && st->res->file_stream_len > 0) {
546 static __thread char file_buf[65536];
547 off_t offset = st->res->file_stream_offset + (off_t)st->body_sent;
548 size_t to_read = st->res->file_stream_len - st->body_sent;
549 if (to_read > sizeof(file_buf)) to_read = sizeof(file_buf);
550 ssize_t nr = pread(st->res->file_stream_fd, file_buf, to_read, offset);
551 if (nr > 0) {
552 ssize_t nw = lsquic_stream_write(stream, file_buf, (size_t)nr);
553 if (nw < 0) {
554 lsquic_stream_close(stream);
555 return;
556 }
557 st->body_sent += (size_t)nw;
558 } else if (nr == 0) {
559 /* EOF: mark everything as sent */
560 st->body_sent = st->res->file_stream_len;
561 } else if (nr < 0) {
562 if (errno == EAGAIN || errno == EWOULDBLOCK) {
563 /* Retry next tick */
564 } else {
565 lsquic_stream_close(stream);
566 return;
567 }
568 }
569 body_len = st->res->file_stream_len;
570 }
571 } else if (st->res->is_ptr_body) {
572 body_len = st->res->ptr_body_len;
573 body_data = (const char *)st->res->ptr_body;
574 } else if (st->res->body) {
575 body_len = st->res->body->size;
576 body_data = st->res->body->data;
577 }
578
579 if (body_data && body_len > 0 && st->body_sent < body_len) {
580 ssize_t n = lsquic_stream_write(stream, body_data + st->body_sent,
581 body_len - st->body_sent);
582 if (n < 0) {
583 lsquic_stream_close(stream);
584 return;
585 }
586 st->body_sent += (size_t)n;
587 }
588
589 if (st->body_sent >= body_len) {
590 st->write_state = 2;
591 lsquic_stream_shutdown(stream, 1);

826cwist_error_t cwist_http3_server_loop(int udp_fd,
827 cwist_http3_context *ctx,
828 cwist_http3_request_handler_func handler,
829 void *user_ctx) {
830 cwist_error_t err = make_error(CWIST_ERR_INT16);
831 if (udp_fd < 0 || !ctx || !ctx->ssl_ctx || !handler) {
832 err.error.err_i16 = -1;
833 return err;
834 }
835
836 struct lsquic_engine_settings settings;
837 lsquic_engine_init_settings(&settings, LSENG_HTTP_SERVER);
838 settings.es_versions = (1 << LSQVER_I001) | (1 << LSQVER_I002);
839 settings.es_init_max_data = 1048576;
840 settings.es_init_max_stream_data_bidi_local = 524288;
841 settings.es_init_max_stream_data_bidi_remote = 524288;
842 settings.es_max_streams_in = 100;
843 settings.es_support_push = ctx->push_enabled;
844 settings.es_allow_migration = ctx->allow_migration ? ctx->allow_migration : 1;
845 settings.es_max_delayed_0rtt_packets = 32;
846 settings.es_datagrams = ctx->datagram_enabled;
847
848 struct lsquic_engine_api api = {
849 .ea_stream_if = &cwist_h3_stream_if,
850 .ea_stream_if_ctx = ctx,
851 .ea_packets_out = cwist_h3_packets_out,
852 .ea_packets_out_ctx = &udp_fd,
853 .ea_get_ssl_ctx = cwist_h3_get_ssl_ctx,
854 .ea_hsi_if = &cwist_h3_hset_if,
855 .ea_hsi_ctx = NULL,
856 .ea_settings = &settings,
857 .ea_alpn = "h3",
858 };
859
860 lsquic_engine_t *engine = lsquic_engine_new(LSENG_HTTP_SERVER, &api);
861 if (!engine) {
862 err.error.err_i16 = -1;
863 return err;
864 }
865
866 ctx->engine = engine;
867 ctx->handler = handler;
868 ctx->user_ctx = user_ctx;
869 ctx->udp_fd = udp_fd;
870 ctx->running = 1;
871
872 printf("[HTTP/3] Listening on UDP socket %d\n", udp_fd);
873
874 struct sockaddr_storage local_addr;
875 socklen_t local_addr_len = sizeof(local_addr);
876 memset(&local_addr, 0, sizeof(local_addr));
877 if (getsockname(udp_fd, (struct sockaddr *)&local_addr, &local_addr_len) != 0) {
878 local_addr_len = 0;
879 }
880
881 /* Make socket non-blocking for polling */
882 int flags = fcntl(udp_fd, F_GETFL, 0);
883 if (flags >= 0) fcntl(udp_fd, F_SETFL, flags | O_NONBLOCK);
884
885 /* Enable ECN reception for congestion control feedback */
886 int on = 1;
887 setsockopt(udp_fd, IPPROTO_IP, IP_RECVTOS, &on, sizeof(on));
888#ifdef IPV6_RECVTCLASS
889 setsockopt(udp_fd, IPPROTO_IPV6, IPV6_RECVTCLASS, &on, sizeof(on));
890#endif
891
892 unsigned char *pkt_buf = malloc(65535);
893 if (!pkt_buf) {
894 lsquic_engine_destroy(engine);
895 ctx->engine = NULL;
896 err.error.err_i16 = -1;
897 return err;
898 }
899
900#ifdef __linux__
901 int epoll_fd = epoll_create1(0);
902 if (epoll_fd < 0) {
903 err.error.err_i16 = -1;
904 lsquic_engine_destroy(engine);
905 ctx->engine = NULL;
906 return err;
907 }
908 struct epoll_event ev;
909 ev.events = EPOLLIN;
910 ev.data.fd = udp_fd;
911 if (epoll_ctl(epoll_fd, EPOLL_CTL_ADD, udp_fd, &ev) < 0) {
912 close(epoll_fd);
913 err.error.err_i16 = -1;
914 lsquic_engine_destroy(engine);
915 ctx->engine = NULL;
916 return err;
917 }
918#endif
919
920 while (ctx && ctx->running && atomic_load(&g_cwist_running)) {
921 int diff = 100000; /* default 100 ms in microseconds */
922 if (lsquic_engine_earliest_adv_tick(engine, &diff)) {
923 if (diff <= 0)
924 diff = 0;
925 else if (diff > 1000000)
926 diff = 1000000;
927 }
928
929#ifdef __linux__
930 struct epoll_event events[1];
931 int pret = epoll_wait(epoll_fd, events, 1, diff / 1000);
932 if (pret < 0) {
933 if (errno == EINTR)
934 continue;
935 break;
936 }
937 if (pret > 0 && (events[0].events & (EPOLLERR | EPOLLHUP))) {
938 fprintf(stderr, "[HTTP/3] UDP socket error, exiting loop.\n");
939 break;
940 }
941 if (pret > 0 && (events[0].events & EPOLLIN)) {
942#else
943 struct pollfd pfd = { .fd = udp_fd, .events = POLLIN };
944 int pret = poll(&pfd, 1, diff / 1000);
945
946 if (pret < 0) {
947 if (errno == EINTR)
948 continue;
949 if (errno == EBADF) {
950 fprintf(stderr, "[HTTP/3] UDP socket closed, exiting loop.\n");
951 break;
952 }
953 /* Other fatal poll errors */
954 break;
955 }
956
957 if (pret > 0) {
958 if (pfd.revents & (POLLERR | POLLNVAL)) {
959 fprintf(stderr, "[HTTP/3] UDP socket error, exiting loop.\n");
960 break;
961 }
962 if (pfd.revents & POLLIN) {
963#endif
964 struct sockaddr_storage peer_addr;
965 socklen_t peer_addr_len = sizeof(peer_addr);
966 struct msghdr msg = {0};
967 struct iovec iov = { pkt_buf, 65535 };
968 msg.msg_name = &peer_addr;
969 msg.msg_namelen = sizeof(peer_addr);
970 msg.msg_iov = &iov;
971 msg.msg_iovlen = 1;
972
973 /* ECN support: allocate cmsg buffer */
974 char cmsg_buf[CMSG_SPACE(sizeof(int))];
975 msg.msg_control = cmsg_buf;
976 msg.msg_controllen = sizeof(cmsg_buf);
977
978 ssize_t nr = recvmsg(udp_fd, &msg, 0);
979 if (nr > 0) {
980 int ecn = 0;
981 for (struct cmsghdr *cmsg = CMSG_FIRSTHDR(&msg);
982 cmsg != NULL;
983 cmsg = CMSG_NXTHDR(&msg, cmsg)) {
984 if (cmsg->cmsg_level == IPPROTO_IP &&
985 cmsg->cmsg_type == IP_TOS) {
986 ecn = *(int *)CMSG_DATA(cmsg) & 0x3;
987 break;
988 }
989#ifdef IPV6_TCLASS
990 if (cmsg->cmsg_level == IPPROTO_IPV6 &&
991 cmsg->cmsg_type == IPV6_TCLASS) {
992 ecn = *(int *)CMSG_DATA(cmsg) & 0x3;
993 break;
994 }
995#endif
996 }
997 lsquic_engine_packet_in(engine, pkt_buf, (size_t)nr,
998 local_addr_len ? (struct sockaddr *)&local_addr : NULL,
999 (struct sockaddr *)&peer_addr,
1000 ctx, ecn);
1001 } else if (nr < 0 && errno != EAGAIN && errno != EWOULDBLOCK) {
1002 if (errno == ECONNREFUSED || errno == ENETUNREACH ||
1003 errno == EHOSTUNREACH) {
1004 /* Transient error, keep going */
1005 } else if (errno == EBADF) {
1006 fprintf(stderr, "[HTTP/3] UDP socket closed.\n");
1007 break;
1008 }
1009 }
1010#ifdef __linux__
1011 }
1012#else
1013 }
1014 }
1015#endif
1016
1017 lsquic_engine_process_conns(engine);
1018 }
1019
1020#ifdef __linux__
1021 if (epoll_fd >= 0) close(epoll_fd);
1022#endif
1023
1024 free(pkt_buf);
1025 if (ctx && ctx->engine) {
1026 lsquic_engine_destroy((lsquic_engine_t *)ctx->engine);
1027 ctx->engine = NULL;
1028 }
1029 err.error.err_i16 = 0;
1030 return err;
1031}

1045int cwist_http3_push_resource(cwist_http_request *req,
1046 const char *path,
1047 const char *content_type) {
1048 if (!req || !req->private_data || !path) return -1;
1049 lsquic_stream_t *stream = (lsquic_stream_t *)req->private_data;
1050 lsquic_conn_t *conn = lsquic_stream_conn(stream);
1051 if (!conn || !lsquic_conn_is_push_enabled(conn)) return -1;
1052
1053 /* Build push headers */
1054 struct lsxpack_header headers_arr[4];
1055 char hbuf[2048];
1056 size_t hbuf_off = 0;
1057 size_t hdr_count = 0;
1058
1059 /* :method = GET */
1060 const char *method = "GET";
1061 size_t mlen = strlen(method);
1062 if (hbuf_off + 7 + 2 + mlen <= sizeof(hbuf)) {
1063 memcpy(hbuf + hbuf_off, ":method", 7);
1064 memcpy(hbuf + hbuf_off + 9, method, mlen);
1065 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
1066 0, 7, 9, mlen);
1067 hbuf_off += 9 + mlen;
1068 hdr_count++;
1069 }
1070
1071 /* :path */
1072 size_t plen = strlen(path);
1073 if (hbuf_off + 5 + 2 + plen <= sizeof(hbuf)) {
1074 memcpy(hbuf + hbuf_off, ":path", 5);
1075 memcpy(hbuf + hbuf_off + 7, path, plen);
1076 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
1077 0, 5, 7, plen);
1078 hbuf_off += 7 + plen;
1079 hdr_count++;
1080 }
1081
1082 /* :authority (copy from original request if available) */
1083 char *authority = cwist_http_header_get(req->headers, "host");
1084 if (!authority) authority = "localhost";
1085 size_t alen = strlen(authority);
1086 if (hbuf_off + 10 + 2 + alen <= sizeof(hbuf)) {
1087 memcpy(hbuf + hbuf_off, ":authority", 10);
1088 memcpy(hbuf + hbuf_off + 12, authority, alen);
1089 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
1090 0, 10, 12, alen);
1091 hbuf_off += 12 + alen;
1092 hdr_count++;
1093 }
1094
1095 /* content-type (optional) */
1096 if (content_type) {
1097 size_t ctlen = strlen(content_type);
1098 size_t ct_name_len = strlen("content-type");
1099 if (hbuf_off + ct_name_len + 2 + ctlen <= sizeof(hbuf)) {
1100 memcpy(hbuf + hbuf_off, "content-type", ct_name_len);
1101 memcpy(hbuf + hbuf_off + ct_name_len + 2, content_type, ctlen);
1102 lsxpack_header_set_offset2(&headers_arr[hdr_count], hbuf + hbuf_off,
1103 0, ct_name_len, ct_name_len + 2, ctlen);
1104 hbuf_off += ct_name_len + 2 + ctlen;
1105 hdr_count++;
1106 }
1107 }
1108
1109 lsquic_http_headers_t headers = {
1110 .count = (unsigned)hdr_count,
1111 .headers = headers_arr,
1112 };
1113
1114 return lsquic_conn_push_stream(conn, NULL, stream, &headers);
1115}
