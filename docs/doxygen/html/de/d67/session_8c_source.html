36static char *base64_encode(const uint8_t *data, size_t len) {
37 size_t out_len = 4 * ((len + 2) / 3);
38 char *out = cwist_alloc(out_len + 1);
39 if (!out) return NULL;
40 for (size_t i = 0, j = 0; i < len; i += 3, j += 4) {
41 uint32_t v = ((uint32_t)data[i]) << 16;
42 if (i + 1 < len) v |= ((uint32_t)data[i + 1]) << 8;
43 if (i + 2 < len) v |= ((uint32_t)data[i + 2]);
44 out[j] = b64[(v >> 18) & 0x3F];
45 out[j + 1] = b64[(v >> 12) & 0x3F];
46 out[j + 2] = (i + 1 < len) ? b64[(v >> 6) & 0x3F] : '=';
47 out[j + 3] = (i + 2 < len) ? b64[v & 0x3F] : '=';
48 }
49 out[out_len] = '\0';
50 return out;
51}

53static int base64_decode(const char *in, uint8_t *out, size_t out_len) {
54 size_t in_len = strlen(in);
55 if (in_len % 4 != 0) return -1;
56 size_t pad = 0;
57 if (in_len > 0 && in[in_len - 1] == '=') pad++;
58 if (in_len > 1 && in[in_len - 2] == '=') pad++;
59 size_t len = (in_len / 4) * 3 - pad;
60 if (len > out_len) return -1;
61
62 int decode[256];
63 for (int i = 0; i < 256; i++) decode[i] = -1;
64 for (int i = 0; i < 64; i++) decode[(unsigned char)b64[i]] = i;
65
66 size_t j = 0;
67 for (size_t i = 0; i < in_len; i += 4) {
68 int a = decode[(unsigned char)in[i]];
69 int b = decode[(unsigned char)in[i + 1]];
70 int c = decode[(unsigned char)in[i + 2]];
71 int d = decode[(unsigned char)in[i + 3]];
72 if (a < 0 || b < 0) return -1;
73 uint32_t v = ((uint32_t)a << 18) | ((uint32_t)b << 12);
74 if (c >= 0) v |= ((uint32_t)c << 6);
75 if (d >= 0) v |= (uint32_t)d;
76 out[j++] = (v >> 16) & 0xFF;
77 if (c >= 0 && in[i + 2] != '=') out[j++] = (v >> 8) & 0xFF;
78 if (d >= 0 && in[i + 3] != '=') out[j++] = v & 0xFF;
79 }
80 return (int)j;
81}

85static bool hmac_sha256(const char *key, size_t key_len,
86 const char *msg, size_t msg_len,
87 uint8_t out[32]) {
88 unsigned int len = 32;
89 unsigned char *r = HMAC(EVP_sha256(),
90 key, (int)key_len,
91 (const unsigned char *)msg, msg_len,
92 out, &len);
93 return r != NULL && len == 32;

98static char *generate_secret(size_t len) {
99 char *secret = cwist_alloc(len * 2 + 1);
100 if (!secret) return NULL;
101 int fd = open("/dev/urandom", O_RDONLY);
102 if (fd < 0) {
103 cwist_free(secret);
104 return NULL;
105 }
106 unsigned char *buf = cwist_alloc(len);
107 if (!buf) {
108 close(fd);
109 cwist_free(secret);
110 return NULL;
111 }
112 ssize_t n = read(fd, buf, len);
113 close(fd);
114 if (n != (ssize_t)len) {
115 cwist_free(buf);
116 cwist_free(secret);
117 return NULL;
118 }
119 static const char hex[] = "0123456789abcdef";
120 for (size_t i = 0; i < len; i++) {
121 secret[i * 2] = hex[buf[i] >> 4];
122 secret[i * 2 + 1] = hex[buf[i] & 0x0F];
123 }
124 secret[len * 2] = '\0';
125 cwist_free(buf);
126 return secret;
127}

167static bool verify_signature(cwist_app *app, const char *payload_b64, const char *sig_b64) {
168 uint8_t sig[32];
169 if (base64_decode(sig_b64, sig, sizeof(sig)) != 32) return false;
170 uint8_t expected[32];
171 if (!hmac_sha256(app->session_secret, strlen(app->session_secret),
172 payload_b64, strlen(payload_b64), expected)) {
173 return false;
174 }
175 return memcmp(sig, expected, 32) == 0;
176}

179 uint8_t payload[8192];
180 int len = base64_decode(payload_b64, payload, sizeof(payload) - 1);
181 if (len < 0) return NULL;
182 payload[len] = '\0';

184 if (!json) return NULL;

186 if (!map) {

190 cJSON *item = NULL;
191 cJSON_ArrayForEach(item, json) {
192 if (cJSON_IsString(item) && item->string && item->valuestring) {
193 cwist_query_map_set(map, item->string, item->valuestring);
194 }

200cwist_session_t *cwist_session_start(cwist_app *app,
201 cwist_http_request *req,
202 cwist_http_response *res) {
203 if (!app || !req || !res) return NULL;
204 if (!app->session_secret && cwist_app_use_session(app, NULL) != 0) return NULL;
205
206 cwist_session_t *session = cwist_alloc(sizeof(*session));
207 if (!session) return NULL;
208 session->app = app;
209 session->req = req;
210 session->data = cwist_query_map_create();
211 session->modified = false;
212 session->invalidated = false;
213 session->loaded = false;
214
215 cwist_query_map *cookies = cwist_query_map_create();
216 if (cookies) {
217 const char *cookie_header = cwist_http_header_get(req->headers, "Cookie");
218 cwist_cookie_parse(cookies, cookie_header);
219 const char *raw = cwist_cookie_get(cookies, app->session_name);
220 if (raw) {
221 char *copy = cwist_strdup(raw);
222 char *dot = strchr(copy, '.');
223 if (dot) {
224 *dot = '\0';
225 if (verify_signature(app, copy, dot + 1)) {
226 cwist_query_map_destroy(session->data);
227 session->data = parse_payload(copy);
228 if (!session->data) session->data = cwist_query_map_create();
229 session->loaded = true;
230 }
231 }
232 cwist_free(copy);
233 }
234 cwist_query_map_destroy(cookies);
235 }
236
237 req->session = session;
238 return session;
239}

272int cwist_session_commit(cwist_session_t *session, cwist_http_response *res) {
273 if (!session || !res) return -1;
274 if (!session->modified && session->loaded) return 0;
275
276 if (session->invalidated) {
277 cwist_cookie_delete(res, session->app->session_name);
278 return 0;
279 }
280
281 json_ctx_t ctx;
282 ctx.json = cJSON_CreateObject();
283 if (!ctx.json) return -1;
284 cwist_query_map_foreach(session->data, add_to_json, &ctx);
285 char *raw = cJSON_PrintUnformatted(ctx.json);
286 cJSON_Delete(ctx.json);
287 if (!raw) return -1;
288
289 char *payload_b64 = base64_encode((const uint8_t *)raw, strlen(raw));
290 free(raw);
291 if (!payload_b64) return -1;
292
293 uint8_t sig[32];
294 if (!hmac_sha256(session->app->session_secret, strlen(session->app->session_secret),
295 payload_b64, strlen(payload_b64), sig)) {
296 cwist_free(payload_b64);
297 return -1;
298 }
299 char *sig_b64 = base64_encode(sig, sizeof(sig));
300 if (!sig_b64) {
301 cwist_free(payload_b64);
302 return -1;
303 }
304
305 char cookie_value[16384];
306 snprintf(cookie_value, sizeof(cookie_value), "%s.%s", payload_b64, sig_b64);
307 cwist_free(payload_b64);
308 cwist_free(sig_b64);
309
310 cwist_cookie_options opts = {0};
311 opts.path = "/";
312 opts.max_age_seconds = session->app->session_max_age;
313 opts.http_only = true;
314 opts.same_site = "Lax";
315
316 int rc = cwist_cookie_set(res, session->app->session_name, cookie_value, &opts);
317 return rc;
318}
