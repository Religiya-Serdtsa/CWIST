24char *cwist_cookie_encode(const char *value) {
25 if (!value) return NULL;
26 size_t len = strlen(value);
27 size_t max = len * 3 + 1;
28 char *out = cwist_alloc(max);
29 if (!out) return NULL;
30 size_t j = 0;
31 for (size_t i = 0; i < len; i++) {
32 unsigned char c = (unsigned char)value[i];
33 if (needs_url_encode((char)c)) {
34 snprintf(out + j, max - j, "%%%02X", c);
35 j += 3;
36 } else {
37 out[j++] = (char)c;
38 }
39 }
40 out[j] = '\0';
41 return out;
42}

44int cwist_cookie_decode(const char *in, char *out, size_t out_len) {
45 if (!in || !out || out_len == 0) return -1;
46 size_t len = strlen(in);
47 size_t j = 0;
48 for (size_t i = 0; i < len; i++) {
49 if (j >= out_len - 1) return -1;
50 if (in[i] == '%' && i + 2 < len) {
51 unsigned int hex;
52 if (sscanf(in + i + 1, "%2x", &hex) != 1) {
53 out[j++] = in[i];
54 continue;
55 }
56 out[j++] = (char)hex;
57 i += 2;
58 } else if (in[i] == '+') {
59 out[j++] = ' ';
60 } else {
61 out[j++] = in[i];
62 }
63 }
64 out[j] = '\0';
65 return (int)j;
66}

68void cwist_cookie_parse(cwist_query_map *map, const char *header) {
69 if (!map || !header || !*header) return;

71 char *buf = cwist_strdup(header);
72 if (!buf) return;

74 char *save = NULL;

76 while (pair) {
77 /* trim leading whitespace */

79 char *eq = strchr(pair, '=');
80 if (eq) {
81 *eq = '\0';
