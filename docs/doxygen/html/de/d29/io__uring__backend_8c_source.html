189void cwist_uring_backend_destroy(cwist_uring_backend_t *be) {
190 if (!be) return;
191 atomic_store_explicit(&be->stopped, true, memory_order_release);
192 if (be->buf_pool.base) {
193 sys_io_uring_register(be->ring_fd, IORING_UNREGISTER_BUFFERS, NULL, 0);
194 munmap(be->buf_pool.base, be->buf_pool.buf_size * be->buf_pool.buf_count);
195 ttak_mem_free(be->buf_pool.bid_map);
196 }
197 if (be->sq.sqes) munmap(be->sq.sqes, be->sq.sqe_sz);
198 if (be->cq.cqes) munmap((char *)be->cq.cqes - be->params.cq_off.cqes, be->cq.ring_sz);
199 if (be->sq.array) munmap((char *)be->sq.array - be->params.sq_off.array, be->sq.ring_sz);
200 if (be->streams) ttak_mem_free(be->streams);
201 if (be->free_stack) ttak_mem_free(be->free_stack);
202 if (be->ring_fd >= 0) close(be->ring_fd);
203 pthread_mutex_destroy(&be->stream_lock);
204 ttak_mem_free(be);
205}

219cwist_core_stream_t *cwist_uring_stream_register(cwist_uring_backend_t *be, int fd, uint32_t proto) {
220 if (!be || fd < 0) return NULL;
221 pthread_mutex_lock(&be->stream_lock);
222 uint32_t top = atomic_load_explicit(&be->free_top, memory_order_acquire);
223 if (top == 0) {
224 pthread_mutex_unlock(&be->stream_lock);
225 return NULL;
226 }
227 uint32_t idx = be->free_stack[--top];
228 atomic_store_explicit(&be->free_top, top, memory_order_release);
229 pthread_mutex_unlock(&be->stream_lock);

232 uint32_t index = s->index;
233 memset(s, 0, sizeof(*s));
234 s->index = index;
235 s->fd = fd;
236 s->protocol = proto;
237 s->generation = atomic_fetch_add_explicit(&be->global_gen, 1, memory_order_relaxed);
238 atomic_init(&s->pending_io_count, 0);

276 if (!be || !stream || stream->is_dead) return false;

355static void process_cqe(cwist_uring_backend_t *be, struct io_uring_cqe *cqe) {
356 uint64_t ud = cqe->user_data;
357 if (!ud) return;
358 uint32_t gen = DECODE_GEN(ud), idx = DECODE_IDX(ud);
359 bool is_send = DECODE_IS_SEND(ud);
360 if (idx >= be->cfg.max_streams) return;
361 cwist_core_stream_t *stream = &be->streams[idx];
362 if (stream->generation != gen) return;
363 stream->deadline_ts = get_time_ms() + 3000;
364 if (cqe->res < 0 && cqe->res != -EAGAIN && cqe->res != -EINTR && cqe->res != -ECANCELED) stream->is_dead = true;
365 cwist_uring_cqe_cb cb = is_send ? stream->send_cb : stream->recv_cb;
366 if (cb) cb(stream, cqe->res, cqe->flags, stream->user_data);
367 if (cqe->flags & IORING_CQE_F_BUFFER) cwist_uring_buf_release(be, (uint16_t)(cqe->flags >> 16));
368 if (atomic_fetch_sub_explicit(&stream->pending_io_count, 1, memory_order_acq_rel) == 1) {
369 if (stream->is_dead) finalize_stream_destruction(be, stream);
370 }
371}

373int cwist_uring_poll_cq(cwist_uring_backend_t *be, unsigned min_wait, int timeout_ms) {
374 (void)timeout_ms;
375 if (!be) return -EINVAL;
376 if (atomic_load_explicit(&be->pending_sqes, memory_order_acquire) > 0) flush_sq_internal(be);
377 unsigned flags = (min_wait > 0) ? IORING_ENTER_GETEVENTS : 0;
378 if (sys_io_uring_enter(be->ring_fd, 0, min_wait, flags, NULL) < 0) return -errno;
379 struct cwist_uring_cq_ring *cq = &be->cq;
380 uint32_t head = atomic_load_explicit((_Atomic uint32_t *)cq->head, memory_order_acquire);
381 uint32_t tail = atomic_load_explicit((_Atomic uint32_t *)cq->tail, memory_order_acquire);
382 int processed = 0;
383 while (head != tail) {
384 process_cqe(be, &cq->cqes[head & *cq->ring_mask]);
385 head++;
386 processed++;
387 }
388 atomic_store_explicit((_Atomic uint32_t *)cq->head, head, memory_order_release);
389 return processed;
390}

392void cwist_uring_backend_run(cwist_uring_backend_t *be) {
393 if (!be) return;
394 atomic_store_explicit(&be->running, true, memory_order_release);
395 uint64_t last_sweep = get_time_ms();
396 while (!atomic_load_explicit(&be->stopped, memory_order_acquire)) {
397 if (atomic_load_explicit(&be->pending_sqes, memory_order_acquire) > 0) flush_sq_internal(be);
398 if (cwist_uring_poll_cq(be, 1, 100) < 0) break;
399 uint64_t now = get_time_ms();
400 if (now - last_sweep >= 1000) {
401 for (uint32_t i = 0; i < be->cfg.max_streams; ++i) {
402 cwist_core_stream_t *s = &be->streams[i];
403 if (s->fd >= 0 && !s->is_dead && now > s->deadline_ts) cwist_uring_stream_unregister(be, s);
404 }
405 last_sweep = now;
406 }
407 }
408 atomic_store_explicit(&be->running, false, memory_order_release);
409}
